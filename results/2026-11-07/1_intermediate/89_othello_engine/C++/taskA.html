34bool inside(int r, int c) {
35 return r >= 0 && r < 8 && c >= 0 && c < 8;
36}
37string initial() {
38 string b(64, '.');
39 b[27] = b[36] = 'W';
40 b[28] = b[35] = 'B';
41 return b;

43vector<int> flips(const string &b, char s, int p) {
44 vector<int> out;
45 if (b[p] != '.')
46 return out;
47 int row = p / 8, col = p % 8;
48 for (int dr = -1; dr <= 1; ++dr)
49 for (int dc = -1; dc <= 1; ++dc) {
50 if (!dr && !dc)
51 continue;
52 int r = row + dr, c = col + dc;
53 vector<int> line;
54 while (inside(r, c) && b[8 * r + c] == other(s)) {
55 line.push_back(8 * r + c);
56 r += dr;
57 c += dc;
58 }
59 if (!line.empty() && inside(r, c) && b[8 * r + c] == s)
60 out.insert(out.end(), line.begin(), line.end());
61 }
62 sort(out.begin(), out.end());
63 return out;
64}
65vector<int> legal(const string &b, char s) {
66 vector<int> ms;
67 for (int p = 0; p < 64; ++p)
68 if (b[p] == '.' && !flips(b, s, p).empty())
69 ms.push_back(p);
70 return ms;
71}
72string place(string b, char s, int p) {
73 for (int q : flips(b, s, p))
74 b[q] = s;
75 b[p] = s;
76 return b;
77}
78int countDisc(const string &b, char s) {
79 return count(b.begin(), b.end(), s);
80}
81bool terminal(const string &b) {
82 return legal(b, 'B').empty() && legal(b, 'W').empty();
83}
84string gameOver(const string &b) {
85 int x = countDisc(b, 'B'), y = countDisc(b, 'W');
86 return "game over B " + to_string(x) + " W " + to_string(y) + " " +
87 (x > y ? "winner B"
88 : x < y ? "winner W"
89 : "draw");
90}
91array<int, 4> evaluate(const string &b, char s) {
92 int w = 0;
93 for (int i = 0; i < 64; ++i)
94 w += b[i] == s ? weights[i] : b[i] == other(s) ? -weights[i] : 0;
95 int own = legal(b, s).size(), enemy = legal(b, other(s)).size();
96 return {w + 10 * (own - enemy), w, own, enemy};
97}
98map<tuple<string, char, int>, int> searchCache, perftCache;
99int search(const string &b, char s, int d) {
100 auto key = make_tuple(b, s, d);
101 auto hit = searchCache.find(key);
102 if (hit != searchCache.end())
103 return hit->second;
104 vector<int> ms = legal(b, s);
105 char o = other(s);
106 int v;
107 if (ms.empty() && legal(b, o).empty())
108 v = (d < 0 ? 1 : 1000) * (countDisc(b, s) - countDisc(b, o));
109 else if (d == 0)
110 v = evaluate(b, s)[0];
111 else if (ms.empty())
112 v = -search(b, o, d < 0 ? -1 : d - 1);
113 else {
114 v = -1000000;
115 for (int p : ms)
116 v = max(v, -search(place(b, s, p), o, d < 0 ? -1 : d - 1));
117 }
118 if (searchCache.size() > 300000)
119 searchCache.clear();
120 searchCache[key] = v;
121 return v;
122}
123using Ranking = pair<int, int>;
124vector<Ranking> ranked(const string &b, char s, int d) {
125 vector<Ranking> rows;
126 for (int p : legal(b, s))
127 rows.emplace_back(-search(place(b, s, p), other(s), d < 0 ? -1 : d - 1), p);
128 sort(rows.begin(), rows.end(), [](Ranking a, Ranking b) {
129 return a.first != b.first ? a.first > b.first : a.second < b.second;
130 });
131 return rows;
132}
133pair<vector<string>, string> variation(string b, char s, int d) {
134 vector<string> steps;
135 while (d != 0 && !terminal(b)) {
136 vector<Ranking> ms = ranked(b, s, d);
137 if (ms.empty())
138 steps.push_back("pass");
139 else {
140 int p = ms[0].second;
141 steps.push_back(name(p));
142 b = place(b, s, p);
143 }
144 s = other(s);
145 if (d > 0)
146 --d;
147 }
148 return {steps, b};
149}
150int perft(const string &b, char s, int d) {
151 if (!d)
152 return 1;
153 auto key = make_tuple(b, s, d);
154 auto hit = perftCache.find(key);
155 if (hit != perftCache.end())
156 return hit->second;
157 vector<int> ms = legal(b, s);
158 int n = 0;
159 if (ms.empty())
160 n = legal(b, other(s)).empty() ? 1 : perft(b, other(s), d - 1);
161 else
162 for (int p : ms)
163 n += perft(place(b, s, p), other(s), d - 1);
164 if (perftCache.size() > 200000)
165 perftCache.clear();
166 perftCache[key] = n;
167 return n;
168}
169pair<int, int> stable(const string &b) {
170 array<bool, 64> set{};
171 array<pair<int, int>, 4> axes = {{{0, 1}, {1, 0}, {1, 1}, {1, -1}}};
172 while (true) {
173 vector<int> found;
174 for (int p = 0; p < 64; ++p) {
175 if (b[p] == '.' || set[p])
176 continue;
177 bool ok = true;
178 for (auto [dr, dc] : axes) {
179 bool full = true, anchor = false;
180 for (int sign : {-1, 1}) {
181 int r = p / 8 + dr * sign, c = p % 8 + dc * sign;
182 if (!inside(r, c) || (set[8 * r + c] && b[8 * r + c] == b[p]))
183 anchor = true;
184 while (inside(r, c)) {
185 if (b[8 * r + c] == '.')
186 full = false;
187 r += dr * sign;
188 c += dc * sign;
189 }
190 }
191 if (!full && !anchor) {
192 ok = false;
193 break;
194 }
195 }
196 if (ok)
197 found.push_back(p);
198 }
199 if (found.empty())
200 break;
201 for (int p : found)
202 set[p] = true;
203 }
204 int black = 0, white = 0;
205 for (int i = 0; i < 64; ++i)
206 if (set[i]) {
207 if (b[i] == 'B')
208 ++black;
209 else
210 ++white;
211 }
212 return {black, white};
213}
214int transform(int p, const string &op) {
215 int r = p / 8, c = p % 8;
216 if (op == "rot90")
217 return 8 * c + 7 - r;
218 if (op == "rot180")
219 return 63 - p;
220 if (op == "rot270")
221 return 8 * (7 - c) + r;
222 if (op == "mirror")
223 return 8 * r + 7 - c;
224 if (op == "flip")
225 return 8 * (7 - r) + c;
226 if (op == "diag")
227 return 8 * c + r;
228 return 8 * (7 - c) + 7 - r;
229}
230string transformBoard(const string &b, const string &op) {
231 string a(64, '.');
232 for (int p = 0; p < 64; ++p)
233 a[transform(p, op)] = b[p];
234 return a;
235}

243string randomGame(string b, char s, Random &rng) {
244 while (true) {
245 vector<int> ms = legal(b, s);
246 if (!ms.empty())
247 b = place(b, s, ms[rng.draw() % ms.size()]);
248 else if (legal(b, other(s)).empty())
249 return b;
250 s = other(s);
251 }

321 bool move(int p, bool quiet = false, int value = 1000001) {
322 Position before = snapshot();
323 char s = side;
324 int k = flips(board, s, p).size();
325 board = place(board, s, p);
326 if (!quiet)
327 emit(string(1, s) + " " + name(p) + " flips " + to_string(k) +
328 (value == 1000001 ? "" : " value " + to_string(value)));
329 side = other(s);
330 bool passed = adjust(quiet);
331 history.push_back({s, p, k, passed, before, snapshot()});
332 redo.clear();
333 return passed;
334 }

341 void record(const string &text) {
342 Othello e;
343 vector<int> ps;
344 int passes = 0;
345 for (size_t i = 0; i < text.size(); i += 2) {
346 int p = square(text.substr(i, 2));
347 string why = e.reason(p);
348 if (!why.empty()) {
349 emit("ILLEGAL move " + to_string(i / 2 + 1) + " " + name(p) + " " + why);
350 return;
351 }
352 ps.push_back(p);
353 passes += e.move(p, true);
354 }
355 restore(e.snapshot());
356 history = e.history;
357 redo.clear();
358 string first = name(ps[0]),
359 op = first == "e6" ? "diag"
360 : first == "d3" ? "anti"
361 : first == "c4" ? "rot180"
362 : "",
363 normal;
364 for (int p : ps)
365 normal += name(op.empty() ? p : transform(p, op));
366 emit("record " + to_string(ps.size()) + " moves " + to_string(passes) + " passes");
367 emit("normal " + normal);
368 emit(status());
369 }
370 void sheet() {
371 string b = history.empty() ? board : history[0].before.board;
372 vector<string> fields;
373 for (char c : b)
374 fields.emplace_back(1, c);
375 for (size_t i = 0; i < history.size(); ++i)
376 fields[history[i].square] = to_string(i + 1);
377 emit(" a b c d e f g h");
378 for (int r = 0; r < 8; ++r) {
379 string line = to_string(r + 1);
380 for (int c = 0; c < 8; ++c) {
381 string s = fields[8 * r + c];

385 }

387 void stats() {
388 int total = 0, passes = 0;
389 for (const Move &r : history) {
390 total += r.count;
391 passes += r.passed;
392 }
393 if (history.empty())
394 emit("moves 0");
395 else
396 emit("moves " + to_string(history.size()) + " flips " + to_string(total) + " avg " +
397 average(total, history.size()) + " passes " + to_string(passes));
398 for (char s : {'B', 'W'}) {
399 int n = 0, sum = 0, max = -1, p = -1, corners = 0;
400 for (const Move &r : history)
401 if (r.side == s) {
402 ++n;
403 sum += r.count;
404 if (r.count > max) {
405 max = r.count;
406 p = r.square;
407 }
408 corners += r.square == 0 || r.square == 7 || r.square == 56 || r.square == 63;
409 }
410 if (!n)
411 emit(string(1, s) + " moves 0");
412 else
413 emit(string(1, s) + " moves " + to_string(n) + " flips " + to_string(sum) +
414 " avg " + average(sum, n) + " max " + to_string(max) + " " + name(p) +
415 " corners " + to_string(corners));
416 }
417 }
418 void autoPlay(int black, int white, int64_t seed = 0) {
419 Random rng{seed};
420 string text;
421 int n = 0, passes = 0;
422 while (!over) {
423 vector<int> ms = legal(board, side);
424 int p = seed ? ms[rng.draw() % ms.size()]
425 : ranked(board, side, side == 'B' ? black : white)[0].second;
426 text += name(p);
427 ++n;
428 passes += move(p, true);
429 }
430 emit(string(seed ? "random" : "autoplay") + " " + to_string(n) + " moves " +
431 to_string(passes) + " passes");
432 emit("moves " + text);
433 emit(gameOver(board));
434 }
435 void monteCarlo(int k, int64_t seed) {
436 Random rng{seed};
437 int best = -1, bp = -1;
438 for (int p : legal(board, side)) {
439 int w = 0, d = 0, l = 0;
440 string start = place(board, side, p);
441 for (int i = 0; i < k; ++i) {
442 string end = randomGame(start, other(side), rng);
443 int diff = countDisc(end, side) - countDisc(end, other(side));
444 if (diff > 0)
445 ++w;
446 else if (diff < 0)
447 ++l;
448 else
449 ++d;
450 }
451 emit(name(p) + " " + to_string(w) + " " + to_string(d) + " " + to_string(l));
452 if (2 * w + d > best) {
453 best = 2 * w + d;
454 bp = p;
455 }
456 }
457 emit("best " + name(bp) + " " + to_string(best));
458 }
459 void execute(const string &text) {
460 istringstream in(text);
461 string cmd, x;
462 in >> cmd;
463 vector<string> a;
464 while (in >> x)
465 a.push_back(x);
466 string error;
467 bool known = zero.count(cmd) || numeric.count(cmd) || cmd == "LOAD" || cmd == "TRANSFORM" ||
468 cmd == "AUTOPLAY" || cmd == "MC" || cmd == "FLIPS" || cmd == "MOVE" ||
469 cmd == "PLAY" || cmd == "RECORD";
470 if (!known)
471 error = "unknown command";
472 else if (cmd == "LOAD") {
473 if (a.size() != 2 || !regex_match(a[0], regex("[.BW]{64}")) ||
474 (a[1] != "B" && a[1] != "W"))
475 error = "bad board";
476 } else if (cmd == "MOVE" || cmd == "FLIPS") {
477 if (a.size() != 1)
478 error = "bad arguments";
479 else if (!regex_match(a[0], regex("[a-h][1-8]")))
480 error = "bad square";
481 } else if (cmd == "PLAY" || cmd == "RECORD") {
482 if (a.size() != 1)
483 error = "bad arguments";
484 else if (a[0].size() > 120 || !regex_match(a[0], regex("([a-h][1-8])+")))
485 error = "bad record";
486 } else if (zero.count(cmd)) {
487 if (!a.empty())
488 error = "bad arguments";
489 } else if (cmd == "TRANSFORM") {
490 if (a.size() != 1 || !transforms.count(a[0]))
491 error = "bad arguments";
492 } else {
493 size_t expected = cmd == "AUTOPLAY" || cmd == "MC" ? 2 : 1;
494 bool valid = a.size() == expected;
495 vector<int64_t> nums;
496 for (const string &s : a) {
497 if (!regex_match(s, regex("[0-9]+")))
498 valid = false;
499 else
500 nums.push_back(stoll(s));
501 }
502 if (valid) {
503 int64_t n = nums[0];
504 if (cmd == "AUTOPLAY")
505 valid = n >= 1 && n <= 4 && nums[1] >= 1 && nums[1] <= 4;
506 else if (cmd == "HINT" || cmd == "PV" || cmd == "AI" || cmd == "ANALYZE")
507 valid = n >= 1 && n <= 4;
508 else if (cmd == "PERFT" || cmd == "DIVIDE")
509 valid = n >= 1 && n <= 8;
510 else if (cmd == "MC")
511 valid = n >= 1 && n <= 100 && nums[1] >= 1 && nums[1] <= 2147483646;
512 else if (cmd == "RANDOM")
513 valid = n >= 1 && n <= 2147483646;
514 else
515 valid = n >= 1;
516 }
517 if (!valid)
518 error = "bad arguments";
519 }
520 if (!error.empty()) {
521 emit("ERROR " + error);
522 return;
523 }
524 if (over && active.count(cmd)) {
525 emit("ERROR game over");
526 return;
527 }
528 int n = !a.empty() && regex_match(a[0], regex("[0-9]+")) ? stoi(a[0]) : 0;
529 if (cmd == "NEW") {
530 reset();
531 emit("OK");
532 } else if (cmd == "LOAD") {
533 board = a[0];
534 side = a[1][0];
535 over = false;
536 history.clear();
537 redo.clear();
538 emit("OK");
539 adjust();
540 } else if (cmd == "DUMP")
541 emit("board " + board + " " + side);
542 else if (cmd == "SHOW") {
543 emit(" a b c d e f g h");
544 vector<int> ms = over ? vector<int>{} : legal(board, side);
545 for (int r = 0; r < 8; ++r) {
546 string row = to_string(r + 1);
547 for (int c = 0; c < 8; ++c) {
548 int p = 8 * r + c;
549 row += " ";
550 row += find(ms.begin(), ms.end(), p) != ms.end() ? '*' : board[p];
551 }
552 emit(row);
553 }
554 emit(status());
555 } else if (cmd == "SCORE")
556 emit("score B " + to_string(countDisc(board, 'B')) + " W " +
557 to_string(countDisc(board, 'W')) + " empty " + to_string(countDisc(board, '.')));
558 else if (cmd == "LEGAL") {
559 vector<int> ms = over ? vector<int>{} : legal(board, side);
560 string out = "legal " + to_string(ms.size());
561 for (int p : ms)
562 out += " " + name(p);
563 emit(out);
564 } else if (cmd == "MOVE" || cmd == "FLIPS") {
565 int p = square(a[0]);
566 string why = reason(p);
567 if (!why.empty())
568 emit("ILLEGAL " + why);
569 else if (cmd == "MOVE")
570 move(p);
571 else {
572 vector<int> f = flips(board, side, p);
573 string out = "flips " + a[0] + " " + to_string(f.size());
574 for (int q : f)
575 out += " " + name(q);
576 emit(out);
577 }
578 } else if (cmd == "PLAY") {
579 for (size_t i = 0; i < a[0].size(); i += 2) {
580 int p = square(a[0].substr(i, 2));
581 string why = reason(p);
582 if (!why.empty()) {
583 emit("ILLEGAL move " + to_string(i / 2 + 1) + " " + name(p) + " " + why);
584 break;
585 }
586 move(p);
587 }
588 } else if (cmd == "RECORD")
589 record(a[0]);
590 else if (cmd == "UNDO") {
591 if (size_t(n) > history.size())
592 emit("ERROR cannot undo");
593 else {
594 auto first = history.end() - n;
595 restore(first->before);
596 redo.insert(redo.begin(), first, history.end());
597 history.erase(first, history.end());
598 emit(status());
599 }
600 } else if (cmd == "REDO") {
601 if (size_t(n) > redo.size())
602 emit("ERROR cannot redo");
603 else {
604 history.insert(history.end(), redo.begin(), redo.begin() + n);
605 restore(redo[n - 1].after);
606 redo.erase(redo.begin(), redo.begin() + n);
607 emit(status());
608 }
609 } else if (cmd == "HISTORY") {
610 emit("history " + to_string(history.size()));
611 for (size_t i = 0; i < history.size(); ++i) {
612 const Move &r = history[i];
613 emit(to_string(i + 1) + " " + r.side + " " + name(r.square) + " " +
614 to_string(r.count) + (r.passed ? " pass" : ""));
615 }
616 } else if (cmd == "SHEET")
617 sheet();
618 else if (cmd == "STATS")
619 stats();
620 else if (cmd == "TRANSFORM") {
621 board = transformBoard(board, a[0]);
622 for (vector<Move> *rows : {&history, &redo})
623 for (Move &r : *rows) {
624 r.square = transform(r.square, a[0]);
625 r.before.board = transformBoard(r.before.board, a[0]);
626 r.after.board = transformBoard(r.after.board, a[0]);
627 }
628 emit("OK");
629 } else if (cmd == "EVAL") {
630 auto e = evaluate(board, side);
631 emit("eval " + to_string(e[0]) + " weights " + to_string(e[1]) + " mobility " +
632 to_string(e[2]) + " " + to_string(e[3]));
633 } else if (cmd == "STABLE") {
634 auto [b, w] = stable(board);
635 emit("stable B " + to_string(b) + " W " + to_string(w));
636 } else if (cmd == "HINT")
637 for (auto [v, p] : ranked(board, side, n))
638 emit(name(p) + " " + to_string(v));
639 else if (cmd == "PV") {
640 auto pv = variation(board, side, n);
641 emit("pv " + to_string(search(board, side, n)) +
642 (pv.first.empty() ? "" : " " + joined(pv.first)));
643 } else if (cmd == "AI") {
644 auto [v, p] = ranked(board, side, n)[0];
645 move(p, false, v);
646 } else if (cmd == "ANALYZE") {
647 map<char, int> loss = {{'B', 0}, {'W', 0}};
648 for (size_t i = 0; i < history.size(); ++i) {
649 const Move &r = history[i];
650 vector<Ranking> rows = ranked(r.before.board, r.side, n);
651 auto [best, bp] = rows[0];
652 int v = find_if(rows.begin(), rows.end(), [&](Ranking row) {
653 return row.second == r.square;
654 })->first;
655 loss[r.side] += best - v;
656 emit(to_string(i + 1) + " " + r.side + " " + name(r.square) + " " + to_string(v) +
657 " best " + name(bp) + " " + to_string(best));
658 }
659 emit("loss B " + to_string(loss['B']) + " W " + to_string(loss['W']));
660 } else if (cmd == "SOLVE") {
661 if (countDisc(board, '.') >= 9)
662 emit("ERROR too many empties");
663 else {
664 for (auto [v, p] : ranked(board, side, -1))
665 emit(name(p) + " " + to_string(v));
666 auto pv = variation(board, side, -1);
667 emit("line " + joined(pv.first));
668 emit("final B " + to_string(countDisc(pv.second, 'B')) + " W " +
669 to_string(countDisc(pv.second, 'W')));
670 }
671 } else if (cmd == "AUTOPLAY")
672 autoPlay(n, stoi(a[1]));
673 else if (cmd == "RANDOM")
674 autoPlay(0, 0, n);
675 else if (cmd == "MC")
676 monteCarlo(n, stoll(a[1]));
677 else if (cmd == "PERFT")
678 emit("perft " + to_string(n) + " " + to_string(perft(board, side, n)));
679 else if (cmd == "DIVIDE") {
680 int total = 0;
681 for (int p : legal(board, side)) {
682 int v = perft(place(board, side, p), other(side), n - 1);
683 emit(name(p) + " " + to_string(v));
684 total += v;
685 }
686 emit("total " + to_string(total));
687 }
688 }
