106flips <- function(b, s, p) {
107 if (b[p + 1] != '.') {
108 return(numeric())
109 }
110 out <- numeric()
111 row <- floor(p / 8)
112 col <- p %% 8
113 for (dr in -1:1) {
114 for (dc in -1:1) {
115 if (dr == 0 && dc == 0) {
116 next
117 }
118 r <- row + dr
119 c <- col + dc
120 line <- numeric()
121 while (inside(r, c) && b[8 * r + c + 1] == other(s)) {
122 line <- c(line, 8 * r + c)
123 r <- r + dr
124 c <- c + dc
125 }
126 if (length(line) && inside(r, c) && b[8 * r + c + 1] == s) {
127 out <- c(out, line)
128 }
129 }
130 }
131 sort(out)
132}

168search <- function(b, s, d) {
169 key <- position_key(b, s, d)
170 if (exists(key, search_cache, inherits = FALSE)) {
171 return(search_cache[[key]])
172 }
173 ms <- legal(b, s)
174 o <- other(s)
175 if (!length(ms) && !length(legal(b, o))) {
176 v <- (if (d < 0) 1 else 1000) * (sum(b == s) - sum(b == o))
177 } else if (d == 0) {
178 v <- evaluate(b, s)[1]
179 } else if (!length(ms)) {
180 v <- -search(b, o, if (d < 0) -1 else d - 1)
181 } else {
182 v <- max(vapply(
183 ms,
184 function(p) -search(place(b, s, p), o, if (d < 0) -1 else d - 1),
185 numeric(1)
186 ))
187 }
188 if (length(search_cache) > 300000) {
189 rm(list = ls(search_cache, all.names = TRUE), envir = search_cache)

246stable <- function(b) {
247 stable <- rep(FALSE, 64)
248 axes <- list(c(0, 1), c(1, 0), c(1, 1), c(1, -1))
249 repeat {
250 found <- numeric()
251 for (p in 0:63) {
252 if (b[p + 1] == '.' || stable[p + 1]) {
253 next
254 }
255 row <- floor(p / 8)
256 col <- p %% 8
257 ok <- TRUE
258 for (axis in axes) {
259 dr <- axis[1]
260 dc <- axis[2]
261 full <- TRUE
262 anchor <- FALSE
263 for (sign in c(-1, 1)) {
264 r <- row + dr * sign
265 c <- col + dc * sign
266 if (
267 !inside(r, c) ||
268 (stable[8 * r + c + 1] && b[8 * r + c + 1] == b[p + 1])
269 ) {
270 anchor <- TRUE
271 }
272 while (inside(r, c)) {
273 if (b[8 * r + c + 1] == '.') {
274 full <- FALSE
275 }
276 r <- r + dr * sign
277 c <- c + dc * sign
278 }
279 }
280 if (!full && !anchor) {
281 ok <- FALSE
282 break
283 }
284 }
285 if (ok) found <- c(found, p)
286 }
287 if (!length(found)) {
288 break
289 }
290 stable[found + 1] <- TRUE
291 }
292 c(sum(stable & b == 'B'), sum(stable & b == 'W'))
293}
294transform_square <- function(p, op) {
295 r <- floor(p / 8)
296 c <- p %% 8
297 switch(
298 op,
299 rot90 = 8 * c + 7 - r,
300 rot180 = 63 - p,
301 rot270 = 8 * (7 - c) + r,
302 mirror = 8 * r + 7 - c,
303 flip = 8 * (7 - r) + c,
304 diag = 8 * c + r,
305 anti = 8 * (7 - c) + 7 - r
306 )
307}

534monte_carlo <- function(e, k, seed) {
535 rng <- new_random(seed)
536 best <- -1
537 bp <- -1
538 for (p in legal(e$board, e$side)) {
539 w <- d <- l <- 0
540 start <- place(e$board, e$side, p)
541 for (i in seq_len(k)) {
542 b <- random_game(start, other(e$side), rng)
543 diff <- sum(b == e$side) - sum(b == other(e$side))
544 if (diff > 0) {
545 w <- w + 1
546 } else if (diff < 0) {
547 l <- l + 1
548 } else {
549 d <- d + 1
550 }
551 }
552 emit(square_name(p), w, d, l)
553 if (2 * w + d > best) {
554 best <- 2 * w + d
555 bp <- p
556 }
557 }
558 emit('best', square_name(bp), best)

560execute <- function(e, text) {
561 a <- strsplit(text, ' ', fixed = TRUE)[[1]]
562 cmd <- a[1]
563 a <- a[-1]
564 error <- ''
565 if (!cmd %in% commands) {
566 error <- 'unknown command'
567 } else if (cmd == 'LOAD') {
568 if (
569 length(a) != 2 || !grepl('^[.BW]{64}$', a[1]) || !a[2] %in% c('B', 'W')
570 ) {
571 error <- 'bad board'
572 }
573 } else if (cmd %in% c('MOVE', 'FLIPS')) {
574 error <- if (length(a) != 1) {
575 'bad arguments'
576 } else if (!grepl('^[a-h][1-8]$', a[1])) {
577 'bad square'
578 } else {
579 ''
580 }
581 } else if (cmd %in% c('PLAY', 'RECORD')) {
582 error <- if (length(a) != 1) {
583 'bad arguments'
584 } else if (nchar(a[1]) > 120 || !grepl('^([a-h][1-8])+$', a[1])) {
585 'bad record'
586 } else {
587 ''
588 }
589 } else if (cmd %in% zero) {
590 if (length(a)) error <- 'bad arguments'
591 } else if (cmd == 'TRANSFORM') {
592 if (length(a) != 1 || !a[1] %in% transforms) error <- 'bad arguments'
593 } else {
594 expected <- if (cmd %in% c('AUTOPLAY', 'MC')) 2 else 1
595 valid <- length(a) == expected && all(grepl('^[0-9]+$', a))
596 if (valid) {
597 ns <- as.numeric(a)
598 n <- ns[1]
599 valid <- switch(
600 cmd,
601 AUTOPLAY = all(ns >= 1 & ns <= 4),
602 HINT = n >= 1 && n <= 4,
603 PV = n >= 1 && n <= 4,
604 AI = n >= 1 && n <= 4,
605 ANALYZE = n >= 1 && n <= 4,
606 PERFT = n >= 1 && n <= 8,
607 DIVIDE = n >= 1 && n <= 8,
608 MC = n >= 1 && n <= 100 && ns[2] >= 1 && ns[2] <= 2147483646,
609 RANDOM = n >= 1 && n <= 2147483646,
610 n >= 1
611 )
612 }
613 if (!valid) error <- 'bad arguments'
614 }
615 if (nchar(error)) {
616 emit('ERROR', error)
617 return()
618 }
619 if (e$over && cmd %in% active) {
620 emit('ERROR game over')
621 return()
622 }
623 n <- if (length(a) && grepl('^[0-9]+$', a[1])) as.numeric(a[1]) else 0
624 if (cmd == 'NEW') {
625 reset(e)
626 emit('OK')
627 } else if (cmd == 'LOAD') {
628 restore(
629 e,
630 list(
631 board = strsplit(a[1], '', fixed = TRUE)[[1]],
632 side = a[2],
633 over = FALSE
634 )
635 )
636 e$history <- list()
637 e$redo <- list()
638 emit('OK')
639 adjust(e)
640 } else if (cmd == 'DUMP') {
641 emit('board', paste0(e$board, collapse = ''), e$side)
642 } else if (cmd == 'SHOW') {
643 emit(' a b c d e f g h')
644 ms <- if (e$over) numeric() else legal(e$board, e$side)
645 for (r in 0:7) {
646 row <- ifelse((8 * r + 0:7) %in% ms, '*', e$board[8 * r + 1:8])
647 emit(r + 1, paste(row, collapse = ' '))
648 }
649 emit(status(e))
650 } else if (cmd == 'SCORE') {
651 emit(
652 'score B',
653 sum(e$board == 'B'),
654 'W',
655 sum(e$board == 'W'),
656 'empty',
657 sum(e$board == '.')
658 )
659 } else if (cmd == 'LEGAL') {
660 ms <- if (e$over) numeric() else legal(e$board, e$side)
661 out <- c('legal', length(ms), square_name(ms))
662 emit(paste(out, collapse = ' '))
663 } else if (cmd %in% c('MOVE', 'FLIPS')) {
664 p <- square_index(a[1])
665 why <- reason(e, p)
666 if (nchar(why)) {
667 emit('ILLEGAL', why)
668 } else if (cmd == 'MOVE') {
669 move(e, p)
670 } else {
671 f <- flips(e$board, e$side, p)
672 emit(paste(c('flips', a[1], length(f), square_name(f)), collapse = ' '))
673 }
674 } else if (cmd == 'PLAY') {
675 for (i in seq(1, nchar(a[1]), by = 2)) {
676 p <- square_index(substr(a[1], i, i + 1))
677 why <- reason(e, p)
678 if (nchar(why)) {
679 emit('ILLEGAL move', (i + 1) / 2, square_name(p), why)
680 break
681 }
682 move(e, p)
683 }
684 } else if (cmd == 'RECORD') {
685 record(e, a[1])
686 } else if (cmd == 'UNDO') {
687 if (n > length(e$history)) {
688 emit('ERROR cannot undo')
689 } else {
690 start <- length(e$history) - n + 1
691 removed <- e$history[seq(start, length(e$history))]
692 restore(e, removed[[1]]$before)
693 e$history <- if (start == 1) list() else e$history[seq_len(start - 1)]
694 e$redo <- c(removed, e$redo)
695 emit(status(e))
696 }
697 } else if (cmd == 'REDO') {
698 if (n > length(e$redo)) {
699 emit('ERROR cannot redo')
700 } else {
701 added <- e$redo[seq_len(n)]
702 e$redo <- if (n == length(e$redo)) {
703 list()
704 } else {
705 e$redo[seq(n + 1, length(e$redo))]
706 }
707 e$history <- c(e$history, added)
708 restore(e, added[[n]]$after)
709 emit(status(e))
710 }
711 } else if (cmd == 'HISTORY') {
712 emit('history', length(e$history))
713 for (i in seq_along(e$history)) {
714 r <- e$history[[i]]
715 out <- c(i, r$side, square_name(r$square), r$count, if (r$passed) 'pass')
716 emit(paste(out, collapse = ' '))
717 }
718 } else if (cmd == 'SHEET') {
719 sheet(e)
720 } else if (cmd == 'STATS') {
721 stats(e)
722 } else if (cmd == 'TRANSFORM') {
723 op <- a[1]
724 e$board <- transform_board(e$board, op)
725 tr <- function(r) {
726 r$square <- transform_square(r$square, op)
727 r$before$board <- transform_board(r$before$board, op)
728 r$after$board <- transform_board(r$after$board, op)
729 r
730 }
731 e$history <- lapply(e$history, tr)
732 e$redo <- lapply(e$redo, tr)
733 emit('OK')
734 } else if (cmd == 'EVAL') {
735 v <- evaluate(e$board, e$side)
736 emit('eval', v[1], 'weights', v[2], 'mobility', v[3], v[4])
737 } else if (cmd == 'STABLE') {
738 v <- stable(e$board)
739 emit('stable B', v[1], 'W', v[2])
740 } else if (cmd == 'HINT') {
741 rows <- ranked(e$board, e$side, n)
742 for (i in seq_len(nrow(rows))) {
743 emit(square_name(rows[i, 2]), rows[i, 1])
744 }
745 } else if (cmd == 'PV') {
746 pv <- variation(e$board, e$side, n)
747 emit(paste(c('pv', search(e$board, e$side, n), pv$steps), collapse = ' '))
748 } else if (cmd == 'AI') {
749 row <- ranked(e$board, e$side, n)[1, ]
750 move(e, row[2], FALSE, row[1])
751 } else if (cmd == 'ANALYZE') {
752 loss <- c(B = 0, W = 0)
