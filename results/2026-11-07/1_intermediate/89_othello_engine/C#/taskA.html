14 "NEW DUMP SHOW SCORE LEGAL HISTORY SHEET STATS EVAL STABLE SOLVE".Split(' ');

21 static char Other(char s) => s == 'B' ? 'W' : 'B';
22 static string Name(int p) => "abcdefgh" [p % 8].ToString() + (p / 8 + 1);
23 static int Square(string s) => (s[1] - '1') * 8 + s[0] - 'a';
24 static bool Inside(int r, int c) => r >= 0 && r < 8 && c >= 0 && c < 8;
25 static string Initial() {
26 char[] b = Enumerable.Repeat('.', 64).ToArray();
27 b[27] = b[36] = 'W';
28 b[28] = b[35] = 'B';
29 return new(b);
30 }
31 static List<int> Flips(string b, char s, int p) {
32 List<int> output = new();
33 if (b[p] != '.')
34 return output;
35 int row = p / 8, col = p % 8;
36 for (int dr = -1; dr <= 1; dr++)
37 for (int dc = -1; dc <= 1; dc++) {
38 if (dr == 0 && dc == 0)
39 continue;
40 int r = row + dr, c = col + dc;
41 List<int> line = new();
42 while (Inside(r, c) && b[8 * r + c] == Other(s)) {
43 line.Add(8 * r + c);
44 r += dr;
45 c += dc;
46 }
47 if (line.Count > 0 && Inside(r, c) && b[8 * r + c] == s)
48 output.AddRange(line);
49 }
50 output.Sort();
51 return output;
52 }
53 static List<int> Legal(string b, char s) =>
54 Enumerable.Range(0, 64).Where(p => b[p] == '.' && Flips(b, s, p).Count > 0).ToList();
55 static string Place(string b, char s, int p) {
56 char[] a = b.ToCharArray();
57 foreach (int q in Flips(b, s, p))
58 a[q] = s;
59 a[p] = s;
60 return new(a);
61 }
62 static int Count(string b, char s) => b.Count(x => x == s);
63 static bool Terminal(string b) => Legal(b, 'B').Count == 0 && Legal(b, 'W').Count == 0;
64 static string GameOver(string b) {
65 int x = Count(b, 'B'), y = Count(b, 'W');
66 return $"game over B {x} W {y} " + (x > y ? "winner B" : x < y ? "winner W" : "draw");
67 }
68 static (int Value, int Weight, int Own, int Enemy) Evaluate(string b, char s) {
69 int w = Enumerable.Range(0, 64).Sum(i => b[i] == s ? Weights[i]
70 : b[i] == Other(s) ? -Weights[i]
71 : 0),
72 own = Legal(b, s).Count, enemy = Legal(b, Other(s)).Count;
73 return (w + 10 * (own - enemy), w, own, enemy);
74 }
75 static readonly Dictionary<(string, char, int), int> SearchCache = new(), PerftCache = new();
76 static int Search(string b, char s, int d) {
77 var key = (b, s, d);
78 if (SearchCache.TryGetValue(key, out int found))
79 return found;
80 List<int> moves = Legal(b, s);
81 char o = Other(s);
82 int v;
83 if (moves.Count == 0 && Legal(b, o).Count == 0)
84 v = (d < 0 ? 1 : 1000) * (Count(b, s) - Count(b, o));
85 else if (d == 0)
86 v = Evaluate(b, s).Value;
87 else if (moves.Count == 0)
88 v = -Search(b, o, d < 0 ? -1 : d - 1);
89 else
90 v = moves.Max(p => -Search(Place(b, s, p), o, d < 0 ? -1 : d - 1));
91 if (SearchCache.Count > 300000)
92 SearchCache.Clear();
93 SearchCache[key] = v;
94 return v;
95 }
96 static List<(int Value, int Square)> Ranked(string b, char s, int d) =>
97 Legal(b, s)
98 .Select(p => (Value: -Search(Place(b, s, p), Other(s), d < 0 ? -1 : d - 1), Square: p))
99 .OrderByDescending(x => x.Value)
100 .ThenBy(x => x.Square)
101 .ToList();
102 static (List<string> Steps, string Board) Variation(string b, char s, int d) {
103 List<string> steps = new();
104 while (d != 0 && !Terminal(b)) {
105 var moves = Ranked(b, s, d);
106 if (moves.Count == 0)
107 steps.Add("pass");
108 else {
109 int p = moves[0].Square;
110 steps.Add(Name(p));
111 b = Place(b, s, p);
112 }
113 s = Other(s);
114 if (d > 0)
115 d--;
116 }
117 return (steps, b);
118 }
119 static int Perft(string b, char s, int d) {
120 if (d == 0)
121 return 1;
122 var key = (b, s, d);
123 if (PerftCache.TryGetValue(key, out int found))
124 return found;
125 List<int> moves = Legal(b, s);
126 char o = Other(s);
127 int n = moves.Count > 0 ? moves.Sum(p => Perft(Place(b, s, p), o, d - 1))
128 : Legal(b, o).Count == 0 ? 1
129 : Perft(b, o, d - 1);
130 if (PerftCache.Count > 200000)
131 PerftCache.Clear();
132 PerftCache[key] = n;
133 return n;
134 }
135 static (int Black, int White) Stable(string b) {
136 bool[] stable = new bool[64];
137 (int, int)[] axes = { (0, 1), (1, 0), (1, 1), (1, -1) };
138 while (true) {
139 List<int> found = new();
140 for (int p = 0; p < 64; p++) {
141 if (b[p] == '.' || stable[p])
142 continue;
143 bool ok = true;
144 foreach (var (dr, dc) in axes) {
145 bool full = true, anchor = false;
146 foreach (int sign in new[] { -1, 1 }) {
147 int r = p / 8 + dr * sign, c = p % 8 + dc * sign;
148 if (!Inside(r, c) || (stable[8 * r + c] && b[8 * r + c] == b[p]))
149 anchor = true;
150 while (Inside(r, c)) {
151 if (b[8 * r + c] == '.')
152 full = false;
153 r += dr * sign;
154 c += dc * sign;
155 }
156 }
157 if (!full && !anchor) {
158 ok = false;
159 break;
160 }
161 }
162 if (ok)
163 found.Add(p);
164 }
165 if (found.Count == 0)
166 break;
167 foreach (int p in found)
168 stable[p] = true;
169 }
170 return (Enumerable.Range(0, 64).Count(i => stable[i] && b[i] == 'B'),
171 Enumerable.Range(0, 64).Count(i => stable[i] && b[i] == 'W'));
172 }
173 static int Transform(int p, string op) {
174 int r = p / 8, c = p % 8;
175 return op switch { "rot90" => 8 * c + 7 - r, "rot180" => 63 - p,
176 "rot270" => 8 * (7 - c) + r, "mirror" => 8 * r + 7 - c,
177 "flip" => 8 * (7 - r) + c, "diag" => 8 * c + r,
178 _ => 8 * (7 - c) + 7 - r };
179 }
180 static string TransformBoard(string b, string op) {
181 char[] a = new char[64];
182 for (int p = 0; p < 64; p++)
183 a[Transform(p, op)] = b[p];
184 return new(a);
185 }
186 class Minstd {
187 long value;
188 public Minstd(long seed) {
189 value = seed;
190 }
191 public long Draw() => value = value * 48271 % 2147483647;
192 }
193 static string RandomGame(string b, char s, Minstd rng) {
194 while (true) {
195 List<int> moves = Legal(b, s);
196 if (moves.Count > 0)
197 b = Place(b, s, moves[(int)(rng.Draw() % moves.Count)]);
198 else if (Legal(b, Other(s)).Count == 0)
199 return b;
200 s = Other(s);
201 }
202 }
203 static string Average(int f, int n) {
204 int v = 100 * f / n;
205 return $"{v/100}.{v%100:00}";
206 }

214 static void Emit(params object[] xs) => Console.WriteLine(string.Join(" ", xs));

229 bool Adjust(bool quiet = false) {
230 if (Legal(board, side).Count > 0)
231 return false;
232 if (Legal(board, Other(side)).Count > 0) {
233 if (!quiet)
234 Emit(side, "passes");
235 side = Other(side);
236 return true;
237 }
238 over = true;
239 if (!quiet)
240 Emit(GameOver(board));
241 return false;
242 }
243 bool PlayMove(int p, bool quiet = false, int? value = null) {
244 Position before = Snapshot();
245 char s = side;
246 int k = Flips(board, s, p).Count;
247 board = Place(board, s, p);
248 if (!quiet) {
249 if (value.HasValue)
250 Emit(s, Name(p), "flips", k, "value", value.Value);
251 else
252 Emit(s, Name(p), "flips", k);
253 }
254 side = Other(s);
255 bool passed = Adjust(quiet);
256 history.Add(new(s, p, k, passed, before, Snapshot()));
257 redo.Clear();
258 return passed;
259 }
260 string Reason(int p) => over ? "game over"
261 : board[p] != '.' ? "occupied"
262 : Flips(board, side, p).Count == 0 ? "no flips"
263 : "";
264 void Record(string text) {
265 Othello e = new();
266 List<int> ps = new();
267 int passes = 0;
268 for (int i = 0; i < text.Length; i += 2) {
269 int p = Square(text.Substring(i, 2));
270 string why = e.Reason(p);
271 if (why != "") {
272 Emit("ILLEGAL move", i / 2 + 1, Name(p), why);
273 return;
274 }
275 ps.Add(p);
276 if (e.PlayMove(p, true))
277 passes++;
278 }
279 Restore(e.Snapshot());
280 history = e.history;
281 redo.Clear();
282 string op = Name(ps[0]) switch { "e6" => "diag", "d3" => "anti", "c4" => "rot180",
283 _ => "" };
284 string normal = string.Concat(ps.Select(p => Name(op == "" ? p : Transform(p, op))));
285 Emit("record", ps.Count, "moves", passes, "passes");
286 Emit("normal", normal);
287 Emit(Status());
288 }
289 void Sheet() {
290 string b = history.Count == 0 ? board : history[0].Before.Board;
291 string[] fields = b.Select(c => c.ToString()).ToArray();
292 for (int i = 0; i < history.Count; i++)
293 fields[history[i].Square] = (i + 1).ToString();
294 Emit(" " + string.Concat("abcdefgh".Select(c => c.ToString().PadLeft(3))));
295 for (int r = 0; r < 8; r++)
296 Emit((r + 1) + string.Concat(fields.Skip(8 * r).Take(8).Select(x => x.PadLeft(3))));
297 }
298 void Stats() {
299 int total = history.Sum(r => r.Count);
300 if (history.Count == 0)
301 Emit("moves 0");
302 else
303 Emit("moves", history.Count, "flips", total, "avg", Average(total, history.Count),
304 "passes", history.Count(r => r.Passed));
305 foreach (char s in new[] { 'B', 'W' }) {
306 List<Move> rows = history.Where(r => r.Side == s).ToList();
307 if (rows.Count == 0) {
308 Emit(s, "moves 0");
309 continue;
310 }
311 int sum = rows.Sum(r => r.Count), max = rows.Max(r => r.Count);
312 Emit(s, "moves", rows.Count, "flips", sum, "avg", Average(sum, rows.Count), "max",
313 max, Name(rows.First(r => r.Count == max).Square), "corners",
314 rows.Count(r => new[] { 0, 7, 56, 63 }.Contains(r.Square)));
315 }
316 }
317 void AutoPlay(int black, int white, long seed = 0) {
318 Minstd rng = new(seed);
319 List<string> ps = new();
320 int passes = 0;
321 while (!over) {
322 List<int> moves = Legal(board, side);
323 int p = seed != 0 ? moves[(int)(rng.Draw() % moves.Count)]
324 : Ranked(board, side, side == 'B' ? black : white)[0].Square;
325 ps.Add(Name(p));
326 if (PlayMove(p, true))
327 passes++;
328 }
329 Emit(seed != 0 ? "random" : "autoplay", ps.Count, "moves", passes, "passes");
330 Emit("moves", string.Concat(ps));
331 Emit(GameOver(board));
332 }
333 void MonteCarlo(int k, long seed) {
334 Minstd rng = new(seed);
335 int best = -1, bp = -1;
336 foreach (int p in Legal(board, side)) {
337 int w = 0, d = 0, l = 0;
338 string start = Place(board, side, p);
339 for (int i = 0; i < k; i++) {
340 string end = RandomGame(start, Other(side), rng);
341 int diff = Count(end, side) - Count(end, Other(side));
342 if (diff > 0)
343 w++;
344 else if (diff < 0)
345 l++;
346 else
347 d++;
348 }
349 Emit(Name(p), w, d, l);
350 if (2 * w + d > best) {
351 best = 2 * w + d;
352 bp = p;
353 }
354 }
355 Emit("best", Name(bp), best);
356 }
357 public void Execute(string text) {
358 string[] words = text.Split(' ');
359 string cmd = words[0];
360 string[] a = words.Skip(1).ToArray();
361 string error = "";
362 if (!Commands.Contains(cmd))
363 error = "unknown command";
364 else if (cmd == "LOAD") {
365 if (a.Length != 2 || !Regex.IsMatch(a[0], @"\A[.BW]{64}\z") ||
366 !new[] { "B", "W" }.Contains(a[1]))
367 error = "bad board";
368 } else if (cmd == "MOVE" || cmd == "FLIPS") {
369 if (a.Length != 1)
370 error = "bad arguments";
371 else if (!Regex.IsMatch(a[0], @"\A[a-h][1-8]\z"))
372 error = "bad square";
373 } else if (cmd == "PLAY" || cmd == "RECORD") {
374 if (a.Length != 1)
375 error = "bad arguments";
376 else if (a[0].Length > 120 || !Regex.IsMatch(a[0], @"\A([a-h][1-8])+\z"))
377 error = "bad record";
378 } else if (Zero.Contains(cmd)) {
379 if (a.Length != 0)
380 error = "bad arguments";
381 } else if (cmd == "TRANSFORM") {
382 if (a.Length != 1 || !Transforms.Contains(a[0]))
383 error = "bad arguments";
384 } else {
385 int expected = cmd == "AUTOPLAY" || cmd == "MC" ? 2 : 1;
386 bool valid = a.Length == expected && a.All(s => Regex.IsMatch(s, @"\A[0-9]+\z"));
387 if (valid) {
388 long[] nums = a.Select(long.Parse).ToArray();
389 long n0 = nums[0];
390 valid = cmd switch { "AUTOPLAY" => nums.All(n => n >= 1 && n <= 4),
391 "HINT" or "PV" or "AI" or "ANALYZE" => n0 >= 1 && n0 <= 4,
392 "PERFT" or "DIVIDE" => n0 >= 1 && n0 <= 8,
393 "MC" => n0 >= 1 && n0 <= 100 && nums[1] >= 1 &&
394 nums[1] <= 2147483646,
395 "RANDOM" => n0 >= 1 && n0 <= 2147483646,
396 _ => n0 >= 1 };
397 }
398 if (!valid)
399 error = "bad arguments";
400 }
401 if (error != "") {
402 Emit("ERROR", error);
403 return;
404 }
405 if (over && Active.Contains(cmd)) {
406 Emit("ERROR game over");
407 return;
408 }
409 int n = a.Length > 0 && int.TryParse(a[0], out int numeric) ? numeric : 0;
410 switch (cmd) {
411 case "NEW":
412 Reset();
413 Emit("OK");
414 break;
415 case "LOAD":
416 board = a[0];
417 side = a[1][0];
418 over = false;
419 history.Clear();
420 redo.Clear();
421 Emit("OK");
422 Adjust();
423 break;
424 case "DUMP":
425 Emit("board", board, side);
426 break;
427 case "SHOW":
428 Emit(" a b c d e f g h");
429 var legal = over ? new List<int>() : Legal(board, side);
430 for (int r = 0; r < 8; r++)
431 Emit((r + 1) + " " +
432 string.Join(" ",
433 Enumerable.Range(0, 8).Select(
434 c => legal.Contains(8 * r + c) ? '*' : board[8 * r + c])));
435 Emit(Status());
436 break;
437 case "SCORE":
438 Emit("score B", Count(board, 'B'), "W", Count(board, 'W'), "empty",
439 Count(board, '.'));
440 break;
441 case "LEGAL":
442 var moves = over ? new List<int>() : Legal(board, side);
443 Emit(string.Join(
444 " ", new[] { "legal", moves.Count.ToString() }.Concat(moves.Select(Name))));
445 break;
446 case "MOVE":
447 case "FLIPS": {
448 int p = Square(a[0]);
449 string why = Reason(p);
450 if (why != "")
451 Emit("ILLEGAL", why);
452 else if (cmd == "MOVE")
453 PlayMove(p);
454 else {
455 List<int> f = Flips(board, side, p);
456 Emit(string.Join(
457 " ", new[] { "flips", a[0], f.Count.ToString() }.Concat(f.Select(Name))));
458 }
459 break;
460 }
461 case "PLAY":
462 for (int i = 0; i < a[0].Length; i += 2) {
463 int p = Square(a[0].Substring(i, 2));
464 string why = Reason(p);
465 if (why != "") {
466 Emit("ILLEGAL move", i / 2 + 1, Name(p), why);
467 break;
468 }
469 PlayMove(p);
470 }
471 break;
472 case "RECORD":
473 Record(a[0]);
474 break;
475 case "UNDO":
476 if (n > history.Count)
477 Emit("ERROR cannot undo");
478 else {
479 List<Move> removed = history.GetRange(history.Count - n, n);
480 Restore(removed[0].Before);
481 history.RemoveRange(history.Count - n, n);
482 redo.InsertRange(0, removed);
483 Emit(Status());
484 }
485 break;
486 case "REDO":
487 if (n > redo.Count)
488 Emit("ERROR cannot redo");
489 else {
490 List<Move> added = redo.GetRange(0, n);
491 redo.RemoveRange(0, n);
492 history.AddRange(added);
493 Restore(added[^1].After);
494 Emit(Status());
495 }
496 break;
497 case "HISTORY":
498 Emit("history", history.Count);
499 for (int i = 0; i < history.Count; i++) {
500 Move r = history[i];
501 Emit($"{i+1} {r.Side} {Name(r.Square)} {r.Count}" + (r.Passed ? " pass" : ""));
502 }
503 break;
504 case "SHEET":
505 Sheet();
506 break;
507 case "STATS":
508 Stats();
509 break;
510 case "TRANSFORM": {
511 string op = a[0];
512 board = TransformBoard(board, op);
513 Move Tr(Move r) =>
514 r with { Square = Transform(r.Square, op),
515 Before = r.Before with { Board = TransformBoard(r.Before.Board, op) },
516 After = r.After with { Board = TransformBoard(r.After.Board, op) } };
517 history = history.Select(Tr).ToList();
518 redo = redo.Select(Tr).ToList();
519 Emit("OK");
520 break;
521 }
522 case "EVAL": {
523 var (v, w, x, y) = Evaluate(board, side);
524 Emit("eval", v, "weights", w, "mobility", x, y);
525 break;
526 }
527 case "STABLE": {
528 var (b, w) = Stable(board);
529 Emit("stable B", b, "W", w);
530 break;
531 }
532 case "HINT":
533 foreach (var (v, p) in Ranked(board, side, n))
534 Emit(Name(p), v);
535 break;
536 case "PV":
537 var pv = Variation(board, side, n);
538 Emit(string.Join(
539 " ", new[] { "pv", Search(board, side, n).ToString() }.Concat(pv.Steps)));
540 break;
541 case "AI": {
542 var (v, p) = Ranked(board, side, n)[0];
543 PlayMove(p, false, v);
544 break;
545 }
546 case "ANALYZE": {
547 Dictionary<char, int> loss = new() { { 'B', 0 }, { 'W', 0 } };
548 for (int i = 0; i < history.Count; i++) {
549 Move r = history[i];
550 var rows = Ranked(r.Before.Board, r.Side, n);
551 var (best, bp) = rows[0];
552 int v = rows.First(x => x.Square == r.Square).Value;
553 loss[r.Side] += best - v;
554 Emit(i + 1, r.Side, Name(r.Square), v, "best", Name(bp), best);
555 }
556 Emit("loss B", loss['B'], "W", loss['W']);
557 break;
558 }
559 case "SOLVE":
560 if (Count(board, '.') >= 9)
561 Emit("ERROR too many empties");
562 else {
563 foreach (var (v, p) in Ranked(board, side, -1))
564 Emit(Name(p), v);
565 var line = Variation(board, side, -1);
566 Emit(string.Join(" ", new[] { "line" }.Concat(line.Steps)));
567 Emit("final B", Count(line.Board, 'B'), "W", Count(line.Board, 'W'));
568 }
569 break;
570 case "AUTOPLAY":
571 AutoPlay(n, int.Parse(a[1]));
572 break;
573 case "RANDOM":
574 AutoPlay(0, 0, n);
575 break;
576 case "MC":
577 MonteCarlo(n, long.Parse(a[1]));
578 break;
579 case "PERFT":
580 Emit("perft", n, Perft(board, side, n));
581 break;
582 case "DIVIDE":
583 int total = 0;
584 foreach (int p in Legal(board, side)) {
585 int v = Perft(Place(board, side, p), Other(side), n - 1);
586 Emit(Name(p), v);
587 total += v;
588 }
589 Emit("total", total);
590 break;
591 }
592 }
593 }

595 int n = int.Parse(Console.ReadLine()!);
596 Othello e = new();
597 for (int i = 0; i < n; i++)
598 e.Execute(Console.ReadLine()!);
599 }
