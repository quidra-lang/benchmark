74def other(s) = s == "B" ? "W" : "B"
75def square_name(p) = "abcdefgh"[p % 8] + (p / 8 + 1).to_s
76def square_index(s) = (s[1].to_i - 1) * 8 + s[0].ord - 97
77def inside(r, c) = r >= 0 && r < 8 && c >= 0 && c < 8
78def initial
79 b = "." * 64
80 b[27] = b[36] = "W"
81 b[28] = b[35] = "B"
82 b
83end

85def flips(b, s, p)
86 return [] unless b[p] == "."
87 row, col = p.divmod(8)
88 out = []
89 (-1..1).each do |dr|
90 (-1..1).each do |dc|
91 next if dr == 0 && dc == 0
92 r = row + dr
93 c = col + dc
94 line = []
95 while inside(r, c) && b[8 * r + c] == other(s)
96 line << 8 * r + c
97 r += dr
98 c += dc
99 end
100
101 out.concat(line) if !line.empty? && inside(r, c) && b[8 * r + c] == s
102 end
103 end
104
105 out.sort
106end

108def legal(b, s) = (0...64).select { |p| b[p] == "." && !flips(b, s, p).empty? }
109def place(b, s, p)
110 a = b.dup
111 flips(b, s, p).each { |q| a[q] = s }
112 a[p] = s
113 a
114end

116def terminal(b) = legal(b, "B").empty? && legal(b, "W").empty?
117def game_over(b)
118 x = b.count("B")
119 y = b.count("W")
120 "game over B #{x} W #{y} #{x > y ? "winner B" : x < y ? "winner W" : "draw"}"
121end

123def evaluate(b, s)
124 w = (0...64).sum { |i| b[i] == s ? WEIGHTS[i] : b[i] == other(s) ? -WEIGHTS[i] : 0 }
125 own = legal(b, s).length
126 enemy = legal(b, other(s)).length
127 [w + 10 * (own - enemy), w, own, enemy]
128end

130SEARCH_CACHE = {}
131PERFT_CACHE = {}
132def search(b, s, d)
133 key = [b, s, d]
134 return SEARCH_CACHE[key] if SEARCH_CACHE.key?(key)
135 ms = legal(b, s)
136 o = other(s)
137 v = if ms.empty? && legal(b, o).empty?
138 (d < 0 ? 1 : 1000) * (b.count(s) - b.count(o))
139 elsif d == 0
140 evaluate(b, s)[0]
141 elsif ms.empty?
142 -search(b, o, d < 0 ? -1 : d - 1)
143 else
144 ms.map { |p| -search(place(b, s, p), o, d < 0 ? -1 : d - 1) }.max
145 end
146
147 SEARCH_CACHE.clear if SEARCH_CACHE.length > 300000
148 SEARCH_CACHE[key] = v
149end
150
151def ranked(b, s, d)
152 legal(b, s).map { |p| [-search(place(b, s, p), other(s), d < 0 ? -1 : d - 1), p] }.sort_by { |v, p| [-v, p] }
153end
154
155def variation(b, s, d)
156 steps = []
157 until d == 0 || terminal(b)
158 ms = ranked(b, s, d)
159 if ms.empty?
160 steps << "pass"
161 else
162 p = ms[0][1]
163 steps << square_name(p)
164 b = place(b, s, p)
165 end
166
167 s = other(s)
168 d -= 1 if d > 0
169 end
170
171 [steps, b]
172end
173
174def perft(b, s, d)
175 return 1 if d == 0
176 key = [b, s, d]
177 return PERFT_CACHE[key] if PERFT_CACHE.key?(key)
178 ms = legal(b, s)
179 o = other(s)
180 n = ms.empty? ? (legal(b, o).empty? ? 1 : perft(b, o, d - 1)) : ms.sum { |p| perft(place(b, s, p), o, d - 1) }
181 PERFT_CACHE.clear if PERFT_CACHE.length > 200000
182 PERFT_CACHE[key] = n
183end

185def stable(b)
186 stable = Array.new(64, false)
187 axes = [[0, 1], [1, 0], [1, 1], [1, -1]]
188 loop do
189 found = (0...64).select do |p|
190 next false if b[p] == "." || stable[p]
191 row, col = p.divmod(8)
192 axes.all? do |dr, dc|
193 full = true
194 anchor = false
195 [-1, 1].each do |sign|
196 r = row + dr * sign
197 c = col + dc * sign
198 anchor = true if !inside(r, c) || (stable[8 * r + c] && b[8 * r + c] == b[p])
199 while inside(r, c)
200 full = false if b[8 * r + c] == "."
201 r += dr * sign
202 c += dc * sign
203 end
204 end
205
206 full || anchor
207 end
208 end
209
210 break if found.empty?
211 found.each { |p| stable[p] = true }
212 end
213
214 %w[B W].map { |s| (0...64).count { |p| stable[p] && b[p] == s } }
215end

217def transform(p, op)
218 r, c = p.divmod(8)
219 case op
220 when "rot90"
221 8 * c + 7 - r
222 when "rot180"
223 63 - p
224 when "rot270"
225 8 * (7 - c) + r
226 when "mirror"
227 8 * r + 7 - c
228 when "flip"
229 8 * (7 - r) + c
230 when "diag"
231 8 * c + r
232 else
233 8 * (7 - c) + 7 - r
234 end
235end

237def transform_board(b, op)
238 a = "." * 64
239 64.times { |p| a[transform(p, op)] = b[p] }
240 a
241end

243class Minstd
244 def initialize(seed) = @value = seed
245 def draw = @value = 48271 * @value % 2147483647
246end

248def random_game(b, s, rng)
249 loop do
250 ms = legal(b, s)
251 if ms.empty?
252 return b if legal(b, other(s)).empty?
253 else
254 b = place(b, s, ms[rng.draw % ms.length])
255 end
256
257 s = other(s)
258 end
259end

261def average(f, n)
262 v = 100 * f / n
263 "#{v / 100}.#{(v % 100).to_s.rjust(2, "0")}"
264end

284 def snapshot = Position.new(@board, @side, @over)
285 def restore(p)
286 @board = p.board
287 @side = p.side

291 def status = @over ? game_over(@board) : @side + " to move"
292 def adjust(quiet = false)
293 return false unless legal(@board, @side).empty?
294 unless legal(@board, other(@side)).empty?
295 emit(@side, "passes") unless quiet
296 @side = other(@side)
297 return true
298 end

300 @over = true
301 emit(game_over(@board)) unless quiet
302 false
303 end

305 def move(p, quiet = false, value = nil)
306 before = snapshot
307 s = @side
308 k = flips(@board, s, p).length
309 @board = place(@board, s, p)
310 emit(s, square_name(p), "flips", k, *(value.nil? ? [] : ["value", value])) unless quiet
311 @side = other(s)
312 passed = adjust(quiet)
313 @history << Move.new(s, p, k, passed, before, snapshot)
314 @redo = []
315 passed
316 end

318 def reason(p) = @over ? "game over" : @board[p] != "." ? "occupied" : flips(@board, @side, p).empty? ? "no flips" : nil
319 def record(text)
320 engine = Othello.new
321 ps = text.scan(/../).map { |s| square_index(s) }
322 passes = 0
323 ps.each_with_index do |p, i|
324 why = engine.reason(p)
325 if why
326 emit("ILLEGAL move", i + 1, square_name(p), why)
327 return
328 end
329
330 passes += 1 if engine.move(p, true)
331 end
332
333 restore(engine.snapshot)
334 @history = engine.history
335 @redo = []
336 op = {"f5" => nil, "e6" => "diag", "d3" => "anti", "c4" => "rot180"}[square_name(ps[0])]
337 emit("record", ps.length, "moves", passes, "passes")
338 emit("normal", ps.map { |p| square_name(op ? transform(p, op) : p) }.join)
339 emit(status)
340 end

342 def sheet
343 fields = (@history.empty? ? @board : @history[0].before.board).chars
344 @history.each_with_index { |r, i| fields[r.square] = (i + 1).to_s }
345 emit(" " + "abcdefgh".chars.map { |s| s.rjust(3) }.join)
346 8.times { |r| emit((r + 1).to_s + fields[8 * r, 8].map { |s| s.rjust(3) }.join) }
347 end

349 def stats
350 total = @history.sum(&:count)
351 @history.empty? ? emit("moves 0") : emit(
352 "moves",
353 @history.length,
354 "flips",
355 total,
356 "avg",
357 average(total, @history.length),
358 "passes",
359 @history.count(&:passed)
360 )
361 %w[B W].each do |s|
362 rows = @history.select { |r| r.side == s }
363 if rows.empty?
364 emit(s, "moves 0")
365 next
366 end
367
368 sum = rows.sum(&:count)
369 max = rows.map(&:count).max
370 emit(
371 s,
372 "moves",
373 rows.length,
374 "flips",
375 sum,
376 "avg",
377 average(sum, rows.length),
378 "max",
379 max,
380 square_name(rows.find { |r| r.count == max }.square),
381 "corners",
382 rows.count { |r| [0, 7, 56, 63].include?(r.square) }
383 )
384 end
385 end

387 def auto(black, white, seed = nil)
388 rng = seed && Minstd.new(seed)
389 ps = []
390 passes = 0
391 until @over
392 ms = legal(@board, @side)
393 p = rng ? ms[rng.draw % ms.length] : ranked(@board, @side, @side == "B" ? black : white)[0][1]
394 ps << square_name(p)
395 passes += 1 if move(p, true)
396 end
397
398 emit(rng ? "random" : "autoplay", ps.length, "moves", passes, "passes")
399 emit("moves", ps.join)
400 emit(game_over(@board))
401 end

403 def monte_carlo(k, seed)
404 rng = Minstd.new(seed)
405 best = -1
406 bp = nil
407 legal(@board, @side).each do |p|
408 w = d = l = 0
409 start = place(@board, @side, p)
410 k.times do
411 b = random_game(start, other(@side), rng)
412 diff = b.count(@side) - b.count(other(@side))
413 if diff > 0
414 w += 1
415 elsif diff < 0
416 l += 1
417 else
418 d += 1
419 end
420 end
421
422 emit(square_name(p), w, d, l)
423 if 2 * w + d > best
424 best = 2 * w + d
425 bp = p
426 end
427 end
428
429 emit("best", square_name(bp), best)
430 end

432 def execute(text)
433 cmd, *a = text.split(" ")
434 error = nil
435 if !COMMANDS.include?(cmd)
436 error = "unknown command"
437 elsif cmd == "LOAD"
438 error = "bad board" unless a.length == 2 && /\A[.BW]{64}\z/.match?(a[0]) && %w[B W].include?(a[1])
439 elsif %w[MOVE FLIPS].include?(cmd)
440 error = a.length != 1 ? "bad arguments" : /\A[a-h][1-8]\z/.match?(a[0]) ? nil : "bad square"
441 elsif %w[PLAY RECORD].include?(cmd)
442 error = a.length != 1 ? "bad arguments" : a[0].length <= 120 && /\A(?:[a-h][1-8])+\z/.match?(a[0]) ? nil : "bad record"
443 elsif ZERO.include?(cmd)
444 error = "bad arguments" unless a.empty?
445 elsif cmd == "TRANSFORM"
446 error = "bad arguments" unless a.length == 1 && TRANSFORMS.include?(a[0])
447 else
448 n = a.map(&:to_i)
449 expected = %w[AUTOPLAY MC].include?(cmd) ? 2 : 1
450 valid = a.length == expected && a.all? { |s| /\A[0-9]+\z/.match?(s) }
451 valid &&= case cmd
452 when "AUTOPLAY"
453 n.all? { |x| (1..4).cover?(x) }
454 when "HINT", "PV", "AI", "ANALYZE"
455 (1..4).cover?(n[0])
456 when "PERFT", "DIVIDE"
457 (1..8).cover?(n[0])
458 when "MC"
459 (1..100).cover?(n[0]) && (1..2147483646).cover?(n[1])
460 when "RANDOM"
461 (1..2147483646).cover?(n[0])
462 else
463 n[0] >= 1
464 end
465
466 error = "bad arguments" unless valid
467 end
468
469 if error
470 emit("ERROR", error)
471 return
472 end
473
474 if ACTIVE.include?(cmd) && @over
475 emit("ERROR game over")
476 return
477 end
478
479 n = a[0].to_i
480 case cmd
481 when "NEW"
482 reset
483 emit("OK")
484 when "LOAD"
485 @board = a[0]
486 @side = a[1]
487 @over = false
488 @history = []
489 @redo = []
490 emit("OK")
491 adjust
492 when "DUMP"
493 emit("board", @board, @side)
494 when "SHOW"
495 emit(" a b c d e f g h")
496 ms = @over ? [] : legal(@board, @side)
497 8.times { |r| emit(r + 1, *(0...8).map { |c| ms.include?(8 * r + c) ? "*" : @board[8 * r + c] }) }
498 emit(status)
499 when "SCORE"
500 emit("score B", @board.count("B"), "W", @board.count("W"), "empty", @board.count("."))
501 when "LEGAL"
502 ms = @over ? [] : legal(@board, @side)
503 emit("legal", ms.length, *ms.map { |p| square_name(p) })
504 when "MOVE", "FLIPS"
505 p = square_index(a[0])
506 why = reason(p)
507 if why
508 emit("ILLEGAL", why)
509 elsif cmd == "MOVE"
510 move(p)
511 else
512 f = flips(@board, @side, p)
513 emit("flips", a[0], f.length, *f.map { |q| square_name(q) })
514 end
515
516 when "PLAY"
517 a[0].scan(/../).each_with_index do |s, i|
518 p = square_index(s)
519 why = reason(p)
520 if why
521 emit("ILLEGAL move", i + 1, s, why)
522 break
523 end
524
525 move(p)
526 end
527
528 when "RECORD"
529 record(a[0])
530 when "UNDO"
531 if n > @history.length
532 emit("ERROR cannot undo")
533 else
534 removed = @history.pop(n)
535 restore(removed[0].before)
536 @redo = removed + @redo
537 emit(status)
538 end
539
540 when "REDO"
541 if n > @redo.length
542 emit("ERROR cannot redo")
543 else
544 added = @redo.shift(n)
545 @history.concat(added)
546 restore(added[-1].after)
547 emit(status)
548 end
549
550 when "HISTORY"
551 emit("history", @history.length)
552 @history.each_with_index { |r, i|
553 emit(i + 1, r.side, square_name(r.square), r.count, *(r.passed ? ["pass"] : []))
554 }
555 when "SHEET"
556 sheet
557 when "STATS"
558 stats
559 when "TRANSFORM"
560 op = a[0]
561 @board = transform_board(@board, op)
562 transform_record = lambda do |r|
563 before = r.before.with(board: transform_board(r.before.board, op))
564 after = r.after.with(board: transform_board(r.after.board, op))
565 r.with(square: transform(r.square, op), before: before, after: after)
566 end
567
568 @history = @history.map(&transform_record)
569 @redo = @redo.map(&transform_record)
570 emit("OK")
571 when "EVAL"
572 v, w, x, y = evaluate(@board, @side)
573 emit("eval", v, "weights", w, "mobility", x, y)
574 when "STABLE"
575 b, w = stable(@board)
576 emit("stable B", b, "W", w)
577 when "HINT"
578 ranked(@board, @side, n).each { |v, p| emit(square_name(p), v) }
579 when "PV"
580 steps, = variation(@board, @side, n)
581 emit("pv", search(@board, @side, n), *steps)
582 when "AI"
583 v, p = ranked(@board, @side, n)[0]
584 move(p, false, v)
585 when "ANALYZE"
586 loss = {"B" => 0, "W" => 0}
587 @history.each_with_index do |r, i|
588 rows = ranked(r.before.board, r.side, n)
589 best, bp = rows[0]
590 v = rows.find { |_, p| p == r.square }[0]
591 loss[r.side] += best - v
592 emit(i + 1, r.side, square_name(r.square), v, "best", square_name(bp), best)
593 end
594
595 emit("loss B", loss["B"], "W", loss["W"])
596 when "SOLVE"
597 if @board.count(".") >= 9
598 emit("ERROR too many empties")
599 else
600 ranked(@board, @side, -1).each { |v, p| emit(square_name(p), v) }
601 steps, b = variation(@board, @side, -1)
602 emit("line", *steps)
603 emit("final B", b.count("B"), "W", b.count("W"))
604 end
605
606 when "AUTOPLAY"
607 auto(n, a[1].to_i)
608 when "RANDOM"
609 auto(0, 0, n)
610 when "MC"
611 monte_carlo(n, a[1].to_i)
612 when "PERFT"
613 emit("perft", n, perft(@board, @side, n))
614 when "DIVIDE"
615 total = 0
616 legal(@board, @side).each { |p|
617 v = perft(place(@board, @side, p), other(@side), n - 1)
618 emit(square_name(p), v)
619 total += v
620 }
621 emit("total", total)
622 end
623 end
624end
