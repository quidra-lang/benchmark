func initial() Board {
	var b Board
	for i := range b {
		b[i] = '.'
	}
	b[27] = 'W'
	b[36] = 'W'
	b[28] = 'B'
	b[35] = 'B'
	return b
func newEngine(out *bufio.Writer) *Othello {
	return &Othello{board: initial(), side: 'B', out: out}
func (e *Othello) reset() {
	e.board = initial()
	e.side = 'B'
	e.over = false
	e.history = nil
	e.redo = nil
	case "NEW":
		e.reset()
		e.emit("OK")
