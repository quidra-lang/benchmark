30func other(s byte) byte {
31 if s == 'B' {
32 return 'W'
33 }
34 return 'B'
35}
36func name(p int) string {
37 return string([]byte{byte('a' + p%8), byte('1' + p/8)})
38}
39func square(s string) int {
40 return int(s[1]-'1')*8 + int(s[0]-'a')
41}
42func inside(r, c int) bool {
43 return r >= 0 && r < 8 && c >= 0 && c < 8
44}
45func initial() Board {
46 var b Board
47 for i := range b {
48 b[i] = '.'
49 }
50 b[27] = 'W'
51 b[36] = 'W'
52 b[28] = 'B'
53 b[35] = 'B'
54 return b
55}
56func flips(b Board, s byte, p int) []int {
57 out := []int{}
58 if b[p] != '.' {
59 return out
60 }
61 row, col := p/8, p%8
62 for dr := -1; dr <= 1; dr++ {
63 for dc := -1; dc <= 1; dc++ {
64 if dr == 0 && dc == 0 {
65 continue
66 }
67 r, c := row+dr, col+dc
68 line := []int{}
69 for inside(r, c) && b[8*r+c] == other(s) {
70 line = append(line, 8*r+c)
71 r += dr
72 c += dc
73 }
74 if len(line) > 0 && inside(r, c) && b[8*r+c] == s {
75 out = append(out, line...)
76 }
77 }
78 }
79 sort.Ints(out)
80 return out
81}
82func legal(b Board, s byte) []int {
83 ms := []int{}
84 for p := range b {
85 if b[p] == '.' && len(flips(b, s, p)) > 0 {
86 ms = append(ms, p)
87 }
88 }
89 return ms
90}
91func place(b Board, s byte, p int) Board {
92 for _, q := range flips(b, s, p) {
93 b[q] = s
94 }
95 b[p] = s
96 return b
97}
98func count(b Board, s byte) int {
99 n := 0
100 for _, x := range b {
101 if x == s {
102 n++
103 }
104 }
105 return n
106}
107func terminal(b Board) bool {
108 return len(legal(b, 'B')) == 0 && len(legal(b, 'W')) == 0
109}
110func gameOver(b Board) string {
111 x, y := count(b, 'B'), count(b, 'W')
112 result := "draw"
113 if x > y {
114 result = "winner B"
115 } else if x < y {
116 result = "winner W"
117 }
118 return fmt.Sprintf("game over B %d W %d %s", x, y, result)
119}
120func evaluate(b Board, s byte) [4]int {
121 w := 0
122 for i, x := range b {
123 if x == s {
124 w += weights[i]
125 } else if x == other(s) {
126 w -= weights[i]
127 }
128 }
129 own, enemy := len(legal(b, s)), len(legal(b, other(s)))
130 return [4]int{w + 10*(own-enemy), w, own, enemy}
131}

135 side byte
136 depth int

139var searchCache = map[key]int{}

142func search(b Board, s byte, d int) int {
143 k := key{b, s, d}
144 if v, ok := searchCache[k]; ok {
145 return v
146 }
147 moves, o := legal(b, s), other(s)
148 v := 0
149 if len(moves) == 0 && len(legal(b, o)) == 0 {
150 scale := 1000
151 if d < 0 {
152 scale = 1
153 }
154 v = scale * (count(b, s) - count(b, o))
155 } else if d == 0 {
156 v = evaluate(b, s)[0]
157 } else {
158 next := d - 1
159 if d < 0 {
160 next = -1
161 }
162 if len(moves) == 0 {
163 v = -search(b, o, next)
164 } else {
165 v = -1000000
166 for _, p := range moves {
167 x := -search(place(b, s, p), o, next)
168 if x > v {
169 v = x
170 }
171 }
172 }
173 }
174 if len(searchCache) > 300000 {
175 clear(searchCache)
176 }
177 searchCache[k] = v
178 return v
179}

181type ranking struct {
182 value, square int
183}
184
185func ranked(b Board, s byte, d int) []ranking {
186 rows := []ranking{}
187 next := d - 1
188 if d < 0 {
189 next = -1
190 }
191 for _, p := range legal(b, s) {
192 rows = append(rows, ranking{-search(place(b, s, p), other(s), next), p})
193 }
194 sort.Slice(rows, func(i, j int) bool {
195 if rows[i].value != rows[j].value {
196 return rows[i].value > rows[j].value
197 }
198 return rows[i].square < rows[j].square
199 })
200 return rows
201}
202func variation(b Board, s byte, d int) ([]string, Board) {
203 steps := []string{}
204 for d != 0 && !terminal(b) {
205 rows := ranked(b, s, d)
206 if len(rows) == 0 {
207 steps = append(steps, "pass")
208 } else {
209 p := rows[0].square
210 steps = append(steps, name(p))
211 b = place(b, s, p)
212 }
213 s = other(s)
214 if d > 0 {
215 d--
216 }
217 }
218 return steps, b
219}
220func perft(b Board, s byte, d int) int {
221 if d == 0 {
222 return 1
223 }
224 k := key{b, s, d}
225 if v, ok := perftCache[k]; ok {
226 return v
227 }
228 moves, o := legal(b, s), other(s)
229 n := 0
230 if len(moves) == 0 {
231 if len(legal(b, o)) == 0 {
232 n = 1
233 } else {
234 n = perft(b, o, d-1)
235 }
236 } else {
237 for _, p := range moves {
238 n += perft(place(b, s, p), o, d-1)
239 }
240 }
241 if len(perftCache) > 200000 {
242 clear(perftCache)
243 }
244 perftCache[k] = n
245 return n
246}
247func stable(b Board) (int, int) {
248 var stable [64]bool
249 axes := [][2]int{{0, 1}, {1, 0}, {1, 1}, {1, -1}}
250 for {
251 found := []int{}
252 for p, x := range b {
253 if x == '.' || stable[p] {
254 continue
255 }
256 ok := true
257 for _, axis := range axes {
258 dr, dc := axis[0], axis[1]
259 full, anchor := true, false
260 for _, sign := range []int{-1, 1} {
261 r, c := p/8+dr*sign, p%8+dc*sign
262 if !inside(r, c) || (stable[8*r+c] && b[8*r+c] == b[p]) {
263 anchor = true
264 }
265 for inside(r, c) {
266 if b[8*r+c] == '.' {
267 full = false
268 }
269 r += dr * sign
270 c += dc * sign
271 }
272 }
273 if !full && !anchor {
274 ok = false
275 break
276 }
277 }
278 if ok {
279 found = append(found, p)
280 }
281 }
282 if len(found) == 0 {
283 break
284 }
285 for _, p := range found {
286 stable[p] = true
287 }
288 }
289 black, white := 0, 0
290 for p, yes := range stable {
291 if yes {
292 if b[p] == 'B' {
293 black++
294 } else {
295 white++
296 }
297 }
298 }
299 return black, white
300}
301func transform(p int, op string) int {
302 r, c := p/8, p%8
303 switch op {
304 case "rot90":
305 return 8*c + 7 - r
306 case "rot180":
307 return 63 - p
308 case "rot270":
309 return 8*(7-c) + r
310 case "mirror":
311 return 8*r + 7 - c
312 case "flip":
313 return 8*(7-r) + c
314 case "diag":
315 return 8*c + r
316 default:
317 return 8*(7-c) + 7 - r
318 }
319}
320func transformBoard(b Board, op string) Board {
321 var a Board
322 for p, x := range b {
323 a[transform(p, op)] = x
324 }
325 return a
326}

336func randomGame(b Board, s byte, rng *minstd) Board {
337 for {
338 moves := legal(b, s)
339 if len(moves) > 0 {
340 b = place(b, s, moves[int(rng.draw()%int64(len(moves)))])
341 } else if len(legal(b, other(s))) == 0 {
342 return b
343 }
344 s = other(s)
345 }

398func (e *Othello) adjust(quiet bool) bool {
399 if len(legal(e.board, e.side)) > 0 {
400 return false
401 }
402 if len(legal(e.board, other(e.side))) > 0 {
403 if !quiet {
404 e.emit(string(e.side) + " passes")
405 }
406 e.side = other(e.side)
407 return true
408 }
409 e.over = true
410 if !quiet {
411 e.emit(gameOver(e.board))
412 }
413 return false

415func (e *Othello) move(p int, quiet bool, value *int) bool {
416 before, s, k := e.snapshot(), e.side, len(flips(e.board, e.side, p))
417 e.board = place(e.board, s, p)
418 if !quiet {
419 text := fmt.Sprintf("%c %s flips %d", s, name(p), k)
420 if value != nil {
421 text += fmt.Sprintf(" value %d", *value)
422 }
423 e.emit(text)
424 }
425 e.side = other(s)
426 passed := e.adjust(quiet)
427 e.history = append(e.history, moveRecord{s, p, k, passed, before, e.snapshot()})
428 e.redo = nil
429 return passed
430}

443func (e *Othello) record(text string) {
444 temp := newEngine(e.out)
445 ps := []int{}
446 passes := 0
447 for i := 0; i < len(text); i += 2 {
448 p := square(text[i : i+2])
449 if why := temp.reason(p); why != "" {
450 e.emit(fmt.Sprintf("ILLEGAL move %d %s %s", i/2+1, name(p), why))
451 return
452 }
453 ps = append(ps, p)
454 if temp.move(p, true, nil) {
455 passes++
456 }
457 }
458 e.restore(temp.snapshot())
459 e.history = temp.history
460 e.redo = nil
461 op := map[string]string{"f5": "", "e6": "diag", "d3": "anti", "c4": "rot180"}[name(ps[0])]
462 var normal strings.Builder
463 for _, p := range ps {
464 if op != "" {
465 p = transform(p, op)
466 }
467 normal.WriteString(name(p))
468 }
469 e.emit(fmt.Sprintf("record %d moves %d passes", len(ps), passes))
470 e.emit("normal " + normal.String())
471 e.emit(e.status())
472}

494func (e *Othello) stats() {
495 total, passes := 0, 0
496 for _, r := range e.history {
497 total += r.count
498 if r.passed {
499 passes++
500 }
501 }
502 if len(e.history) == 0 {
503 e.emit("moves 0")
504 } else {
505 e.emit(fmt.Sprintf("moves %d flips %d avg %s passes %d", len(e.history), total, average(total, len(e.history)), passes))
506 }
507 for _, s := range []byte{'B', 'W'} {
508 n, sum, maximum, p, corners := 0, 0, -1, -1, 0
509 for _, r := range e.history {
510 if r.side == s {
511 n++
512 sum += r.count
513 if r.count > maximum {
514 maximum = r.count
515 p = r.square
516 }
517 if r.square == 0 || r.square == 7 || r.square == 56 || r.square == 63 {
518 corners++
519 }
520 }
521 }
522 if n == 0 {
523 e.emit(string(s) + " moves 0")
524 } else {
525 e.emit(fmt.Sprintf("%c moves %d flips %d avg %s max %d %s corners %d", s, n, sum, average(sum, n), maximum, name(p), corners))
526 }
527 }
528}
529func (e *Othello) auto(black, white int, seed int64) {
530 rng := minstd{seed}
531 ps := []string{}
532 passes := 0
533 for !e.over {
534 moves := legal(e.board, e.side)
535 p := 0
536 if seed != 0 {
537 p = moves[int(rng.draw()%int64(len(moves)))]
538 } else {
539 depth := black
540 if e.side == 'W' {
541 depth = white
542 }
543 p = ranked(e.board, e.side, depth)[0].square
544 }
545 ps = append(ps, name(p))
546 if e.move(p, true, nil) {
547 passes++
548 }
549 }
550 kind := "autoplay"
551 if seed != 0 {
552 kind = "random"
553 }
554 e.emit(fmt.Sprintf("%s %d moves %d passes", kind, len(ps), passes))
555 e.emit("moves " + strings.Join(ps, ""))
556 e.emit(gameOver(e.board))
557}
558func (e *Othello) monteCarlo(k int, seed int64) {
559 rng := minstd{seed}
560 best, bp := -1, -1
561 for _, p := range legal(e.board, e.side) {
562 w, d, l := 0, 0, 0
563 start := place(e.board, e.side, p)
564 for i := 0; i < k; i++ {
565 b := randomGame(start, other(e.side), &rng)
566 diff := count(b, e.side) - count(b, other(e.side))
567 if diff > 0 {
568 w++
569 } else if diff < 0 {
570 l++
571 } else {
572 d++
573 }
574 }
575 e.emit(fmt.Sprintf("%s %d %d %d", name(p), w, d, l))
576 if 2*w+d > best {
577 best = 2*w + d
578 bp = p
579 }
580 }
581 e.emit(fmt.Sprintf("best %s %d", name(bp), best))
582}
583func validSquare(s string) bool {
584 return len(s) == 2 && s[0] >= 'a' && s[0] <= 'h' && s[1] >= '1' && s[1] <= '8'
585}
586func digits(s string) bool {
587 if s == "" {
588 return false

590 for _, c := range s {

592 return false

595 return true

597func (e *Othello) execute(text string) {
598 words := strings.Fields(text)
599 cmd, a := words[0], words[1:]
600 problem := ""
601 if !zero[cmd] && !numeric[cmd] && !extra[cmd] {
602 problem = "unknown command"
603 } else if cmd == "LOAD" {
604 valid := len(a) == 2
605 if valid {
606 valid = len(a[0]) == 64 && (a[1] == "B" || a[1] == "W")
607 for _, c := range a[0] {
608 if c != '.' && c != 'B' && c != 'W' {
609 valid = false
610 }
611 }
612 }
613 if !valid {
614 problem = "bad board"
615 }
616 } else if cmd == "MOVE" || cmd == "FLIPS" {
617 if len(a) != 1 {
618 problem = "bad arguments"
619 } else if !validSquare(a[0]) {
620 problem = "bad square"
621 }
622 } else if cmd == "PLAY" || cmd == "RECORD" {
623 if len(a) != 1 {
624 problem = "bad arguments"
625 } else {
626 valid := len(a[0]) >= 2 && len(a[0]) <= 120 && len(a[0])%2 == 0
627 for i := 0; i+1 < len(a[0]); i += 2 {
628 if !validSquare(a[0][i : i+2]) {
629 valid = false
630 }
631 }
632 if !valid {
633 problem = "bad record"
634 }
635 }
636 } else if zero[cmd] {
637 if len(a) != 0 {
638 problem = "bad arguments"
639 }
640 } else if cmd == "TRANSFORM" {
641 if len(a) != 1 || !transforms[a[0]] {
642 problem = "bad arguments"
643 }
644 } else {
645 expected := 1
646 if cmd == "AUTOPLAY" || cmd == "MC" {
647 expected = 2
648 }
649 valid := len(a) == expected
650 ns := []int64{}
651 for _, s := range a {
652 if !digits(s) {
653 valid = false
654 } else {
655 x, err := strconv.ParseInt(s, 10, 64)
656 if err != nil {
657 valid = false
658 } else {
659 ns = append(ns, x)
660 }
661 }
662 }
663 if valid {
664 n := ns[0]
665 switch cmd {
666 case "AUTOPLAY":
667 valid = n >= 1 && n <= 4 && ns[1] >= 1 && ns[1] <= 4
668 case "HINT", "PV", "AI", "ANALYZE":
669 valid = n >= 1 && n <= 4
670 case "PERFT", "DIVIDE":
671 valid = n >= 1 && n <= 8
672 case "MC":
673 valid = n >= 1 && n <= 100 && ns[1] >= 1 && ns[1] <= 2147483646
674 case "RANDOM":
675 valid = n >= 1 && n <= 2147483646
676 default:
677 valid = n >= 1
678 }
679 }
680 if !valid {
681 problem = "bad arguments"
682 }
683 }
684 if problem != "" {
685 e.emit("ERROR " + problem)
686 return
687 }
688 if e.over && active[cmd] {
689 e.emit("ERROR game over")
690 return
691 }
692 n := 0
693 if len(a) > 0 {
694 n, _ = strconv.Atoi(a[0])
695 }
696 switch cmd {
697 case "NEW":
698 e.reset()
699 e.emit("OK")
700 case "LOAD":
701 copy(e.board[:], a[0])
702 e.side = a[1][0]
703 e.over = false
704 e.history = nil
705 e.redo = nil
706 e.emit("OK")
707 e.adjust(false)
708 case "DUMP":
709 e.emit("board " + string(e.board[:]) + " " + string(e.side))
710 case "SHOW":
711 e.emit(" a b c d e f g h")
712 ms := map[int]bool{}
713 if !e.over {
714 for _, p := range legal(e.board, e.side) {
715 ms[p] = true
716 }
717 }
718 for r := 0; r < 8; r++ {
719 row := strconv.Itoa(r + 1)
720 for c := 0; c < 8; c++ {
721 x := e.board[8*r+c]
722 if ms[8*r+c] {
723 x = '*'
724 }
725 row += " " + string(x)
726 }
727 e.emit(row)
728 }
729 e.emit(e.status())
730 case "SCORE":
731 e.emit(fmt.Sprintf("score B %d W %d empty %d", count(e.board, 'B'), count(e.board, 'W'), count(e.board, '.')))
732 case "LEGAL":
733 moves := []int{}
734 if !e.over {
735 moves = legal(e.board, e.side)
736 }
737 line := "legal " + strconv.Itoa(len(moves))
738 for _, p := range moves {
739 line += " " + name(p)
740 }
741 e.emit(line)
742 case "MOVE", "FLIPS":
743 p := square(a[0])
744 if why := e.reason(p); why != "" {
745 e.emit("ILLEGAL " + why)
746 } else if cmd == "MOVE" {
747 e.move(p, false, nil)
748 } else {
749 f := flips(e.board, e.side, p)
750 line := fmt.Sprintf("flips %s %d", a[0], len(f))
751 for _, q := range f {
752 line += " " + name(q)
753 }
754 e.emit(line)
755 }
756 case "PLAY":
757 for i := 0; i < len(a[0]); i += 2 {
758 p := square(a[0][i : i+2])
759 if why := e.reason(p); why != "" {
760 e.emit(fmt.Sprintf("ILLEGAL move %d %s %s", i/2+1, name(p), why))
761 break
762 }
763 e.move(p, false, nil)
764 }
765 case "RECORD":
766 e.record(a[0])
767 case "UNDO":
768 if n > len(e.history) {
769 e.emit("ERROR cannot undo")
770 } else {
771 start := len(e.history) - n
772 removed := append([]moveRecord(nil), e.history[start:]...)
773 e.restore(removed[0].before)
774 e.history = e.history[:start]
775 e.redo = append(removed, e.redo...)
776 e.emit(e.status())
777 }
778 case "REDO":
779 if n > len(e.redo) {
780 e.emit("ERROR cannot redo")
781 } else {
782 added := append([]moveRecord(nil), e.redo[:n]...)
783 e.redo = e.redo[n:]
784 e.history = append(e.history, added...)
785 e.restore(added[n-1].after)
786 e.emit(e.status())
787 }
788 case "HISTORY":
789 e.emit("history " + strconv.Itoa(len(e.history)))
790 for i, r := range e.history {
791 line := fmt.Sprintf("%d %c %s %d", i+1, r.side, name(r.square), r.count)
792 if r.passed {
793 line += " pass"
794 }
795 e.emit(line)
796 }
797 case "SHEET":
798 e.sheet()
799 case "STATS":
800 e.stats()
801 case "TRANSFORM":
802 op := a[0]
803 e.board = transformBoard(e.board, op)
804 tr := func(r moveRecord) moveRecord {
805 r.square = transform(r.square, op)
806 r.before.board = transformBoard(r.before.board, op)
807 r.after.board = transformBoard(r.after.board, op)
808 return r
809 }
810 for i, r := range e.history {
811 e.history[i] = tr(r)
812 }
813 for i, r := range e.redo {
814 e.redo[i] = tr(r)
815 }
816 e.emit("OK")
817 case "EVAL":
818 v := evaluate(e.board, e.side)
819 e.emit(fmt.Sprintf("eval %d weights %d mobility %d %d", v[0], v[1], v[2], v[3]))
820 case "STABLE":
821 b, w := stable(e.board)
822 e.emit(fmt.Sprintf("stable B %d W %d", b, w))
823 case "HINT":
824 for _, r := range ranked(e.board, e.side, n) {
825 e.emit(fmt.Sprintf("%s %d", name(r.square), r.value))
826 }
827 case "PV":
828 steps, _ := variation(e.board, e.side, n)
829 out := []string{"pv", strconv.Itoa(search(e.board, e.side, n))}
830 e.emit(strings.Join(append(out, steps...), " "))
831 case "AI":
832 r := ranked(e.board, e.side, n)[0]
833 e.move(r.square, false, &r.value)
834 case "ANALYZE":
835 loss := map[byte]int{'B': 0, 'W': 0}
836 for i, r := range e.history {
837 rows := ranked(r.before.board, r.side, n)
838 best, bp := rows[0].value, rows[0].square
839 value := 0
840 for _, row := range rows {
841 if row.square == r.square {
842 value = row.value
843 break
844 }
845 }
846 loss[r.side] += best - value
847 e.emit(fmt.Sprintf("%d %c %s %d best %s %d", i+1, r.side, name(r.square), value, name(bp), best))
848 }
849 e.emit(fmt.Sprintf("loss B %d W %d", loss['B'], loss['W']))
850 case "SOLVE":
851 if count(e.board, '.') >= 9 {
852 e.emit("ERROR too many empties")
853 } else {
854 for _, r := range ranked(e.board, e.side, -1) {
855 e.emit(fmt.Sprintf("%s %d", name(r.square), r.value))
856 }
857 steps, b := variation(e.board, e.side, -1)
858 e.emit(strings.Join(append([]string{"line"}, steps...), " "))
859 e.emit(fmt.Sprintf("final B %d W %d", count(b, 'B'), count(b, 'W')))
860 }
861 case "AUTOPLAY":
862 white, _ := strconv.Atoi(a[1])
863 e.auto(n, white, 0)
864 case "RANDOM":
865 e.auto(0, 0, int64(n))
866 case "MC":
867 seed, _ := strconv.ParseInt(a[1], 10, 64)
868 e.monteCarlo(n, seed)
869 case "PERFT":
870 e.emit(fmt.Sprintf("perft %d %d", n, perft(e.board, e.side, n)))
871 case "DIVIDE":
872 total := 0
873 for _, p := range legal(e.board, e.side) {
874 v := perft(place(e.board, e.side, p), other(e.side), n-1)
875 e.emit(fmt.Sprintf("%s %d", name(p), v))
876 total += v
877 }
878 e.emit("total " + strconv.Itoa(total))
879 }
880}
