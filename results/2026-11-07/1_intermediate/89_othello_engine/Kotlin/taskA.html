9fun name(p:Int)="${'a'+p%8}${p/8+1}"
10fun square(s:String)=(s[1]-'1')*8+s[0]-'a'
11fun inside(r:Int,c:Int)=r in 0..7&&c in 0..7
12fun initial():String {
13 val b=CharArray(64){
14 '.'
15 };
16 b[27]='W';

18 b[28]='B';

22fun flips(b:String,s:Char,p:Int):List<Int>{
23 if(b[p]!='.')return emptyList();
24 val out=mutableListOf<Int>();
25 val row=p/8;
26 val col=p%8;
27 for(dr in -1..1)for(dc in -1..1){
28 if(dr==0&&dc==0)continue;
29 var r=row+dr;
30 var c=col+dc;
31 val line=mutableListOf<Int>();
32 while(inside(r,c)&&b[8*r+c]==other(s)){
33 line+=8*r+c;
34 r+=dr;
35 c+=dc
36 };
37 if(line.isNotEmpty()&&inside(r,c)&&b[8*r+c]==s)out+=line
38 };
39 return out.sorted()
40}
41fun legal(b:String,s:Char)=(0..63).filter{
42 b[it]=='.'&&flips(b,s,it).isNotEmpty()
43}
44fun place(b:String,s:Char,p:Int):String{
45 val a=b.toCharArray();
46 for(q in flips(b,s,p))a[q]=s;
47 a[p]=s;
48 return a.concatToString()
49}
50fun terminal(b:String)=legal(b,'B').isEmpty()&&legal(b,'W').isEmpty()
51fun gameOver(b:String):String{
52 val x=b.count{
53 it=='B'
54 };
55 val y=b.count{
56 it=='W'
57 };
58 return "game over B $x W $y ${if(x>y)"winner B"else if(x<y)"winner W"else "draw"}"
59}
60fun evaluate(b:String,s:Char):List<Int>{
61 val w=(0..63).sumOf{
62 if(b[it]==s)weights[it]else if(b[it]==other(s))-weights[it]else 0
63 };
64 val own=legal(b,s).size;
65 val enemy=legal(b,other(s)).size;
66 return listOf(w+10*(own-enemy),w,own,enemy)
67}
68val searchCache=HashMap<Triple<String,Char,Int>,Int>();
69val perftCache=HashMap<Triple<String,Char,Int>,Int>()
70fun search(b:String,s:Char,d:Int):Int{
71 val key=Triple(b,s,d);
72 searchCache[key]?.let{
73 return it
74 };
75 val moves=legal(b,s);
76 val o=other(s);
77 val v=if(moves.isEmpty()&&legal(b,o).isEmpty())(if(d<0)1 else 1000)*(b.count{
78 it==s
79 }-b.count{
80 it==o
81 })else if(d==0)evaluate(b,s)[0]else if(moves.isEmpty())-search(b,o,if(d<0)-1 else d-1)else moves.maxOf{
82 -search(place(b,s,it),o,if(d<0)-1 else d-1)
83 };
84 if(searchCache.size>300000)searchCache.clear();
85 searchCache[key]=v;
86 return v
87}
88fun ranked(b:String,s:Char,d:Int)=legal(b,s).map{
89 -search(place(b,s,it),other(s),if(d<0)-1 else d-1) to it
90}.sortedWith(compareByDescending<Pair<Int,Int>>{
91 it.first
92}.thenBy{
93 it.second
94})
95fun variation(start:String,side:Char,depth:Int):Pair<List<String>,String>{
96 var b=start;
97 var s=side;
98 var d=depth;
99 val steps=mutableListOf<String>();
100 while(d!=0&&!terminal(b)){
101 val moves=ranked(b,s,d);
102 if(moves.isEmpty())steps+="pass"else{
103 val p=moves[0].second;
104 steps+=name(p);
105 b=place(b,s,p)
106 };
107 s=other(s);
108 if(d>0)d--
109 };
110 return steps to b
111}
112fun perft(b:String,s:Char,d:Int):Int{
113 if(d==0)return 1;
114 val key=Triple(b,s,d);
115 perftCache[key]?.let{
116 return it
117 };
118 val moves=legal(b,s);
119 val o=other(s);
120 val n=if(moves.isNotEmpty())moves.sumOf{
121 perft(place(b,s,it),o,d-1)
122 }else if(legal(b,o).isEmpty())1 else perft(b,o,d-1);
123 if(perftCache.size>200000)perftCache.clear();
124 perftCache[key]=n;
125 return n
126}
127fun stable(b:String):Pair<Int,Int>{
128 val stable=BooleanArray(64);
129 val axes=listOf(0 to 1,1 to 0,1 to 1,1 to -1);
130 while(true){
131 val found=mutableListOf<Int>();
132 for(p in 0..63){
133 if(b[p]=='.'||stable[p])continue;
134 var ok=true;
135 for((dr,dc) in axes){
136 var full=true;
137 var anchor=false;
138 for(sign in listOf(-1,1)){
139 var r=p/8+dr*sign;
140 var c=p%8+dc*sign;
141 if(!inside(r,c)||(stable[8*r+c]&&b[8*r+c]==b[p]))anchor=true;
142 while(inside(r,c)){
143 if(b[8*r+c]=='.')full=false;
144 r+=dr*sign;
145 c+=dc*sign
146 }
147 };
148 if(!full&&!anchor){
149 ok=false;
150 break
151 }
152 };
153 if(ok)found+=p
154 };
155 if(found.isEmpty())break;
156 for(p in found)stable[p]=true
157 };
158 return (0..63).count{
159 stable[it]&&b[it]=='B'
160 } to (0..63).count{
161 stable[it]&&b[it]=='W'
162 }
163}
164fun transform(p:Int,op:String):Int{
165 val r=p/8;
166 val c=p%8;
167 return when(op){
168 "rot90"->8*c+7-r;
169 "rot180"->63-p;
170 "rot270"->8*(7-c)+r;
171 "mirror"->8*r+7-c;
172 "flip"->8*(7-r)+c;
173 "diag"->8*c+r;
174 else->8*(7-c)+7-r
175 }
176}
177fun transformBoard(b:String,op:String):String{
178 val a=CharArray(64);
179 for(p in 0..63)a[transform(p,op)]=b[p];
180 return a.concatToString()
181}

188fun randomGame(start:String,side:Char,rng:Minstd):String{
189 var b=start;
190 var s=side;
191 while(true){
192 val moves=legal(b,s);
193 if(moves.isNotEmpty())b=place(b,s,moves[(rng.draw()%moves.size).toInt()])else if(legal(b,other(s)).isEmpty())return b;
194 s=other(s)
195 }
196}

224 fun adjust(quiet:Boolean=false):Boolean{
225 if(legal(board,side).isNotEmpty())return false;
226 if(legal(board,other(side)).isNotEmpty()){
227 if(!quiet)emit(side,"passes");
228 side=other(side);

235 fun move(p:Int,quiet:Boolean=false,value:Int?=null):Boolean{
236 val before=snapshot();
237 val s=side;
238 val k=flips(board,s,p).size;
239 board=place(board,s,p);
240 if(!quiet){
241 if(value==null)emit(s,name(p),"flips",k)else emit(s,name(p),"flips",k,"value",value)
242 };
243 side=other(s);
244 val passed=adjust(quiet);
245 history+=Move(s,p,k,passed,before,snapshot());
246 redo.clear();
247 return passed
248 }
249 fun reason(p:Int)=if(over)"game over"else if(board[p]!='.')"occupied"else if(flips(board,side,p).isEmpty())"no flips"else ""
250 fun record(text:String){
251 val e=Othello();
252 val ps=mutableListOf<Int>();
253 var passes=0;
254 for(i in text.indices step 2){
255 val p=square(text.substring(i,i+2));
256 val why=e.reason(p);

258 emit("ILLEGAL move",i/2+1,name(p),why);

263 };
264 restore(e.snapshot());

266 redo.clear();
267 val op=when(name(ps[0])){

270 "c4"->"rot180";
271 else->""
272 };
273 emit("record",ps.size,"moves",passes,"passes");
274 emit("normal",ps.joinToString(""){
275 name(if(op.isEmpty())it else transform(it,op))
276 });
277 emit(status())
278 }

321 fun auto(black:Int,white:Int,seed:Long=0){
322 val rng=Minstd(seed);
323 val ps=mutableListOf<String>();
324 var passes=0;
325 while(!over){
326 val moves=legal(board,side);
327 val p=if(seed!=0L)moves[(rng.draw()%moves.size).toInt()]else ranked(board,side,if(side=='B')black else white)[0].second;
328 ps+=name(p);
329 if(move(p,true))passes++
330 };
331 emit(if(seed!=0L)"random"else "autoplay",ps.size,"moves",passes,"passes");
332 emit("moves",ps.joinToString(""));
333 emit(gameOver(board))
334 }
335 fun monteCarlo(k:Int,seed:Long){
336 val rng=Minstd(seed);
337 var best=-1;
338 var bp=-1;
339 for(p in legal(board,side)){
340 var w=0;
341 var d=0;
342 var l=0;
343 val start=place(board,side,p);
344 repeat(k){
345 val b=randomGame(start,other(side),rng);
346 val diff=b.count{
347 it==side
348 }-b.count{
349 it==other(side)
350 };
351 if(diff>0)w++ else if(diff<0)l++ else d++
352 };
353 emit(name(p),w,d,l);
354 if(2*w+d>best){
355 best=2*w+d;
356 bp=p
357 }
358 };
359 emit("best",name(bp),best)
360 }
361 fun execute(text:String){
362 val words=text.split(' ');
363 val cmd=words[0];
364 val a=words.drop(1);
365 var error=""
366 if(cmd !in commands)error="unknown command"
367 else if(cmd=="LOAD"){
368 if(a.size!=2||!Regex("[.BW]{64}").matches(a[0])||a[1] !in listOf("B","W"))error="bad board"
369 }
370 else if(cmd in listOf("MOVE","FLIPS")){
371 error=if(a.size!=1)"bad arguments"else if(!Regex("[a-h][1-8]").matches(a[0]))"bad square"else ""
372 }
373 else if(cmd in listOf("PLAY","RECORD")){
374 error=if(a.size!=1)"bad arguments"else if(a[0].length>120||!Regex("([a-h][1-8])+").matches(a[0]))"bad record"else ""
375 }
376 else if(cmd in zero){
377 if(a.isNotEmpty())error="bad arguments"
378 }
379 else if(cmd=="TRANSFORM"){
380 if(a.size!=1||a[0] !in transforms)error="bad arguments"
381 }
382 else{
383 val expected=if(cmd in listOf("AUTOPLAY","MC"))2 else 1;
384 var valid=a.size==expected&&a.all{
385 Regex("[0-9]+").matches(it)
386 };
387 if(valid){
388 val ns=a.map{
389 it.toLong()
390 };
391 val n=ns[0];
392 valid=when(cmd){
393 "AUTOPLAY"->ns.all{
394 it in 1L..4L
395 };
396 "HINT","PV","AI","ANALYZE"->n in 1L..4L;
397 "PERFT","DIVIDE"->n in 1L..8L;
398 "MC"->n in 1L..100L&&ns[1] in 1L..2147483646L;
399 "RANDOM"->n in 1L..2147483646L;
400 else->n>=1
401 }
402 };
403 if(!valid)error="bad arguments"
404 }
405 if(error.isNotEmpty()){
406 emit("ERROR",error);
407 return
408 };
409 if(over&&cmd in active){
410 emit("ERROR game over");
411 return
412 };
413 val n=a.firstOrNull()?.toIntOrNull()?:0
414 when(cmd){
415 "NEW"->{
416 reset();
417 emit("OK")
418 }
419 "LOAD"->{
420 board=a[0];
421 side=a[1][0];
422 over=false;
423 history.clear();
424 redo.clear();
425 emit("OK");
426 adjust()
427 }
428 "DUMP"->emit("board",board,side)
429 "SHOW"->{
430 emit(" a b c d e f g h");
431 val moves=if(over)emptyList()else legal(board,side);
432 for(r in 0..7)emit((r+1).toString()+" "+(0..7).joinToString(" "){
433 c->(if(8*r+c in moves)'*'else board[8*r+c]).toString()
434 });
435 emit(status())
436 }
437 "SCORE"->emit("score B",board.count{
438 it=='B'
439 },"W",board.count{
440 it=='W'
441 },"empty",board.count{
442 it=='.'
443 })
444 "LEGAL"->{
445 val moves=if(over)emptyList()else legal(board,side);
446 emit((listOf("legal",moves.size.toString())+moves.map(::name)).joinToString(" "))
447 }
448 "MOVE","FLIPS"->{
449 val p=square(a[0]);
450 val why=reason(p);
451 if(why.isNotEmpty())emit("ILLEGAL",why)else if(cmd=="MOVE")move(p)else{
452 val f=flips(board,side,p);
453 emit((listOf("flips",a[0],f.size.toString())+f.map(::name)).joinToString(" "))
454 }
455 }
456 "PLAY"->{
457 for(i in a[0].indices step 2){
458 val p=square(a[0].substring(i,i+2));
459 val why=reason(p);
460 if(why.isNotEmpty()){
461 emit("ILLEGAL move",i/2+1,name(p),why);
462 break
463 };
464 move(p)
465 }
466 }
467 "RECORD"->record(a[0])
468 "UNDO"->{
469 if(n>history.size)emit("ERROR cannot undo")else{
470 val removed=history.takeLast(n);
471 restore(removed[0].before);
472 history=history.dropLast(n).toMutableList();
473 redo=(removed+redo).toMutableList();
474 emit(status())
475 }
476 }
477 "REDO"->{
478 if(n>redo.size)emit("ERROR cannot redo")else{
479 val added=redo.take(n);
480 redo=redo.drop(n).toMutableList();
481 history.addAll(added);
482 restore(added.last().after);
483 emit(status())
484 }
485 }
486 "HISTORY"->{
487 emit("history",history.size);
488 history.forEachIndexed{
489 i,r->emit("${i+1} ${r.side} ${name(r.square)} ${r.count}"+(if(r.passed)" pass"else ""))
490 }
491 }
492 "SHEET"->sheet();
493 "STATS"->stats()
494 "TRANSFORM"->{
495 val op=a[0];
496 board=transformBoard(board,op);
497 fun tr(r:Move)=r.copy(square=transform(r.square,op),before=r.before.copy(board=transformBoard(r.before.board,op)),after=r.after.copy(board=transformBoard(r.after.board,op)));
498 history=history.map(::tr).toMutableList();
499 redo=redo.map(::tr).toMutableList();
500 emit("OK")
501 }
502 "EVAL"->{
503 val(v,w,x,y)=evaluate(board,side);
504 emit("eval",v,"weights",w,"mobility",x,y)
505 }
506 "STABLE"->{
507 val(b,w)=stable(board);
508 emit("stable B",b,"W",w)
509 }
510 "HINT"->ranked(board,side,n).forEach{
511 (v,p)->emit(name(p),v)
512 }
513 "PV"->{
514 val(steps,_)=variation(board,side,n);
515 emit((listOf("pv",search(board,side,n).toString())+steps).joinToString(" "))
516 }
517 "AI"->{
518 val(v,p)=ranked(board,side,n)[0];
519 move(p,false,v)
520 }
521 "ANALYZE"->{
522 val loss=mutableMapOf('B' to 0,'W' to 0);
523 history.forEachIndexed{
524 i,r->val rows=ranked(r.before.board,r.side,n);
525 val(best,bp)=rows[0];
526 val v=rows.first{
527 it.second==r.square
528 }.first;
529 loss[r.side]=loss.getValue(r.side)+best-v;
530 emit(i+1,r.side,name(r.square),v,"best",name(bp),best)
531 };
532 emit("loss B",loss.getValue('B'),"W",loss.getValue('W'))
533 }
534 "SOLVE"->{
535 if(board.count{
536 it=='.'
537 }>=9)emit("ERROR too many empties")else{
538 ranked(board,side,-1).forEach{
539 (v,p)->emit(name(p),v)
540 };
541 val(steps,b)=variation(board,side,-1);
542 emit((listOf("line")+steps).joinToString(" "));
543 emit("final B",b.count{
544 it=='B'
545 },"W",b.count{
546 it=='W'
547 })
548 }
549 }
550 "AUTOPLAY"->auto(n,a[1].toInt());
551 "RANDOM"->auto(0,0,n.toLong());
552 "MC"->monteCarlo(n,a[1].toLong())
553 "PERFT"->emit("perft",n,perft(board,side,n))
554 "DIVIDE"->{
555 var total=0;
556 for(p in legal(board,side)){
557 val v=perft(place(board,side,p),other(side),n-1);
558 emit(name(p),v);
559 total+=v
560 };
561 emit("total",total)
562 }
563 }
564 }
565}
