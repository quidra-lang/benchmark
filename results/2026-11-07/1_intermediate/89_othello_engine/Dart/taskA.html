100List<int> flips(String b, String s, int p) {
101 if (b[p] != '.') return [];
102 var out = <int>[];
103 var row = p ~/ 8;
104 var col = p % 8;
105 for (var dr = -1; dr <= 1; dr++) {
106 for (var dc = -1; dc <= 1; dc++) {
107 if (dr == 0 && dc == 0) continue;
108 var r = row + dr;
109 var c = col + dc;
110 var line = <int>[];
111 while (inside(r, c) && b[8 * r + c] == other(s)) {
112 line.add(8 * r + c);
113 r += dr;
114 c += dc;
115 }
116 if (line.isNotEmpty && inside(r, c) && b[8 * r + c] == s)
117 out.addAll(line);
118 }
119 }
120 out.sort();
121 return out;
122}

148 var w = 0;
149 for (var i = 0; i < 64; i++)

162int search(String b, String s, int d) {
163 var key = (b, s, d);
164 if (searchCache.containsKey(key)) return searchCache[key]!;
165 var moves = legal(b, s);
166 var o = other(s);
167 int v;
168 if (moves.isEmpty && legal(b, o).isEmpty) {
169 v = (d < 0 ? 1 : 1000) * (count(b, s) - count(b, o));
170 } else if (d == 0) {
171 v = evaluate(b, s).$1;
172 } else if (moves.isEmpty) {
173 v = -search(b, o, d < 0 ? -1 : d - 1);
174 } else {
175 v = -1000000;
176 for (var p in moves)
177 v = math.max(v, -search(place(b, s, p), o, d < 0 ? -1 : d - 1));
178 }
179 if (searchCache.length > 300000) searchCache.clear();
180 searchCache[key] = v;
181 return v;
182}

212int perft(String b, String s, int d) {
213 if (d == 0) return 1;
214 var key = (b, s, d);
215 if (perftCache.containsKey(key)) return perftCache[key]!;
216 var moves = legal(b, s);
217 var o = other(s);
218 var n = 0;
219 if (moves.isEmpty) {
220 n = legal(b, o).isEmpty ? 1 : perft(b, o, d - 1);
221 } else {
222 for (var p in moves) n += perft(place(b, s, p), o, d - 1);
223 }
224 if (perftCache.length > 200000) perftCache.clear();
225 perftCache[key] = n;
226 return n;
227}

229(int, int) stable(String b) {
230 var stable = List.filled(64, false);
231 const axes = [(0, 1), (1, 0), (1, 1), (1, -1)];
232 while (true) {
233 var found = <int>[];
234 for (var p = 0; p < 64; p++) {
235 if (b[p] == '.' || stable[p]) continue;
236 var ok = true;
237 for (var (dr, dc) in axes) {
238 var full = true;
239 var anchor = false;
240 for (var sign in [-1, 1]) {
241 var r = p ~/ 8 + dr * sign;
242 var c = p % 8 + dc * sign;
243 if (!inside(r, c) || (stable[8 * r + c] && b[8 * r + c] == b[p]))
244 anchor = true;
245 while (inside(r, c)) {
246 if (b[8 * r + c] == '.') full = false;
247 r += dr * sign;
248 c += dc * sign;
249 }
250 }
251 if (!full && !anchor) {
252 ok = false;
253 break;
254 }
255 }
256 if (ok) found.add(p);
257 }
258 if (found.isEmpty) break;
259 for (var p in found) stable[p] = true;
260 }
261 return (
262 [
263 for (var i = 0; i < 64; i++)
264 if (stable[i] && b[i] == 'B') i,
265 ].length,
266 [
267 for (var i = 0; i < 64; i++)
268 if (stable[i] && b[i] == 'W') i,
269 ].length,
270 );
271}

273int transform(int p, String op) {
274 var r = p ~/ 8;
275 var c = p % 8;
276 return switch (op) {
277 'rot90' => 8 * c + 7 - r,
278 'rot180' => 63 - p,
279 'rot270' => 8 * (7 - c) + r,
280 'mirror' => 8 * r + 7 - c,
281 'flip' => 8 * (7 - r) + c,
282 'diag' => 8 * c + r,
283 _ => 8 * (7 - c) + 7 - r,
284 };
285}

535 void execute(String text) {
536 var words = text.split(' ');
537 var cmd = words[0];
538 var a = words.sublist(1);
539 var error = '';
540 if (!commands.contains(cmd)) {
541 error = 'unknown command';
542 } else if (cmd == 'LOAD') {
543 if (a.length != 2 ||
544 !RegExp(r'^[.BW]{64}$').hasMatch(a[0]) ||
545 !['B', 'W'].contains(a[1]))
546 error = 'bad board';
547 } else if (['MOVE', 'FLIPS'].contains(cmd)) {
548 if (a.length != 1) {
549 error = 'bad arguments';
550 } else if (!RegExp(r'^[a-h][1-8]$').hasMatch(a[0])) {
551 error = 'bad square';
552 }
553 } else if (['PLAY', 'RECORD'].contains(cmd)) {
554 if (a.length != 1) {
555 error = 'bad arguments';
556 } else if (a[0].length > 120 ||
557 !RegExp(r'^([a-h][1-8])+$').hasMatch(a[0])) {
558 error = 'bad record';
559 }
560 } else if (zero.contains(cmd)) {
561 if (a.isNotEmpty) error = 'bad arguments';
562 } else if (cmd == 'TRANSFORM') {
563 if (a.length != 1 || !transforms.contains(a[0])) error = 'bad arguments';
564 } else {
565 var expected = ['AUTOPLAY', 'MC'].contains(cmd) ? 2 : 1;
566 var valid =
567 a.length == expected &&
568 a.every((s) => RegExp(r'^[0-9]+$').hasMatch(s));
569 if (valid) {
570 var ns = a.map(int.parse).toList();
571 var n = ns[0];
572 valid = switch (cmd) {
573 'AUTOPLAY' => ns.every((n) => n >= 1 && n <= 4),
574 'HINT' || 'PV' || 'AI' || 'ANALYZE' => n >= 1 && n <= 4,
575 'PERFT' || 'DIVIDE' => n >= 1 && n <= 8,
576 'MC' => n >= 1 && n <= 100 && ns[1] >= 1 && ns[1] <= 2147483646,
577 'RANDOM' => n >= 1 && n <= 2147483646,
578 _ => n >= 1,
579 };
580 }
581 if (!valid) error = 'bad arguments';
582 }
583 if (error.isNotEmpty) {
584 emit(['ERROR', error]);
585 return;
586 }
587 if (over && active.contains(cmd)) {
588 emit(['ERROR game over']);
589 return;
590 }
591 var n = a.isEmpty ? 0 : int.tryParse(a[0]) ?? 0;
592 switch (cmd) {
593 case 'NEW':
594 {
595 reset();
596 emit(['OK']);
597 break;
598 }
599 case 'LOAD':
600 {
601 board = a[0];
602 side = a[1];
603 over = false;
604 history = [];
605 redo = [];
606 emit(['OK']);
607 adjust();
608 break;
609 }
610 case 'DUMP':
611 {
612 emit(['board', board, side]);
613 break;
614 }
615 case 'SHOW':
616 {
617 emit([' a b c d e f g h']);
618 var ms = over ? <int>[] : legal(board, side);
619 for (var r = 0; r < 8; r++)
620 emit([
621 r + 1,
622 for (var c = 0; c < 8; c++)
623 ms.contains(8 * r + c) ? '*' : board[8 * r + c],
624 ]);
625 emit([status()]);
626 break;
627 }
628 case 'SCORE':
629 {
630 emit([
631 'score B',
632 count(board, 'B'),
633 'W',
634 count(board, 'W'),
635 'empty',
636 count(board, '.'),
637 ]);
638 break;
639 }
640 case 'LEGAL':
641 {
642 var ms = over ? <int>[] : legal(board, side);
643 emit(['legal', ms.length, ...ms.map(name)]);
644 break;
645 }
646 case 'MOVE':
647 case 'FLIPS':
648 {
649 var p = square(a[0]);
650 var why = reason(p);
651 if (why.isNotEmpty) {
652 emit(['ILLEGAL', why]);
653 } else if (cmd == 'MOVE') {
654 move(p);
655 } else {
656 var f = flips(board, side, p);
657 emit(['flips', a[0], f.length, ...f.map(name)]);
658 }
659 break;
660 }
661 case 'PLAY':
662 {
663 for (var i = 0; i < a[0].length; i += 2) {
664 var p = square(a[0].substring(i, i + 2));
665 var why = reason(p);
666 if (why.isNotEmpty) {
667 emit(['ILLEGAL move', i ~/ 2 + 1, name(p), why]);
668 break;
669 }
670 move(p);
671 }
672 break;
673 }
674 case 'RECORD':
675 {
676 record(a[0]);
677 break;
678 }
679 case 'UNDO':
680 {
681 if (n > history.length) {
682 emit(['ERROR cannot undo']);
683 } else {
684 var removed = history.sublist(history.length - n);
685 restore(removed[0].before);
686 history.removeRange(history.length - n, history.length);
687 redo = [...removed, ...redo];
688 emit([status()]);
689 }
690 break;
691 }
692 case 'REDO':
693 {
694 if (n > redo.length) {
695 emit(['ERROR cannot redo']);
696 } else {
697 var added = redo.sublist(0, n);
698 redo.removeRange(0, n);
699 history.addAll(added);
700 restore(added.last.after);
701 emit([status()]);
702 }
703 break;
704 }
705 case 'HISTORY':
706 {
707 emit(['history', history.length]);
708 for (var i = 0; i < history.length; i++) {
709 var r = history[i];
710 emit([
711 i + 1,
712 r.side,
713 name(r.square),
714 r.count,
715 if (r.passed) 'pass',
716 ]);
717 }
718 break;
719 }
720 case 'SHEET':
721 {
722 sheet();
723 break;
724 }
725 case 'STATS':
726 {
727 stats();
728 break;
729 }
730 case 'TRANSFORM':
731 {
732 var op = a[0];
733 board = transformBoard(board, op);
734 Move tr(Move r) => Move(
735 r.side,
736 transform(r.square, op),
737 r.count,
738 r.passed,
739 Position(
740 transformBoard(r.before.board, op),
741 r.before.side,
742 r.before.over,
743 ),
744 Position(
745 transformBoard(r.after.board, op),
746 r.after.side,
747 r.after.over,
748 ),
749 );
750 history = history.map(tr).toList();
751 redo = redo.map(tr).toList();
752 emit(['OK']);
753 break;
754 }
755 case 'EVAL':
756 {
757 var (v, w, x, y) = evaluate(board, side);
758 emit(['eval', v, 'weights', w, 'mobility', x, y]);
759 break;
760 }
761 case 'STABLE':
762 {
763 var (b, w) = stable(board);
764 emit(['stable B', b, 'W', w]);
765 break;
766 }
767 case 'HINT':
768 {
769 for (var (v, p) in ranked(board, side, n)) emit([name(p), v]);
770 break;
771 }
772 case 'PV':
773 {
774 var (steps, _) = variation(board, side, n);
775 emit(['pv', search(board, side, n), ...steps]);
776 break;
777 }
778 case 'AI':
779 {
780 var (v, p) = ranked(board, side, n)[0];
781 move(p, false, v);
782 break;
783 }
784 case 'ANALYZE':
785 {
786 var loss = {'B': 0, 'W': 0};
787 for (var i = 0; i < history.length; i++) {
788 var r = history[i];
789 var rows = ranked(history[i].before.board, history[i].side, n);
790 var (best, bp) = rows[0];
791 var value = rows.firstWhere((x) => x.$2 == r.square).$1;
792 loss[r.side] = loss[r.side]! + best - value;
793 emit([
794 i + 1,
795 r.side,
796 name(r.square),
797 value,
798 'best',
799 name(bp),
800 best,
801 ]);
802 }
803 emit(['loss B', loss['B']!, 'W', loss['W']!]);
804 break;
805 }
806 case 'SOLVE':
807 {
808 if (count(board, '.') >= 9) {
809 emit(['ERROR too many empties']);
810 } else {
811 for (var (v, p) in ranked(board, side, -1)) emit([name(p), v]);
812 var (steps, b) = variation(board, side, -1);
813 emit(['line', ...steps]);
814 emit(['final B', count(b, 'B'), 'W', count(b, 'W')]);
815 }
816 break;
817 }
818 case 'AUTOPLAY':
819 {
820 autoplay(n, int.parse(a[1]));
821 break;
822 }
823 case 'RANDOM':
824 {
825 autoplay(0, 0, n);
826 break;
827 }
828 case 'MC':
829 {
830 monteCarlo(n, int.parse(a[1]));
831 break;
832 }
833 case 'PERFT':
834 {
835 emit(['perft', n, perft(board, side, n)]);
836 break;
837 }
838 case 'DIVIDE':
839 {
840 var total = 0;
841 for (var p in legal(board, side)) {
842 var value = perft(place(board, side, p), other(side), n - 1);
843 emit([name(p), value]);
844 total += value;
845 }
846 emit(['total', total]);
847 break;
848 }
849 }
850 }
851}
