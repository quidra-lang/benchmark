50def initial_board():
51 board = ["."] * 64
52 for square, color in (("d4", "W"), ("e5", "W"), ("d5", "B"), ("e4", "B")):
53 board[square_index(square)] = color

57@lru_cache(maxsize=200000)
58def flips(board, color, square):
59 if board[square] != ".":
60 return ()
61 row, column = divmod(square, 8)
62 enemy = other(color)
63 result = []
64 for dr, dc in DIRS:
65 r, c = row + dr, column + dc
66 line = []
67 while on_board(r, c) and board[8 * r + c] == enemy:
68 line.append(8 * r + c)
69 r += dr
70 c += dc
71 if line and on_board(r, c) and board[8 * r + c] == color:
72 result.extend(line)
73 return tuple(sorted(result))

83 chars = list(board)
84 chars[square] = color

94def game_over(board):
95 black, white = board.count("B"), board.count("W")
96 result = "winner B" if black > white else "winner W" if white > black else "draw"
97 return f"game over B {black} W {white} {result}"

100def evaluate(board, side):
101 opponent = other(side)
102 weight = sum(w if board[i] == side else -w if board[i] == opponent else 0
103 for i, w in enumerate(WEIGHTS))
104 own = len(legal(board, side))
105 enemy = len(legal(board, opponent))
106 return weight + 10 * (own - enemy), weight, own, enemy

109@lru_cache(maxsize=500000)
110def search(board, side, depth):
111 moves = legal(board, side)
112 opponent = other(side)
113 if not moves and not legal(board, opponent):
114 return 1000 * (board.count(side) - board.count(opponent))
115 if depth == 0:
116 return evaluate(board, side)[0]
117 if not moves:
118 return -search(board, opponent, depth - 1)
119 return max(-search(place(board, side, move), opponent, depth - 1)
120 for move in moves)

124 opponent = other(side)
125 return sorted(

132@lru_cache(maxsize=500000)
133def exact(board, side):
134 moves, opponent = legal(board, side), other(side)
135 if not moves and not legal(board, opponent):
136 return board.count(side) - board.count(opponent)
137 if not moves:
138 return -exact(board, opponent)
139 return max(-exact(place(board, side, move), opponent) for move in moves)

143 opponent = other(side)
144 return sorted(
145 ((-exact(place(board, side, move), opponent), move)

151def variation(board, side, depth=None):
152 steps = []
153 while not terminal(board) and (depth is None or depth > 0):
154 choices = legal(board, side)
155 if choices:
156 move = (exact_moves(board, side) if depth is None
157 else scored_moves(board, side, depth))[0][1]
158 steps.append(square_name(move))
159 board = place(board, side, move)
160 else:
161 steps.append("pass")
162 side = other(side)
163 if depth is not None:
164 depth -= 1
165 return steps, board

168@lru_cache(maxsize=200000)
169def perft(board, side, depth):
170 if depth == 0:
171 return 1
172 moves, opponent = legal(board, side), other(side)
173 if not moves:
174 return 1 if not legal(board, opponent) else perft(board, opponent, depth - 1)
175 return sum(perft(place(board, side, move), opponent, depth - 1)
176 for move in moves)

179def stable_counts(board):
180 stable = set()
181 while True:
182 new = set()
183 for square, disc in enumerate(board):
184 if disc == "." or square in stable:
185 continue
186 row, column = divmod(square, 8)
187 for axis in AXES:
188 full_line = True
189 anchored = False
190 for dr, dc in axis:
191 r, c = row + dr, column + dc
192 if not on_board(r, c):
193 anchored = True
194 elif 8 * r + c in stable and board[8 * r + c] == disc:
195 anchored = True
196 while on_board(r, c):
197 if board[8 * r + c] == ".":
198 full_line = False
199 r += dr
200 c += dc
201 if not full_line and not anchored:
202 break
203 else:
204 new.add(square)
205 if not new:
206 break
207 stable.update(new)
208 return (sum(board[i] == "B" for i in stable),
209 sum(board[i] == "W" for i in stable))

212def transform_square(square, name):
213 row, col = divmod(square, 8)
214 if name == "rot90":
215 row, col = col, 7 - row
216 elif name == "rot180":
217 row, col = 7 - row, 7 - col
218 elif name == "rot270":
219 row, col = 7 - col, row
220 elif name == "mirror":
221 col = 7 - col
222 elif name == "flip":
223 row = 7 - row
224 elif name == "diag":
225 row, col = col, row
226 else:
227 row, col = 7 - col, 7 - row
228 return row * 8 + col

231def transform_board(board, name):
232 result = ["."] * 64
233 for square, disc in enumerate(board):
234 result[transform_square(square, name)] = disc
235 return "".join(result)

272 while True:
273 moves = legal(board, side)

275 square = moves[random.draw() % len(moves)]

287class Othello:
288 def __init__(self):
289 self.board = initial_board()
290 self.side = "B"
291 self.over = False
292 self.history = []

297 self.board = initial_board()

306 def snapshot(self):

310 self.board, self.side, self.over = position.board, position.side, position.over

312 def status(self):
313 return game_over(self.board) if self.over else self.side + " to move"

315 def adjust(self, quiet=False):
316 if legal(self.board, self.side):
317 return False
318 if legal(self.board, other(self.side)):
319 if not quiet:
320 self.emit(self.side, "passes")
321 self.side = other(self.side)
322 return True
323 self.over = True
324 if not quiet:
325 self.emit(game_over(self.board))
326 return False

328 def move(self, square, quiet=False, value=None):
329 before = self.snapshot()
330 side = self.side
331 count = len(flips(self.board, side, square))
332 self.board = place(self.board, side, square)
333 if not quiet:
334 if value is None:
335 self.emit(side, square_name(square), "flips", count)
336 else:
337 self.emit(side, square_name(square), "flips", count, "value", value)
338 self.side = other(side)
339 passed = self.adjust(quiet)
340 self.history.append(Move(side, square, count, passed, before, self.snapshot()))
341 self.redo.clear()
342 return passed

344 def reason(self, square):
345 if self.over:
346 return "game over"
347 if self.board[square] != ".":
348 return "occupied"
349 if not flips(self.board, self.side, square):
350 return "no flips"
351 return None

353 def read_record(self, text):
354 engine = Othello()
355 positions = [square_index(text[i:i + 2]) for i in range(0, len(text), 2)]
356 passes = 0
357 for number, square in enumerate(positions, 1):
358 reason = engine.reason(square)
359 if reason:
360 self.emit("ILLEGAL", "move", number, square_name(square), reason)
361 return
362 passes += engine.move(square, quiet=True)
363 self.board, self.side, self.over = engine.board, engine.side, engine.over
364 self.history, self.redo = engine.history, []
365 first = square_name(positions[0])
366 operation = {"f5": None, "e6": "diag", "d3": "anti", "c4": "rot180"}[first]
367 normalized = "".join(
368 square_name(transform_square(square, operation)) if operation
369 else square_name(square) for square in positions
370 )
371 self.emit("record", len(positions), "moves", passes, "passes")
372 self.emit("normal", normalized)
373 self.emit(self.status())

384 def sheet(self):
385 board = self.history[0].before.board if self.history else self.board
386 fields = list(board)

388 fields[record.square] = str(number)

393 def stats(self):
394 if self.history:
395 total = sum(row.count for row in self.history)
396 passes = sum(row.passed for row in self.history)
397 self.emit("moves", len(self.history), "flips", total,
398 "avg", average(total, len(self.history)), "passes", passes)
399 else:
400 self.emit("moves", 0)
401 for side in "BW":
402 rows = [row for row in self.history if row.side == side]
403 if not rows:
404 self.emit(side, "moves", 0)
405 continue
406 flipped = sum(row.count for row in rows)
407 maximum = max(row.count for row in rows)
408 first = next(row for row in rows if row.count == maximum)
409 corners = sum(row.square in (0, 7, 56, 63) for row in rows)
410 self.emit(side, "moves", len(rows), "flips", flipped,
411 "avg", average(flipped, len(rows)), "max", maximum,
412 square_name(first.square), "corners", corners)

414 def analyze(self, depth):
415 losses = {"B": 0, "W": 0}
416 for index, record in enumerate(self.history, 1):
417 rankings = scored_moves(record.before.board, record.side, depth)
418 best, best_square = rankings[0]
419 value = next(score for score, square in rankings if square == record.square)
420 losses[record.side] += best - value
421 self.emit(index, record.side, square_name(record.square), value,
422 "best", square_name(best_square), best)
423 self.emit("loss", "B", losses["B"], "W", losses["W"])

425 def auto(self, black, white, seed=None):
426 generator = Random(seed) if seed is not None else None
427 names, passes = [], 0
428 while not self.over:
429 if generator is not None:
430 choices = legal(self.board, self.side)
431 square = choices[generator.draw() % len(choices)]
432 else:
433 depth = black if self.side == "B" else white
434 square = scored_moves(self.board, self.side, depth)[0][1]
435 names.append(square_name(square))
436 passes += self.move(square, quiet=True)
437 self.emit("random" if generator else "autoplay", len(names), "moves", passes, "passes")
438 self.emit("moves", "".join(names))
439 self.emit(game_over(self.board))

441 def monte_carlo(self, samples, seed):
442 generator = Random(seed)
443 opponent = other(self.side)
444 results = []
445 for move in legal(self.board, self.side):
446 start = place(self.board, self.side, move)
447 wins = draws = losses = 0
448 for _ in range(samples):
449 final = random_game(start, opponent, generator)
450 score = final.count(self.side) - final.count(opponent)
451 if score > 0:
452 wins += 1
453 elif score < 0:
454 losses += 1
455 else:
456 draws += 1
457 self.emit(square_name(move), wins, draws, losses)
458 results.append((2 * wins + draws, move))
459 best, square = min(results, key=lambda row: (-row[0], row[1]))
460 self.emit("best", square_name(square), best)

462 def valid_command(self, command, a):
463 if command == "NEW":
464 self.reset()
465 self.emit("OK")
466 elif command == "LOAD":
467 self.board, self.side, self.over = a[0], a[1], False
468 self.history, self.redo = [], []
469 self.emit("OK")
470 self.adjust()
471 elif command == "DUMP":
472 self.emit("board", self.board, self.side)
473 elif command == "SHOW":
474 self.print_board()
475 elif command == "SCORE":
476 self.emit("score", "B", self.board.count("B"), "W", self.board.count("W"),
477 "empty", self.board.count("."))
478 elif command == "LEGAL":
479 moves = legal(self.board, self.side) if not self.over else ()
480 self.emit("legal", len(moves), *(square_name(move) for move in moves))
481 elif command in SQUARE_ARGS:
482 square = square_index(a[0])
483 reason = self.reason(square)
484 if reason:
485 self.emit("ILLEGAL", reason)
486 elif command == "FLIPS":
487 changes = flips(self.board, self.side, square)
488 self.emit("flips", a[0], len(changes),
489 *(square_name(s) for s in changes))
490 else:
491 self.move(square)
492 elif command == "PLAY":
493 text = a[0]
494 for number, offset in enumerate(range(0, len(text), 2), 1):
495 square = square_index(text[offset:offset + 2])
496 reason = self.reason(square)
497 if reason:
498 self.emit("ILLEGAL", "move", number, square_name(square), reason)
499 break
500 self.move(square)
501 elif command == "RECORD":
502 self.read_record(a[0])
503 elif command == "UNDO":
504 count = int(a[0])
505 if count > len(self.history):
506 self.emit("ERROR", "cannot undo")
507 else:
508 removed = self.history[-count:]
509 self.restore(removed[0].before)
510 del self.history[-count:]
511 self.redo = removed + self.redo
512 self.emit(self.status())
513 elif command == "REDO":
514 count = int(a[0])
515 if count > len(self.redo):
516 self.emit("ERROR", "cannot redo")
517 else:
518 added = self.redo[:count]
519 del self.redo[:count]
520 self.history.extend(added)
521 self.restore(added[-1].after)
522 self.emit(self.status())
523 elif command == "HISTORY":
524 self.emit("history", len(self.history))
525 for number, record in enumerate(self.history, 1):
526 parts = [number, record.side, square_name(record.square), record.count]
527 if record.passed:
528 parts.append("pass")
529 self.emit(*parts)
530 elif command == "SHEET":
531 self.sheet()
532 elif command == "STATS":
533 self.stats()
534 elif command == "TRANSFORM":
535 self.board = transform_board(self.board, a[0])
536 self.history = [transform_move(row, a[0]) for row in self.history]
537 self.redo = [transform_move(row, a[0]) for row in self.redo]
538 self.emit("OK")
539 elif command == "EVAL":
540 value, weights, own, enemy = evaluate(self.board, self.side)
541 self.emit("eval", value, "weights", weights, "mobility", own, enemy)
542 elif command == "STABLE":
543 black, white = stable_counts(self.board)
544 self.emit("stable", "B", black, "W", white)
545 elif command == "HINT":
546 for value, square in scored_moves(self.board, self.side, int(a[0])):
547 self.emit(square_name(square), value)
548 elif command == "PV":
549 depth = int(a[0])
550 value = search(self.board, self.side, depth)
551 steps, _ = variation(self.board, self.side, depth)
552 self.emit("pv", value, *steps)
553 elif command == "AI":
554 value, square = scored_moves(self.board, self.side, int(a[0]))[0]
555 self.move(square, value=value)
556 elif command == "ANALYZE":
557 self.analyze(int(a[0]))
558 elif command == "SOLVE":
559 if self.board.count(".") >= 9:
560 self.emit("ERROR", "too many empties")
561 else:
562 for value, square in exact_moves(self.board, self.side):
563 self.emit(square_name(square), value)
564 steps, final = variation(self.board, self.side)
565 self.emit("line", *steps)
566 self.emit("final", "B", final.count("B"), "W", final.count("W"))
567 elif command == "AUTOPLAY":
568 self.auto(int(a[0]), int(a[1]))
569 elif command == "RANDOM":
570 self.auto(0, 0, seed=int(a[0]))
571 elif command == "MC":
572 self.monte_carlo(int(a[0]), int(a[1]))
573 elif command == "PERFT":
574 depth = int(a[0])
575 self.emit("perft", depth, perft(self.board, self.side, depth))
576 elif command == "DIVIDE":
577 depth = int(a[0])
578 total = 0
579 for square in legal(self.board, self.side):
580 count = perft(place(self.board, self.side, square),
581 other(self.side), depth - 1)
582 self.emit(square_name(square), count)
583 total += count
584 self.emit("total", total)

586 def execute(self, text):
587 command, *args = text.split(" ")
588 if command not in COMMANDS:
589 self.emit("ERROR", "unknown command")
590 return
591 if command == "LOAD":
592 valid = len(args) == 2 and len(args[0]) == 64 and \
593 set(args[0]) <= set(".BW") and args[1] in ("B", "W")
594 if not valid:
595 self.emit("ERROR", "bad board")
596 return
597 elif command in SQUARE_ARGS:
598 if len(args) != 1:
599 self.emit("ERROR", "bad arguments")
600 return
601 if re.fullmatch(r"[a-h][1-8]", args[0]) is None:
602 self.emit("ERROR", "bad square")
603 return
604 elif command in RECORD_ARGS:
605 if len(args) != 1:
606 self.emit("ERROR", "bad arguments")
607 return
608 record = args[0]
609 valid = 2 <= len(record) <= 120 and len(record) % 2 == 0 and \
610 all(re.fullmatch(r"[a-h][1-8]", record[i:i + 2])
611 for i in range(0, len(record), 2))
612 if not valid:
613 self.emit("ERROR", "bad record")
614 return
615 elif command in ZERO_ARGS:
616 if args:
617 self.emit("ERROR", "bad arguments")
618 return
619 elif command == "TRANSFORM":
620 if len(args) != 1 or args[0] not in TRANSFORMS:
621 self.emit("ERROR", "bad arguments")
622 return
623 else:
624 expected = 2 if command in ("AUTOPLAY", "MC") else 1
625 if len(args) != expected or not all(re.fullmatch(r"[0-9]+", x) for x in args):
626 self.emit("ERROR", "bad arguments")
627 return
628 numbers = list(map(int, args))
629 if command == "AUTOPLAY":
630 valid = all(1 <= n <= 4 for n in numbers)
631 elif command in ("HINT", "PV", "AI", "ANALYZE"):
632 valid = 1 <= numbers[0] <= 4
633 elif command in ("PERFT", "DIVIDE"):
634 valid = 1 <= numbers[0] <= 8
635 elif command == "MC":
636 valid = 1 <= numbers[0] <= 100 and 1 <= numbers[1] <= 2147483646
637 elif command == "RANDOM":
638 valid = 1 <= numbers[0] <= 2147483646
639 else:
640 valid = numbers[0] >= 1
641 if not valid:
642 self.emit("ERROR", "bad arguments")
643 return
644 if command in ACTIVE_ONLY and self.over:
645 self.emit("ERROR", "game over")
646 return
647 self.valid_command(command, args)
