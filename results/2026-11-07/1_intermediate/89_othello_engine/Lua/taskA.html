9local transforms=set('rot90 rot180 rot270 mirror flip diag anti')
10local zero=set('NEW DUMP SHOW SCORE LEGAL HISTORY SHEET STATS EVAL STABLE SOLVE')
11local numeric=set('UNDO REDO HINT PV AI ANALYZE RANDOM PERFT DIVIDE')
12local active=set('FLIPS MOVE PLAY HINT PV AI SOLVE AUTOPLAY RANDOM MC DIVIDE')
13local extra=set('LOAD TRANSFORM AUTOPLAY MC FLIPS MOVE PLAY RECORD')
14local function other(s)
15 return s=='B' and 'W' or 'B'
16end
17local function name(p)
18 return string.char(97+p%8)..(p//8+1)
19end
20local function square(s)
21 return (tonumber(s:sub(2,2))-1)*8+s:byte(1)-97
22end
23local function inside(r,c)
24 return r>=0 and r<8 and c>=0 and c<8
25end
26local function initial()
27 local b={};
28 for i=1,64 do
29 b[i]='.'
30 end;
31 b[28]='W';
32 b[37]='W';
33 b[29]='B';
34 b[36]='B';
35 return table.concat(b)
36end
37local function disc(b,p)
38 return b:sub(p+1,p+1)
39end
40local function flips(b,s,p)
41 if disc(b,p)~='.' then
42 return {}
43 end;
44 local out,row,col={},p//8,p%8
45 for dr=-1,1 do
46 for dc=-1,1 do
47 if dr~=0 or dc~=0 then
48 local r,c=row+dr,col+dc;
49 local line={};
50 while inside(r,c) and disc(b,8*r+c)==other(s) do
51 line[#line+1]=8*r+c;
52 r=r+dr;
53 c=c+dc
54 end;
55 if #line>0 and inside(r,c) and disc(b,8*r+c)==s then
56 for _,q in ipairs(line) do
57 out[#out+1]=q
58 end
59 end
60 end
61 end
62 end;
63 table.sort(out);
64 return out
65end
66local function legal(b,s)
67 local ms={};
68 for p=0,63 do
69 if disc(b,p)=='.' and #flips(b,s,p)>0 then
70 ms[#ms+1]=p
71 end
72 end;
73 return ms
74end
75local function place(b,s,p)
76 local a={};
77 for i=1,64 do
78 a[i]=b:sub(i,i)
79 end;
80 for _,q in ipairs(flips(b,s,p)) do
81 a[q+1]=s
82 end;
83 a[p+1]=s;
84 return table.concat(a)
85end
86local function count(b,s)
87 local n=0;
88 for i=1,64 do
89 if b:sub(i,i)==s then
90 n=n+1
91 end
92 end;
93 return n
94end
95local function terminal(b)
96 return #legal(b,'B')==0 and #legal(b,'W')==0
97end
98local function gameOver(b)
99 local x,y=count(b,'B'),count(b,'W');
100 return 'game over B '..x..' W '..y..' '..(x>y and 'winner B' or x<y and 'winner W' or 'draw')
101end
102local function evaluate(b,s)
103 local w=0;
104 for i=1,64 do
105 local x=b:sub(i,i);
106 w=w+(x==s and weights[i] or x==other(s) and -weights[i] or 0)
107 end;
108 local own,enemy=#legal(b,s),#legal(b,other(s));
109 return w+10*(own-enemy),w,own,enemy
110end
111local searchCache,searchSize={},0
112local function search(b,s,d)
113 local key=b..s..':'..d;
114 if searchCache[key] then
115 return searchCache[key]
116 end;
117 local ms,o=legal(b,s),other(s);
118 local v
119 if #ms==0 and #legal(b,o)==0 then
120 v=(d<0 and 1 or 1000)*(count(b,s)-count(b,o))
121 elseif d==0 then
122 v=evaluate(b,s)
123 elseif #ms==0 then
124 v=-search(b,o,d<0 and -1 or d-1)
125 else
126 v=-1000000;
127 for _,p in ipairs(ms) do
128 v=math.max(v,-search(place(b,s,p),o,d<0 and -1 or d-1))
129 end
130 end
131 if searchSize>300000 then
132 searchCache={};
133 searchSize=0
134 end;
135 searchCache[key]=v;
136 searchSize=searchSize+1;
137 return v
138end
139local function ranked(b,s,d)
140 local rows={};
141 for _,p in ipairs(legal(b,s)) do
142 rows[#rows+1]={-search(place(b,s,p),other(s),d<0 and -1 or d-1),p}
143 end;
144 table.sort(rows,function(a,b)
145 return a[1]>b[1] or (a[1]==b[1] and a[2]<b[2])
146 end);
147 return rows
148end
149local function variation(b,s,d)
150 local steps={};
151 while d~=0 and not terminal(b) do
152 local ms=ranked(b,s,d);
153 if #ms>0 then
154 local p=ms[1][2];
155 steps[#steps+1]=name(p);
156 b=place(b,s,p)
157 else
158 steps[#steps+1]='pass'
159 end;
160 s=other(s);
161 if d>0 then
162 d=d-1
163 end
164 end;
165 return steps,b
166end
167local perftCache,perftSize={},0
168local function perft(b,s,d)
169 if d==0 then
170 return 1
171 end;
172 local key=b..s..d;
173 if perftCache[key] then
174 return perftCache[key]
175 end;
176 local ms,o=legal(b,s),other(s);
177 local n=0;
178 if #ms==0 then
179 n=#legal(b,o)==0 and 1 or perft(b,o,d-1)
180 else
181 for _,p in ipairs(ms) do
182 n=n+perft(place(b,s,p),o,d-1)
183 end
184 end;
185 if perftSize>200000 then
186 perftCache={};
187 perftSize=0
188 end;
189 perftCache[key]=n;
190 perftSize=perftSize+1;
191 return n
192end
193local function stable(b)
194 local stable={};
195 local axes={{0,1},{1,0},{1,1},{1,-1}}
196 while true do
197 local found={};
198 for p=0,63 do
199 if disc(b,p)~='.' and not stable[p] then
200 local row,col=p//8,p%8;
201 local ok=true;
202 for _,axis in ipairs(axes) do
203 local dr,dc=axis[1],axis[2];
204 local full,anchor=true,false;
205 for _,sign in ipairs({-1,1}) do
206 local r,c=row+dr*sign,col+dc*sign;
207 if not inside(r,c) or (stable[8*r+c] and disc(b,8*r+c)==disc(b,p)) then
208 anchor=true
209 end;
210 while inside(r,c) do
211 if disc(b,8*r+c)=='.' then
212 full=false
213 end;
214 r=r+dr*sign;
215 c=c+dc*sign
216 end
217 end;
218 if not full and not anchor then
219 ok=false;
220 break
221 end
222 end;
223 if ok then
224 found[#found+1]=p
225 end
226 end
227 end;
228 if #found==0 then
229 break
230 end;
231 for _,p in ipairs(found) do
232 stable[p]=true
233 end
234 end
235 local black,white=0,0;
236 for p in pairs(stable) do
237 if disc(b,p)=='B' then
238 black=black+1
239 else
240 white=white+1
241 end
242 end;
243 return black,white
244end
245local function transform(p,op)
246 local r,c=p//8,p%8;
247 if op=='rot90' then
248 return 8*c+7-r
249 elseif op=='rot180' then
250 return 63-p
251 elseif op=='rot270' then
252 return 8*(7-c)+r
253 elseif op=='mirror' then
254 return 8*r+7-c
255 elseif op=='flip' then
256 return 8*(7-r)+c
257 elseif op=='diag' then
258 return 8*c+r
259 else
260 return 8*(7-c)+7-r
261 end
262end
263local function transformBoard(b,op)
264 local a={};
265 for p=0,63 do
266 a[transform(p,op)+1]=disc(b,p)
267 end;
268 return table.concat(a)
269end
270local function random(seed)
271 return {value=seed,draw=function(self)
272 self.value=self.value*48271%2147483647;
273 return self.value
274 end
275 }
276end
277local function randomGame(b,s,rng)
278 while true do
279 local ms=legal(b,s);
280 if #ms>0 then
281 b=place(b,s,ms[rng:draw()%#ms+1])
282 elseif #legal(b,other(s))==0 then
283 return b
284 end;
285 s=other(s)
286 end
287end

291end

298end

306function Engine:reset()
307 self.board=initial();
308 self.side='B';
309 self.over=false;
310 self.history={};
311 self.redo={}
312end
313function Engine:snapshot()
314 return {board=self.board,side=self.side,over=self.over}
315end
316function Engine:restore(p)
317 self.board=p.board;
318 self.side=p.side;
319 self.over=p.over
320end
321function Engine:status()
322 return self.over and gameOver(self.board) or self.side..' to move'
323end
324function Engine:adjust(quiet)
325 if #legal(self.board,self.side)>0 then
326 return false
327 end;
328 if #legal(self.board,other(self.side))>0 then
329 if not quiet then
330 emit(self.side,'passes')
331 end;
332 self.side=other(self.side);
333 return true
334 end;
335 self.over=true;
336 if not quiet then
337 emit(gameOver(self.board))
338 end;
339 return false
340end
341function Engine:move(p,quiet,value)
342 local before,s,k=self:snapshot(),self.side,#flips(self.board,self.side,p);
343 self.board=place(self.board,s,p);
344 if not quiet then
345 if value then
346 emit(s,name(p),'flips',k,'value',value)
347 else
348 emit(s,name(p),'flips',k)
349 end
350 end;
351 self.side=other(s);
352 local passed=self:adjust(quiet);
353 self.history[#self.history+1]={side=s,square=p,count=k,passed=passed,before=before,after=self:snapshot()};
354 self.redo={};
355 return passed
356end
357function Engine:reason(p)
358 if self.over then
359 return 'game over'
360 elseif disc(self.board,p)~='.' then
361 return 'occupied'
362 elseif #flips(self.board,self.side,p)==0 then
363 return 'no flips'
364 end;
365 return nil
366end
367function Engine:record(text)
368 local e,ps,passes=Engine.new(),{},0;
369 for i=1,#text,2 do
370 local s=text:sub(i,i+1);
371 local p=square(s);
372 local why=e:reason(p);
373 if why then
374 emit('ILLEGAL move',(i+1)//2,s,why);
375 return
376 end;
377 ps[#ps+1]=p;
378 if e:move(p,true) then
379 passes=passes+1
380 end
381 end;
382 self:restore(e:snapshot());
383 self.history=e.history;
384 self.redo={};
385 local op=({f5='',e6='diag',d3='anti',c4='rot180'})[name(ps[1])];
386 local normal={};
387 for _,p in ipairs(ps) do
388 normal[#normal+1]=name(op=='' and p or transform(p,op))
389 end;
390 emit('record',#ps,'moves',passes,'passes');
391 emit('normal',table.concat(normal));
392 emit(self:status())
393end
394function Engine:sheet()
395 local b=#self.history>0 and self.history[1].before.board or self.board;
396 local fields={};
397 for i=1,64 do
398 fields[i]=b:sub(i,i)
399 end;
400 for i,r in ipairs(self.history) do
401 fields[r.square+1]=tostring(i)
402 end;
403 emit(' a b c d e f g h');
404 for r=0,7 do
405 local row=tostring(r+1);
406 for c=0,7 do
407 row=row..string.format('%3s',fields[8*r+c+1])
408 end;
409 emit(row)
410 end
411end
412function Engine:stats()
413 local n,total,passes=#self.history,0,0;
414 for _,r in ipairs(self.history) do
415 total=total+r.count;
416 if r.passed then
417 passes=passes+1
418 end
419 end;
420 if n==0 then
421 emit('moves 0')
422 else
423 emit('moves',n,'flips',total,'avg',average(total,n),'passes',passes)
424 end;
425 for _,s in ipairs({'B','W'}) do
426 local k,sum,max,p,corners=0,0,-1,nil,0;
427 for _,r in ipairs(self.history) do
428 if r.side==s then
429 k=k+1;
430 sum=sum+r.count;
431 if r.count>max then
432 max=r.count;
433 p=r.square
434 end;
435 if r.square==0 or r.square==7 or r.square==56 or r.square==63 then
436 corners=corners+1
437 end
438 end
439 end;
440 if k==0 then
441 emit(s,'moves 0')
442 else
443 emit(s,'moves',k,'flips',sum,'avg',average(sum,k),'max',max,name(p),'corners',corners)
444 end
445 end
446end
447function Engine:auto(black,white,seed)
448 local rng=seed and random(seed);
449 local ps,passes={},0;
450 while not self.over do
451 local ms=legal(self.board,self.side);
452 local p=rng and ms[rng:draw()%#ms+1] or ranked(self.board,self.side,self.side=='B' and black or white)[1][2];
453 ps[#ps+1]=name(p);
454 if self:move(p,true) then
455 passes=passes+1
456 end
457 end;
458 emit(rng and 'random' or 'autoplay',#ps,'moves',passes,'passes');
459 emit('moves',table.concat(ps));
460 emit(gameOver(self.board))
461end
462function Engine:monteCarlo(k,seed)
463 local rng=random(seed);
464 local best,bp=-1,nil;
465 for _,p in ipairs(legal(self.board,self.side)) do
466 local w,d,l=0,0,0;
467 local start=place(self.board,self.side,p);
468 for _=1,k do
469 local b=randomGame(start,other(self.side),rng);
470 local diff=count(b,self.side)-count(b,other(self.side));
471 if diff>0 then
472 w=w+1
473 elseif diff<0 then
474 l=l+1
475 else
476 d=d+1
477 end
478 end;
479 emit(name(p),w,d,l);
480 if 2*w+d>best then
481 best=2*w+d;
482 bp=p
483 end
484 end;
485 emit('best',name(bp),best)
486end
487function Engine:execute(text)
488 local a={};
489 for s in text:gmatch('%S+') do
490 a[#a+1]=s
491 end;
492 local cmd=table.remove(a,1);
493 local err
494 if not zero[cmd] and not numeric[cmd] and not extra[cmd] then
495 err='unknown command'
496 elseif cmd=='LOAD' then
497 if #a~=2 or #a[1]~=64 or a[1]:find('[^.BW]') or (a[2]~='B' and a[2]~='W') then
498 err='bad board'
499 end
500 elseif cmd=='MOVE' or cmd=='FLIPS' then
501 if #a~=1 then
502 err='bad arguments'
503 elseif not a[1]:match('^[a-h][1-8]$') then
504 err='bad square'
505 end
506 elseif cmd=='PLAY' or cmd=='RECORD' then
507 if #a~=1 then
508 err='bad arguments'
509 else
510 local t=a[1];
511 local valid=#t>=2 and #t<=120 and #t%2==0;
512 for i=1,#t,2 do
513 if not t:sub(i,i+1):match('^[a-h][1-8]$') then
514 valid=false
515 end
516 end;
517 if not valid then
518 err='bad record'
519 end
520 end
521 elseif zero[cmd] then
522 if #a~=0 then
523 err='bad arguments'
524 end
525 elseif cmd=='TRANSFORM' then
526 if #a~=1 or not transforms[a[1]] then
527 err='bad arguments'
528 end
529 else
530 local expected=(cmd=='AUTOPLAY' or cmd=='MC') and 2 or 1;
531 local valid=#a==expected;
532 local ns={};
533 for i,s in ipairs(a) do
534 if not s:match('^%d+$') then
535 valid=false
536 else
537 ns[i]=tonumber(s)
538 end
539 end;
540 if valid then
541 local n=ns[1];
542 if cmd=='AUTOPLAY' then
543 valid=n>=1 and n<=4 and ns[2]>=1 and ns[2]<=4
544 elseif cmd=='HINT' or cmd=='PV' or cmd=='AI' or cmd=='ANALYZE' then
545 valid=n>=1 and n<=4
546 elseif cmd=='PERFT' or cmd=='DIVIDE' then
547 valid=n>=1 and n<=8
548 elseif cmd=='MC' then
549 valid=n>=1 and n<=100 and ns[2]>=1 and ns[2]<=2147483646
550 elseif cmd=='RANDOM' then
551 valid=n>=1 and n<=2147483646
552 else
553 valid=n>=1
554 end
555 end;
556 if not valid then
557 err='bad arguments'
558 end
559 end
560 if err then
561 emit('ERROR',err);
562 return
563 end;
564 if self.over and active[cmd] then
565 emit('ERROR game over');
566 return
567 end;
568 local n=tonumber(a[1])
569 if cmd=='NEW' then
570 self:reset();
571 emit('OK')
572 elseif cmd=='LOAD' then
573 self.board=a[1];
574 self.side=a[2];
575 self.over=false;
576 self.history={};
577 self.redo={};
578 emit('OK');
579 self:adjust()
580 elseif cmd=='DUMP' then
581 emit('board',self.board,self.side)
582 elseif cmd=='SHOW' then
583 emit(' a b c d e f g h');
584 local ms={};
585 if not self.over then
586 for _,p in ipairs(legal(self.board,self.side)) do
587 ms[p]=true
588 end
589 end;
590 for r=0,7 do
591 local row={r+1};
592 for c=0,7 do
593 row[#row+1]=ms[8*r+c] and '*' or disc(self.board,8*r+c)
594 end;
595 emit(table.unpack(row))
596 end;
597 emit(self:status())
598 elseif cmd=='SCORE' then
599 emit('score B',count(self.board,'B'),'W',count(self.board,'W'),'empty',count(self.board,'.'))
600 elseif cmd=='LEGAL' then
601 local ms=self.over and {} or legal(self.board,self.side);
602 local out={'legal',#ms};
603 for _,p in ipairs(ms) do
604 out[#out+1]=name(p)
605 end;
606 emit(table.unpack(out))
607 elseif cmd=='MOVE' or cmd=='FLIPS' then
608 local p=square(a[1]);
609 local why=self:reason(p);
610 if why then
611 emit('ILLEGAL',why)
612 elseif cmd=='MOVE' then
613 self:move(p)
614 else
615 local f=flips(self.board,self.side,p);
616 local out={'flips',a[1],#f};
617 for _,q in ipairs(f) do
618 out[#out+1]=name(q)
619 end;
620 emit(table.unpack(out))
621 end
622 elseif cmd=='PLAY' then
623 for i=1,#a[1],2 do
624 local p=square(a[1]:sub(i,i+1));
625 local why=self:reason(p);
626 if why then
627 emit('ILLEGAL move',(i+1)//2,name(p),why);
628 break
629 end;
630 self:move(p)
631 end
632 elseif cmd=='RECORD' then
633 self:record(a[1])
634 elseif cmd=='UNDO' then
635 if n>#self.history then
636 emit('ERROR cannot undo')
637 else
638 local start=#self.history-n+1;
639 self:restore(self.history[start].before);
640 local red={};
641 for i=start,#self.history do
642 red[#red+1]=self.history[i]
643 end;
644 for _,r in ipairs(self.redo) do
645 red[#red+1]=r
646 end;
647 for i=#self.history,start,-1 do
648 self.history[i]=nil
649 end;
650 self.redo=red;
651 emit(self:status())
652 end
653 elseif cmd=='REDO' then
654 if n>#self.redo then
655 emit('ERROR cannot redo')
656 else
657 for _=1,n do
658 self.history[#self.history+1]=table.remove(self.redo,1)
659 end;
660 self:restore(self.history[#self.history].after);
661 emit(self:status())
662 end
663 elseif cmd=='HISTORY' then
664 emit('history',#self.history);
665 for i,r in ipairs(self.history) do
666 if r.passed then
667 emit(i,r.side,name(r.square),r.count,'pass')
668 else
669 emit(i,r.side,name(r.square),r.count)
670 end
671 end
672 elseif cmd=='SHEET' then
673 self:sheet()
674 elseif cmd=='STATS' then
675 self:stats()
676 elseif cmd=='TRANSFORM' then
677 local op=a[1];
678 self.board=transformBoard(self.board,op);
679 local function tr(p)
680 return {board=transformBoard(p.board,op),side=p.side,over=p.over}
681 end;
682 for _,rows in ipairs({self.history,self.redo}) do
683 for _,r in ipairs(rows) do
684 r.square=transform(r.square,op);
685 r.before=tr(r.before);
686 r.after=tr(r.after)
687 end
688 end;
689 emit('OK')
690 elseif cmd=='EVAL' then
691 local v,w,x,y=evaluate(self.board,self.side);
692 emit('eval',v,'weights',w,'mobility',x,y)
693 elseif cmd=='STABLE' then
694 local b,w=stable(self.board);
695 emit('stable B',b,'W',w)
696 elseif cmd=='HINT' then
697 for _,row in ipairs(ranked(self.board,self.side,n)) do
698 emit(name(row[2]),row[1])
699 end
700 elseif cmd=='PV' then
701 local steps=variation(self.board,self.side,n);
702 local out={'pv',search(self.board,self.side,n)};
703 for _,s in ipairs(steps) do
704 out[#out+1]=s
705 end;
706 emit(table.unpack(out))
707 elseif cmd=='AI' then
708 local row=ranked(self.board,self.side,n)[1];
709 self:move(row[2],false,row[1])
710 elseif cmd=='ANALYZE' then
711 local loss={B=0,W=0};
712 for i,r in ipairs(self.history) do
713 local rows=ranked(r.before.board,r.side,n);
714 local best,bp=rows[1][1],rows[1][2];
715 local value;
716 for _,row in ipairs(rows) do
717 if row[2]==r.square then
718 value=row[1];
719 break
720 end
721 end;
722 loss[r.side]=loss[r.side]+best-value;
723 emit(i,r.side,name(r.square),value,'best',name(bp),best)
724 end;
725 emit('loss B',loss.B,'W',loss.W)
726 elseif cmd=='SOLVE' then
727 if count(self.board,'.')>=9 then
728 emit('ERROR too many empties')
729 else
730 for _,row in ipairs(ranked(self.board,self.side,-1)) do
731 emit(name(row[2]),row[1])
732 end;
733 local steps,b=variation(self.board,self.side,-1);
734 local out={'line'};
735 for _,s in ipairs(steps) do
736 out[#out+1]=s
737 end;
738 emit(table.unpack(out));
