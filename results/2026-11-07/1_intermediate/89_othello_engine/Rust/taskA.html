12fn opponent(side: u8) -> u8 {
13 if side == b'B' { b'W' } else { b'B' }

16fn initial() -> Board {
17 let mut board = [b'.'; 64];

19 board[36] = b'W';
20 board[28] = b'B';
21 board[35] = b'B';
22 board

25fn name(p: usize) -> String {
26 format!("{}{}", (b'a' + (p % 8) as u8) as char, p / 8 + 1)
27}

29fn square(s: &str) -> usize {
30 (s.as_bytes()[1] - b'1') as usize * 8 + (s.as_bytes()[0] - b'a') as usize
31}

33fn inside(row: i32, col: i32) -> bool {
34 (0..8).contains(&row) && (0..8).contains(&col)
35}

37fn flips(board: &Board, side: u8, p: usize) -> Vec<usize> {
38 let mut result = Vec::new();
39 if board[p] != b'.' {
40 return result;
41 }
42 let row = (p / 8) as i32;
43 let col = (p % 8) as i32;
44 for dr in -1..=1 {
45 for dc in -1..=1 {
46 if dr == 0 && dc == 0 {
47 continue;
48 }
49 let mut r = row + dr;
50 let mut c = col + dc;
51 let mut path = Vec::new();
52 while inside(r, c) && board[(r * 8 + c) as usize] == opponent(side) {
53 path.push((r * 8 + c) as usize);
54 r += dr;
55 c += dc;
56 }
57 if !path.is_empty() && inside(r, c) && board[(r * 8 + c) as usize] == side {
58 result.extend(path);
59 }
60 }
61 }
62 result.sort_unstable();
63 result
64}

66fn legal(board: &Board, side: u8) -> Vec<usize> {
67 (0..64)
68 .filter(|&p| board[p] == b'.' && !flips(board, side, p).is_empty())
69 .collect()
70}

72fn place(board: &Board, side: u8, p: usize) -> Board {
73 let mut next = *board;
74 for q in flips(board, side, p) {
75 next[q] = side;
76 }
77 next[p] = side;
78 next
79}

81fn count(board: &Board, piece: u8) -> usize {
82 board.iter().filter(|&&x| x == piece).count()

85fn terminal(board: &Board) -> bool {

89fn game_over(board: &Board) -> String {
90 let black = count(board, b'B');
91 let white = count(board, b'W');
92 let winner = if black > white {
93 "winner B"
94 } else if black < white {
95 "winner W"
96 } else {
97 "draw"
98 };
99 format!("game over B {black} W {white} {winner}")
100}

102fn evaluate(board: &Board, side: u8) -> [i32; 4] {
103 let mut weights = 0;
104 for p in 0..64 {
105 if board[p] == side {
106 weights += WEIGHTS[p];
107 } else if board[p] == opponent(side) {
108 weights -= WEIGHTS[p];
109 }
110 }
111 let own = legal(board, side).len() as i32;
112 let other = legal(board, opponent(side)).len() as i32;
113 [weights + 10 * (own - other), weights, own, other]
114}

116#[derive(Default)]
117struct Solver {
118 search_cache: HashMap<(Board, u8, i32), i32>,
119 perft_cache: HashMap<(Board, u8, u32), u64>,
120}
121
122impl Solver {
123 fn search(&mut self, board: &Board, side: u8, depth: i32) -> i32 {
124 let key = (*board, side, depth);
125 if let Some(&value) = self.search_cache.get(&key) {
126 return value;
127 }
128 let moves = legal(board, side);
129 let other = opponent(side);
130 let value = if moves.is_empty() && legal(board, other).is_empty() {
131 let factor = if depth < 0 { 1 } else { 1000 };
132 factor * (count(board, side) as i32 - count(board, other) as i32)
133 } else if depth == 0 {
134 evaluate(board, side)[0]
135 } else if moves.is_empty() {
136 -self.search(board, other, if depth < 0 { -1 } else { depth - 1 })
137 } else {
138 moves
139 .into_iter()
140 .map(|p| {
141 -self.search(
142 &place(board, side, p),
143 other,
144 if depth < 0 { -1 } else { depth - 1 },
145 )
146 })
147 .max()
148 .unwrap()
149 };
150 if self.search_cache.len() > 300_000 {
151 self.search_cache.clear();
152 }
153 self.search_cache.insert(key, value);
154 value
155 }
156
157 fn ranked(&mut self, board: &Board, side: u8, depth: i32) -> Vec<(i32, usize)> {
158 let mut results: Vec<_> = legal(board, side)
159 .into_iter()
160 .map(|p| {
161 (
162 -self.search(
163 &place(board, side, p),
164 opponent(side),
165 if depth < 0 { -1 } else { depth - 1 },
166 ),
167 p,
168 )
169 })
170 .collect();
171 results.sort_by_key(|&(value, p)| (-value, p));
172 results
173 }

175 fn variation(
176 &mut self,
177 mut board: Board,
178 mut side: u8,
179 mut depth: i32,
180 ) -> (Vec<String>, Board) {
181 let mut steps = Vec::new();
182 while depth != 0 && !terminal(&board) {
183 let moves = self.ranked(&board, side, depth);
184 if let Some(&(_, p)) = moves.first() {
185 steps.push(name(p));
186 board = place(&board, side, p);
187 } else {
188 steps.push("pass".to_string());
189 }
190 side = opponent(side);
191 if depth > 0 {
192 depth -= 1;
193 }
194 }
195 (steps, board)
196 }
197
198 fn perft(&mut self, board: &Board, side: u8, depth: u32) -> u64 {
199 if depth == 0 {
200 return 1;
201 }
202 let key = (*board, side, depth);
203 if let Some(&result) = self.perft_cache.get(&key) {
204 return result;
205 }
206 let moves = legal(board, side);
207 let other = opponent(side);
208 let result = if moves.is_empty() {
209 if legal(board, other).is_empty() {
210 1
211 } else {
212 self.perft(board, other, depth - 1)
213 }
214 } else {
215 moves
216 .into_iter()
217 .map(|p| self.perft(&place(board, side, p), other, depth - 1))
218 .sum()
219 };
220 if self.perft_cache.len() > 200_000 {
221 self.perft_cache.clear();
222 }
223 self.perft_cache.insert(key, result);
224 result
225 }
226}

228fn stable(board: &Board) -> (usize, usize) {
229 let mut set = [false; 64];
230 let axes = [(0, 1), (1, 0), (1, 1), (1, -1)];
231 loop {
232 let mut added = Vec::new();
233 for p in 0..64 {
234 if board[p] == b'.' || set[p] {
235 continue;
236 }
237 let mut valid = true;
238 for (dr, dc) in axes {
239 let mut full = true;
240 let mut anchor = false;
241 for sign in [-1, 1] {
242 let mut r = p as i32 / 8 + dr * sign;
243 let mut c = p as i32 % 8 + dc * sign;
244 if !inside(r, c)
245 || (set[(r * 8 + c) as usize] && board[(r * 8 + c) as usize] == board[p])
246 {
247 anchor = true;
248 }
249 while inside(r, c) {
250 if board[(r * 8 + c) as usize] == b'.' {
251 full = false;
252 }
253 r += dr * sign;
254 c += dc * sign;
255 }
256 }
257 if !full && !anchor {
258 valid = false;
259 break;
260 }
261 }
262 if valid {
263 added.push(p);
264 }
265 }
266 if added.is_empty() {
267 break;
268 }
269 for p in added {
270 set[p] = true;
271 }
272 }
273 let black = (0..64).filter(|&p| set[p] && board[p] == b'B').count();
274 let white = (0..64).filter(|&p| set[p] && board[p] == b'W').count();
275 (black, white)
276}

278fn transformed_square(p: usize, operation: &str) -> usize {
279 let r = p / 8;
280 let c = p % 8;
281 match operation {
282 "rot90" => 8 * c + 7 - r,
283 "rot180" => 63 - p,
284 "rot270" => 8 * (7 - c) + r,
285 "mirror" => 8 * r + 7 - c,
286 "flip" => 8 * (7 - r) + c,
287 "diag" => 8 * c + r,
288 _ => 8 * (7 - c) + 7 - r,
289 }
290}
291
292fn transformed_board(board: &Board, operation: &str) -> Board {
293 let mut result = [b'.'; 64];
294 for p in 0..64 {
295 result[transformed_square(p, operation)] = board[p];
296 }
297 result
298}

315fn random_game(mut board: Board, mut side: u8, random: &mut Random) -> Board {
316 loop {
317 let moves = legal(&board, side);
318 if !moves.is_empty() {
319 let p = moves[(random.draw() as usize) % moves.len()];
320 board = place(&board, side, p);
321 } else if legal(&board, opponent(side)).is_empty() {
322 return board;
323 }
324 side = opponent(side);
325 }
326}

563 fn automatic(&mut self, black: i32, white: i32, seed: Option<u64>) {
564 let mut random = seed.map(Random::new);
565 let mut played = String::new();

567 let mut moves = 0;
568 while !self.over {

592 fn monte_carlo(&self, games: usize, seed: u64) {
593 let mut random = Random::new(seed);
594 let mut best = None;
595 for p in legal(&self.board, self.side) {
596 let mut wins = 0;
597 let mut draws = 0;
598 let mut losses = 0;
599 let first = place(&self.board, self.side, p);
600 for _ in 0..games {
601 let last = random_game(first, opponent(self.side), &mut random);
602 let difference =
603 count(&last, self.side) as i32 - count(&last, opponent(self.side)) as i32;
604 if difference > 0 {
605 wins += 1;
606 } else if difference == 0 {
607 draws += 1;
608 } else {
609 losses += 1;
610 }
611 }
612 println!("{} {} {} {}", name(p), wins, draws, losses);
613 let points = 2 * wins + draws;
614 if best.map_or(true, |(_, value)| points > value) {
615 best = Some((p, points));
616 }
617 }
618 if let Some((p, value)) = best {
619 println!("best {} {}", name(p), value);
620 }
621 }

623 fn execute(&mut self, line: &str) {
624 let mut parts = line.split(' ');
625 let command = parts.next().unwrap_or("");
626 let args: Vec<&str> = parts.collect();
627 let zero = [
628 "NEW", "DUMP", "SHOW", "SCORE", "LEGAL", "HISTORY", "SHEET", "STATS", "EVAL", "STABLE",
629 "SOLVE",
630 ];
631 let numeric = [
632 "UNDO", "REDO", "HINT", "PV", "AI", "ANALYZE", "RANDOM", "PERFT", "DIVIDE",
633 ];
634 let active = [
635 "FLIPS", "MOVE", "PLAY", "HINT", "PV", "AI", "SOLVE", "AUTOPLAY", "RANDOM", "MC",
636 "DIVIDE",
637 ];
638 let known = zero.contains(&command)
639 || numeric.contains(&command)
640 || [
641 "LOAD",
642 "TRANSFORM",
643 "AUTOPLAY",
644 "MC",
645 "FLIPS",
646 "MOVE",
647 "PLAY",
648 "RECORD",
649 ]
650 .contains(&command);
651 if !known {
652 println!("ERROR unknown command");
653 return;
654 }
655 let mut error = None;
656 if command == "LOAD" {
657 if args.len() != 2
658 || args[0].len() != 64
659 || !args[0].bytes().all(|c| matches!(c, b'.' | b'B' | b'W'))
660 || !["B", "W"].contains(&args[1])
661 {
662 error = Some("bad board");
663 }
664 } else if command == "MOVE" || command == "FLIPS" {
665 if args.len() != 1 {
666 error = Some("bad arguments");
667 } else {
668 let bytes = args[0].as_bytes();
669 if bytes.len() != 2
670 || !(b'a'..=b'h').contains(&bytes[0])
671 || !(b'1'..=b'8').contains(&bytes[1])

673 error = Some("bad square");
674 }
675 }
676 } else if command == "PLAY" || command == "RECORD" {
677 if args.len() != 1 {
678 error = Some("bad arguments");
679 } else {
680 let bytes = args[0].as_bytes();
681 if bytes.is_empty()
682 || bytes.len() > 120
683 || bytes.len() % 2 != 0
684 || !bytes.chunks_exact(2).all(|pair| {
685 (b'a'..=b'h').contains(&pair[0]) && (b'1'..=b'8').contains(&pair[1])
686 })
687 {
688 error = Some("bad record");
689 }
690 }
691 } else if zero.contains(&command) {
692 if !args.is_empty() {
693 error = Some("bad arguments");
694 }
695 } else if command == "TRANSFORM" {
696 if args.len() != 1
697 || ![
698 "rot90", "rot180", "rot270", "mirror", "flip", "diag", "anti",
699 ]
700 .contains(&args[0])
701 {
702 error = Some("bad arguments");
703 }
704 } else {
705 let expected = if command == "MC" || command == "AUTOPLAY" {
706 2
707 } else {
708 1
709 };
710 let valid_numbers = args.len() == expected
711 && args.iter().all(|arg| {
712 !arg.is_empty()
713 && arg.bytes().all(|c| c.is_ascii_digit())
714 && arg.parse::<u64>().is_ok()
715 });
716 if !valid_numbers {
717 error = Some("bad arguments");
718 } else {
719 let a = args[0].parse::<u64>().unwrap();
720 let b = args.get(1).map(|s| s.parse::<u64>().unwrap()).unwrap_or(0);
721 let valid = match command {
722 "AUTOPLAY" => (1..=4).contains(&a) && (1..=4).contains(&b),
723 "HINT" | "PV" | "AI" | "ANALYZE" => (1..=4).contains(&a),
724 "PERFT" | "DIVIDE" => (1..=8).contains(&a),
725 "MC" => (1..=100).contains(&a) && (1..=2147483646).contains(&b),
726 "RANDOM" => (1..=2147483646).contains(&a),
727 _ => a >= 1,
728 };
729 if !valid {
730 error = Some("bad arguments");
731 }
732 }
733 }
734 if let Some(message) = error {
735 println!("ERROR {}", message);
736 return;
737 }
738 if self.over && active.contains(&command) {
739 println!("ERROR game over");
740 return;
741 }
742 let value = args
743 .first()
744 .and_then(|s| s.parse::<usize>().ok())
745 .unwrap_or(0);
746 match command {
747 "NEW" => {
748 self.reset();
749 println!("OK");
750 }
751 "LOAD" => {
752 self.board.copy_from_slice(args[0].as_bytes());
753 self.side = args[1].as_bytes()[0];
754 self.over = false;
755 self.history.clear();
756 self.redo.clear();
757 println!("OK");
758 self.adjust(false);
759 }
760 "DUMP" => println!(
761 "board {} {}",
762 String::from_utf8_lossy(&self.board),
763 self.side as char
764 ),
765 "SHOW" => {
766 println!(" a b c d e f g h");
767 let moves = if self.over {
768 Vec::new()
769 } else {
770 legal(&self.board, self.side)
771 };
772 for r in 0..8 {
773 let mut text = (r + 1).to_string();
774 for c in 0..8 {
775 let p = r * 8 + c;
776 text.push(' ');
777 text.push(if moves.contains(&p) {
778 '*'
779 } else {
780 self.board[p] as char
781 });
782 }
783 println!("{}", text);
784 }
785 println!("{}", self.status());
786 }
787 "SCORE" => println!(
788 "score B {} W {} empty {}",
789 count(&self.board, b'B'),
790 count(&self.board, b'W'),
791 count(&self.board, b'.')
792 ),
793 "LEGAL" => {
794 let moves = if self.over {
795 Vec::new()
796 } else {
797 legal(&self.board, self.side)
798 };
799 let mut text = format!("legal {}", moves.len());
800 for p in moves {
801 text.push_str(&format!(" {}", name(p)));
802 }
803 println!("{}", text);
804 }
805 "MOVE" | "FLIPS" => {
806 let p = square(args[0]);
807 if let Some(message) = self.reason(p) {
808 println!("ILLEGAL {}", message);
809 } else if command == "MOVE" {
810 self.play(p, false, None);
811 } else {
812 let flipped = flips(&self.board, self.side, p);
813 let mut line = format!("flips {} {}", args[0], flipped.len());
814 for q in flipped {
815 line.push_str(&format!(" {}", name(q)));
816 }
817 println!("{}", line);
818 }
819 }
820 "PLAY" => {
821 for (index, pair) in args[0].as_bytes().chunks_exact(2).enumerate() {
822 let p = (pair[1] - b'1') as usize * 8 + (pair[0] - b'a') as usize;
823 if let Some(message) = self.reason(p) {
824 println!("ILLEGAL move {} {} {}", index + 1, name(p), message);
825 break;
826 }
827 self.play(p, false, None);
828 }
829 }
830 "RECORD" => self.record(args[0]),
831 "UNDO" => {
832 if value > self.history.len() {
833 println!("ERROR cannot undo");
834 } else {
835 let removed = self.history.split_off(self.history.len() - value);
836 self.restore(removed[0].before);
837 let mut redo = removed;
838 redo.append(&mut self.redo);
839 self.redo = redo;
840 println!("{}", self.status());
841 }
842 }
843 "REDO" => {
844 if value > self.redo.len() {
845 println!("ERROR cannot redo");
846 } else {
847 let tail = self.redo.split_off(value);
848 let added = std::mem::replace(&mut self.redo, tail);
849 self.restore(added.last().unwrap().after);
850 self.history.extend(added);
851 println!("{}", self.status());
852 }
853 }
854 "HISTORY" => {
855 println!("history {}", self.history.len());
856 for (i, row) in self.history.iter().enumerate() {
857 println!(
858 "{} {} {} {}{}",
859 i + 1,
860 row.side as char,
861 name(row.square),
862 row.count,
863 if row.passed { " pass" } else { "" }
864 );
865 }
866 }
867 "SHEET" => self.sheet(),
868 "STATS" => self.stats(),
869 "TRANSFORM" => {
870 let operation = args[0];
871 self.board = transformed_board(&self.board, operation);
872 for row in self.history.iter_mut().chain(self.redo.iter_mut()) {
873 row.square = transformed_square(row.square, operation);
874 row.before.board = transformed_board(&row.before.board, operation);
875 row.after.board = transformed_board(&row.after.board, operation);
876 }
877 println!("OK");
878 }
879 "EVAL" => {
880 let [value, weights, own, other] = evaluate(&self.board, self.side);
881 println!(
882 "eval {} weights {} mobility {} {}",
883 value, weights, own, other
884 );
885 }
886 "STABLE" => {
887 let (black, white) = stable(&self.board);
888 println!("stable B {} W {}", black, white);
889 }
890 "HINT" => {
891 for (score, p) in self.solver.ranked(&self.board, self.side, value as i32) {
892 println!("{} {}", name(p), score);
893 }
894 }
895 "PV" => {
896 let score = self.solver.search(&self.board, self.side, value as i32);
897 let (moves, _) = self.solver.variation(self.board, self.side, value as i32);
898 println!("pv {} {}", score, moves.join(" "));
899 }
900 "AI" => {
901 let (score, p) = self.solver.ranked(&self.board, self.side, value as i32)[0];
902 self.play(p, false, Some(score));
903 }
904 "ANALYZE" => {
905 let mut losses = [0, 0];
906 for (i, row) in self.history.iter().enumerate() {
907 let moves = self
908 .solver
909 .ranked(&row.before.board, row.side, value as i32);
910 let (best, best_square) = moves[0];
911 let played = moves.iter().find(|(_, p)| *p == row.square).unwrap().0;
912 losses[if row.side == b'B' { 0 } else { 1 }] += best - played;
913 println!(
914 "{} {} {} {} best {} {}",
915 i + 1,
916 row.side as char,
917 name(row.square),
918 played,
919 name(best_square),
920 best
921 );
922 }
923 println!("loss B {} W {}", losses[0], losses[1]);
924 }
925 "SOLVE" => {
926 if count(&self.board, b'.') >= 9 {
927 println!("ERROR too many empties");
928 } else {
929 for (score, p) in self.solver.ranked(&self.board, self.side, -1) {
930 println!("{} {}", name(p), score);
931 }
932 let (moves, board) = self.solver.variation(self.board, self.side, -1);
933 println!("line {}", moves.join(" "));
934 println!("final B {} W {}", count(&board, b'B'), count(&board, b'W'));
935 }
936 }
937 "AUTOPLAY" => self.automatic(value as i32, args[1].parse::<i32>().unwrap(), None),
938 "RANDOM" => self.automatic(0, 0, Some(value as u64)),
939 "MC" => self.monte_carlo(value, args[1].parse::<u64>().unwrap()),
940 "PERFT" => println!(
941 "perft {} {}",
942 value,
943 self.solver.perft(&self.board, self.side, value as u32)
944 ),
945 "DIVIDE" => {
946 let mut total = 0;
947 for p in legal(&self.board, self.side) {
948 let amount = self.solver.perft(
949 &place(&self.board, self.side, p),
950 opponent(self.side),
951 value as u32 - 1,
952 );
953 println!("{} {}", name(p), amount);
954 total += amount;
955 }
956 println!("total {}", total);
957 }
958 _ => {}
959 }
960 }
961}
