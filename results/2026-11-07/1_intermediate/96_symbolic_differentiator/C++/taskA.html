16struct Rational {
17 int64_t p, q;
18 Rational(int64_t n = 0, int64_t d = 1) {
19 if (!d) throw runtime_error("division by zero");
20 if (d<0) {
21 n = -n;
22 d = -d;
23 }
24 int64_t g = gcd(n, d);
25 p = n/g;
26 q = d/g;
27 }
28 Rational operator+(Rational b) const {
29 return {
30 p*b.q+b.p*q, q*b.q
31 };
32 }
33 Rational operator*(Rational b) const {
34 return {
35 p*b.p, q*b.q
36 };
37 }
38 Rational operator-() const {
39 return {
40 -p, q
41 };
42 }
43 bool operator == (Rational b) const {
44 return p == b.p && q == b.q;
45 }
46 bool operator<(Rational b) const {
47 return p*b.q<b.p*q;
48 }
49};

75int compare(const Expr&a, const Expr&b) {
76 if (a.kind != b.kind) return int(a.kind)-int(b.kind);
77 if (a.kind == Number) return a.q == b.q?0:a.q<b.q?-1:1;
78 if (a.name != b.name) return a.name<b.name?-1:1;
79 for (size_t i = 0; i<min(a.args.size(), b.args.size()); ++i) {
80 int c = compare(a.args[i], b.args[i]);
81 if (c) return c;
82 }
83 return a.args.size() == b.args.size()?0:a.args.size()<b.args.size()?-1:1;
84}

131int termCompare(const Expr&a, const Expr&b) {
132 vector<Expr>x = coefficient(a).second, y = coefficient(b).second, bases;
133 for (const Expr&f:x) bases.push_back(base(f));
134 for (const Expr&f:y) bases.push_back(base(f));
135 sort(bases.begin(), bases.end(), [](const Expr&s, const Expr&t) {
136 return compare(s, t)<0;
137 });
138 bases.erase(unique(bases.begin(), bases.end()), bases.end());
139 for (const Expr&f:bases) {
140 Expr u = zero, v = zero;
141 for (const Expr&t:x) if (base(t) == f) u = exponent(t);
142 for (const Expr&t:y) if (base(t) == f) v = exponent(t);
143 int c = exponentCompare(v, u);
144 if (c) return c;
145 }
146 return 0;
147}

152Expr add(vector<Expr>operands) {
153 vector<Expr>terms, monomials;
154 vector<vector<Expr>>keys;
155 vector<Rational>values;
156 for (const Expr&f:operands) {
157 if (f.kind == Sum) terms.insert(terms.end(), f.args.begin(), f.args.end());
158 else if (f.kind == Product && f.args.size() == 2 && f.args[0].kind == Number && f.args[1].kind == Sum) {
159 for (const Expr&t:f.args[1].args) terms.push_back(mul({
160 f.args[0], t
161 }));
162 } else terms.push_back(f);
163 }
164 for (const Expr&f:terms) {
165 auto [c, m] = coefficient(f);
166 size_t i = 0;
167 while (i<keys.size() && keys[i] != m)++i;
168 if (i == keys.size()) {
169 keys.push_back(m);
170 values.push_back(c);
171 } else values[i] = values[i]+c;
172 }
173 vector<Expr>built;
174 for (size_t i = 0; i<keys.size(); ++i) if (values[i].p) built.push_back(product(values[i], keys[i]));
175 sort(built.begin(), built.end(), [](const Expr&a, const Expr&b) {
176 return termCompare(a, b)<0;
177 });
178 return built.empty()?zero:built.size() == 1?built[0]:node(Sum, built);
179}
180Expr mul(vector<Expr>operands) {
181 vector<Expr>factors, bases;
182 vector<vector<Expr>>groups;
183 Rational c(1);
184 for (const Expr&f:operands) if (f.kind == Product) factors.insert(factors.end(), f.args.begin(), f.args.end());
185 else factors.push_back(f);
186 for (const Expr&f:factors) if (f.kind == Number) c = c*f.q;
187 else {
188 Expr b = base(f);
189 size_t i = 0;
190 while (i<bases.size() && !(bases[i] == b))++i;
191 if (i == bases.size()) {
192 bases.push_back(b);
193 groups.push_back({
194 });
195 }
196 groups[i].push_back(f);
197 }
198 if (!c.p) return zero;
199 vector<Expr>r;
200 for (size_t i = 0; i<groups.size(); ++i) {
201 vector<Expr>xs;
202 for (const Expr&f:groups[i]) xs.push_back(exponent(f));
203 r.push_back(groups[i].size() == 1?groups[i][0]:power(bases[i], add(xs)));
204 }
205 bool regroup = false;
206 for (size_t i = 0; i<r.size(); ++i) {
207 if (r[i].kind == Number || r[i].kind == Product) regroup = true;
208 for (size_t j = 0; j<i; ++j) if (base(r[i]) == base(r[j])) regroup = true;
209 }
210 if (regroup) {
211 r.insert(r.begin(), num(c));
212 return mul(r);
213 }
214 sort(r.begin(), r.end(), [](const Expr&a, const Expr&b) {
215 return compare(base(a), base(b))<0;
216 });
217 return product(c, r);
218}
219int64_t root(int64_t v, int64_t d) {
220 if (v == 1) return 1;
221 if (d>20) return -1;
222 int64_t r = llround(pow(double(v), 1.0/d));
223 for (int64_t s = max<int64_t>(0, r-1); s <= r+1; ++s) {
224 int64_t a = 1;
225 for (int64_t i = 0; i<d && a <= v; ++i) a *= s;
226 if (a == v) return s;
227 }
228 return -1;
229}
230Expr rationalPower(Rational b, int64_t n) {
231 if (n<0) {
232 b = Rational(b.q, b.p);
233 n = -n;
234 }
235 Rational r(1);
236 while (n) {
237 if (n%2) r = r*b;
238 n /= 2;
239 if (n) b = b*b;
240 }
241 return num(r);
242}
243Expr power(Expr b, Expr x) {
244 if (x == zero) return one;
245 if (x == one) return b;
246 if (b.kind == Number) {
247 if (b == one) return one;
248 if (x.kind != Number) return node(Power, {
249 b, x
250 });
251 if (!b.q.p) {
252 if (x.q.p>0) return zero;
253 throw runtime_error("division by zero");
254 }
255 if (integer(x)) return rationalPower(b.q, x.q.p);
256 if (b.q.p>0) {
257 int64_t s = root(b.q.p, x.q.q), t = root(b.q.q, x.q.q);
258 if (s >= 0 && t >= 0) return rationalPower({
259 s, t
260 }, x.q.p);
261 }
262 }
263 else if (integer(x)) {
264 if (b.kind == Power) return power(b.args[0], mul({
265 b.args[1], x
266 }));
267 if (b.kind == Product) {
268 vector<Expr>a;
269 for (const Expr&f:b.args) a.push_back(power(f, x));
270 return mul(a);
271 }
272 }
273 return node(Power, {
274 b, x
275 });
276}
277Expr fn(string s, Expr a) {
278 if (s == "sqrt") return power(a, half);
279 if (a == zero && (s == "sin" || s == "tan")) return zero;
280 if (a == zero && (s == "cos" || s == "exp")) return one;
281 if (a == one && s == "ln") return zero;
282 if ((s == "sin" || s == "cos" || s == "tan") && coefficient(a).first.p<0) {
283 Expr r = fn(s, mul({
284 minusOne, a
285 }));
286 return s == "cos"?r:mul({
287 minusOne, r
288 });
289 }
290 if (a.kind == Function && ((s == "exp" && a.name == "ln") || (s == "ln" && a.name == "exp"))) return a.args[0];
291 return node(Function, {
292 a
293 }, s);
294}

304Expr derivative(const Expr&f, const string&v) {
305 if (f.kind == Number) return zero;
306 if (f.kind == Variable) return f.name == v?one:zero;
307 if (f.kind == Sum) {
308 vector<Expr>a;
309 for (const Expr&t:f.args) a.push_back(derivative(t, v));
310 return add(a);
311 }
312 if (f.kind == Product) {
313 vector<Expr>a;
314 for (size_t i = 0; i<f.args.size(); ++i) {
315 vector<Expr>p = f.args;
316 p[i] = derivative(p[i], v);
317 a.push_back(mul(p));
318 }
319 return add(a);
320 }
321 if (f.kind == Power) {
322 Expr b = f.args[0], x = f.args[1];
323 if (!variables(x).count(v)) return mul({
324 x, power(b, add({
325 x, minusOne
326 })), derivative(b, v)
327 });
328 if (!variables(b).count(v)) return mul({
329 f, fn("ln", b), derivative(x, v)
330 });
331 return mul({
332 f, add({
333 mul({
334 derivative(x, v), fn("ln", b)
335 }), mul({
336 x, derivative(b, v), power(b, minusOne)
337 })
338 })
339 });
340 }
341 Expr a = f.args[0], h = f.name == "sin"?fn("cos", a):f.name == "cos"?mul({
342 minusOne, fn("sin", a)
343 }):f.name == "tan"?power(fn("cos", a), num(-2)):f.name == "exp"?f:power(a, minusOne);
344 return mul({
345 h, derivative(a, v)
346 });
347}

373Expr expand(const Expr&f) {
374 if (f.kind == Variable || f.kind == Number) return f;
375 vector<Expr>a;
376 for (const Expr&t:f.args) a.push_back(expand(t));
377 if (f.kind == Product) {
378 Expr r = a[0];
379 for (size_t i = 1; i<a.size(); ++i) r = distribute(r, a[i]);

381 }
382 if (f.kind == Power && a[0].kind == Sum && integer(a[1]) && a[1].q.p>0) {
383 Expr r = a[0];

385 return r;
386 }
387 return rebuild(f, a);

389double numeric(const Expr&f) {
390 if (f.kind == Number) return double(f.q.p)/f.q.q;
391 if (f.kind == Sum || f.kind == Product) {
392 double r = f.kind == Sum?0:1;
393 for (const Expr&t:f.args) {
394 double x = numeric(t);
395 if (f.kind == Sum) r += x;
396 else r *= x;
397 }
398 return r;
399 }
400 if (f.kind == Power) {
401 double u = numeric(f.args[0]), z = numeric(f.args[1]);
402 bool integral = f.args[1].kind == Number?integer(f.args[1]):z == trunc(z);
403 if (u<0 && !integral) throw runtime_error("domain error");
404 return f.args[1] == half?sqrt(u):pow(u, z);
405 }
406 double a = numeric(f.args[0]);
407 if (f.name == "ln") {
408 if (a <= 0) throw runtime_error("domain error");
409 return log(a);
410 }
411 return f.name == "sin"?sin(a):f.name == "cos"?cos(a):f.name == "tan"?tan(a):exp(a);
412}

435string quotient(Rational c, const vector<Expr>&a) {
436 vector<string>top, bottom;
437 if (abs(c.p) != 1) top.push_back(to_string(abs(c.p)));
438 if (c.q != 1) bottom.push_back(to_string(c.q));
439 for (const Expr&f:a) if (negative(f)) {
440 Expr r = num(-f.args[1].q);
441 bottom.push_back(r == one?factor(f.args[0]):showPower(f.args[0], r));
442 } else top.push_back(factor(f));
443 string s = top.empty()?"1":join(top, "*");
444 if (bottom.size() == 1) s += "/"+bottom[0];
445 else if (!bottom.empty()) s += "/("+join(bottom, "*")+")";
446 return c.p<0?"-"+s:s;
447}
448string show(const Expr&f) {
449 if (f.kind == Number) return to_string(f.q.p)+(f.q.q == 1?"":"/"+to_string(f.q.q));
450 if (f.kind == Variable) return f.name;
451 if (f.kind == Function) return f.name+"("+show(f.args[0])+")";
452 if (f.kind == Power) return negative(f)?quotient(1, {
453 f
454 }):showPower(f.args[0], f.args[1]);
455 if (f.kind == Product) {
456 auto [c, a] = coefficient(f);
457 return quotient(c, a);
458 }
459 string r = show(f.args[0]);
460 for (size_t i = 1; i<f.args.size(); ++i) {
461 string s = show(f.args[i]);
462 r += s[0] == '-'?" - "+s.substr(1):" + "+s;
463 }
464 return r;
465}

474class Parser {
475 vector<string>tokens;
476 size_t position = 0;
477 const map<string, Expr>&names;
478 string peek() {
479 return position<tokens.size()?tokens[position]:"";
480 }
481 [[noreturn]] void unexpected() {
482 throw runtime_error(position == tokens.size()?"unexpected end of expression":"unexpected '"+peek()+"'");
483 }
484 bool accept(string s) {
485 if (peek() != s) return false;
486 ++position;
487 return true;
488 }
489 void expect(string s) {
490 if (!accept(s)) unexpected();
491 }
492 Syntax atomTree() {
493 string t = peek();
494 if (t.empty()) unexpected();
495 if (accept("(")) {
496 Syntax a = expression();
497 expect(")");
498 return a;
499 }
500 if (!isalnum(t[0])) unexpected();
501 ++position;
502 if (isdigit(t[0])) return {
503 "", num(stoll(t)), {
504 }
505 };
506 if (isupper(t[0])) {
507 auto i = names.find(t);
508 if (i == names.end()) throw runtime_error("undefined name '"+t+"'");
509 return {
510 "", i->second, {
511 }
512 };
513 }
514 if (!functions.count(t)) return {
515 "", node(Variable, {
516 }, t), {
517 }
518 };
519 expect("(");
520 Syntax a = expression();
521 expect(")");
522 return {
523 t, zero, {
524 a
525 }
526 };
527 }
528 Syntax unary() {
529 if (accept("-")) return {
530 "neg", zero, {
531 unary()
532 }
533 };
534 Syntax a = atomTree();
535 if (accept("^")) return {
536 "^", zero, {
537 a, unary()
538 }
539 };
540 return a;
541 }
542 Syntax term() {
543 Syntax a = unary();
544 while (peek() == "*" || peek() == "/") {
545 string op = tokens[position++];
546 a = {
547 op, zero, {
548 a, unary()
549 }
550 };
551 }
552 return a;
553 }
554 Syntax expression() {
555 Syntax a = term();
556 while (peek() == "+" || peek() == "-") {
557 string op = tokens[position++];
558 a = {
559 op, zero, {
560 a, term()
561 }
562 };
563 }
564 return a;
565 }
566 Expr evaluate(const Syntax&t) {
567 if (t.op.empty()) return t.value;
568 vector<Expr>a;
569 for (const Syntax&s:t.args) a.push_back(evaluate(s));
570 if (t.op == "+") return add(a);
571 if (t.op == "-") return add({
572 a[0], mul({
573 minusOne, a[1]
574 })
575 });
576 if (t.op == "*") return mul(a);
577 if (t.op == "/") return mul({
578 a[0], power(a[1], minusOne)
579 });
580 if (t.op == "^") return power(a[0], a[1]);
581 if (t.op == "neg") return mul({
582 minusOne, a[0]
583 });
584 return fn(t.op, a[0]);
585 }
586 public:
587 Parser(const string&text, const map<string, Expr>&n):names(n) {
588 for (size_t p = 0; p<text.size(); ) {
589 char c = text[p];
590 if (c == ' ') {
591 ++p;
592 continue;
593 }
594 size_t start = p++;
595 if (isdigit(c)) {
596 while (p<text.size() && isdigit(text[p]))++p;
597 } else if (islower(c)) {
598 while (p<text.size() && (islower(text[p]) || isdigit(text[p])))++p;
599 } else if (isupper(c)) {
600 while (p<text.size() && (isupper(text[p]) || isdigit(text[p])))++p;
601 } else if (string("+-*/^()").find(c) == string::npos) throw runtime_error(string("invalid character '")+c+"'");
602 tokens.push_back(text.substr(start, p-start));
603 }
604 }
605 Expr parse() {
606 Syntax t = expression();
607 if (position<tokens.size()) unexpected();
608 return evaluate(t);
609 }

672vector<string>command(string line, map<string, Expr>&names) {
673 size_t p = line.find(' ');
674 string c = line.substr(0, p), arg = p == string::npos?"":line.substr(p+1);
675 if (c == "LET") {
676 size_t i = arg.find('=');
677 if (i == string::npos) throw runtime_error("wrong number of arguments");
678 string s = trim(arg.substr(0, i));
679 if (s.empty() || !isupper(s[0]) || !all_of(s.begin(), s.end(), [](char ch) {
680 return isupper(ch) || isdigit(ch);
681 })) throw runtime_error("invalid name '"+s+"'");
682 Expr f = Parser(trim(arg.substr(i+1)), names).parse();
683 names[s] = f;
684 return {
685 s+" = "+show(f)
686 };
687 }
688 auto sig = signatures.find(c);
689 if (sig == signatures.end()) throw runtime_error("unknown command "+c);
690 vector<string>fields = p == string::npos?vector<string> {
691 }:split(arg, ';');
692 for (string&s:fields) s = trim(s);
693 if (c == "EVAL"?(fields.size() != 1 && fields.size() != 2):fields.size() != sig->second.size()) throw runtime_error("wrong number of arguments");
694 vector<Expr>forms(fields.size(), zero);
695 int n = 0;
696 for (size_t i = 0; i<fields.size(); ++i) {
697 char k = sig->second[i];
698 string s = fields[i];
699 if (k == 'e') forms[i] = Parser(s, names).parse();
700 else if (k == 'v' && !variable(s)) throw runtime_error("invalid variable '"+s+"'");
701 else if (k == 'n') {
702 if (s.size() != 1 || s[0]<'1' || s[0]>'8') throw runtime_error("invalid order '"+s+"'");
703 n = s[0]-'0';
704 } else if (k == 'c') forms[i] = literal(s);
705 }
706 Expr f = forms[0];
707 string v = fields.size()>1?fields[1]:"";
708 vector<string>out;
709 if (c == "SIMPLIFY") return {
710 "simplified: "+show(f)
711 };
712 if (c == "EXPAND") return {
713 "expanded: "+show(expand(f))
714 };
715 if (c == "DIFF") return {
716 "d/d"+v+" = "+show(derivative(f, v))
717 };
718 if (c == "DIFFN") {
719 for (int k = 1; k <= n; ++k) {
720 f = derivative(f, v);
721 out.push_back("d^"+to_string(k)+"/d"+v+"^"+to_string(k)+" = "+show(f));
722 }
723 return out;
724 }
725 if (c == "GRAD" || c == "HESSIAN") {
726 set<string>vs = variables(f);
727 if (vs.empty()) return {
728 "no variables"
729 };
730 for (auto i = vs.begin(); i != vs.end(); ++i) {
731 Expr d = derivative(f, *i);
732 if (c == "GRAD") out.push_back("d/d"+*i+" = "+show(d));
733 else for (auto j = i; j != vs.end(); ++j) out.push_back("d2/d"+*i+"d"+*j+" = "+show(derivative(d, *j)));
734 }
735 return out;
736 }
737 if (c == "IMPLICIT") {
738 string w = fields[2];
739 if (v == w) throw runtime_error("same variable");
740 return {
741 "d"+w+"/d"+v+" = "+show(mul({
742 minusOne, derivative(f, v), power(derivative(f, w), minusOne)
743 }))
744 };
745 }
746 if (c == "SUBST") return {
747 "substituted: "+show(substitute(f, {
748 {
749 v, forms[2]
750 }
751 }))
752 };
753 if (c == "EVAL") {
754 map<string, Expr>values;
755 for (const string&word:fields.size() == 2?split(fields[1], ' '):vector<string> {
756 }) {
757 if (word.empty()) continue;
758 size_t i = word.find('=');
759 string s = word.substr(0, i), t = i == string::npos?"":word.substr(i+1);
760 if (i == string::npos || !variable(s) || !rationalLiteral(t)) throw runtime_error("invalid assignment '"+word+"'");
761 Expr q = literal(t);
762 if (values.count(s)) throw runtime_error("duplicate variable "+s);
763 values[s] = q;
764 }
765 for (const string&s:variables(f)) if (!values.count(s)) throw runtime_error("no value for "+s);
766 Expr r = substitute(f, values);
767 if (r.kind == Number) return {
768 "exact: "+show(r)
769 };
770 ostringstream s;
771 s<<fixed<<setprecision(6)<<numeric(r);
772 return {
773 "approx: "+s.str()
774 };
775 }
776 Expr x = forms[2];
777 map<string, Expr>point = {
778 {
779 v, x
780 }
781 };
782 if (c == "TANGENT") {
783 Expr slope = substitute(derivative(f, v), point), y = substitute(f, point), z = node(Variable, {
784 }, v);
785 string t = "tangent: y = "+show(add({
786 mul({
787 slope, z
788 }), y, mul({
789 num(-x.q), slope
790 })
791 }));
792 if (slope == zero) return {
793 t, "normal: "+v+" = "+show(x)
794 };
795 Expr k = power(slope, minusOne);
796 return {
797 t, "normal: y = "+show(add({
798 mul({
799 minusOne, k, z
800 }), y, mul({
801 x, k
802 })
803 }))
804 };
805 }
806 if (c == "LIMIT") {
807 vector<Expr>top, bottom;
808 for (const Expr&t:f.kind == Product?f.args:vector<Expr> {
809 f
810 }) if (negative(t)) bottom.push_back(power(t.args[0], num(-t.args[1].q)));
811 else top.push_back(t);
812 Expr g = mul(top), h = mul(bottom);
813 for (int i = 0; i <= 5; ++i) {
814 Expr u = substitute(g, point), b = substitute(h, point);
815 if (!(b == zero)) return {
816 "limit: "+show(mul({
817 u, power(b, minusOne)
818 }))
819 };
820 if (!(u == zero)) return {
821 "limit: unbounded"
822 };
823 if (i<5) {
824 g = derivative(g, v);
825 h = derivative(h, v);
826 }
827 }
828 return {
829 "limit: unknown"
830 };
831 }
832 vector<Expr>ds = {
833 f
834 }, ps;
835 for (int i = 0; i<n; ++i) ds.push_back(derivative(ds.back(), v));
836 int factorial = 1;
837 for (int i = 0; i <= n; ++i) {
838 if (i) factorial *= i;
839 Expr co = mul({
840 substitute(ds[i], point), num(Rational(1, factorial))
841 });
842 ps.push_back(mul({
843 co, power(add({
844 node(Variable, {
845 }, v), num(-x.q)
846 }), num(i))
847 }));
848 }
849 return {
850 "taylor: "+show(expand(add(ps)))
851 };
852}
853int main() {
854 int n;
855 cin>>n;
856 string line;
857 getline(cin, line);
858 map<string, Expr>names;
859 while (n-- && getline(cin, line)) {
860 try {
861 for (const string&s:command(line, names)) cout<<s<<'\n';
862 } catch (const runtime_error&e) {
863 cout<<"error: "<<e.what()<<'\n';
864 }
865 }
866}
