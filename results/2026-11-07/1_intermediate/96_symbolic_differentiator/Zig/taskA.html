40fn gcd(a0: i128, b0: i128) i128 {
41 var a: i128 = @intCast(@abs(a0));
42 var b: i128 = @intCast(@abs(b0));
43 while (b != 0) {
44 const r = @rem(a, b);
45 a = b;
46 b = r;
47 }
48 return a;
49}

51fn rational(p0: i128, q0: i128) Rational {
52 var p = p0;
53 var q = q0;
54 if (q < 0) {
55 p = -p;
56 q = -q;
57 }
58 const d = gcd(p, q);
59 return .{ .p = @divTrunc(p, d), .q = @divTrunc(q, d) };
60}
61
62fn plus(a: Rational, b: Rational) Rational {
63 return rational(a.p * b.q + b.p * a.q, a.q * b.q);
64}
65
66fn times(a: Rational, b: Rational) Rational {
67 return rational(a.p * b.p, a.q * b.q);
68}
69
70fn negate(a: Rational) Rational {
71 return .{ .p = -a.p, .q = a.q };
72}

135 return if (f.k == 3) f.a[0] else f;

142 fn compare(a: *Expr, b: *Expr) i32 {
143 if (a.k != b.k) return comparison(u8, a.k, b.k);
144 if (a.k == 5) return comparison(i128, a.q.p * b.q.q, b.q.p * a.q.q);
145 const c = std.mem.order(u8, a.s, b.s);
146 if (c != .eq) return if (c == .lt) -1 else 1;
147 for (a.a, 0..) |item, i| {
148 if (i >= b.a.len) break;
149 const k = compare(item, b.a[i]);
150 if (k != 0) return k;
151 }
152 return comparison(usize, a.a.len, b.a.len);
153 }
154
155 fn equal(a: *Expr, b: *Expr) bool {
156 return compare(a, b) == 0;
157 }

159 const Coefficient = struct {
160 c: Rational,
161 factors: []const *Expr,
162 };
163
164 fn coefficient(f: *Expr) Coefficient {
165 if (f.k == 5) return .{ .c = f.q, .factors = &.{} };
166 if (f.k == 2) {
167 if (f.a[0].k == 5) return .{ .c = f.a[0].q, .factors = f.a[1..] };
168 return .{ .c = .{ .p = 1 }, .factors = f.a };
169 }
170 return .{ .c = .{ .p = 1 }, .factors = &.{f} };
171 }

173 fn product(m: *Machine, c: Rational, factors: []const *Expr) !*Expr {
174 if (factors.len == 0) return m.num(c.p, c.q);
175 if (c.p == c.q) return if (factors.len == 1) factors[0] else m.node(2, factors, "");
176 var collected: Expressions = .empty;
177 try collected.append(m.gpa, try m.num(c.p, c.q));
178 try collected.appendSlice(m.gpa, factors);
179 return m.node(2, collected.items, "");
180 }

182 fn compareExponent(a: *Expr, b: *Expr) i32 {
183 if (a.k == 5 and b.k != 5) return -1;
184 if (a.k != 5 and b.k == 5) return 1;
185 return compare(a, b);
186 }

188 fn termOrder(m: *Machine, first: *Expr, second: *Expr) i32 {
189 const a = coefficient(first).factors;
190 const b = coefficient(second).factors;
191 var i: usize = 0;
192 var j: usize = 0;
193 while (i < a.len or j < b.len) {
194 const order: i32 = if (i == a.len) 1 else if (j == b.len) -1 else compare(base(a[i]), base(b[j]));
195 var x = m.zero;
196 var y = m.zero;
197 if (order <= 0) {
198 x = m.exponent(a[i]);
199 i += 1;
200 }
201 if (order >= 0) {
202 y = m.exponent(b[j]);
203 j += 1;
204 }
205 const cmp = compareExponent(x, y);
206 if (cmp != 0) return -cmp;
207 }
208 return 0;
209 }

211 fn lessTerm(m: *Machine, a: *Expr, b: *Expr) bool {
212 return m.termOrder(a, b) < 0;
213 }

215 fn lessFactor(_: void, a: *Expr, b: *Expr) bool {
216 return compare(base(a), base(b)) < 0;

219 fn sameSequence(a: []const *Expr, b: []const *Expr) bool {
220 if (a.len != b.len) return false;
221 for (a, b) |x, y| if (!equal(x, y)) return false;
222 return true;
223 }

226 c: Rational,
227 factors: []const *Expr,
228 };

230 fn add(m: *Machine, values: []const *Expr) anyerror!*Expr {
231 var terms: Expressions = .empty;
232 for (values) |f| {
233 if (f.k == 1) {
234 try terms.appendSlice(m.gpa, f.a);
235 } else if (f.k == 2 and f.a.len == 2 and f.a[0].k == 5 and f.a[1].k == 1) {
236 for (f.a[1].a) |term| try terms.append(m.gpa, try m.mul(&.{ f.a[0], term }));
237 } else {
238 try terms.append(m.gpa, f);
239 }
240 }
241 var groups: std.ArrayList(SumGroup) = .empty;
242 for (terms.items) |f| {
243 const info = coefficient(f);
244 var found = false;
245 for (groups.items) |*g| {
246 if (sameSequence(g.factors, info.factors)) {
247 g.c = plus(g.c, info.c);
248 found = true;
249 break;
250 }
251 }
252 if (!found) try groups.append(m.gpa, .{ .c = info.c, .factors = info.factors });
253 }
254 var result: Expressions = .empty;
255 for (groups.items) |g| if (g.c.p != 0) {
256 try result.append(m.gpa, try m.product(g.c, g.factors));
257 };
258 std.mem.sort(*Expr, result.items, m, lessTerm);
259 if (result.items.len == 0) return m.zero;
260 if (result.items.len == 1) return result.items[0];
261 return m.node(1, result.items, "");
262 }

269 fn mul(m: *Machine, input: []const *Expr) anyerror!*Expr {
270 var pending = try m.gpa.dupe(*Expr, input);
271 while (true) {
272 var factors: Expressions = .empty;
273 for (pending) |f| {
274 if (f.k == 2) try factors.appendSlice(m.gpa, f.a) else try factors.append(m.gpa, f);
275 }
276 var c = Rational{ .p = 1 };
277 var groups: std.ArrayList(ProductGroup) = .empty;
278 for (factors.items) |f| {
279 if (f.k == 5) {
280 c = times(c, f.q);
281 } else {
282 const b = base(f);
283 var found = false;
284 for (groups.items) |*g| if (equal(g.b, b)) {
285 try g.values.append(m.gpa, f);
286 found = true;
287 break;
288 };
289 if (!found) {
290 var parts: Expressions = .empty;
291 try parts.append(m.gpa, f);
292 try groups.append(m.gpa, .{ .b = b, .values = parts });
293 }
294 }
295 }
296 if (c.p == 0) return m.zero;
297 var reduced: Expressions = .empty;
298 for (groups.items) |group| {
299 if (group.values.items.len == 1) {
300 try reduced.append(m.gpa, group.values.items[0]);
301 } else {
302 var exponents: Expressions = .empty;
303 for (group.values.items) |f| try exponents.append(m.gpa, m.exponent(f));
304 try reduced.append(m.gpa, try m.power(group.b, try m.add(exponents.items)));
305 }
306 }
307 var retry = false;
308 for (reduced.items, 0..) |f, i| {
309 if (f.k == 2 or f.k == 5) retry = true;
310 for (reduced.items[0..i]) |other| {
311 if (equal(base(f), base(other))) retry = true;
312 }
313 }
314 if (retry) {
315 var again: Expressions = .empty;
316 try again.append(m.gpa, try m.num(c.p, c.q));
317 try again.appendSlice(m.gpa, reduced.items);
318 pending = try again.toOwnedSlice(m.gpa);
319 continue;
320 }
321 std.mem.sort(*Expr, reduced.items, {}, lessFactor);
322 return m.product(c, reduced.items);
323 }
324 }

326 fn exactRoot(value: i128, degree: i128) ?i128 {
327 if (value == 1) return 1;
328 if (degree > 20 or degree <= 0 or value < 0) return null;
329 var candidate: i128 = 0;
330 while (candidate <= 1000) : (candidate += 1) {
331 var product_value: i128 = 1;
332 var k: i128 = 0;
333 while (k < degree) : (k += 1) {
334 if (candidate != 0 and product_value > @divTrunc(value, candidate)) {
335 product_value = value + 1;
336 break;
337 }
338 product_value *= candidate;
339 if (product_value > value) break;
340 }
341 if (product_value == value) return candidate;
342 if (product_value > value) break;
343 }
344 return null;
345 }

347 fn rationalPower(m: *Machine, fraction: Rational, exp: i128) anyerror!*Expr {
348 var b = if (exp < 0) rational(fraction.q, fraction.p) else fraction;
349 var n: u128 = @intCast(@abs(exp));
350 var value = Rational{ .p = 1 };
351 while (n != 0) {
352 if (n & 1 != 0) value = times(value, b);
353 n >>= 1;
354 if (n != 0) b = times(b, b);
355 }
356 return m.num(value.p, value.q);
357 }

359 fn power(m: *Machine, b: *Expr, x: *Expr) anyerror!*Expr {
360 if (equal(x, m.zero)) return m.one;
361 if (equal(x, m.one)) return b;
362 if (b.k == 5) {
363 if (equal(b, m.one)) return m.one;
364 if (x.k == 5) {
365 if (b.q.p == 0) {
366 if (x.q.p > 0) return m.zero;
367 return m.fail("division by zero", .{});
368 }
369 if (isInteger(x)) return m.rationalPower(b.q, x.q.p);
370 if (b.q.p > 0) {
371 if (exactRoot(b.q.p, x.q.q)) |p| {
372 if (exactRoot(b.q.q, x.q.q)) |q| {
373 return m.rationalPower(rational(p, q), x.q.p);
374 }
375 }
376 }
377 }
378 } else if (isInteger(x)) {
379 if (b.k == 3) return m.power(b.a[0], try m.mul(&.{ b.a[1], x }));
380 if (b.k == 2) {
381 var factors: Expressions = .empty;
382 for (b.a) |f| try factors.append(m.gpa, try m.power(f, x));
383 return m.mul(factors.items);
384 }
385 }
386 return m.node(3, &.{ b, x }, "");
387 }

389 fn function(m: *Machine, name: Text, a: *Expr) anyerror!*Expr {
390 if (eql(name, "sqrt")) return m.power(a, m.half);
391 if (equal(a, m.zero)) {
392 if (among(name, &.{ "sin", "tan" })) return m.zero;
393 if (among(name, &.{ "cos", "exp" })) return m.one;
394 }
395 if (eql(name, "ln") and equal(a, m.one)) return m.zero;
396 if (among(name, &.{ "sin", "cos", "tan" }) and coefficient(a).c.p < 0) {
397 const positive = try m.mul(&.{ m.minus, a });
398 const transformed = try m.function(name, positive);
399 return if (eql(name, "cos")) transformed else m.mul(&.{ m.minus, transformed });
400 }
401 if (a.k == 4 and ((eql(name, "ln") and eql(a.s, "exp")) or
402 (eql(name, "exp") and eql(a.s, "ln")))) return a.a[0];
403 return m.node(4, &.{a}, name);
404 }

430 fn derivative(m: *Machine, f: *Expr, variable: Text) anyerror!*Expr {
431 if (f.k == 5) return m.zero;
432 if (f.k == 0) return if (eql(f.s, variable)) m.one else m.zero;
433 if (f.k == 1) {
434 var a: Expressions = .empty;
435 for (f.a) |term| try a.append(m.gpa, try m.derivative(term, variable));
436 return m.add(a.items);
437 }
438 if (f.k == 2) {
439 var terms: Expressions = .empty;
440 for (f.a, 0..) |_, i| {
441 var factors: Expressions = .empty;
442 for (f.a, 0..) |g, j|
443 try factors.append(m.gpa, if (i == j) try m.derivative(g, variable) else g);
444 try terms.append(m.gpa, try m.mul(factors.items));
445 }
446 return m.add(terms.items);
447 }
448 if (f.k == 3) {
449 const b = f.a[0];
450 const x = f.a[1];
451 if (!try m.containsVariable(x, variable)) return m.mul(&.{
452 x, try m.power(b, try m.add(&.{ x, m.minus })), try m.derivative(b, variable),
453 });
454 if (!try m.containsVariable(b, variable)) return m.mul(&.{
455 f, try m.function("ln", b), try m.derivative(x, variable),
456 });
457 return m.mul(&.{ f, try m.add(&.{
458 try m.mul(&.{ try m.derivative(x, variable), try m.function("ln", b) }),
459 try m.mul(&.{ x, try m.derivative(b, variable), try m.power(b, m.minus) }),
460 }) });
461 }
462 const a = f.a[0];
463 const factor = if (eql(f.s, "sin"))
464 try m.function("cos", a)
465 else if (eql(f.s, "cos"))
466 try m.mul(&.{ m.minus, try m.function("sin", a) })
467 else if (eql(f.s, "tan"))
468 try m.power(try m.function("cos", a), try m.num(-2, 1))
469 else if (eql(f.s, "exp"))
470 f
471 else
472 try m.power(a, m.minus);
473 return m.mul(&.{ factor, try m.derivative(a, variable) });
474 }

476 fn substitute(m: *Machine, f: *Expr, names: []const Text, replacements: []const *Expr) anyerror!*Expr {
477 if (f.k == 5) return f;
478 if (f.k == 0) {
479 for (names, 0..) |name, i| if (eql(f.s, name)) return replacements[i];
480 return f;
481 }
482 var args: Expressions = .empty;
483 for (f.a) |child| try args.append(m.gpa, try m.substitute(child, names, replacements));
484 if (f.k == 1) return m.add(args.items);
485 if (f.k == 2) return m.mul(args.items);
486 if (f.k == 3) return m.power(args.items[0], args.items[1]);
487 return m.function(f.s, args.items[0]);
488 }

490 fn distribute(m: *Machine, a: *Expr, b: *Expr) anyerror!*Expr {
491 const x: []const *Expr = if (a.k == 1) a.a else &.{a};
492 const y: []const *Expr = if (b.k == 1) b.a else &.{b};
493 var terms: Expressions = .empty;
494 for (x) |first| for (y) |second|
495 try terms.append(m.gpa, try m.mul(&.{ first, second }));
496 return m.add(terms.items);
497 }

499 fn expand(m: *Machine, f: *Expr) anyerror!*Expr {
500 if (f.k == 0 or f.k == 5) return f;
501 var args: Expressions = .empty;
502 for (f.a) |child| try args.append(m.gpa, try m.expand(child));
503 if (f.k == 1) return m.add(args.items);
504 if (f.k == 4) return m.function(f.s, args.items[0]);
505 if (f.k == 2) {
506 var result = args.items[0];
507 for (args.items[1..]) |b| result = try m.distribute(result, b);
508 return result;
509 }
510 const b = args.items[0];
511 const x = args.items[1];
512 if (b.k == 1 and isInteger(x) and x.q.p > 0) {
513 var result = b;
514 var k: i128 = 1;
515 while (k < x.q.p) : (k += 1) result = try m.distribute(result, b);
516 return result;
517 }
518 return m.power(b, x);
519 }

521 fn numericalValue(m: *Machine, f: *Expr) anyerror!f64 {
522 if (f.k == 5) return @as(f64, @floatFromInt(f.q.p)) / @as(f64, @floatFromInt(f.q.q));
523 if (f.k == 1 or f.k == 2) {
524 var result: f64 = if (f.k == 1) 0.0 else 1.0;
525 for (f.a) |child| {
526 const value = try m.numericalValue(child);
527 if (f.k == 1) result += value else result *= value;
528 }
529 return result;
530 }
531 if (f.k == 3) {
532 const u = try m.numericalValue(f.a[0]);
533 const z = try m.numericalValue(f.a[1]);
534 if (u < 0 and !(if (f.a[1].k == 5) isInteger(f.a[1]) else @floor(z) == z))
535 return m.fail("domain error", .{});
536 return if (equal(f.a[1], m.half)) @sqrt(u) else std.math.pow(f64, u, z);
537 }
538 const a = try m.numericalValue(f.a[0]);
539 if (eql(f.s, "sin")) return @sin(a);
540 if (eql(f.s, "cos")) return @cos(a);
541 if (eql(f.s, "tan")) return std.math.tan(a);
542 if (eql(f.s, "exp")) return @exp(a);
543 if (a <= 0) return m.fail("domain error", .{});
544 return @log(a);
545 }

568 fn quotient(m: *Machine, c: Rational, factors: []const *Expr) anyerror!Text {
569 var top: Strings = .empty;
570 var bottom: Strings = .empty;
571 const p: i128 = @intCast(@abs(c.p));
572 if (p != 1) try top.append(m.gpa, try m.formatted("{d}", .{p}));
573 if (c.q != 1) try bottom.append(m.gpa, try m.formatted("{d}", .{c.q}));
574 for (factors) |f| {
575 if (negative(f)) {
576 const exponent = try m.num(-f.a[1].q.p, f.a[1].q.q);
577 try bottom.append(m.gpa, if (equal(exponent, m.one))
578 try m.showFactor(f.a[0])
579 else
580 try m.showPower(f.a[0], exponent));
581 } else {
582 try top.append(m.gpa, try m.showFactor(f));
583 }
584 }
585 const numerator = if (top.items.len == 0) "1" else try m.join(top.items, "*");
586 const result = if (bottom.items.len == 0)
587 numerator
588 else if (bottom.items.len == 1)
589 try m.formatted("{s}/{s}", .{ numerator, bottom.items[0] })
590 else
591 try m.formatted("{s}/({s})", .{ numerator, try m.join(bottom.items, "*") });
592 return if (c.p < 0) m.formatted("-{s}", .{result}) else result;
593 }

595 fn show(m: *Machine, f: *Expr) anyerror!Text {
596 if (f.k == 5) return if (f.q.q == 1)
597 m.formatted("{d}", .{f.q.p})
598 else
599 m.formatted("{d}/{d}", .{ f.q.p, f.q.q });
600 if (f.k == 0) return f.s;
601 if (f.k == 4) return m.formatted("{s}({s})", .{ f.s, try m.show(f.a[0]) });
602 if (f.k == 3) return if (negative(f))
603 m.quotient(.{ .p = 1 }, &.{f})
604 else
605 m.showPower(f.a[0], f.a[1]);
606 if (f.k == 2) {
607 const info = coefficient(f);
608 return m.quotient(info.c, info.factors);
609 }
610 var text: Text = try m.show(f.a[0]);
611 for (f.a[1..]) |term| {
612 const next = try m.show(term);
613 text = if (std.mem.startsWith(u8, next, "-"))
614 try m.formatted("{s} - {s}", .{ text, next[1..] })
615 else
616 try m.formatted("{s} + {s}", .{ text, next });
617 }
618 return text;
619 }

621 fn tokenize(m: *Machine, text: Text) anyerror![]Text {
622 var tokens: Strings = .empty;
623 var i: usize = 0;
624 while (i < text.len) {
625 const c = text[i];
626 if (c == ' ') {
627 i += 1;
628 continue;
629 }
630 const start = i;
631 if (std.ascii.isDigit(c)) {
632 while (i < text.len and std.ascii.isDigit(text[i])) i += 1;
633 } else if (std.ascii.isLower(c)) {
634 while (i < text.len and (std.ascii.isLower(text[i]) or std.ascii.isDigit(text[i]))) i += 1;
635 } else if (std.ascii.isUpper(c)) {
636 while (i < text.len and (std.ascii.isUpper(text[i]) or std.ascii.isDigit(text[i]))) i += 1;
637 } else if (std.mem.indexOfScalar(u8, "-+*/^()", c) != null) {
638 i += 1;
639 } else {
640 return m.fail("invalid character '{c}'", .{c});
641 }
642 try tokens.append(m.gpa, text[start..i]);
643 }
644 return tokens.toOwnedSlice(m.gpa);
645 }
646
647 fn tree(m: *Machine, name: Text, args: []const *Tree) !*Tree {
648 const ptr = try m.gpa.create(Tree);
649 ptr.* = .{ .operation = .{ .name = name, .a = try m.gpa.dupe(*Tree, args) } };
650 return ptr;
651 }
652
653 fn leaf(m: *Machine, value: *Expr) !*Tree {
654 const ptr = try m.gpa.create(Tree);
655 ptr.* = .{ .value = value };
656 return ptr;
657 }
658
659 fn evaluateTree(m: *Machine, value: *Tree) anyerror!*Expr {
660 switch (value.*) {
661 .value => |f| return f,
662 .operation => |op| {
663 var a: Expressions = .empty;
664 for (op.a) |child| try a.append(m.gpa, try m.evaluateTree(child));
665 const x = a.items;
666 if (eql(op.name, "+")) return m.add(x);
667 if (eql(op.name, "-")) return m.add(&.{ x[0], try m.mul(&.{ m.minus, x[1] }) });
668 if (eql(op.name, "*")) return m.mul(x);
669 if (eql(op.name, "/")) return m.mul(&.{ x[0], try m.power(x[1], m.minus) });
670 if (eql(op.name, "^")) return m.power(x[0], x[1]);
671 if (eql(op.name, "neg")) return m.mul(&.{ m.minus, x[0] });
672 return m.function(op.name, x[0]);
673 },
674 }
675 }

677 const Parser = struct {
678 m: *Machine,
679 tokens: []Text,
680 position: usize = 0,
681
682 fn peek(p: *Parser) Text {
683 return if (p.position < p.tokens.len) p.tokens[p.position] else "";
684 }
685
686 fn unexpected(p: *Parser) anyerror {
687 return if (p.position == p.tokens.len)
688 p.m.fail("unexpected end of expression", .{})
689 else
690 p.m.fail("unexpected '{s}'", .{p.peek()});
691 }
692
693 fn accept(p: *Parser, s: Text) bool {
694 if (!eql(p.peek(), s)) return false;
695 p.position += 1;
696 return true;
697 }
698
699 fn expect(p: *Parser, s: Text) anyerror!void {
700 if (!p.accept(s)) return p.unexpected();
701 }
702
703 fn atom(p: *Parser) anyerror!*Tree {
704 if (p.accept("(")) {
705 const f = try p.expression();
706 try p.expect(")");
707 return f;
708 }
709 const word = p.peek();
710 if (word.len == 0) return p.unexpected();
711 if (std.ascii.isDigit(word[0])) {
712 p.position += 1;
713 return p.m.leaf(try p.m.num(try std.fmt.parseInt(i128, word, 10), 1));
714 }
715 if (std.ascii.isUpper(word[0])) {
716 p.position += 1;
717 const definition = p.m.names.get(word) orelse
718 return p.m.fail("undefined name '{s}'", .{word});
719 return p.m.leaf(definition);
720 }
721 if (std.ascii.isLower(word[0])) {
722 p.position += 1;
723 if (!among(word, &.{ "sin", "cos", "tan", "exp", "ln", "sqrt" })) {
724 return p.m.leaf(try p.m.node(0, &.{}, word));
725 }
726 try p.expect("(");
727 const a = try p.expression();
728 try p.expect(")");
729 return p.m.tree(word, &.{a});
730 }
731 return p.unexpected();
732 }
733
734 fn unary(p: *Parser) anyerror!*Tree {
735 if (p.accept("-")) return p.m.tree("neg", &.{try p.unary()});
736 const a = try p.atom();
737 return if (p.accept("^")) p.m.tree("^", &.{ a, try p.unary() }) else a;
738 }
739
740 fn term(p: *Parser) anyerror!*Tree {
741 var a = try p.unary();
742 while (eql(p.peek(), "*") or eql(p.peek(), "/")) {
743 const op = p.peek();
744 p.position += 1;
745 a = try p.m.tree(op, &.{ a, try p.unary() });
746 }
747 return a;
748 }
749
750 fn expression(p: *Parser) anyerror!*Tree {
751 var a = try p.term();
752 while (eql(p.peek(), "+") or eql(p.peek(), "-")) {
753 const op = p.peek();
754 p.position += 1;
755 a = try p.m.tree(op, &.{ a, try p.term() });
756 }
757 return a;
758 }
759 };
760
761 fn parse(m: *Machine, expression: Text) anyerror!*Expr {
762 var reader = Parser{ .m = m, .tokens = try m.tokenize(expression) };
763 const result = try reader.expression();
764 if (reader.position != reader.tokens.len) return reader.unexpected();
765 return m.evaluateTree(result);
766 }

768 fn validVariable(name: Text) bool {
769 if (name.len == 0 or !std.ascii.isLower(name[0]) or
770 among(name, &.{ "sin", "cos", "tan", "exp", "ln", "sqrt" })) return false;
771 for (name[1..]) |c| if (!std.ascii.isLower(c) and !std.ascii.isDigit(c)) return false;
772 return true;
773 }
774
775 fn validName(name: Text) bool {
776 if (name.len == 0 or !std.ascii.isUpper(name[0])) return false;
777 for (name[1..]) |c| if (!std.ascii.isUpper(c) and !std.ascii.isDigit(c)) return false;
778 return true;
779 }
780
781 fn literal(m: *Machine, text: Text) anyerror!*Expr {
782 if (text.len == 0) return m.fail("invalid number '{s}'", .{text});
783 const start: usize = if (text[0] == '-') 1 else 0;
784 if (start == text.len) return m.fail("invalid number '{s}'", .{text});
785 var slash: ?usize = null;
786 for (text[start..], start..) |c, i| {
787 if (c == '/') {
788 if (slash != null) return m.fail("invalid number '{s}'", .{text});
789 slash = i;
790 } else if (!std.ascii.isDigit(c)) return m.fail("invalid number '{s}'", .{text});
791 }
792 const numerator = text[0 .. slash orelse text.len];
793 const denominator = if (slash) |i| text[i + 1 ..] else "1";
794 if (numerator.len == start or denominator.len == 0) return m.fail("invalid number '{s}'", .{text});
795 if (denominator[0] == '/' or numerator[numerator.len - 1] == '/') return m.fail("invalid number '{s}'", .{text});
796 return m.num(try std.fmt.parseInt(i128, numerator, 10), try std.fmt.parseInt(i128, denominator, 10));
797 }
798
799 fn checkedVariable(m: *Machine, name: Text) anyerror!Text {
800 if (!validVariable(name)) return m.fail("invalid variable '{s}'", .{name});
801 return name;
802 }
803
804 fn order(m: *Machine, text: Text) anyerror!usize {
805 if (text.len != 1 or text[0] < '1' or text[0] > '8') return m.fail("invalid order '{s}'", .{text});
806 return text[0] - '0';
807 }

809 fn command(m: *Machine, line_text: Text) anyerror!void {
810 const split = std.mem.indexOfScalar(u8, line_text, ' ');
811 const command_name = if (split) |p| line_text[0..p] else line_text;
812 const arg = if (split) |p| line_text[p + 1 ..] else "";
813 if (eql(command_name, "LET")) {
814 const idx = std.mem.indexOfScalar(u8, arg, '=') orelse
815 return m.fail("wrong number of arguments", .{});
816 const name = std.mem.trim(u8, arg[0..idx], " ");
817 if (!validName(name)) return m.fail("invalid name '{s}'", .{name});
818 const expr = try m.parse(std.mem.trim(u8, arg[idx + 1 ..], " "));
819 try m.names.put(m.gpa, name, expr);
820 try m.line(try m.formatted("{s} = {s}", .{ name, try m.show(expr) }));
821 return;
822 }
823
824 const kind: Text = if (among(command_name, &.{ "SIMPLIFY", "EXPAND", "GRAD", "HESSIAN" }))
825 "e"
826 else if (eql(command_name, "DIFF")) "ev" else if (eql(command_name, "DIFFN")) "evn" else if (eql(command_name, "IMPLICIT")) "evv" else if (eql(command_name, "SUBST")) "eve" else if (eql(command_name, "EVAL")) "ea" else if (among(command_name, &.{ "TANGENT", "LIMIT" })) "evc" else if (eql(command_name, "TAYLOR")) "evcn" else return m.fail("unknown command {s}", .{command_name});
827
828 var fields: Strings = .empty;
829 if (split != null) {
830 var iterator = std.mem.splitScalar(u8, arg, ';');
831 while (iterator.next()) |part| try fields.append(m.gpa, std.mem.trim(u8, part, " "));
832 }
833 if (eql(command_name, "EVAL")) {
834 if (fields.items.len != 1 and fields.items.len != 2) return m.fail("wrong number of arguments", .{});
835 } else if (fields.items.len != kind.len) return m.fail("wrong number of arguments", .{});
836
837 var exprs: Expressions = .empty;
838 var strings: Strings = .empty;
839 var orders: std.ArrayList(usize) = .empty;
840 for (fields.items, 0..) |field, i| {
841 const field_type = kind[i];
842 if (field_type == 'e') {
843 try exprs.append(m.gpa, try m.parse(field));
844 } else if (field_type == 'v') {
845 try strings.append(m.gpa, try m.checkedVariable(field));
846 } else if (field_type == 'n') {
847 try orders.append(m.gpa, try m.order(field));
848 } else if (field_type == 'c') {
849 try exprs.append(m.gpa, try m.literal(field));
850 }
851 }
852 var f = exprs.items[0];
853 const variable = if (strings.items.len > 0) strings.items[0] else "";
854 if (eql(command_name, "SIMPLIFY")) {
855 try m.line(try m.formatted("simplified: {s}", .{try m.show(f)}));
856 } else if (eql(command_name, "EXPAND")) {
857 try m.line(try m.formatted("expanded: {s}", .{try m.show(try m.expand(f))}));
858 } else if (eql(command_name, "DIFF")) {
859 try m.line(try m.formatted("d/d{s} = {s}", .{ variable, try m.show(try m.derivative(f, variable)) }));
860 } else if (eql(command_name, "DIFFN")) {
861 const n = orders.items[0];
862 for (1..n + 1) |k| {
863 f = try m.derivative(f, variable);
864 try m.line(try m.formatted("d^{d}/d{s}^{d} = {s}", .{ k, variable, k, try m.show(f) }));
865 }
866 } else if (eql(command_name, "GRAD") or eql(command_name, "HESSIAN")) {
867 const vars = try m.variables(f);
868 if (vars.len == 0) {
869 try m.line("no variables");
870 return;
871 }
872 for (vars, 0..) |v, i| {
873 const first = try m.derivative(f, v);
874 if (eql(command_name, "GRAD")) {
875 try m.line(try m.formatted("d/d{s} = {s}", .{ v, try m.show(first) }));
876 } else {
877 for (vars[i..]) |w| {
878 try m.line(try m.formatted("d2/d{s}d{s} = {s}", .{
879 v, w, try m.show(try m.derivative(first, w)),
880 }));
881 }
882 }
883 }
884 } else if (eql(command_name, "IMPLICIT")) {
885 const other = strings.items[1];
886 if (eql(variable, other)) return m.fail("same variable", .{});
887 const value = try m.mul(&.{
888 m.minus, try m.derivative(f, variable),
889 try m.power(try m.derivative(f, other), m.minus),
890 });
891 try m.line(try m.formatted("d{s}/d{s} = {s}", .{ other, variable, try m.show(value) }));
892 } else if (eql(command_name, "SUBST")) {
893 const result = try m.substitute(f, &.{variable}, &.{exprs.items[1]});
894 try m.line(try m.formatted("substituted: {s}", .{try m.show(result)}));
895 } else if (eql(command_name, "EVAL")) {
896 var assignments: Strings = .empty;
897 var values: Expressions = .empty;
898 if (fields.items.len == 2) {
899 var words = std.mem.tokenizeScalar(u8, fields.items[1], ' ');
900 while (words.next()) |word| {
901 const equal_pos = std.mem.indexOfScalar(u8, word, '=') orelse
902 return m.fail("invalid assignment '{s}'", .{word});
903 const name = word[0..equal_pos];
904 const lit = word[equal_pos + 1 ..];
905 if (!validVariable(name) or lit.len == 0) return m.fail("invalid assignment '{s}'", .{word});
906 if (!validLiteral(lit)) return m.fail("invalid assignment '{s}'", .{word});
907 const value = try m.literal(lit);
908 if (among(name, assignments.items)) return m.fail("duplicate variable {s}", .{name});
909 try assignments.append(m.gpa, name);
910 try values.append(m.gpa, value);
911 }
912 }
913 for (try m.variables(f)) |name| {
914 if (!among(name, assignments.items)) return m.fail("no value for {s}", .{name});
915 }
916 const result = try m.substitute(f, assignments.items, values.items);
917 if (result.k == 5) {
918 try m.line(try m.formatted("exact: {s}", .{try m.show(result)}));
919 } else {
920 const value = try m.numericalValue(result);
921 const rounded = @floor(@abs(value) * 1000000.0 + 0.5);
922 const scaled: i128 = @intFromFloat(rounded);
923 const whole = @divTrunc(scaled, 1000000);
924 const decimals = @mod(scaled, 1000000);
925 try m.line(try m.formatted("approx: {s}{d}.{d:0>6}", .{
926 if (value < 0) @as(Text, "-") else "", whole, decimals,
927 }));
928 }
929 } else {
930 const point = exprs.items[1];
931 if (eql(command_name, "TANGENT")) {
932 const slope = try m.substitute(try m.derivative(f, variable), &.{variable}, &.{point});
933 const y = try m.substitute(f, &.{variable}, &.{point});
934 const v = try m.node(0, &.{}, variable);
935 const neg = try m.num(-point.q.p, point.q.q);
936 const tangent = try m.add(&.{
937 try m.mul(&.{ slope, v }), y, try m.mul(&.{ neg, slope }),
938 });
939 try m.line(try m.formatted("tangent: y = {s}", .{try m.show(tangent)}));
940 if (equal(slope, m.zero)) {
941 try m.line(try m.formatted("normal: {s} = {s}", .{ variable, try m.show(point) }));
942 } else {
943 const reciprocal = try m.power(slope, m.minus);
944 const normal = try m.add(&.{
945 try m.mul(&.{ m.minus, reciprocal, v }), y,
946 try m.mul(&.{ point, reciprocal }),
947 });
948 try m.line(try m.formatted("normal: y = {s}", .{try m.show(normal)}));
949 }
950 } else if (eql(command_name, "LIMIT")) {
951 var tops: Expressions = .empty;
952 var bottoms: Expressions = .empty;
953 const parts = if (f.k == 2) f.a else &.{f};
954 for (parts) |part| {
955 if (negative(part)) try bottoms.append(m.gpa, try m.power(
956 part.a[0],
957 try m.num(-part.a[1].q.p, part.a[1].q.q),
958 )) else try tops.append(m.gpa, part);
959 }
960 var numerator = try m.mul(tops.items);
961 var denominator = try m.mul(bottoms.items);
962 for (0..6) |k| {
963 const a = try m.substitute(numerator, &.{variable}, &.{point});
964 const b = try m.substitute(denominator, &.{variable}, &.{point});
965 if (!equal(b, m.zero)) {
966 const result = try m.mul(&.{ a, try m.power(b, m.minus) });
967 try m.line(try m.formatted("limit: {s}", .{try m.show(result)}));
968 return;
969 }
970 if (!equal(a, m.zero)) {
971 try m.line("limit: unbounded");
972 return;
973 }
974 if (k == 5) {
975 try m.line("limit: unknown");
976 return;
977 }
978 numerator = try m.derivative(numerator, variable);
979 denominator = try m.derivative(denominator, variable);
980 }
981 } else if (eql(command_name, "TAYLOR")) {
982 const n = orders.items[0];
983 var derivatives: Expressions = .empty;
984 try derivatives.append(m.gpa, f);
985 for (0..n) |_| {
986 try derivatives.append(m.gpa, try m.derivative(derivatives.items[derivatives.items.len - 1], variable));
987 }
988 const negative_point = try m.num(-point.q.p, point.q.q);
989 const offset = try m.add(&.{ try m.node(0, &.{}, variable), negative_point });
990 var factorial: i128 = 1;
991 var terms: Expressions = .empty;
992 for (0..n + 1) |k| {
993 if (k > 0) factorial *= @intCast(k);
994 const value = try m.substitute(derivatives.items[k], &.{variable}, &.{point});
995 const coefficient_value = try m.mul(&.{ value, try m.num(1, factorial) });
996 try terms.append(m.gpa, try m.mul(&.{
997 coefficient_value, try m.power(offset, try m.num(@intCast(k), 1)),
998 }));
999 }
1000 try m.line(try m.formatted("taylor: {s}", .{try m.show(try m.expand(try m.add(terms.items)))}));
1001 }
1002 }
1003 }

1005 fn validLiteral(text: Text) bool {
1006 if (text.len == 0) return false;
1007 const first: usize = if (text[0] == '-') 1 else 0;
1008 if (first == text.len) return false;
1009 var slash: ?usize = null;
1010 for (text[first..], first..) |c, i| {
1011 if (c == '/') {
1012 if (slash != null) return false;
1013 slash = i;
1014 } else if (!std.ascii.isDigit(c)) return false;
1015 }
1016 return if (slash) |i| i > first and i + 1 < text.len else true;
1017 }
1018};
