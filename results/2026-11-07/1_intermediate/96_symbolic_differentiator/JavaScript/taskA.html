10function rational(p, q = 1) {

55function termCompare(a, b) {
56 const x = coefficient(a)[1],
57 y = coefficient(b)[1];
58 const bases = [...x, ...y]
59 .map(base)
60 .sort(compare)
61 .filter((f, i, a) => !i || !eq(f, a[i - 1]));

64 v = y.find((t) => eq(base(t), f));

66 if (c) return c;

70function add(...operands) {
71 const terms = [];
72 for (const f of operands) {
73 if (f.k === 1) terms.push(...f.a);
74 else if (f.k === 2 && f.a.length === 2 && f.a[0].k === 5 && f.a[1].k === 1)
75 for (const t of f.a[1].a) terms.push(mul(f.a[0], t));
76 else terms.push(f);
77 }
78 const groups = [];
79 for (const f of terms) {
80 const [c, m] = coefficient(f);
81 const g = groups.find((g) => g.m.length === m.length && g.m.every((t, i) => eq(t, m[i])));
82 if (g) g.c = plus(g.c, c);
83 else groups.push({ c, m });
84 }
85 const built = groups
86 .filter((g) => g.c[0])
87 .map((g) => product(g.c, g.m))
88 .sort(termCompare);
89 return !built.length ? zero : built.length === 1 ? built[0] : node(1, built);

91function mul(...operands) {
92 const factors = operands.flatMap((f) => (f.k === 2 ? f.a : [f]));
93 let c = [1, 1];
94 const groups = [];
95 for (const f of factors) {
96 if (f.k === 5) c = times(c, f.q);
97 else {
98 const g = groups.find((g) => eq(g.b, base(f)));
99 if (g) g.a.push(f);
100 else groups.push({ b: base(f), a: [f] });
101 }
102 }
103 if (!c[0]) return zero;
104 const r = groups.map((g) =>
105 g.a.length === 1 ? g.a[0] : power(g.b, add(...g.a.map(exponent))),
106 );
107 if (
108 r.some((f) => f.k === 5 || f.k === 2) ||
109 r.some((f, i) => r.slice(0, i).some((g) => eq(base(f), base(g))))
110 )
111 return mul(num(...c), ...r);
112 return product(
113 c,
114 r.sort((a, b) => compare(base(a), base(b))),
115 );
116}
117function root(v, d) {
118 if (v === 1) return 1;
119 if (d > 20) return null;
120 const r = Math.round(v ** (1 / d));
121 for (let i = Math.max(0, r - 1); i <= r + 1; i++) if (i ** d === v) return i;
122 return null;
123}
124function rationalPower(b, n) {
125 if (n < 0) return rationalPower([b[1], b[0]], -n);
126 let r = [1, 1];
127 while (n) {
128 if (n % 2) r = times(r, b);
129 n = Math.floor(n / 2);
130 if (n) b = times(b, b);
131 }
132 return num(...r);

134function power(b, x) {
135 if (eq(x, zero)) return one;
136 if (eq(x, one)) return b;
137 if (b.k === 5) {
138 if (eq(b, one)) return one;
139 if (x.k !== 5) return node(3, [b, x]);
140 if (!b.q[0]) {
141 if (x.q[0] > 0) return zero;
142 fail("division by zero");
143 }
144 if (integer(x)) return rationalPower(b.q, x.q[0]);
145 if (b.q[0] > 0) {
146 const s = root(b.q[0], x.q[1]),
147 t = root(b.q[1], x.q[1]);
148 if (s !== null && t !== null) return rationalPower([s, t], x.q[0]);
149 }
150 } else if (integer(x)) {
151 if (b.k === 3) return power(b.a[0], mul(b.a[1], x));
152 if (b.k === 2) return mul(...b.a.map((f) => power(f, x)));
153 }
154 return node(3, [b, x]);
155}
156function fn(s, a) {
157 if (s === "sqrt") return power(a, half);
158 if (eq(a, zero) && ["sin", "tan", "cos", "exp"].includes(s))
159 return ["sin", "tan"].includes(s) ? zero : one;
160 if (s === "ln" && eq(a, one)) return zero;
161 if (["sin", "cos", "tan"].includes(s) && coefficient(a)[0][0] < 0) {
162 const t = fn(s, mul(minus, a));
163 return s === "cos" ? t : mul(minus, t);
164 }
165 if (a.k === 4 && ((s === "ln" && a.s === "exp") || (s === "exp" && a.s === "ln")))
166 return a.a[0];
167 return node(4, [a], s);
168}
169function variables(f) {
170 const r = new Set(f.k === 0 ? [f.s] : []);
171 for (const a of f.a) for (const s of variables(a)) r.add(s);
172 return [...r].sort();
173}
174function derivative(f, v) {
175 if (f.k === 5) return zero;
176 if (f.k === 0) return f.s === v ? one : zero;
177 if (f.k === 1) return add(...f.a.map((t) => derivative(t, v)));
178 if (f.k === 2)
179 return add(
180 ...f.a.map((t, i) => mul(...f.a.map((g, j) => (i === j ? derivative(g, v) : g)))),
181 );
182 if (f.k === 3) {
183 const [b, x] = f.a;
184 if (!variables(x).includes(v)) return mul(x, power(b, add(x, minus)), derivative(b, v));
185 if (!variables(b).includes(v)) return mul(f, fn("ln", b), derivative(x, v));
186 return mul(
187 f,
188 add(mul(derivative(x, v), fn("ln", b)), mul(x, derivative(b, v), power(b, minus))),
189 );
190 }
191 const a = f.a[0];
192 const h =
193 f.s === "sin"
194 ? fn("cos", a)
195 : f.s === "cos"
196 ? mul(minus, fn("sin", a))
197 : f.s === "tan"
198 ? power(fn("cos", a), num(-2))
199 : f.s === "exp"
200 ? f
201 : power(a, minus);

203}

256function quotient(c, a) {

258 q = c[1],

285function parse(text, names) {
286 const tokens = [];
287 const re = /[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*/^()]| +|./g;
288 for (const t of text.match(re) || []) {
289 if (t[0] === " ") continue;
290 if (!/^[a-zA-Z0-9+*/^()-]+$/.test(t)) fail("invalid character '" + t + "'");
291 tokens.push(t);
292 }
293 let p = 0;
294 const peek = () => tokens[p] || "";
295 const unexpected = () =>
296 fail(p === tokens.length ? "unexpected end of expression" : "unexpected '" + peek() + "'");
297 function accept(s) {
298 if (peek() === s) {
299 p++;
300 return true;
301 }
302 return false;
303 }
304 function expect(s) {
305 if (!accept(s)) unexpected();
306 }
307 function atomTree() {
308 const t = peek();
309 if (!t) unexpected();
310 if (accept("(")) {
311 const a = expression();
312 expect(")");
313 return a;
314 }
315 if (!/^[a-zA-Z0-9]/.test(t)) unexpected();
316 p++;
317 if (/^[0-9]/.test(t)) return { f: num(Number(t)) };
318 if (/^[A-Z]/.test(t)) {
319 if (!names.has(t)) fail("undefined name '" + t + "'");
320 return { f: names.get(t) };
321 }
322 if (!functions.has(t)) return { f: node(0, [], t) };
323 expect("(");
324 const a = expression();
325 expect(")");
326 return { op: t, a: [a] };
327 }
328 function unary() {
329 if (accept("-")) return { op: "neg", a: [unary()] };
330 const a = atomTree();
331 return accept("^") ? { op: "^", a: [a, unary()] } : a;
332 }
333 function term() {
334 let a = unary();
335 while (peek() === "*" || peek() === "/") {
336 const op = tokens[p++];
337 a = { op, a: [a, unary()] };
338 }
339 return a;
340 }
341 function expression() {
342 let a = term();
343 while (peek() === "+" || peek() === "-") {
344 const op = tokens[p++];
345 a = { op, a: [a, term()] };
346 }
347 return a;
348 }
349 function evaluate(t) {
350 if (t.f) return t.f;
351 const a = t.a.map(evaluate);
352 switch (t.op) {
353 case "+":
354 return add(...a);
355 case "-":
356 return add(a[0], mul(minus, a[1]));
357 case "*":
358 return mul(...a);
359 case "/":
360 return mul(a[0], power(a[1], minus));
361 case "^":
362 return power(...a);
363 case "neg":
364 return mul(minus, a[0]);
365 default:
366 return fn(t.op, a[0]);
367 }
368 }
369 const tree = expression();
370 if (p < tokens.length) unexpected();
371 return evaluate(tree);
372}
