4data class Rational(val p: Long, val q: Long) : Comparable<Rational> {
5 operator fun plus(b: Rational) = rational(p * b.q + b.p * q, q * b.q)
6
7 operator fun times(b: Rational) = rational(p * b.p, q * b.q)
8
9 operator fun unaryMinus() = rational(-p, q)
10
11 override fun compareTo(b: Rational) = (p * b.q).compareTo(b.p * q)
12}
13
14fun rational(n: Long, d: Long = 1): Rational {
15 if (d == 0L) error("division by zero")
16 var p = n
17 var q = d
18 if (q < 0) {
19 p = -p
20 q = -q
21 }
22 var a = abs(p)
23 var b = q
24 while (b != 0L) {
25 val t = a % b
26 a = b
27 b = t
28 }
29 return Rational(p / a, q / a)
30}

32data class Expr(val kind: Int, val q: Rational, val name: String, val args: List<Expr>)
33
34data class Parts(val coefficient: Rational, val factors: List<Expr>)
35
36data class Syntax(val op: String, val value: Expr? = null, val args: List<Syntax> = emptyList())
37
38fun number(q: Rational) = Expr(5, q, "", emptyList())
39
40fun num(p: Long, q: Long = 1) = number(rational(p, q))
41
42fun num(p: Int) = num(p.toLong())
43
44fun node(k: Int, a: List<Expr> = emptyList(), s: String = "") = Expr(k, rational(0), s, a.toList())

46val zero = num(0)
47val one = num(1)
48val minus = num(-1)
49val half = num(1L, 2)
50val functions = setOf("sin", "cos", "tan", "exp", "ln", "sqrt")
51
52fun compare(a: Expr, b: Expr): Int {
53 if (a.kind != b.kind) return a.kind.compareTo(b.kind)
54 if (a.kind == 5) return a.q.compareTo(b.q)
55 val c = a.name.compareTo(b.name)
56 if (c != 0) return c
57 for (i in 0 until min(a.args.size, b.args.size)) {
58 val c = compare(a.args[i], b.args[i])
59 if (c != 0) return c
60 }
61 return a.args.size.compareTo(b.args.size)
62}
63
64fun integer(f: Expr) = f.kind == 5 && f.q.q == 1L
65
66fun negative(f: Expr) = f.kind == 3 && f.args[1].kind == 5 && f.args[1].q.p < 0
67
68fun base(f: Expr) = if (f.kind == 3) f.args[0] else f
69
70fun exponent(f: Expr) = if (f.kind == 3) f.args[1] else one
71
72fun coefficient(f: Expr): Parts =
73 when {
74 f.kind == 5 -> Parts(f.q, emptyList())
75 f.kind == 2 && f.args[0].kind == 5 -> Parts(f.args[0].q, f.args.drop(1))
76 f.kind == 2 -> Parts(rational(1), f.args)
77 else -> Parts(rational(1), listOf(f))
78 }
79
80fun product(c: Rational, a: List<Expr>): Expr {
81 if (a.isEmpty()) return number(c)
82 val r = if (c != rational(1)) listOf(number(c)) + a else a
83 return if (r.size == 1) r[0] else node(2, r)
84}

86fun exponentCompare(a: Expr, b: Expr): Int {
87 if (a.kind == 5 && b.kind != 5) return -1
88 if (a.kind != 5 && b.kind == 5) return 1
89 return compare(a, b)
90}

92fun termCompare(a: Expr, b: Expr): Int {
93 val x = coefficient(a).factors.associate { base(it) to exponent(it) }
94 val y = coefficient(b).factors.associate { base(it) to exponent(it) }
95 for (b in (x.keys + y.keys).sortedWith(::compare)) {
96 val c = exponentCompare(y[b] ?: zero, x[b] ?: zero)
97 if (c != 0) return c
98 }
99 return 0
100}

102fun add(vararg a: Expr) = add(a.toList())
103
104fun add(operands: List<Expr>): Expr {
105 val terms = operands.flatMap { f ->
106 when {
107 f.kind == 1 -> f.args
108 f.kind == 2 && f.args.size == 2 && f.args[0].kind == 5 && f.args[1].kind == 1 ->
109 f.args[1].args.map { mul(f.args[0], it) }
110 else -> listOf(f)
111 }
112 }
113 val groups = linkedMapOf<List<Expr>, Rational>()
114 for (f in terms) {
115 val (c, m) = coefficient(f)
116 groups[m] = (groups[m] ?: rational(0)) + c
117 }
118 val built =
119 groups
120 .mapNotNull { (m, c) -> if (c.p == 0L) null else product(c, m) }
121 .sortedWith(::termCompare)
122 return if (built.isEmpty()) zero else if (built.size == 1) built[0] else node(1, built)
123}
124
125fun mul(vararg a: Expr) = mul(a.toList())
126
127fun mul(operands: List<Expr>): Expr {
128 var c = rational(1)
129 val groups = linkedMapOf<Expr, MutableList<Expr>>()
130 for (f in operands.flatMap { if (it.kind == 2) it.args else listOf(it) }) {
131 if (f.kind == 5) c *= f.q else groups.getOrPut(base(f)) { mutableListOf() }.add(f)
132 }
133 if (c.p == 0L) return zero
134 val r = groups.map { (b, g) -> if (g.size == 1) g[0] else power(b, add(g.map(::exponent))) }
135 if (r.any { it.kind == 2 || it.kind == 5 } || r.map(::base).distinct().size < r.size)
136 return mul(listOf(number(c)) + r)
137 return product(c, r.sortedWith { a, b -> compare(base(a), base(b)) })
138}

140fun root(v: Long, d: Long): Long? {
141 if (v == 1L) return 1
142 if (d > 20) return null
143 val r = v.toDouble().pow(1.0 / d).roundToLong()
144 for (s in max(0, r - 1)..r + 1) {
145 var a = 1L
146 var i = 0L
147 while (i < d && a <= v) {
148 a *= s
149 i++
150 }
151 if (a == v) return s
152 }
153 return null
154}

156fun rationalPower(value: Rational, exponent: Long): Expr {
157 var b = value
158 var n = exponent
159 if (n < 0) {
160 b = rational(b.q, b.p)
161 n = -n
162 }
163 var r = rational(1)
164 while (n > 0) {
165 if (n % 2 != 0L) r *= b
166 n /= 2
167 if (n > 0) b *= b
168 }
169 return number(r)
170}

172fun power(b: Expr, x: Expr): Expr {
173 if (x == zero) return one
174 if (x == one) return b
175 if (b.kind == 5) {
176 if (b == one) return one
177 if (x.kind != 5) return node(3, listOf(b, x))
178 if (b.q.p == 0L) {
179 if (x.q.p > 0) return zero
180 error("division by zero")
181 }
182 if (integer(x)) return rationalPower(b.q, x.q.p)
183 if (b.q.p > 0) {
184 val s = root(b.q.p, x.q.q)
185 val t = root(b.q.q, x.q.q)
186 if (s != null && t != null) return rationalPower(rational(s, t), x.q.p)
187 }
188 } else if (integer(x)) {
189 if (b.kind == 3) return power(b.args[0], mul(b.args[1], x))
190 if (b.kind == 2) return mul(b.args.map { power(it, x) })
191 }
192 return node(3, listOf(b, x))
193}

195fun fn(s: String, a: Expr): Expr {
196 if (s == "sqrt") return power(a, half)
197 if (a == zero && s in setOf("sin", "tan")) return zero
198 if (a == zero && s in setOf("cos", "exp")) return one
199 if (a == one && s == "ln") return zero
200 if (s in setOf("sin", "cos", "tan") && coefficient(a).coefficient.p < 0) {
201 val r = fn(s, mul(minus, a))
202 return if (s == "cos") r else mul(minus, r)
203 }
204 if (a.kind == 4 && ((s == "exp" && a.name == "ln") || (s == "ln" && a.name == "exp")))
205 return a.args[0]
206 return node(4, listOf(a), s)
207}

209fun variables(f: Expr): Set<String> =
210 if (f.kind == 0) setOf(f.name) else f.args.flatMap { variables(it) }.toSortedSet()
211
212fun derivative(f: Expr, v: String): Expr {
213 if (f.kind == 5) return zero
214 if (f.kind == 0) return if (f.name == v) one else zero
215 if (f.kind == 1) return add(f.args.map { derivative(it, v) })
216 if (f.kind == 2)
217 return add(
218 f.args.indices.map { i ->
219 mul(f.args.mapIndexed { j, t -> if (i == j) derivative(t, v) else t })
220 },
221 )
222 if (f.kind == 3) {
223 val (b, x) = f.args
224 if (v !in variables(x)) return mul(x, power(b, add(x, minus)), derivative(b, v))
225 if (v !in variables(b)) return mul(f, fn("ln", b), derivative(x, v))
226 return mul(
227 f,
228 add(mul(derivative(x, v), fn("ln", b)), mul(x, derivative(b, v), power(b, minus))),
229 )
230 }
231 val a = f.args[0]
232 val h =
233 when (f.name) {
234 "sin" -> fn("cos", a)
235 "cos" -> mul(minus, fn("sin", a))
236 "tan" -> power(fn("cos", a), num(-2))
237 "exp" -> f
238 else -> power(a, minus)
239 }
240 return mul(h, derivative(a, v))
241}

243fun rebuild(f: Expr, a: List<Expr>): Expr =
244 when (f.kind) {
245 1 -> add(a)
246 2 -> mul(a)
247 3 -> power(a[0], a[1])
248 else -> fn(f.name, a[0])
249 }

251fun substitute(f: Expr, m: Map<String, Expr>): Expr =
252 when (f.kind) {
253 0 -> m[f.name] ?: f
254 5 -> f
255 else -> rebuild(f, f.args.map { substitute(it, m) })
256 }

258fun terms(f: Expr) = if (f.kind == 1) f.args else listOf(f)
259
260fun distribute(a: Expr, b: Expr) = add(terms(a).flatMap { x -> terms(b).map { y -> mul(x, y) } })
261
262fun expand(f: Expr): Expr {
263 if (f.kind == 0 || f.kind == 5) return f
264 val a = f.args.map(::expand)
265 if (f.kind == 2) return a.reduce(::distribute)
266 if (f.kind == 3 && a[0].kind == 1 && integer(a[1]) && a[1].q.p > 0) {
267 var r = a[0]
268 for (i in 1 until a[1].q.p) r = distribute(r, a[0])
269 return r
270 }
271 return rebuild(f, a)
272}

274fun numeric(f: Expr): Double {
275 if (f.kind == 5) return f.q.p.toDouble() / f.q.q
276 if (f.kind == 1) return f.args.fold(0.0) { r, t -> r + numeric(t) }
277 if (f.kind == 2) return f.args.fold(1.0) { r, t -> r * numeric(t) }
278 if (f.kind == 3) {
279 val u = numeric(f.args[0])
280 val z = numeric(f.args[1])
281 val integral = if (f.args[1].kind == 5) integer(f.args[1]) else z == floor(z)
282 if (u < 0 && !integral) error("domain error")
283 return if (f.args[1] == half) sqrt(u) else u.pow(z)
284 }
285 val a = numeric(f.args[0])
286 if (f.name == "ln" && a <= 0) error("domain error")
287 return when (f.name) {
288 "sin" -> sin(a)
289 "cos" -> cos(a)
290 "tan" -> tan(a)
291 "exp" -> exp(a)
292 else -> ln(a)
293 }
294}

296fun atomic(f: Expr) =
297 f.kind == 0 || f.kind == 4 || (integer(f) && f.q.p >= 0) || (f.kind == 3 && f.args[1] == half)
298
299fun atom(f: Expr) = if (atomic(f)) show(f) else "(${show(f)})"
300
301fun showPower(b: Expr, x: Expr) = if (x == half) "sqrt(${show(b)})" else "${atom(b)}^${atom(x)}"
302
303fun factor(f: Expr) =
304 if (f.kind == 1) "(${show(f)})"
305 else if (f.kind == 3) showPower(f.args[0], f.args[1]) else show(f)

307fun quotient(c: Rational, a: List<Expr>): String {
308 val top = mutableListOf<String>()
309 val bottom = mutableListOf<String>()
310 if (abs(c.p) != 1L) top.add(abs(c.p).toString())
311 if (c.q != 1L) bottom.add(c.q.toString())
312 for (f in a) {
313 if (negative(f)) {
314 val r = number(-f.args[1].q)
315 bottom.add(if (r == one) factor(f.args[0]) else showPower(f.args[0], r))
316 } else top.add(factor(f))
317 }
318 var s = if (top.isEmpty()) "1" else top.joinToString("*")
319 if (bottom.size == 1) s += "/" + bottom[0]
320 else if (bottom.isNotEmpty()) s += "/(" + bottom.joinToString("*") + ")"
321 return (if (c.p < 0) "-" else "") + s
322}

324fun show(f: Expr): String {
325 if (f.kind == 5) return f.q.p.toString() + (if (f.q.q == 1L) "" else "/${f.q.q}")
326 if (f.kind == 0) return f.name
327 if (f.kind == 4) return "${f.name}(${show(f.args[0])})"
328 if (f.kind == 3)
329 return if (negative(f)) quotient(rational(1), listOf(f))
330 else showPower(f.args[0], f.args[1])
331 if (f.kind == 2) {
332 val (c, a) = coefficient(f)
333 return quotient(c, a)
334 }
335 return f.args.map(::show).reduce { a, b ->
336 a + if (b.startsWith('-')) " - " + b.substring(1) else " + " + b
337 }
338}

340class Parser(text: String, val names: Map<String, Expr>) {
341 private val tokens = mutableListOf<String>()
342 private var position = 0
343
344 init {
345 for (m in Regex("[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*/^()]| +|.").findAll(text)) {
346 val t = m.value
347 if (t.startsWith(' ')) continue
348 if (!Regex("[a-zA-Z0-9+*/^()-]+").matches(t)) error("invalid character '$t'")
349 tokens.add(t)
350 }
351 }
352
353 private fun peek() = tokens.getOrElse(position) { "" }
354
355 private fun unexpected(): Nothing =
356 error(
357 if (position == tokens.size) "unexpected end of expression"
358 else "unexpected '${peek()}'",
359 )
360
361 private fun accept(s: String): Boolean {
362 if (peek() != s) return false
363 position++
364 return true
365 }
366
367 private fun expect(s: String) {
368 if (!accept(s)) unexpected()
369 }
370
371 private fun atomTree(): Syntax {
372 val t = peek()
373 if (t.isEmpty()) unexpected()
374 if (accept("(")) {
375 val a = expression()
376 expect(")")
377 return a
378 }
379 if (!t[0].isLetterOrDigit()) unexpected()
380 position++
381 if (t[0].isDigit()) return Syntax("", num(t.toLong()))
382 if (t[0].isUpperCase()) return Syntax("", names[t] ?: error("undefined name '$t'"))
383 if (t !in functions) return Syntax("", node(0, s = t))
384 expect("(")
385 val a = expression()
386 expect(")")
387 return Syntax(t, args = listOf(a))
388 }
389
390 private fun unary(): Syntax {
391 if (accept("-")) return Syntax("neg", args = listOf(unary()))
392 val a = atomTree()
393 return if (accept("^")) Syntax("^", args = listOf(a, unary())) else a
394 }
395
396 private fun term(): Syntax {
397 var a = unary()
398 while (peek() in setOf("*", "/")) {
399 val op = tokens[position++]
400 a = Syntax(op, args = listOf(a, unary()))
401 }
402 return a
403 }
404
405 private fun expression(): Syntax {
406 var a = term()
407 while (peek() in setOf("+", "-")) {
408 val op = tokens[position++]
409 a = Syntax(op, args = listOf(a, term()))
410 }
411 return a
412 }
413
414 private fun evaluate(t: Syntax): Expr {
415 t.value?.let {
416 return it
417 }
418 val a = t.args.map(::evaluate)
419 return when (t.op) {
420 "+" -> add(a)
421 "-" -> add(a[0], mul(minus, a[1]))
422 "*" -> mul(a)
423 "/" -> mul(a[0], power(a[1], minus))
424 "^" -> power(a[0], a[1])
425 "neg" -> mul(minus, a[0])
426 else -> fn(t.op, a[0])
427 }
428 }
429
430 fun parse(): Expr {
431 val t = expression()
432 if (position < tokens.size) unexpected()
433 return evaluate(t)
434 }
435}

437fun variable(s: String) = Regex("[a-z][a-z0-9]*").matches(s) && s !in functions

439fun rationalLiteral(s: String) = Regex("-?[0-9]+(?:/[0-9]+)?").matches(s)

441fun literal(s: String): Expr {
442 if (!rationalLiteral(s)) error("invalid number '$s'")

447val signatures =
448 mapOf(
449 "SIMPLIFY" to "e",
450 "EXPAND" to "e",
451 "DIFF" to "ev",
452 "DIFFN" to "evn",
453 "GRAD" to "e",
454 "HESSIAN" to "e",
455 "IMPLICIT" to "evv",
456 "SUBST" to "eve",
457 "EVAL" to "ea",
458 "TANGENT" to "evc",
459 "LIMIT" to "evc",
460 "TAYLOR" to "evcn",

463fun command(line: String, names: MutableMap<String, Expr>): List<String> {
464 val p = line.indexOf(' ')
465 val c = if (p < 0) line else line.substring(0, p)
466 val arg = if (p < 0) "" else line.substring(p + 1)
467 if (c == "LET") {
468 val i = arg.indexOf('=')
469 if (i < 0) error("wrong number of arguments")
470 val s = arg.substring(0, i).trim(' ')
471 if (!Regex("[A-Z][A-Z0-9]*").matches(s)) error("invalid name '$s'")
472 val f = Parser(arg.substring(i + 1).trim(' '), names).parse()
473 names[s] = f
474 return listOf("$s = ${show(f)}")
475 }
476 val sig = signatures[c] ?: error("unknown command $c")
477 val fields = if (p < 0) emptyList() else arg.split(';').map { it.trim(' ') }
478 if (if (c == "EVAL") fields.size !in 1..2 else fields.size != sig.length)
479 error("wrong number of arguments")
480 val forms = MutableList(fields.size) { zero }
481 var n = 0
482 for (i in fields.indices) {
483 val s = fields[i]
484 when (sig[i]) {
485 'e' -> forms[i] = Parser(s, names).parse()
486 'v' -> if (!variable(s)) error("invalid variable '$s'")
487 'n' -> {
488 if (!Regex("[1-8]").matches(s)) error("invalid order '$s'")
489 n = s.toInt()
490 }
491 'c' -> forms[i] = literal(s)
492 }
493 }
494 var f = forms[0]
495 val v = fields.getOrElse(1) { "" }
496 val out = mutableListOf<String>()
497 if (c == "SIMPLIFY") return listOf("simplified: ${show(f)}")
498 if (c == "EXPAND") return listOf("expanded: ${show(expand(f))}")
499 if (c == "DIFF") return listOf("d/d$v = ${show(derivative(f,v))}")
500 if (c == "DIFFN") {
501 for (k in 1..n) {
502 f = derivative(f, v)
503 out.add("d^$k/d$v^$k = ${show(f)}")
504 }
505 return out
506 }
507 if (c == "GRAD" || c == "HESSIAN") {
508 val vs = variables(f).sorted()
509 if (vs.isEmpty()) return listOf("no variables")
510 for (i in vs.indices) {
511 val d = derivative(f, vs[i])
512 if (c == "GRAD") out.add("d/d${vs[i]} = ${show(d)}")
513 else
514 for (j in i until vs.size) out.add(
515 "d2/d${vs[i]}d${vs[j]} = ${show(derivative(d,vs[j]))}",
516 )
517 }
518 return out
519 }
520 if (c == "IMPLICIT") {
521 val w = fields[2]
522 if (v == w) error("same variable")
523 return listOf("d$w/d$v = ${show(mul(minus,derivative(f,v),power(derivative(f,w),minus)))}")
524 }
525 if (c == "SUBST") return listOf("substituted: ${show(substitute(f,mapOf(v to forms[2])))}")
526 if (c == "EVAL") {
527 val values = mutableMapOf<String, Expr>()
528 for (word in v.split(' ').filter { it.isNotEmpty() }) {
529 val i = word.indexOf('=')
530 val s = if (i < 0) word else word.substring(0, i)
531 val t = if (i < 0) "" else word.substring(i + 1)
532 if (i < 0 || !variable(s) || !rationalLiteral(t)) error("invalid assignment '$word'")
533 val q = literal(t)
534 if (s in values) error("duplicate variable $s")
535 values[s] = q
536 }
537 for (s in variables(f).sorted()) if (s !in values) error("no value for $s")
538 val r = substitute(f, values)
539 return listOf(
540 if (r.kind == 5) "exact: ${show(r)}"
541 else String.format(Locale.ROOT, "approx: %.6f", numeric(r)),
542 )
543 }
544 val x = forms[2]
545 val point = mapOf(v to x)
546 if (c == "TANGENT") {
547 val m = substitute(derivative(f, v), point)
548 val y = substitute(f, point)
549 val z = node(0, s = v)
550 val t = "tangent: y = ${show(add(mul(m,z),y,mul(number(-x.q),m)))}"
551 if (m == zero) return listOf(t, "normal: $v = ${show(x)}")
552 val k = power(m, minus)
553 return listOf(t, "normal: y = ${show(add(mul(minus,k,z),y,mul(x,k)))}")
554 }
555 if (c == "LIMIT") {
556 val top = mutableListOf<Expr>()
557 val bottom = mutableListOf<Expr>()
558 for (t in if (f.kind == 2) f.args else listOf(f)) {
559 if (negative(t)) bottom.add(power(t.args[0], number(-t.args[1].q))) else top.add(t)
560 }
561 var g = mul(top)
562 var h = mul(bottom)
563 for (k in 0..5) {
564 val u = substitute(g, point)
565 val b = substitute(h, point)
566 if (b != zero) return listOf("limit: ${show(mul(u,power(b,minus)))}")
567 if (u != zero) return listOf("limit: unbounded")
568 if (k < 5) {
569 g = derivative(g, v)
570 h = derivative(h, v)
571 }
572 }
573 return listOf("limit: unknown")
574 }
575 val ds = mutableListOf(f)
576 repeat(n) { ds.add(derivative(ds.last(), v)) }
577 val ps = mutableListOf<Expr>()
578 var fac = 1L
579 for (k in 0..n) {
580 if (k > 0) fac *= k
581 val co = mul(substitute(ds[k], point), num(1L, fac))
582 ps.add(mul(co, power(add(node(0, s = v), number(-x.q)), num(k))))
583 }
584 return listOf("taylor: ${show(expand(add(ps)))}")
585}

587fun main() {
588 val input = System.`in`.bufferedReader()
589 val names = mutableMapOf<String, Expr>()
590 repeat(input.readLine().toInt()) {
591 val line = input.readLine()
592 try {
593 for (s in command(line, names)) println(s)
594 } catch (e: IllegalStateException) {
595 println("error: ${e.message}")
596 }
597 }
598}
