int store(Context &context, Expression expression)
    index := int(#context.nodes)
    context.nodes = context.nodes.append(expression)
    return index

    divisor := gcd_value(p, q)
    return store(&context, Expression("number", p / divisor, q / divisor))
int constant(Context &context, int value)
    return rational(&context, value)
