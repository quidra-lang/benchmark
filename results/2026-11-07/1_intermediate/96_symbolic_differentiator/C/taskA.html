60static long long gcd_ll(long long a, long long b) {
61 a = llabs(a);
62 b = llabs(b);
63 while (b) {
64 long long t = a % b;
65 a = b;
66 b = t;
67 }
68 return a ? a : 1;
69}

71static void fraction(long long *p, long long *q) {
72 if (!*q) errorf("division by zero");
73 if (*q < 0) *p = -*p, *q = -*q;
74 long long g = gcd_ll(*p, *q);
75 *p /= g;
76 *q /= g;
77}

124static int compare(const Node *left, const Node *right) {
125 if (left == right) return 0;
126 if (left->kind != right->kind) return left->kind < right->kind ? -1 : 1;
127 if (left->kind == NUMBER) {
128 long long a = left->p * right->q, b = right->p * left->q;
129 return (a > b) - (a < b);
130 }
131 if (left->kind == VARIABLE || left->kind == FUNCTION) {
132 int c = strcmp(left->name, right->name);
133 if (c) return (c > 0) - (c < 0);
134 }
135 int len = left->length < right->length ? left->length : right->length;
136 for (int i = 0; i < len; ++i) {
137 int c = compare(left->children[i], right->children[i]);
138 if (c) return c;
139 }
140 return (left->length > right->length) - (left->length < right->length);
141}

159static Term coefficient(Node *f) {
160 Term t;
161 t.p = 1;
162 t.q = 1;
163 t.length = 0;
164 t.parts = NULL;
165 if (f->kind == NUMBER) {
166 t.p = f->p;
167 t.q = f->q;
168 } else if (f->kind == PRODUCT) {
169 t.parts = f->children;
170 t.length = f->length;
171 if (f->children[0]->kind == NUMBER) {
172 t.p = f->children[0]->p;
173 t.q = f->children[0]->q;
174 ++t.parts;
175 --t.length;
176 }
177 } else {
178 t.parts = malloc(sizeof(Node *));
179 t.parts[0] = f;
180 t.length = 1;
181 }
182 return t;
183}

217static int monomial_equal(const Term *a, const Term *b) {
218 if (a->length != b->length) return 0;
219 for (int i = 0; i < a->length; ++i) if (!equal(a->parts[i], b->parts[i])) return 0;
220 return 1;

223static Node *build_product(long long p, long long q, int count, Node **args) {
224 if (!p) return zero;
225 if (!count) return number(p, q);
226 if (p == q && count == 1) return args[0];
227 Node *all[MAX_CHILDREN];
228 int n = 0;
229 if (p != q) all[n++] = number(p, q);
230 for (int i = 0; i < count; ++i) all[n++] = args[i];
231 return make_node(PRODUCT, n, all, NULL);
232}

234static void flatten_add(Node *f, Node **terms, int *count) {
235 if (f->kind == SUM) {
236 for (int i = 0; i < f->length; ++i) terms[(*count)++] = f->children[i];
237 } else if (f->kind == PRODUCT && f->length == 2 && f->children[0]->kind == NUMBER && f->children[1]->kind == SUM) {
238 for (int i = 0; i < f->children[1]->length; ++i) {
239 terms[(*count)++] = multiply_two(f->children[0], f->children[1]->children[i]);
240 }
241 } else {
242 terms[(*count)++] = f;
243 }
244}

252static int term_compare(const void *ap, const void *bp) {
253 const Node *a = *(Node *const *)ap, *b = *(Node *const *)bp;
254 Term at = coefficient((Node *)a), bt = coefficient((Node *)b);
255 Node *bases[MAX_CHILDREN * 2];
256 int count = 0;
257 for (int i = 0; i < at.length; ++i) {
258 Node *candidate = base(at.parts[i]);
259 int found = 0;
260 for (int j = 0; j < count; ++j) found |= equal(candidate, bases[j]);
261 if (!found) bases[count++] = candidate;
262 }
263 for (int i = 0; i < bt.length; ++i) {
264 Node *candidate = base(bt.parts[i]);
265 int found = 0;
266 for (int j = 0; j < count; ++j) found |= equal(candidate, bases[j]);
267 if (!found) bases[count++] = candidate;
268 }
269 for (int i = 1; i < count; ++i) {
270 Node *current = bases[i];
271 int j = i;
272 while (j > 0 && compare(bases[j - 1], current) > 0) bases[j] = bases[j - 1], --j;
273 bases[j] = current;
274 }
275 for (int i = 0; i < count; ++i) {
276 Node *ax = zero, *bx = zero;
277 for (int j = 0; j < at.length; ++j) if (equal(base(at.parts[j]), bases[i])) ax = exponent(at.parts[j]);
278 for (int j = 0; j < bt.length; ++j) if (equal(base(bt.parts[j]), bases[i])) bx = exponent(bt.parts[j]);
279 int c = exponent_compare(bx, ax);
280 if (c) return c;
281 }
282 return 0;
283}

285static Node *add(int count, Node **args) {
286 Node *terms[MAX_CHILDREN];
287 int term_count = 0;
288 for (int i = 0; i < count; ++i) flatten_add(args[i], terms, &term_count);
289 Term groups[MAX_CHILDREN];
290 int group_count = 0;
291 for (int i = 0; i < term_count; ++i) {
292 Term next = coefficient(terms[i]);
293 int index = -1;
294 for (int j = 0; j < group_count; ++j) if (monomial_equal(&next, &groups[j])) {index = j; break;}
295 if (index < 0) groups[group_count++] = next;
296 else add_fraction(&groups[index].p, &groups[index].q, next.p, next.q);
297 }
298 Node *built[MAX_CHILDREN];
299 int length = 0;
300 for (int i = 0; i < group_count; ++i) {
301 if (groups[i].p) built[length++] = build_product(groups[i].p, groups[i].q, groups[i].length, groups[i].parts);
302 }
303 qsort(built, (size_t)length, sizeof(Node *), term_compare);
304 if (!length) return zero;
305 if (length == 1) return built[0];
306 return make_node(SUM, length, built, NULL);
307}

320static Node *multiply(int count, Node **args) {
321 Node *factors[MAX_CHILDREN];
322 int factor_count = 0;
323 for (int i = 0; i < count; ++i) flatten_multiply(args[i], factors, &factor_count);
324 long long p = 1, q = 1;
325 FactorGroup groups[MAX_CHILDREN];
326 int group_count = 0;
327 for (int i = 0; i < factor_count; ++i) {
328 Node *f = factors[i];
329 if (f->kind == NUMBER) {
330 mul_fraction(&p, &q, f->p, f->q);
331 continue;
332 }
333 Node *b = base(f);
334 int index = -1;
335 for (int j = 0; j < group_count; ++j) if (equal(groups[j].base, b)) {index = j; break;}
336 if (index < 0) {
337 groups[group_count].base = b;
338 groups[group_count].original = f;
339 groups[group_count].exponent = exponent(f);
340 groups[group_count].count = 1;
341 ++group_count;
342 } else {
343 groups[index].exponent = add_two(groups[index].exponent, exponent(f));
344 ++groups[index].count;
345 }
346 }
347 if (!p) return zero;
348 Node *built[MAX_CHILDREN];
349 for (int i = 0; i < group_count; ++i) {
350 built[i] = groups[i].count == 1 ? groups[i].original : power(groups[i].base, groups[i].exponent);
351 }
352 int retry = 0;
353 for (int i = 0; i < group_count; ++i) {
354 if (built[i]->kind == NUMBER || built[i]->kind == PRODUCT) retry = 1;
355 for (int j = 0; j < i; ++j) if (equal(base(built[i]), base(built[j]))) retry = 1;
356 }
357 if (retry) {
358 Node *again[MAX_CHILDREN];
359 again[0] = number(p, q);
360 for (int i = 0; i < group_count; ++i) again[i + 1] = built[i];
361 return multiply(group_count + 1, again);
362 }
363 qsort(built, (size_t)group_count, sizeof(Node *), compare_bases);
364 return build_product(p, q, group_count, built);
365}

367static long long integer_root(long long value, long long degree) {
368 if (value == 1) return 1;
369 if (degree > 20) return -1;
370 long long rounded = llround(pow((double)value, 1.0 / (double)degree));
371 for (long long r = rounded > 0 ? rounded - 1 : 0; r <= rounded + 1; ++r) {
372 long long total = 1;
373 for (long long i = 0; i < degree; ++i) {
374 if (r && total > value / r) {total = value + 1; break;}
375 total *= r;
376 }
377 if (total == value) return r;
378 }
379 return -1;
380}

382static Node *rational_power(long long p, long long q, long long exp) {
383 if (exp < 0) {
384 long long t = p;
385 p = q;
386 q = t;
387 exp = -exp;
388 }
389 long long a = 1, b = 1;
390 while (exp) {
391 if (exp % 2) mul_fraction(&a, &b, p, q);
392 exp /= 2;
393 if (exp) mul_fraction(&p, &q, p, q);
394 }
395 return number(a, b);
396}

398static Node *power(Node *b, Node *x) {
399 if (equal(x, zero)) return one;
400 if (equal(x, one)) return b;
401 if (b->kind == NUMBER) {
402 if (equal(b, one)) return one;
403 if (x->kind != NUMBER) return power_node(b, x);
404 if (!b->p) {
405 if (x->p > 0) return zero;
406 errorf("division by zero");
407 }
408 if (is_integer(x)) return rational_power(b->p, b->q, x->p);
409 if (b->p > 0) {
410 long long a = integer_root(b->p, x->q);
411 long long c = integer_root(b->q, x->q);
412 if (a >= 0 && c >= 0) return rational_power(a, c, x->p);
413 }
414 } else if (is_integer(x)) {
415 if (b->kind == POWER) return power(b->children[0], multiply_two(b->children[1], x));
416 if (b->kind == PRODUCT) {
417 Node *parts[MAX_CHILDREN];
418 for (int i = 0; i < b->length; ++i) parts[i] = power(b->children[i], x);
419 return multiply(b->length, parts);
420 }
421 }
422 return power_node(b, x);
423}

425static Node *function(const char *name, Node *arg) {
426 if (!strcmp(name, "sqrt")) return power(arg, half);
427 if (equal(arg, zero)) {
428 if (!strcmp(name, "sin") || !strcmp(name, "tan")) return zero;
429 if (!strcmp(name, "cos") || !strcmp(name, "exp")) return one;
430 }
431 if (!strcmp(name, "ln") && equal(arg, one)) return zero;
432 if (!strcmp(name, "sin") || !strcmp(name, "cos") || !strcmp(name, "tan")) {
433 Term c = coefficient(arg);
434 if (c.p < 0) {
435 Node *f = function(name, multiply_two(minus_one, arg));
436 return !strcmp(name, "cos") ? f : multiply_two(minus_one, f);
437 }
438 }
439 if (arg->kind == FUNCTION) {
440 if ((!strcmp(name, "ln") && !strcmp(arg->name, "exp")) ||
441 (!strcmp(name, "exp") && !strcmp(arg->name, "ln"))) return arg->children[0];
442 }
443 return function_node(name, arg);
444}

446static int variable_name(const char *name) {
447 if (!islower((unsigned char)*name)) return 0;
448 for (const char *p = name + 1; *p; ++p) if (!islower((unsigned char)*p) && !isdigit((unsigned char)*p)) return 0;
449 static const char *const banned[] = {"sin","cos","tan","exp","ln","sqrt"};
450 for (int i = 0; i < 6; ++i) if (!strcmp(name, banned[i])) return 0;
451 return 1;
452}

465static void tokenize(const char *text) {
466 tokens_count = token_position = 0;
467 for (int i = 0; text[i];) {
468 if (text[i] == ' ') {++i; continue;}
469 int begin = i;
470 if (isdigit((unsigned char)text[i])) while (isdigit((unsigned char)text[i])) ++i;
471 else if (islower((unsigned char)text[i])) while (islower((unsigned char)text[i]) || isdigit((unsigned char)text[i])) ++i;
472 else if (isupper((unsigned char)text[i])) while (isupper((unsigned char)text[i]) || isdigit((unsigned char)text[i])) ++i;
473 else if (strchr("+-*/^()", text[i])) ++i;
474 else errorf("invalid character '%c'", text[i]);
475 int length = i - begin;
476 memcpy(tokens[tokens_count], text + begin, (size_t)length);
477 tokens[tokens_count][length] = '\0';
478 ++tokens_count;
479 }
480}

504static Node *parse_atom(void) {
505 if (accept("(")) {
506 Node *f = parse_expression();
507 expect(")");
508 return f;
509 }
510 const char *text = peek();
511 if (!text[0] || !isalnum((unsigned char)text[0])) unexpected();
512 ++token_position;
513 if (isdigit((unsigned char)text[0])) return integer(atoll(text));
514 if (isupper((unsigned char)text[0])) {
515 Node *f = definition(text);
516 if (!f) errorf("undefined name '%s'", text);
517 return f;
518 }
519 if (variable_name(text)) return variable(text);
520 expect("(");
521 Node *inner = parse_expression();
522 expect(")");
523 return function(text, inner);
524}

526static Node *parse_unary(void) {
527 if (accept("-")) return multiply_two(minus_one, parse_unary());
528 Node *left = parse_atom();
529 if (accept("^")) return power(left, parse_unary());
530 return left;
531}

533static Node *parse_term(void) {
534 Node *left = parse_unary();
535 while (!strcmp(peek(), "*") || !strcmp(peek(), "/")) {
536 int div = accept("/");
537 if (!div) expect("*");
538 Node *right = parse_unary();
539 left = div ? multiply_two(left, power(right, minus_one)) : multiply_two(left, right);
540 }
541 return left;
542}
543
544static Node *parse_expression(void) {
545 Node *left = parse_term();
546 while (!strcmp(peek(), "+") || !strcmp(peek(), "-")) {
547 int subtraction = accept("-");
548 if (!subtraction) expect("+");
549 Node *right = parse_term();
550 left = add_two(left, subtraction ? multiply_two(minus_one, right) : right);
551 }
552 return left;
553}

555static Node *parse(const char *text) {
556 tokenize(text);
557 Node *result = parse_expression();
558 if (token_position != tokens_count) unexpected();
559 return result;
560}

587static Node *derivative(Node *f, const char *name) {
588 if (f->kind == NUMBER) return zero;
589 if (f->kind == VARIABLE) return !strcmp(f->name, name) ? one : zero;
590 if (f->kind == SUM) {
591 Node *terms[MAX_CHILDREN];
592 for (int i = 0; i < f->length; ++i) terms[i] = derivative(f->children[i], name);
593 return add(f->length, terms);
594 }
595 if (f->kind == PRODUCT) {
596 Node *terms[MAX_CHILDREN];
597 for (int i = 0; i < f->length; ++i) {
598 Node *factors[MAX_CHILDREN];
599 for (int j = 0; j < f->length; ++j) {
600 factors[j] = i == j ? derivative(f->children[j], name) : f->children[j];
601 }
602 terms[i] = multiply(f->length, factors);
603 }
604 return add(f->length, terms);
605 }
606 if (f->kind == POWER) {
607 Node *b = f->children[0], *x = f->children[1];
608 if (!contains_variable(x, name)) {
609 return multiply_three(x, power(b, add_two(x, minus_one)), derivative(b, name));
610 }
611 if (!contains_variable(b, name)) {
612 return multiply_three(f, function("ln", b), derivative(x, name));
613 }
614 Node *first = multiply_two(derivative(x, name), function("ln", b));
615 Node *second = multiply_three(x, derivative(b, name), power(b, minus_one));
616 return multiply_two(f, add_two(first, second));
617 }
618 Node *a = f->children[0], *h;
619 if (!strcmp(f->name, "sin")) h = function("cos", a);
620 else if (!strcmp(f->name, "cos")) h = multiply_two(minus_one, function("sin", a));
621 else if (!strcmp(f->name, "tan")) h = power(function("cos", a), integer(-2));
622 else if (!strcmp(f->name, "exp")) h = f;
623 else h = power(a, minus_one);
624 return multiply_two(h, derivative(a, name));
625}

627static Node *substitute(Node *f, int count, const char **names, Node **values) {
628 if (f->kind == VARIABLE) {
629 for (int i = 0; i < count; ++i) if (!strcmp(f->name, names[i])) return values[i];
630 return f;
631 }
632 if (f->kind == NUMBER) return f;
633 Node *children[MAX_CHILDREN];
634 for (int i = 0; i < f->length; ++i) children[i] = substitute(f->children[i], count, names, values);
635 if (f->kind == SUM) return add(f->length, children);
636 if (f->kind == PRODUCT) return multiply(f->length, children);
637 if (f->kind == POWER) return power(children[0], children[1]);
638 return function(f->name, children[0]);
639}

647static Node *distribute(Node *a, Node *b) {
648 Node *left = a->kind == SUM ? a : NULL, *right = b->kind == SUM ? b : NULL;
649 int n = left ? left->length : 1, m = right ? right->length : 1;
650 Node *terms[MAX_CHILDREN];
651 int count = 0;
652 for (int i = 0; i < n; ++i) {
653 for (int j = 0; j < m; ++j) {
654 terms[count++] = multiply_two(left ? left->children[i] : a, right ? right->children[j] : b);
655 }
656 }
657 return add(count, terms);
658}

660static Node *expand(Node *f) {
661 if (f->kind == NUMBER || f->kind == VARIABLE) return f;
662 Node *children[MAX_CHILDREN];
663 for (int i = 0; i < f->length; ++i) children[i] = expand(f->children[i]);
664 if (f->kind == SUM) return add(f->length, children);
665 if (f->kind == PRODUCT) {
666 Node *result = children[0];
667 for (int i = 1; i < f->length; ++i) result = distribute(result, children[i]);
668 return result;
669 }
670 if (f->kind == FUNCTION) return function(f->name, children[0]);
671 if (children[0]->kind == SUM && is_integer(children[1]) && children[1]->p > 0) {
672 Node *result = children[0];
673 for (long long i = 1; i < children[1]->p; ++i) result = distribute(result, children[0]);
674 return result;
675 }
676 return power(children[0], children[1]);
677}

679static double numerical_value(Node *f) {
680 if (f->kind == NUMBER) return (double)f->p / (double)f->q;
681 if (f->kind == SUM) {
682 double result = 0.0;
683 for (int i = 0; i < f->length; ++i) result += numerical_value(f->children[i]);
684 return result;
685 }
686 if (f->kind == PRODUCT) {
687 double result = 1.0;
688 for (int i = 0; i < f->length; ++i) result *= numerical_value(f->children[i]);
689 return result;
690 }
691 if (f->kind == POWER) {
692 double a = numerical_value(f->children[0]);
693 double b = numerical_value(f->children[1]);
694 int integral = f->children[1]->kind == NUMBER ? is_integer(f->children[1]) : floor(b) == b;
695 if (a < 0.0 && !integral) errorf("domain error");
696 return equal(f->children[1], half) ? sqrt(a) : pow(a, b);
697 }
698 double a = numerical_value(f->children[0]);
699 if (!strcmp(f->name, "ln")) {
700 if (a <= 0.0) errorf("domain error");
701 return log(a);
702 }
703 if (!strcmp(f->name, "sin")) return sin(a);
704 if (!strcmp(f->name, "cos")) return cos(a);
705 if (!strcmp(f->name, "tan")) return tan(a);
706 return exp(a);
707}

744static void push_factor(char *list, int *count, Node *f, int negative_power) {
745 if (*count) strcat(list, "*");
746 if (negative_power && f->kind == POWER) {
747 Node *positive = number(-f->children[1]->p, f->children[1]->q);
748 if (equal(positive, one)) render_factor(f->children[0], list);
749 else render_power(f->children[0], positive, list);
750 } else render_factor(f, list);
751 ++*count;

754static void render_quotient(long long p, long long q, int count, Node **factors, char *output) {
755 char numerator[MAX_TEXT] = "", denominator[MAX_TEXT] = "";
756 int numerators = 0, denominators = 0;
757 if (llabs(p) != 1) {
758 sprintf(numerator, "%lld", llabs(p));
759 ++numerators;
760 }
761 if (q != 1) {
762 sprintf(denominator, "%lld", q);
763 ++denominators;
764 }
765 for (int i = 0; i < count; ++i) {
766 Node *f = factors[i];
767 if (f->kind == POWER && f->children[1]->kind == NUMBER && f->children[1]->p < 0) {
768 push_factor(denominator, &denominators, f, 1);
769 } else push_factor(numerator, &numerators, f, 0);
770 }
771 if (p < 0) strcat(output, "-");
772 strcat(output, numerators ? numerator : "1");
773 if (denominators) {
774 strcat(output, denominators == 1 ? "/" : "/(");
775 strcat(output, denominator);
776 if (denominators != 1) strcat(output, ")");
777 }
778}

780static void render(Node *f, char *buffer) {
781 if (f->kind == NUMBER) {
782 char digits[64];
783 if (f->q == 1) sprintf(digits, "%lld", f->p);
784 else sprintf(digits, "%lld/%lld", f->p, f->q);
785 strcat(buffer, digits);
786 return;
787 }
788 if (f->kind == VARIABLE) {
789 strcat(buffer, f->name);
790 return;
791 }
792 if (f->kind == FUNCTION) {
793 strcat(buffer, f->name);
794 strcat(buffer, "(");
795 render(f->children[0], buffer);
796 strcat(buffer, ")");
797 return;
798 }
799 if (f->kind == POWER) {
800 if (f->children[1]->kind == NUMBER && f->children[1]->p < 0) {
801 Node *items[] = {f};
802 render_quotient(1, 1, 1, items, buffer);
803 } else render_power(f->children[0], f->children[1], buffer);
804 return;
805 }
806 if (f->kind == PRODUCT) {
807 Term t = coefficient(f);
808 render_quotient(t.p, t.q, t.length, t.parts, buffer);
809 return;
810 }
811 for (int i = 0; i < f->length; ++i) {
812 char term[MAX_TEXT] = "";
813 render(f->children[i], term);
814 if (i) {
815 if (term[0] == '-') {
816 strcat(buffer, " - ");
817 strcat(buffer, term + 1);
818 } else {
819 strcat(buffer, " + ");
820 strcat(buffer, term);
821 }
822 } else strcat(buffer, term);
823 }
824}

832static char *trim(char *text) {
833 while (*text == ' ') ++text;
834 size_t length = strlen(text);
835 while (length && text[length - 1] == ' ') text[--length] = '\0';
836 return text;
837}

839static int parse_literal(const char *text, Node **result) {
840 const char *p = text;
841 int negative = 0;
842 if (*p == '-') {negative = 1; ++p;}
843 if (!isdigit((unsigned char)*p)) return 0;
844 long long numerator = 0, denominator = 1;
845 while (isdigit((unsigned char)*p)) numerator = numerator * 10 + *p++ - '0';
846 if (*p == '/') {
847 ++p;
848 if (!isdigit((unsigned char)*p)) return 0;
849 denominator = 0;
850 while (isdigit((unsigned char)*p)) denominator = denominator * 10 + *p++ - '0';
851 }
852 if (*p) return 0;
853 *result = number(negative ? -numerator : numerator, denominator);
854 return 1;
855}

857static Node *literal(const char *text) {
858 Node *result;
859 if (!parse_literal(text, &result)) errorf("invalid number '%s'", text);
860 return result;
861}

863static void assignment(const char *text, char name[202], Node **value) {
864 const char *equal_sign = strchr(text, '=');
865 if (!equal_sign || equal_sign - text >= 201) errorf("invalid assignment '%s'", text);
866 size_t length = (size_t)(equal_sign - text);
867 memcpy(name, text, length);
868 name[length] = '\0';
869 if (!variable_name(name)) errorf("invalid assignment '%s'", text);
870 Node *parsed;
871 if (!parse_literal(equal_sign + 1, &parsed)) errorf("invalid assignment '%s'", text);
872 *value = parsed;
873}

875static void command(char *line) {
876 char *space = strchr(line, ' '), *arguments = "";
877 if (space) *space = '\0', arguments = space + 1;
878 const char *cmd = line;
879 static const char *const commands[] = {
880 "LET","SIMPLIFY","EXPAND","DIFF","DIFFN","GRAD","HESSIAN","IMPLICIT",
881 "SUBST","EVAL","TANGENT","LIMIT","TAYLOR"
882 };
883 int kind = -1;
884 for (int i = 0; i < 13; ++i) if (!strcmp(cmd, commands[i])) kind = i;
885 if (kind < 0) errorf("unknown command %s", cmd);
886 if (kind == 0) {
887 char *equal_sign = strchr(arguments, '=');
888 if (!equal_sign) errorf("wrong number of arguments");
889 *equal_sign = '\0';
890 char *name = trim(arguments), *expression = trim(equal_sign + 1);
891 if (!definition_name(name)) errorf("invalid name '%s'", name);
892 Node *result = parse(expression);
893 int index = -1;
894 for (int i = 0; i < definitions_count; ++i) if (!strcmp(definitions[i].name, name)) index = i;
895 if (index < 0) index = definitions_count++;
896 strcpy(definitions[index].name, name);
897 definitions[index].expression = result;
898 printf("%s = %s\n", name, notation(result));
899 return;
900 }
901 char *fields[5];
902 int count = 0;
903 if (space) {
904 fields[count++] = arguments;
905 for (char *p = arguments; *p; ++p) {
906 if (*p == ';') {
907 *p = '\0';
908 fields[count++] = p + 1;
909 if (count > 4) errorf("wrong number of arguments");
910 }
911 }
912 for (int i = 0; i < count; ++i) fields[i] = trim(fields[i]);
913 }
914 int expected = kind == 1 || kind == 2 || kind == 5 || kind == 6 ? 1 :
915 kind == 3 ? 2 : kind == 12 ? 4 : 3;
916 if (kind == 9) {
917 if (count != 1 && count != 2) errorf("wrong number of arguments");
918 } else if (count != expected) errorf("wrong number of arguments");
919 Node *f = parse(fields[0]);
920 if (kind == 1) {printf("simplified: %s\n", notation(f)); return;}
921 if (kind == 2) {printf("expanded: %s\n", notation(expand(f))); return;}
922 if (kind == 5 || kind == 6) {
923 char names[MAX_CHILDREN][202];
924 int n = variables(f, names);
925 if (!n) {puts("no variables"); return;}
926 for (int i = 0; i < n; ++i) {
927 Node *d = derivative(f, names[i]);
928 if (kind == 5) printf("d/d%s = %s\n", names[i], notation(d));
929 else for (int j = i; j < n; ++j) {
930 printf("d2/d%sd%s = %s\n", names[i], names[j], notation(derivative(d, names[j])));
931 }
932 }
933 return;
934 }
935 if (kind == 9) {
936 char assignments_buffer[256];
937 strcpy(assignments_buffer, count == 2 ? fields[1] : "");
938 const char *names[MAX_TOKENS];
939 Node *values[MAX_TOKENS];
940 char identifiers[MAX_TOKENS][202];
941 int n = 0;
942 for (char *word = strtok(assignments_buffer, " "); word; word = strtok(NULL, " ")) {
943 assignment(word, identifiers[n], &values[n]);
944 for (int i = 0; i < n; ++i) if (!strcmp(identifiers[i], identifiers[n])) errorf("duplicate variable %s", identifiers[n]);
945 names[n] = identifiers[n];
946 ++n;
947 }
948 char required[MAX_CHILDREN][202];
949 int vars = variables(f, required);
950 for (int j = 0; j < vars; ++j) {
951 int assigned = 0;
952 for (int i = 0; i < n; ++i) if (!strcmp(required[j], names[i])) assigned = 1;
953 if (!assigned) errorf("no value for %s", required[j]);
954 }
955 Node *result = substitute(f, n, names, values);
956 if (result->kind == NUMBER) printf("exact: %s\n", notation(result));
957 else printf("approx: %.6f\n", numerical_value(result));
958 return;
959 }
960 if (!variable_name(fields[1])) errorf("invalid variable '%s'", fields[1]);
961 const char *v = fields[1];
962 if (kind == 3) {
963 printf("d/d%s = %s\n", v, notation(derivative(f, v)));
964 return;
965 }
966 if (kind == 4) {
967 if (strlen(fields[2]) != 1 || fields[2][0] < '1' || fields[2][0] > '8') errorf("invalid order '%s'", fields[2]);
968 int order = fields[2][0] - '0';
969 for (int k = 1; k <= order; ++k) {
970 f = derivative(f, v);
971 printf("d^%d/d%s^%d = %s\n", k, v, k, notation(f));
972 }
973 return;
974 }
975 if (kind == 7) {
976 if (!variable_name(fields[2])) errorf("invalid variable '%s'", fields[2]);
977 const char *w = fields[2];
978 if (!strcmp(v, w)) errorf("same variable");
979 Node *result = multiply_three(minus_one, derivative(f, v), power(derivative(f, w), minus_one));
980 printf("d%s/d%s = %s\n", w, v, notation(result));
981 return;
982 }
983 if (kind == 8) {
984 Node *replacement = parse(fields[2]);
985 printf("substituted: %s\n", notation(replace(f, v, replacement)));
986 return;
987 }
988 Node *point = literal(fields[2]);
989 if (kind == 10) {
990 Node *slope = replace(derivative(f, v), v, point);
991 Node *intercept = replace(f, v, point);
992 Node *symbol = variable(v);
993 Node *tangent = add(3, (Node *[]){multiply_two(slope, symbol), intercept,
994 multiply_two(number(-point->p, point->q), slope)});
995 printf("tangent: y = %s\n", notation(tangent));
996 if (equal(slope, zero)) printf("normal: %s = %s\n", v, notation(point));
997 else {
998 Node *inverse = power(slope, minus_one);
999 Node *normal = add(3, (Node *[]){multiply_three(minus_one, inverse, symbol), intercept,
1000 multiply_two(point, inverse)});
1001 printf("normal: y = %s\n", notation(normal));
1002 }
1003 return;
1004 }
1005 if (kind == 11) {
1006 Node *tops[MAX_CHILDREN], *bottoms[MAX_CHILDREN];
1007 int top_count = 0, bottom_count = 0;
1008 int n = f->kind == PRODUCT ? f->length : 1;
1009 for (int i = 0; i < n; ++i) {
1010 Node *part = f->kind == PRODUCT ? f->children[i] : f;
1011 if (part->kind == POWER && part->children[1]->kind == NUMBER && part->children[1]->p < 0) {
1012 Node *exp = number(-part->children[1]->p, part->children[1]->q);
1013 bottoms[bottom_count++] = power(part->children[0], exp);
1014 } else tops[top_count++] = part;
1015 }
1016 Node *numerator = multiply(top_count, tops), *denominator = multiply(bottom_count, bottoms);
1017 for (int k = 0; k <= 5; ++k) {
1018 Node *a = replace(numerator, v, point);
1019 Node *b = replace(denominator, v, point);
1020 if (!equal(b, zero)) {
1021 printf("limit: %s\n", notation(multiply_two(a, power(b, minus_one))));
1022 return;
1023 }
1024 if (!equal(a, zero)) {puts("limit: unbounded"); return;}
1025 if (k < 5) {
1026 numerator = derivative(numerator, v);
1027 denominator = derivative(denominator, v);
1028 }
1029 }
1030 puts("limit: unknown");
1031 return;
1032 }
1033 if (strlen(fields[3]) != 1 || fields[3][0] < '1' || fields[3][0] > '8') errorf("invalid order '%s'", fields[3]);
1034 int order = fields[3][0] - '0';
1035 Node *current = f, *terms[MAX_CHILDREN];
1036 long long factorial = 1;
1037 for (int k = 0; k <= order; ++k) {
1038 if (k) factorial *= k;
1039 Node *c = multiply_two(replace(current, v, point), number(1, factorial));
1040 Node *difference = add_two(variable(v), number(-point->p, point->q));
1041 terms[k] = multiply_two(c, power(difference, integer(k)));
1042 if (k != order) current = derivative(current, v);
1043 }
1044 printf("taylor: %s\n", notation(expand(add(order + 1, terms))));
1045}

1047int main(void) {
1048 zero = integer(0);
1049 one = integer(1);
1050 minus_one = integer(-1);
1051 half = number(1, 2);
1052 char line[256];
1053 if (!fgets(line, sizeof(line), stdin)) return 0;
1054 int count = atoi(line);
1055 for (int i = 0; i < count && fgets(line, sizeof(line), stdin); ++i) {
1056 size_t n = strlen(line);
1057 while (n && (line[n - 1] == '\n' || line[n - 1] == '\r')) line[--n] = '\0';
1058 if (!setjmp(recovery)) command(line);
1059 else printf("error: %s\n", failure);
1060 }
1061 return 0;
1062}
