1FUNCTIONS = %w[sin cos tan exp ln sqrt].freeze
2Expr = Data.define(:kind, :value, :items)
3def num(p, q = 1) = Expr.new(5, Rational(p, q), [])
4def tree(k, a, s = nil) = Expr.new(k, s, a)
5ZERO, ONE, MINUS, HALF = num(0), num(1), num(-1), num(1, 2)
6def integer?(f) = f.kind == 5 && f.value.denominator == 1
7def negative?(f) = f.kind == 3 && f.items[1].kind == 5 && f.items[1].value < 0
8def base(f) = f.kind == 3 ? f.items[0] : f
9def exponent(f) = f.kind == 3 ? f.items[1] : ONE
10def coefficient(f)
11 return [f.value, []] if f.kind == 5
12 return f.items[0].kind == 5 ? [f.items[0].value, f.items.drop(1)] : [Rational(1), f.items] if f.kind == 2
13 [Rational(1), [f]]
14end

16def order(f)
17 return [f.kind, f.value] if [0, 5].include?(f.kind)
18 return [4, f.value, order(f.items[0])] if f.kind == 4
19 [f.kind, f.items.map { |a| order(a) }]
20end
21
22def exponent_compare(a, b)
23 return -1 if a.kind == 5 && b.kind != 5
24 return 1 if a.kind != 5 && b.kind == 5
25 order(a) <=> order(b)
26end
27
28def term_compare(a, b)
29 x, y = coefficient(a)[1], coefficient(b)[1]
30 (x + y).map { |f| base(f) }.uniq.sort_by { |f| order(f) }.each do |f|
31 s, t = x.find { |g| base(g) == f }, y.find { |g| base(g) == f }
32 c = exponent_compare(t ? exponent(t) : ZERO, s ? exponent(s) : ZERO)
33 return c unless c.zero?
34 end
35
36 0
37end

39def product(c, a)
40 return num(c) if a.empty?
41 return tree(2, [num(c), *a]) unless c == 1
42 a.length == 1 ? a[0] : tree(2, a)
43end

45def add(*operands)
46 terms = operands.flat_map do |f|
47 if f.kind == 1
48 f.items
49 elsif f.kind == 2 && f.items.length == 2 && f.items[0].kind == 5 && f.items[1].kind == 1
50 f.items[1].items.map { |t| mul(f.items[0], t) }
51 else
52 [f]
53 end
54 end
55
56 groups = Hash.new(0)
57 terms.each { |f|
58 c, m = coefficient(f)
59 groups[m] += c
60 }
61 built = groups.filter_map { |m, c| product(c, m) unless c.zero? }.sort { |a, b| term_compare(a, b) }
62 built.empty? ? ZERO : built.length == 1 ? built[0] : tree(1, built)
63end
64
65def mul(*operands)
66 c = Rational(1)
67 groups = {}
68 operands.flat_map { |f| f.kind == 2 ? f.items : [f] }.each do |f|
69 if f.kind == 5
70 c *= f.value
71 else
72 (groups[base(f)] ||= []) << f
73 end
74 end
75
76 return ZERO if c.zero?
77 results = groups.map { |b, a| a.length == 1 ? a[0] : power(b, add(*a.map { |f| exponent(f) })) }
78 if results.any? { |f| [2, 5].include?(f.kind) } || results.map { |f| base(f) }.uniq.length < results.length
79 return mul(num(c), *results)
80 end
81
82 product(c, results.sort_by { |f| order(base(f)) })
83end

85def root(v, d)
86 return 1 if v == 1
87 return nil if d > 20
88 r = (v ** (1.0 / d)).round
89 ([0, r - 1].max..r + 1).find { |s| s ** d == v }
90end

92def power(b, x)
93 return ONE if x == ZERO
94 return b if x == ONE
95 if b.kind == 5
96 return ONE if b == ONE
97 return tree(3, [b, x]) unless x.kind == 5
98 if b.value.zero?
99 return ZERO if x.value.positive?
100 raise "division by zero"
101 end
102
103 return num(b.value ** x.value.numerator) if integer?(x)
104 if b.value.positive?
105 s, t = root(b.value.numerator, x.value.denominator), root(b.value.denominator, x.value.denominator)
106 return num(Rational(s, t) ** x.value.numerator) if s && t
107 end
108 elsif integer?(x)
109 return power(b.items[0], mul(b.items[1], x)) if b.kind == 3
110 return mul(*b.items.map { |f| power(f, x) }) if b.kind == 2
111 end
112
113 tree(3, [b, x])
114end
115
116def fn(s, a)
117 return power(a, HALF) if s == "sqrt"
118 return ZERO if a == ZERO && %w[sin tan].include?(s)
119 return ONE if a == ZERO && %w[cos exp].include?(s)
120 return ZERO if a == ONE && s == "ln"
121 if %w[sin cos tan].include?(s) && coefficient(a)[0].negative?
122 r = fn(s, mul(MINUS, a))
123 return s == "cos" ? r : mul(MINUS, r)
124 end
125
126 return a.items[0] if a.kind == 4 && ((s == "exp" && a.value == "ln") || (s == "ln" && a.value == "exp"))
127 tree(4, [a], s)
128end

130def variables(f) = (f.kind == 0 ? [f.value] : f.items.flat_map { |a| variables(a) }).uniq.sort
131
132def derivative(f, v)
133 case f.kind
134 when 5
135 ZERO
136 when 0
137 f.value == v ? ONE : ZERO
138 when 1
139 add(*f.items.map { |a| derivative(a, v) })
140 when 2
141 add(*f.items.each_index.map { |i| mul(*f.items.each_with_index.map { |a, j| i == j ? derivative(a, v) : a }) })
142 when 3
143 b, x = f.items
144 return mul(x, power(b, add(x, MINUS)), derivative(b, v)) unless variables(x).include?(v)
145 return mul(f, fn("ln", b), derivative(x, v)) unless variables(b).include?(v)
146 mul(f, add(mul(derivative(x, v), fn("ln", b)), mul(x, derivative(b, v), power(b, MINUS))))
147 when 4
148 a = f.items[0]
149 h = case f.value
150 when "sin"
151 fn("cos", a)
152 when "cos"
153 mul(MINUS, fn("sin", a))
154 when "tan"
155 power(fn("cos", a), num(-2))
156 when "exp"
157 f
158 else
159 power(a, MINUS)
160 end
161
162 mul(h, derivative(a, v))
163 end
164end

166def rebuild(f, a)
167 case f.kind
168 when 1
169 add(*a)
170 when 2
171 mul(*a)
172 when 3
173 power(*a)
174 else
175 fn(f.value, a[0])
176 end
177end

179def substitute(f, values)
180 return values.fetch(f.value, f) if f.kind == 0
181 return f if f.kind == 5
182 rebuild(f, f.items.map { |a| substitute(a, values) })
183end

185def terms(f) = f.kind == 1 ? f.items : [f]
186def distribute(a, b) = add(*terms(a).flat_map { |x| terms(b).map { |y| mul(x, y) } })
187def expand(f)
188 return f if [0, 5].include?(f.kind)
189 a = f.items.map { |g| expand(g) }
190 return a.reduce { |x, y| distribute(x, y) } if f.kind == 2
191 if f.kind == 3 && a[0].kind == 1 && integer?(a[1]) && a[1].value.positive?
192 return (a[1].value.numerator - 1).times.reduce(a[0]) { |r, _| distribute(r, a[0]) }
193 end
194
195 rebuild(f, a)
196end

198def numeric(f)
199 return f.value.to_f if f.kind == 5
200 return f.items.sum { |a| numeric(a) } if f.kind == 1
201 return f.items.reduce(1.0) { |r, a| r * numeric(a) } if f.kind == 2
202 if f.kind == 3
203 u, z = f.items.map { |a| numeric(a) }
204 integral = f.items[1].kind == 5 ? integer?(f.items[1]) : z == z.truncate
205 raise "domain error" if u < 0 && !integral
206 return f.items[1] == HALF ? Math.sqrt(u) : u ** z
207 end
208
209 a = numeric(f.items[0])
210 raise "domain error" if f.value == "ln" && a <= 0
211 Math.public_send(f.value == "ln" ? :log : f.value, a)
212end

214def atomic?(f) = [0, 4].include?(f.kind) || (integer?(f) && f.value >= 0) || (f.kind == 3 && f.items[1] == HALF)
215def atom(f) = atomic?(f) ? show(f) : "(#{show(f)})"
216def show_power(b, x) = x == HALF ? "sqrt(#{show(b)})" : "#{atom(b)}^#{atom(x)}"
217def factor(f) = f.kind == 1 ? "(#{show(f)})" : f.kind == 3 ? show_power(*f.items) : show(f)
218def quotient(c, a)
219 p, q = c.abs.numerator, c.denominator
220 top, bottom = p == 1 ? [] : [p.to_s], q == 1 ? [] : [q.to_s]
221 a.each do |f|
222 if negative?(f)
223 r = num(-f.items[1].value)
224 bottom << (r == ONE ? factor(f.items[0]) : show_power(f.items[0], r))
225 else
226 top << factor(f)
227 end
228 end
229
230 text = top.empty? ? "1" : top.join("*")
231 text += bottom.length == 1 ? "/#{bottom[0]}" : "/(#{bottom.join("*")})" unless bottom.empty?
232 c.negative? ? "-" + text : text
233end
234
235def show(f)
236 case f.kind
237 when 5
238 f.value.denominator == 1 ? f.value.numerator.to_s : f.value.to_s
239 when 0
240 f.value
241 when 4
242 "#{f.value}(#{show(f.items[0])})"
243 when 3
244 negative?(f) ? quotient(Rational(1), [f]) : show_power(*f.items)
245 when 2
246 quotient(*coefficient(f))
247 else
248 f.items.map { |a| show(a) }.reduce { |a, b| a + (b.start_with?("-") ? " - " + b[1..] : " + " + b) }
249 end
250end

252class Parser
253 def initialize(text, names)
254 @tokens = []
255 text.scan(/[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*\/^()]| +|./).each do |t|
256 next if t.start_with?(" ")
257 raise "invalid character '#{t}'" unless /\A[a-zA-Z0-9+*\/^()-]+\z/.match?(t)
258 @tokens << t
259 end
260
261 @names, @p = names, 0
262 end
263
264 def peek = @tokens[@p]
265 def unexpected = raise peek ? "unexpected '#{peek}'" : "unexpected end of expression"
266 def accept(s)
267 return false unless peek == s
268 @p += 1
269 true
270 end
271
272 def expect(s) = accept(s) || unexpected
273 def atom
274 t = peek
275 unexpected unless t
276 if accept("(")
277 a = expression
278 expect(")")
279 return a
280 end
281
282 unexpected unless /\A[a-zA-Z0-9]/.match?(t)
283 @p += 1
284 return num(t.to_i) if /\A[0-9]/.match?(t)
285 return @names.fetch(t) { raise "undefined name '#{t}'" } if /\A[A-Z]/.match?(t)
286 return tree(0, [], t) unless FUNCTIONS.include?(t)
287 expect("(")
288 a = expression
289 expect(")")
290 [t, a]
291 end
292
293 def unary
294 return ["neg", unary] if accept("-")
295 a = atom
296 accept("^") ? ["^", a, unary] : a
297 end
298
299 def term
300 a = unary
301 while %w[* /].include?(peek)
302 op = peek
303 @p += 1
304 a = [op, a, unary]
305 end
306
307 a
308 end
309
310 def expression
311 a = term
312 while %w[+ -].include?(peek)
313 op = peek
314 @p += 1
315 a = [op, a, term]
316 end
317
318 a
319 end
320
321 def evaluate(t)
322 return t if t.is_a?(Expr)
323 op, *a = t
324 a.map! { |x| evaluate(x) }
325 case op
326 when "+"
327 add(*a)
328 when "-"
329 add(a[0], mul(MINUS, a[1]))
330 when "*"
331 mul(*a)
332 when "/"
333 mul(a[0], power(a[1], MINUS))
334 when "^"
335 power(*a)
336 when "neg"
337 mul(MINUS, a[0])
338 else
339 fn(op, a[0])
340 end
341 end
342
343 def parse
344 t = expression
345 unexpected if peek
346 evaluate(t)
347 end
348end

350def variable?(s) = /\A[a-z][a-z0-9]*\z/.match?(s) && !FUNCTIONS.include?(s)
351def literal(s)
352 raise "invalid number '#{s}'" unless /\A-?[0-9]+(?:\/[0-9]+)?\z/.match?(s)
353 p, q = s.split("/").map(&:to_i)
354 raise "division by zero" if q == 0
355 num(p, q || 1)
356end

358SIGNATURES = {
359 "SIMPLIFY" => "e",
360 "EXPAND" => "e",
361 "DIFF" => "ev",
362 "DIFFN" => "evn",
363 "GRAD" => "e",
364 "HESSIAN" => "e",
365 "IMPLICIT" => "evv",
366 "SUBST" => "eve",
367 "EVAL" => "ea",
368 "TANGENT" => "evc",
369 "LIMIT" => "evc",
370 "TAYLOR" => "evcn"
371}.freeze
372def command(line, names)
373 c, separator, argument = line.partition(" ")
374 if c == "LET"
375 raise "wrong number of arguments" unless argument.include?("=")
376 name, _, text = argument.partition("=")
377 name = name.strip
378 raise "invalid name '#{name}'" unless /\A[A-Z][A-Z0-9]*\z/.match?(name)
379 f = Parser.new(text.strip, names).parse
380 names[name] = f
381 return ["#{name} = #{show(f)}"]
382 end
383
384 raise "unknown command #{c}" unless SIGNATURES.key?(c)
385 fields = separator.empty? ? [] : argument.split(";", -1).map(&:strip)
386 sig = SIGNATURES[c]
387 raise "wrong number of arguments" unless c == "EVAL" ? [1, 2].include?(fields.length) : fields.length == sig.length
388 a = fields.each_with_index.map do |s, i|
389 case sig[i]
390 when "e"
391 Parser.new(s, names).parse
392 when "v"
393 raise "invalid variable '#{s}'" unless variable?(s)
394 s
395 when "n"
396 raise "invalid order '#{s}'" unless /\A[1-8]\z/.match?(s)
397 s.to_i
398 when "c"
399 literal(s)
400 else
401 s
402 end
403 end
404
405 f, v, x, n = a
406 case c
407 when "SIMPLIFY"
408 ["simplified: #{show(f)}"]
409 when "EXPAND"
410 ["expanded: #{show(expand(f))}"]
411 when "DIFF"
412 ["d/d#{v} = #{show(derivative(f, v))}"]
413 when "DIFFN"
414 (1..x).map { |k|
415 f = derivative(f, v)
416 "d^#{k}/d#{v}^#{k} = #{show(f)}"
417 }
418 when "GRAD", "HESSIAN"
419 vs = variables(f)
420 return ["no variables"] if vs.empty?
421 vs.each_with_index.flat_map do |s, i|
422 d = derivative(f, s)
423 c == "GRAD" ? ["d/d#{s} = #{show(d)}"] : vs.drop(i).map { |t| "d2/d#{s}d#{t} = #{show(derivative(d, t))}" }
424 end
425
426 when "IMPLICIT"
427 raise "same variable" if v == x
428 ["d#{x}/d#{v} = #{show(mul(MINUS, derivative(f, v), power(derivative(f, x), MINUS)))}"]
429 when "SUBST"
430 ["substituted: #{show(substitute(f, {v => x}))}"]
431 when "EVAL"
432 values = {}
433 (v || "").split.each do |word|
434 m = /\A([a-z][a-z0-9]*)=(-?[0-9]+(?:\/[0-9]+)?)\z/.match(word)
435 raise "invalid assignment '#{word}'" unless m && variable?(m[1])
436 q = literal(m[2])
437 raise "duplicate variable #{m[1]}" if values.key?(m[1])
438 values[m[1]] = q
439 end
440
441 variables(f).each { |s| raise "no value for #{s}" unless values.key?(s) }
442 r = substitute(f, values)
443 [r.kind == 5 ? "exact: #{show(r)}" : format("approx: %.6f", numeric(r))]
444 when "TANGENT"
445 m = substitute(derivative(f, v), {v => x})
446 y = substitute(f, {v => x})
447 variable = tree(0, [], v)
448 tangent = "tangent: y = #{show(add(mul(m, variable), y, mul(num(-x.value), m)))}"
449 return [tangent, "normal: #{v} = #{show(x)}"] if m == ZERO
450 k = power(m, MINUS)
451 [tangent, "normal: y = #{show(add(mul(MINUS, k, variable), y, mul(x, k)))}"]
452 when "LIMIT"
453 top, bottom = [], []
454 (f.kind == 2 ? f.items : [f]).each { |t|
455 negative?(t) ? bottom << power(t.items[0], num(-t.items[1].value)) : top << t
456 }
457 g, h = mul(*top), mul(*bottom)
458 6.times do |k|
459 u, b = substitute(g, {v => x}), substitute(h, {v => x})
460 return ["limit: #{show(mul(u, power(b, MINUS)))}"] unless b == ZERO
461 return ["limit: unbounded"] unless u == ZERO
462 g, h = derivative(g, v), derivative(h, v) if k < 5
463 end
464
465 ["limit: unknown"]
466 when "TAYLOR"
467 ds = [f]
468 n.times { ds << derivative(ds[-1], v) }
469 factorial = 1
470 ps = ds.each_with_index.map do |d, k|
471 factorial *= k if k > 0
472 mul(mul(substitute(d, {v => x}), num(1, factorial)), power(add(tree(0, [], v), num(-x.value)), num(k)))
473 end
474
475 ["taylor: #{show(expand(add(*ps)))}"]
476 end
477end
