7 record Rational(long p, long q) implements Comparable<Rational> {
8 Rational {
9 if (q == 0) throw new IllegalArgumentException("division by zero");
10 if (q<0) {
11 p = -p;
12 q = -q;
13 }
14 long a = Math.abs(p), b = q;
15 while (b != 0) {
16 long t = a%b;
17 a = b;
18 b = t;
19 }
20 p /= a;
21 q /= a;
22 }
23 Rational(long p) {
24 this(p, 1);
25 }
26 Rational add(Rational b) {
27 return new Rational(p*b.q+b.p*q, q*b.q);
28 }
29 Rational multiply(Rational b) {
30 return new Rational(p*b.p, q*b.q);
31 }
32 Rational negate() {
33 return new Rational(-p, q);
34 }
35 public int compareTo(Rational b) {
36 return Long.compare(p*b.q, b.p*q);
37 }
38 }

62 static int compare(Expr a, Expr b) {
63 if (a.kind != b.kind) return Integer.compare(a.kind, b.kind);
64 if (a.kind == 5) return a.q.compareTo(b.q);
65 int c = a.name.compareTo(b.name);

68 c = compare(a.args.get(i), b.args.get(i));
69 if (c != 0) return c;
70 }

102 static int termCompare(Expr a, Expr b) {
103 Map<Expr, Expr>x = new HashMap<>(), y = new HashMap<>();
104 for (Expr f:coefficient(a).factors) x.put(base(f), exponent(f));
105 for (Expr f:coefficient(b).factors) y.put(base(f), exponent(f));
106 Set<Expr>bases = new TreeSet<>(Main::compare);
107 bases.addAll(x.keySet());
108 bases.addAll(y.keySet());
109 for (Expr f:bases) {
110 int c = exponentCompare(y.getOrDefault(f, ZERO), x.getOrDefault(f, ZERO));
111 if (c != 0) return c;
112 }
113 return 0;

118 static Expr add(List<Expr>operands) {
119 List<Expr>terms = new ArrayList<>();
120 for (Expr f:operands) {
121 if (f.kind == 1) terms.addAll(f.args);
122 else if (f.kind == 2 && f.args.size() == 2 && f.args.get(0).kind == 5 && f.args.get(1).kind == 1) {
123 for (Expr t:f.args.get(1).args) terms.add(mul(f.args.get(0), t));
124 } else terms.add(f);
125 }
126 Map<List<Expr>, Rational>groups = new LinkedHashMap<>();
127 for (Expr f:terms) {
128 Parts p = coefficient(f);
129 groups.merge(p.factors, p.coefficient, Rational::add);
130 }
131 List<Expr>built = new ArrayList<>();
132 groups.forEach((m, c) -> {
133 if (c.p != 0) built.add(product(c, m));
134 });
135 built.sort(Main::termCompare);
136 return built.isEmpty()?ZERO:built.size() == 1?built.get(0):node(1, built);
137 }

141 static Expr mul(List<Expr>operands) {
142 Rational c = new Rational(1);
143 Map<Expr, List<Expr>>groups = new LinkedHashMap<>();
144 List<Expr>factors = new ArrayList<>();
145 for (Expr f:operands) factors.addAll(f.kind == 2?f.args:List.of(f));
146 for (Expr f:factors) if (f.kind == 5) c = c.multiply(f.q);
147 else groups.computeIfAbsent(base(f), b -> new ArrayList<>()).add(f);
148 if (c.p == 0) return ZERO;
149 List<Expr>results = new ArrayList<>();
150 groups.forEach((b, g) -> results.add(g.size() == 1?g.get(0):power(b, add(g.stream().map(Main::exponent).toList()))));
151 Set<Expr>bases = new HashSet<>();
152 boolean regroup = false;
153 for (Expr f:results) {
154 if (f.kind == 2 || f.kind == 5 || !bases.add(base(f))) regroup = true;
155 }
156 if (regroup) {
157 results.add(0, number(c));
158 return mul(results);
159 }
160 results.sort((a, b) -> compare(base(a), base(b)));
161 return product(c, results);
162 }
163 static Long root(long v, long d) {
164 if (v == 1) return 1L;
165 if (d>20) return null;
166 long r = Math.round(Math.pow(v, 1.0/d));
167 for (long s = Math.max(0, r-1); s <= r+1; s++) {
168 long a = 1;
169 for (long i = 0; i<d && a <= v; i++) a *= s;
170 if (a == v) return s;
171 }
172 return null;
173 }
174 static Expr rationalPower(Rational b, long n) {
175 if (n<0) {
176 b = new Rational(b.q, b.p);
177 n = -n;
178 }
179 Rational r = new Rational(1);
180 while (n>0) {
181 if (n%2 != 0) r = r.multiply(b);
182 n /= 2;
183 if (n>0) b = b.multiply(b);
184 }
185 return number(r);
186 }
187 static Expr power(Expr b, Expr x) {
188 if (x.equals(ZERO)) return ONE;
189 if (x.equals(ONE)) return b;
190 if (b.kind == 5) {
191 if (b.equals(ONE)) return ONE;
192 if (x.kind != 5) return node(3, List.of(b, x));
193 if (b.q.p == 0) {
194 if (x.q.p>0) return ZERO;
195 throw new IllegalArgumentException("division by zero");
196 }
197 if (integer(x)) return rationalPower(b.q, x.q.p);
198 if (b.q.p>0) {
199 Long s = root(b.q.p, x.q.q), t = root(b.q.q, x.q.q);
200 if (s != null && t != null) return rationalPower(new Rational(s, t), x.q.p);
201 }
202 }
203 else if (integer(x)) {
204 if (b.kind == 3) return power(b.args.get(0), mul(b.args.get(1), x));
205 if (b.kind == 2) return mul(b.args.stream().map(f -> power(f, x)).toList());
206 }
207 return node(3, List.of(b, x));
208 }
209 static Expr fn(String s, Expr a) {
210 if (s.equals("sqrt")) return power(a, HALF);
211 if (a.equals(ZERO) && Set.of("sin", "tan").contains(s)) return ZERO;
212 if (a.equals(ZERO) && Set.of("cos", "exp").contains(s)) return ONE;
213 if (a.equals(ONE) && s.equals("ln")) return ZERO;
214 if (Set.of("sin", "cos", "tan").contains(s) && coefficient(a).coefficient.p<0) {
215 Expr r = fn(s, mul(MINUS, a));
216 return s.equals("cos")?r:mul(MINUS, r);
217 }
218 if (a.kind == 4 && ((s.equals("exp") && a.name.equals("ln")) || (s.equals("ln") && a.name.equals("exp")))) return a.args.get(0);
219 return node(4, List.of(a), s);
220 }

227 static Expr derivative(Expr f, String v) {
228 if (f.kind == 5) return ZERO;
229 if (f.kind == 0) return f.name.equals(v)?ONE:ZERO;
230 if (f.kind == 1) return add(f.args.stream().map(t -> derivative(t, v)).toList());
231 if (f.kind == 2) {
232 List<Expr>r = new ArrayList<>();
233 for (int i = 0; i<f.args.size(); i++) {
234 List<Expr>a = new ArrayList<>(f.args);
235 a.set(i, derivative(a.get(i), v));
236 r.add(mul(a));
237 }
238 return add(r);
239 }
240 if (f.kind == 3) {
241 Expr b = f.args.get(0), x = f.args.get(1);
242 if (!variables(x).contains(v)) return mul(x, power(b, add(x, MINUS)), derivative(b, v));
243 if (!variables(b).contains(v)) return mul(f, fn("ln", b), derivative(x, v));
244 return mul(f, add(mul(derivative(x, v), fn("ln", b)), mul(x, derivative(b, v), power(b, MINUS))));
245 }
246 Expr a = f.args.get(0), h = switch (f.name) {
247 case "sin" -> fn("cos", a);
248 case "cos" -> mul(MINUS, fn("sin", a));
249 case "tan" -> power(fn("cos", a), num(-2));
250 case "exp" -> f;
251 default -> power(a, MINUS);
252 };
253 return mul(h, derivative(a, v));
254 }

276 static Expr expand(Expr f) {
277 if (f.kind == 0 || f.kind == 5) return f;
278 List<Expr>a = f.args.stream().map(Main::expand).toList();
279 if (f.kind == 2) {

291 static double numeric(Expr f) {
292 if (f.kind == 5) return (double) f.q.p/f.q.q;
293 if (f.kind == 1 || f.kind == 2) {
294 double r = f.kind == 1?0:1;
295 for (Expr t:f.args) {
296 double x = numeric(t);
297 if (f.kind == 1) r += x;
298 else r *= x;
299 }
300 return r;
301 }
302 if (f.kind == 3) {
303 double u = numeric(f.args.get(0)), z = numeric(f.args.get(1));
304 boolean integral = f.args.get(1).kind == 5?integer(f.args.get(1)):z == Math.rint(z);
305 if (u<0 && !integral) throw new IllegalArgumentException("domain error");
306 return f.args.get(1).equals(HALF)?Math.sqrt(u):Math.pow(u, z);
307 }
308 double a = numeric(f.args.get(0));
309 if (f.name.equals("ln") && a <= 0) throw new IllegalArgumentException("domain error");
310 return switch (f.name) {
311 case "sin" -> Math.sin(a);
312 case "cos" -> Math.cos(a);
313 case "tan" -> Math.tan(a);
314 case "exp" -> Math.exp(a);
315 default -> Math.log(a);
316 };
317 }

330 static String quotient(Rational c, List<Expr>a) {
331 List<String>top = new ArrayList<>(), bottom = new ArrayList<>();
332 if (Math.abs(c.p) != 1) top.add(Long.toString(Math.abs(c.p)));
333 if (c.q != 1) bottom.add(Long.toString(c.q));
334 for (Expr f:a) if (negative(f)) {
335 Expr r = number(f.args.get(1).q.negate());
336 bottom.add(r.equals(ONE)?factor(f.args.get(0)):showPower(f.args.get(0), r));
337 } else top.add(factor(f));
338 String s = top.isEmpty()?"1":String.join("*", top);
339 if (bottom.size() == 1) s += "/"+bottom.get(0);
340 else if (!bottom.isEmpty()) s += "/("+String.join("*", bottom)+")";
341 return(c.p<0?"-":"")+s;
342 }
343 static String show(Expr f) {
344 if (f.kind == 5) return f.q.p+(f.q.q == 1?"":"/"+f.q.q);
345 if (f.kind == 0) return f.name;
346 if (f.kind == 4) return f.name+"("+show(f.args.get(0))+")";
347 if (f.kind == 3) return negative(f)?quotient(new Rational(1), List.of(f)):showPower(f.args.get(0), f.args.get(1));
348 if (f.kind == 2) {
349 Parts p = coefficient(f);
350 return quotient(p.coefficient, p.factors);
351 }
352 String s = show(f.args.get(0));
353 for (int i = 1; i<f.args.size(); i++) {
354 String t = show(f.args.get(i));
355 s += t.startsWith("-")?" - "+t.substring(1):" + "+t;
356 }
357 return s;

359 static final class Parser {
360 final List<String>tokens = new ArrayList<>();
361 final Map<String, Expr>names;
362 int position;
363 Parser(String text, Map<String, Expr>names) {
364 this.names = names;
365 Matcher m = Pattern.compile("[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*/^()]| +|.").matcher(text);
366 while (m.find()) {
367 String t = m.group();
368 if (t.startsWith(" ")) continue;
369 if (!t.matches("[a-zA-Z0-9+*/^()-]+")) throw new IllegalArgumentException("invalid character '"+t+"'");
370 tokens.add(t);
371 }
372 }
373 String peek() {
374 return position<tokens.size()?tokens.get(position):"";
375 }
376 void unexpected() {
377 throw new IllegalArgumentException(position == tokens.size()?"unexpected end of expression":"unexpected '"+peek()+"'");
378 }
379 boolean accept(String s) {
380 if (!peek().equals(s)) return false;
381 position++;
382 return true;
383 }
384 void expect(String s) {
385 if (!accept(s)) unexpected();
386 }
387 Syntax atomTree() {
388 String t = peek();
389 if (t.isEmpty()) unexpected();
390 if (accept("(")) {
391 Syntax a = expression();
392 expect(")");
393 return a;
394 }
395 if (!Character.isLetterOrDigit(t.charAt(0))) unexpected();
396 position++;
397 if (Character.isDigit(t.charAt(0))) return new Syntax("", num(Long.parseLong(t)), List.of());
398 if (Character.isUpperCase(t.charAt(0))) {
399 if (!names.containsKey(t)) throw new IllegalArgumentException("undefined name '"+t+"'");
400 return new Syntax("", names.get(t), List.of());
401 }
402 if (!FUNCTIONS.contains(t)) return new Syntax("", node(0, List.of(), t), List.of());
403 expect("(");
404 Syntax a = expression();
405 expect(")");
406 return new Syntax(t, null, List.of(a));
407 }
408 Syntax unary() {
409 if (accept("-")) return new Syntax("neg", null, List.of(unary()));
410 Syntax a = atomTree();
411 return accept("^")?new Syntax("^", null, List.of(a, unary())):a;
412 }
413 Syntax term() {
414 Syntax a = unary();
415 while (peek().equals("*") || peek().equals("/")) {
416 String op = tokens.get(position++);
417 a = new Syntax(op, null, List.of(a, unary()));
418 }
419 return a;
420 }
421 Syntax expression() {
422 Syntax a = term();
423 while (peek().equals("+") || peek().equals("-")) {
424 String op = tokens.get(position++);
425 a = new Syntax(op, null, List.of(a, term()));
426 }
427 return a;
428 }
429 Expr evaluate(Syntax t) {
430 if (t.value != null) return t.value;
431 List<Expr>a = t.args.stream().map(this::evaluate).toList();
432 return switch (t.op) {
433 case "+" -> add(a);
434 case "-" -> add(a.get(0), mul(MINUS, a.get(1)));
435 case "*" -> mul(a);
436 case "/" -> mul(a.get(0), power(a.get(1), MINUS));
437 case "^" -> power(a.get(0), a.get(1));
438 case "neg" -> mul(MINUS, a.get(0));
439 default -> fn(t.op, a.get(0));
440 };
441 }
442 Expr parse() {
443 Syntax t = expression();
444 if (position<tokens.size()) unexpected();
445 return evaluate(t);
446 }
447 }

459 static final Map<String, String>SIGNATURES = Map.ofEntries(Map.entry("SIMPLIFY", "e"), Map.entry("EXPAND", "e"), Map.entry("DIFF", "ev"), Map.entry("DIFFN", "evn"), Map.entry("GRAD", "e"), Map.entry("HESSIAN", "e"), Map.entry("IMPLICIT", "evv"), Map.entry("SUBST", "eve"), Map.entry("EVAL", "ea"), Map.entry("TANGENT", "evc"), Map.entry("LIMIT", "evc"), Map.entry("TAYLOR", "evcn"));
460 static List<String>command(String line, Map<String, Expr>names) {
461 int p = line.indexOf(' ');
462 String c = p<0?line:line.substring(0, p), arg = p<0?"":line.substring(p+1);
463 if (c.equals("LET")) {
464 int i = arg.indexOf('=');
465 if (i<0) throw new IllegalArgumentException("wrong number of arguments");
466 String s = arg.substring(0, i).trim();
467 if (!s.matches("[A-Z][A-Z0-9]*")) throw new IllegalArgumentException("invalid name '"+s+"'");
468 Expr f = new Parser(arg.substring(i+1).trim(), names).parse();
469 names.put(s, f);
470 return List.of(s+" = "+show(f));
471 }
472 String sig = SIGNATURES.get(c);
473 if (sig == null) throw new IllegalArgumentException("unknown command "+c);
474 String[]fields = p<0?new String[0]:Arrays.stream(arg.split(";", -1)).map(String::trim).toArray(String[]::new);
475 if (c.equals("EVAL")?(fields.length != 1 && fields.length != 2):fields.length != sig.length()) throw new IllegalArgumentException("wrong number of arguments");
476 Expr[]forms = new Expr[fields.length];
477 Arrays.fill(forms, ZERO);
478 int n = 0;
479 for (int i = 0; i<fields.length; i++) {
480 String s = fields[i];
481 switch (sig.charAt(i)) {
482 case 'e' -> forms[i] = new Parser(s, names).parse();
483 case 'v' -> {
484 if (!variable(s)) throw new IllegalArgumentException("invalid variable '"+s+"'");
485 }
486 case 'n' -> {
487 if (!s.matches("[1-8]")) throw new IllegalArgumentException("invalid order '"+s+"'");
488 n = Integer.parseInt(s);
489 }
490 case 'c' -> forms[i] = literal(s);
491 }
492 }
493 Expr f = forms[0];
494 String v = fields.length>1?fields[1]:"";
495 List<String>out = new ArrayList<>();
496 if (c.equals("SIMPLIFY")) return List.of("simplified: "+show(f));
497 if (c.equals("EXPAND")) return List.of("expanded: "+show(expand(f)));
498 if (c.equals("DIFF")) return List.of("d/d"+v+" = "+show(derivative(f, v)));
499 if (c.equals("DIFFN")) {
500 for (int k = 1; k <= n; k++) {
501 f = derivative(f, v);
502 out.add("d^"+k+"/d"+v+"^"+k+" = "+show(f));
503 }
504 return out;
505 }
506 if (c.equals("GRAD") || c.equals("HESSIAN")) {
507 List<String>vs = new ArrayList<>(variables(f));
508 if (vs.isEmpty()) return List.of("no variables");
509 for (int i = 0; i<vs.size(); i++) {
510 Expr d = derivative(f, vs.get(i));
511 if (c.equals("GRAD")) out.add("d/d"+vs.get(i)+" = "+show(d));
512 else for (int j = i; j<vs.size(); j++) out.add("d2/d"+vs.get(i)+"d"+vs.get(j)+" = "+show(derivative(d, vs.get(j))));
513 }
514 return out;
515 }
516 if (c.equals("IMPLICIT")) {
517 String w = fields[2];
518 if (v.equals(w)) throw new IllegalArgumentException("same variable");
519 return List.of("d"+w+"/d"+v+" = "+show(mul(MINUS, derivative(f, v), power(derivative(f, w), MINUS))));
520 }
521 if (c.equals("SUBST")) return List.of("substituted: "+show(substitute(f, Map.of(v, forms[2]))));
522 if (c.equals("EVAL")) {
523 Map<String, Expr>values = new HashMap<>();
524 for (String word:v.split(" ")) {
525 if (word.isEmpty()) continue;
526 int i = word.indexOf('=');
527 String s = i<0?word:word.substring(0, i), t = i<0?"":word.substring(i+1);
528 if (i<0 || !variable(s) || !rationalLiteral(t)) throw new IllegalArgumentException("invalid assignment '"+word+"'");
529 Expr q = literal(t);
530 if (values.containsKey(s)) throw new IllegalArgumentException("duplicate variable "+s);
531 values.put(s, q);
532 }
533 for (String s:variables(f)) if (!values.containsKey(s)) throw new IllegalArgumentException("no value for "+s);
534 Expr r = substitute(f, values);
535 return List.of(r.kind == 5?"exact: "+show(r):String.format(Locale.ROOT, "approx: %.6f", numeric(r)));
536 }
537 Expr x = forms[2];
538 Map<String, Expr>point = Map.of(v, x);
539 if (c.equals("TANGENT")) {
540 Expr m = substitute(derivative(f, v), point), y = substitute(f, point), z = node(0, List.of(), v);
541 String t = "tangent: y = "+show(add(mul(m, z), y, mul(number(x.q.negate()), m)));
542 if (m.equals(ZERO)) return List.of(t, "normal: "+v+" = "+show(x));
543 Expr k = power(m, MINUS);
544 return List.of(t, "normal: y = "+show(add(mul(MINUS, k, z), y, mul(x, k))));
545 }
546 if (c.equals("LIMIT")) {
547 List<Expr>top = new ArrayList<>(), bottom = new ArrayList<>();
548 for (Expr t:f.kind == 2?f.args:List.of(f)) if (negative(t)) bottom.add(power(t.args.get(0), number(t.args.get(1).q.negate())));
549 else top.add(t);
550 Expr g = mul(top), h = mul(bottom);
551 for (int i = 0; i <= 5; i++) {
552 Expr u = substitute(g, point), b = substitute(h, point);
553 if (!b.equals(ZERO)) return List.of("limit: "+show(mul(u, power(b, MINUS))));
554 if (!u.equals(ZERO)) return List.of("limit: unbounded");
555 if (i<5) {
556 g = derivative(g, v);
557 h = derivative(h, v);
558 }
559 }
560 return List.of("limit: unknown");
561 }
562 List<Expr>ds = new ArrayList<>(List.of(f)), ps = new ArrayList<>();

568 ps.add(mul(co, power(add(node(0, List.of(), v), number(x.q.negate())), num(i))));
569 }
