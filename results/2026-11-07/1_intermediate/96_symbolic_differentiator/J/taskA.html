673 case. 'EVAL' do.
674 ns=.0$a:
675 vals=.0$a:
676 if. n=2 do.

688 end.

693 m=.sigma subst v derivative f
694 y0=.sigma subst f

702 end.

704 if. 2=kind f do. fs=.value f else. fs=.,<f end.
705 ns=.0$a:
706 ds=.0$a:

708 g=.>a
709 negative=.0
710 if. 3=kind g do. e=.exponent g
711 if. isnum e do. negative=.0>value e end.
712 end.
713 if. negative do. ds=.ds,<(base g)pow N -value e else. ns=.ns,a end.
714 end.
715 g=.mul ns
716 h=.mul ds
717 for_k. i.6 do.
718 a=.sigma subst g
719 b=.sigma subst h
720 if. -.iszero b do. out=.,<'limit: ',P mul a;<b pow N _1 break. end.
721 if. -.iszero a do. out=.,<'limit: unbounded' break. end.
722 if. k=5 do. out=.,<'limit: unknown' break. end.
723 g=.v derivative g
724 h=.v derivative h
725 end.
726 case. 'TAYLOR' do.
727 ts=.0$a:
728 fac=.1x
729 for_k. i.order+1 do.
730 if. k do. f=.v derivative f
731 fac=.fac*k end.
732 ck=.mul (sigma subst f);<N %fac
733 pk=.mul ck;<(add (V v);<N -c)pow N k
734 ts=.ts,<pk
735 end.
736 out=.,<'taylor: ',P expand add ts
737 end.
738 out
