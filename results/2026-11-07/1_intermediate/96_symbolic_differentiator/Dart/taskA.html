11final class Rational implements Comparable<Rational> {
12 final int p, q;
13 factory Rational(int p, [int q = 1]) {
14 if (q == 0) fail('division by zero');
15 if (q < 0) {
16 p = -p;
17 q = -q;
18 }
19 final g = p.gcd(q);
20 return Rational._(p ~/ g, q ~/ g);
21 }
22 const Rational._(this.p, this.q);
23 Rational operator +(Rational b) => Rational(p * b.q + b.p * q, q * b.q);
24 Rational operator *(Rational b) => Rational(p * b.p, q * b.q);
25 Rational operator -() => Rational(-p, q);
26 @override
27 int compareTo(Rational b) => (p * b.q).compareTo(b.p * q);
28 @override
29 bool operator ==(Object b) => b is Rational && p == b.p && q == b.q;
30 @override
31 int get hashCode => Object.hash(p, q);
32}

36final class Expr {
37 final Kind kind;
38 final Rational value;
39 final String name;
40 final List<Expr> args;

44Expr number(Rational q) => Expr(Kind.number, q, '', []);
45Expr num(int p, [int q = 1]) => number(Rational(p, q));
46Expr node(Kind k, List<Expr> a, [String s = '']) => Expr(k, Rational(0), s, a);
47final zero = num(0), one = num(1), minus = num(-1), half = num(1, 2);
48const functions = {'sin', 'cos', 'tan', 'exp', 'ln', 'sqrt'};
49int compare(Expr a, Expr b) {
50 if (a.kind != b.kind) return a.kind.index.compareTo(b.kind.index);
51 if (a.kind == Kind.number) return a.value.compareTo(b.value);
52 final c = a.name.compareTo(b.name);
53 if (c != 0) return c;
54 for (var i = 0; i < math.min(a.args.length, b.args.length); i++) {
55 final c = compare(a.args[i], b.args[i]);
56 if (c != 0) return c;
57 }
58 return a.args.length.compareTo(b.args.length);
59}

61bool equal(Expr a, Expr b) => compare(a, b) == 0;
62bool integer(Expr f) => f.kind == Kind.number && f.value.q == 1;
63bool negative(Expr f) =>
64 f.kind == Kind.power &&
65 f.args[1].kind == Kind.number &&
66 f.args[1].value.p < 0;
67Expr base(Expr f) => f.kind == Kind.power ? f.args[0] : f;
68Expr exponent(Expr f) => f.kind == Kind.power ? f.args[1] : one;
69(Rational, List<Expr>) coefficient(Expr f) {
70 if (f.kind == Kind.number) return (f.value, []);
71 if (f.kind == Kind.product)
72 return f.args[0].kind == Kind.number
73 ? (f.args[0].value, f.args.sublist(1))
74 : (Rational(1), f.args);
75 return (Rational(1), [f]);
76}

78Expr product(Rational c, List<Expr> a) {
79 if (a.isEmpty) return number(c);
80 if (c != Rational(1)) a = [number(c), ...a];
81 return a.length == 1 ? a[0] : node(Kind.product, a);
82}

84int exponentCompare(Expr a, Expr b) {
85 if (a.kind == Kind.number && b.kind != Kind.number) return -1;
86 if (a.kind != Kind.number && b.kind == Kind.number) return 1;
87 return compare(a, b);

90int termCompare(Expr a, Expr b) {
91 final x = coefficient(a).$2, y = coefficient(b).$2, bases = <Expr>[];
92 for (final t in [...x, ...y]) {
93 final b = base(t);
94 if (!bases.any((s) => equal(s, b))) bases.add(b);
95 }
96 bases.sort(compare);
97 for (final b in bases) {
98 var u = zero, v = zero;
99 for (final t in x) {
100 if (equal(base(t), b)) u = exponent(t);
101 }
102 for (final t in y) {
103 if (equal(base(t), b)) v = exponent(t);
104 }
105 final c = exponentCompare(v, u);
106 if (c != 0) return c;
107 }
108 return 0;
109}

111bool sameList(List<Expr> a, List<Expr> b) =>
112 a.length == b.length &&
113 List.generate(a.length, (i) => equal(a[i], b[i])).every((v) => v);
114Expr add(List<Expr> operands) {
115 final terms = <Expr>[];
116 for (final f in operands) {
117 if (f.kind == Kind.sum) {
118 terms.addAll(f.args);
119 } else if (f.kind == Kind.product &&
120 f.args.length == 2 &&
121 f.args[0].kind == Kind.number &&
122 f.args[1].kind == Kind.sum) {
123 for (final t in f.args[1].args) {
124 terms.add(mul([f.args[0], t]));
125 }
126 } else {
127 terms.add(f);
128 }
129 }
130 final keys = <List<Expr>>[], values = <Rational>[];
131 for (final f in terms) {
132 final (c, m) = coefficient(f);
133 final i = keys.indexWhere((a) => sameList(a, m));
134 if (i < 0) {
135 keys.add(m);
136 values.add(c);
137 } else {
138 values[i] = values[i] + c;
139 }
140 }
141 final built = [
142 for (var i = 0; i < keys.length; i++)
143 if (values[i].p != 0) product(values[i], keys[i]),
144 ]..sort(termCompare);
145 return built.isEmpty
146 ? zero
147 : built.length == 1
148 ? built[0]
149 : node(Kind.sum, built);
150}

152Expr mul(List<Expr> operands) {
153 var c = Rational(1);
154 final bases = <Expr>[], groups = <List<Expr>>[];
155 for (final f in operands.expand(
156 (t) => t.kind == Kind.product ? t.args : [t],
157 )) {
158 if (f.kind == Kind.number) {
159 c = c * f.value;
160 } else {
161 final b = base(f);
162 var i = bases.indexWhere((t) => equal(t, b));
163 if (i < 0) {
164 bases.add(b);
165 groups.add([]);
166 i = groups.length - 1;
167 }
168 groups[i].add(f);
169 }
170 }
171 if (c.p == 0) return zero;
172 final r = [
173 for (var i = 0; i < groups.length; i++)
174 groups[i].length == 1
175 ? groups[i][0]
176 : power(bases[i], add(groups[i].map(exponent).toList())),
177 ];
178 var regroup = r.any((f) => f.kind == Kind.number || f.kind == Kind.product);
179 for (var i = 0; i < r.length; i++) {
180 for (var j = 0; j < i; j++) {
181 if (equal(base(r[i]), base(r[j]))) regroup = true;
182 }
183 }
184 if (regroup) return mul([number(c), ...r]);
185 r.sort((a, b) => compare(base(a), base(b)));
186 return product(c, r);
187}

189int? root(int v, int d) {
190 if (v == 1) return 1;
191 if (d > 20) return null;
192 final r = math.pow(v, 1 / d).round();
193 for (var s = math.max(0, r - 1); s <= r + 1; s++) {
194 var a = 1;
195 for (var i = 0; i < d && a <= v; i++) {
196 a *= s;
197 }
198 if (a == v) return s;
199 }
200 return null;
201}

203Expr rationalPower(Rational b, int n) {
204 if (n < 0) {
205 b = Rational(b.q, b.p);
206 n = -n;
207 }
208 var r = Rational(1);
209 while (n > 0) {
210 if (n.isOdd) r = r * b;
211 n = n ~/ 2;
212 if (n > 0) b = b * b;
213 }
214 return number(r);
215}

217Expr power(Expr b, Expr x) {
218 if (equal(x, zero)) return one;
219 if (equal(x, one)) return b;
220 if (b.kind == Kind.number) {
221 if (equal(b, one)) return one;
222 if (x.kind != Kind.number) return node(Kind.power, [b, x]);
223 if (b.value.p == 0) {
224 if (x.value.p > 0) return zero;
225 fail('division by zero');
226 }
227 if (integer(x)) return rationalPower(b.value, x.value.p);
228 if (b.value.p > 0) {
229 final s = root(b.value.p, x.value.q), t = root(b.value.q, x.value.q);
230 if (s != null && t != null)
231 return rationalPower(Rational(s, t), x.value.p);
232 }
233 } else if (integer(x)) {
234 if (b.kind == Kind.power) return power(b.args[0], mul([b.args[1], x]));
235 if (b.kind == Kind.product)
236 return mul(b.args.map((f) => power(f, x)).toList());
237 }
238 return node(Kind.power, [b, x]);
239}

241Expr fn(String s, Expr a) {
242 if (s == 'sqrt') return power(a, half);
243 if (equal(a, zero) && {'sin', 'tan'}.contains(s)) return zero;
244 if (equal(a, zero) && {'cos', 'exp'}.contains(s)) return one;
245 if (equal(a, one) && s == 'ln') return zero;
246 if ({'sin', 'cos', 'tan'}.contains(s) && coefficient(a).$1.p < 0) {
247 final r = fn(s, mul([minus, a]));
248 return s == 'cos' ? r : mul([minus, r]);
249 }
250 if (a.kind == Kind.function &&
251 ((s == 'exp' && a.name == 'ln') || (s == 'ln' && a.name == 'exp')))
252 return a.args[0];
253 return node(Kind.function, [a], s);
254}

256List<String> variables(Expr f) {
257 final r = <String>{if (f.kind == Kind.variable) f.name};
258 for (final a in f.args) {
259 r.addAll(variables(a));
260 }
261 return r.toList()..sort();

264Expr derivative(Expr f, String v) {
265 if (f.kind == Kind.number) return zero;
266 if (f.kind == Kind.variable) return f.name == v ? one : zero;
267 if (f.kind == Kind.sum)
268 return add(f.args.map((t) => derivative(t, v)).toList());
269 if (f.kind == Kind.product)
270 return add([
271 for (var i = 0; i < f.args.length; i++)
272 mul([
273 for (var j = 0; j < f.args.length; j++)
274 i == j ? derivative(f.args[j], v) : f.args[j],
275 ]),
276 ]);
277 if (f.kind == Kind.power) {
278 final b = f.args[0], x = f.args[1];
279 if (!variables(x).contains(v))
280 return mul([
281 x,
282 power(b, add([x, minus])),
283 derivative(b, v),
284 ]);
285 if (!variables(b).contains(v))
286 return mul([f, fn('ln', b), derivative(x, v)]);
287 return mul([
288 f,
289 add([
290 mul([derivative(x, v), fn('ln', b)]),
291 mul([x, derivative(b, v), power(b, minus)]),
292 ]),
293 ]);
294 }
295 final a = f.args[0],
296 h = switch (f.name) {
297 'sin' => fn('cos', a),
298 'cos' => mul([minus, fn('sin', a)]),
299 'tan' => power(fn('cos', a), num(-2)),
300 'exp' => f,
301 _ => power(a, minus),
302 };
303 return mul([h, derivative(a, v)]);
304}

306Expr rebuild(Expr f, List<Expr> a) => switch (f.kind) {
307 Kind.sum => add(a),
308 Kind.product => mul(a),
309 Kind.power => power(a[0], a[1]),
310 _ => fn(f.name, a[0]),
311};
312Expr substitute(Expr f, Map<String, Expr> m) {
313 if (f.kind == Kind.variable) return m[f.name] ?? f;
314 if (f.kind == Kind.number) return f;
315 return rebuild(f, f.args.map((t) => substitute(t, m)).toList());
316}

318List<Expr> terms(Expr f) => f.kind == Kind.sum ? f.args : [f];
319Expr distribute(Expr a, Expr b) => add([
320 for (final x in terms(a))
321 for (final y in terms(b)) mul([x, y]),
322]);
323Expr expand(Expr f) {
324 if (f.kind == Kind.variable || f.kind == Kind.number) return f;
325 final a = f.args.map(expand).toList();
326 if (f.kind == Kind.product) return a.reduce(distribute);
327 if (f.kind == Kind.power &&
328 a[0].kind == Kind.sum &&
329 integer(a[1]) &&
330 a[1].value.p > 0) {
331 var r = a[0];
332 for (var i = 1; i < a[1].value.p; i++) {
333 r = distribute(r, a[0]);
334 }
335 return r;
336 }
337 return rebuild(f, a);
338}

340double numeric(Expr f) {
341 if (f.kind == Kind.number) return f.value.p / f.value.q;
342 if (f.kind == Kind.sum) return f.args.fold(0.0, (r, t) => r + numeric(t));
343 if (f.kind == Kind.product) return f.args.fold(1.0, (r, t) => r * numeric(t));
344 if (f.kind == Kind.power) {
345 final u = numeric(f.args[0]), z = numeric(f.args[1]);
346 final integral = f.args[1].kind == Kind.number
347 ? integer(f.args[1])
348 : z == z.truncateToDouble();
349 if (u < 0 && !integral) fail('domain error');
350 return equal(f.args[1], half) ? math.sqrt(u) : math.pow(u, z).toDouble();
351 }
352 final a = numeric(f.args[0]);
353 if (f.name == 'ln' && a <= 0) fail('domain error');
354 return switch (f.name) {
355 'sin' => math.sin(a),
356 'cos' => math.cos(a),
357 'tan' => math.tan(a),
358 'exp' => math.exp(a),
359 _ => math.log(a),
360 };
361}

363bool atomic(Expr f) =>
364 f.kind == Kind.variable ||
365 f.kind == Kind.function ||
366 (integer(f) && f.value.p >= 0) ||
367 (f.kind == Kind.power && equal(f.args[1], half));
368String atom(Expr f) => atomic(f) ? show(f) : '(${show(f)})';
369String showPower(Expr b, Expr x) =>
370 equal(x, half) ? 'sqrt(${show(b)})' : '${atom(b)}^${atom(x)}';
371String factor(Expr f) => f.kind == Kind.sum
372 ? '(${show(f)})'
373 : f.kind == Kind.power
374 ? showPower(f.args[0], f.args[1])
375 : show(f);
376String quotient(Rational c, List<Expr> a) {
377 final top = <String>[if (c.p.abs() != 1) '${c.p.abs()}'],
378 bottom = <String>[if (c.q != 1) '${c.q}'];
379 for (final f in a) {
380 if (negative(f)) {
381 final r = number(-f.args[1].value);
382 bottom.add(equal(r, one) ? factor(f.args[0]) : showPower(f.args[0], r));
383 } else {
384 top.add(factor(f));
385 }
386 }
387 var s = top.isEmpty ? '1' : top.join('*');
388 if (bottom.length == 1) {
389 s += '/' + bottom[0];
390 } else if (bottom.isNotEmpty) {
391 s += '/(${bottom.join('*')})';
392 }
393 return (c.p < 0 ? '-' : '') + s;
394}

396String show(Expr f) {
397 if (f.kind == Kind.number)
398 return '${f.value.p}' + (f.value.q == 1 ? '' : '/${f.value.q}');
399 if (f.kind == Kind.variable) return f.name;
400 if (f.kind == Kind.function) return '${f.name}(${show(f.args[0])})';
401 if (f.kind == Kind.power)
402 return negative(f)
403 ? quotient(Rational(1), [f])
404 : showPower(f.args[0], f.args[1]);
405 if (f.kind == Kind.product) {
406 final (c, a) = coefficient(f);
407 return quotient(c, a);
408 }
409 return f.args
410 .map(show)
411 .reduce(
412 (a, b) => a + (b.startsWith('-') ? ' - ' + b.substring(1) : ' + ' + b),
413 );
414}

416final class Syntax {
417 final String op;
418 final Expr? value;
419 final List<Syntax> args;
420 Syntax(this.op, this.value, [this.args = const []]);
421}

423final class Parser {
424 final Map<String, Expr> names;
425 final tokens = <String>[];
426 int position = 0;
427 Parser(String text, this.names) {
428 for (final m in RegExp(
429 r'[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*/^()]| +|.',
430 ).allMatches(text)) {
431 final t = m[0]!;
432 if (t.startsWith(' ')) continue;
433 if (!RegExp(r'^[a-zA-Z0-9+*/^()-]+$').hasMatch(t))
434 fail("invalid character '$t'");
435 tokens.add(t);
436 }
437 }
438 String get peek => position < tokens.length ? tokens[position] : '';
439 Never unexpected() => fail(
440 position == tokens.length
441 ? 'unexpected end of expression'
442 : "unexpected '$peek'",
443 );
444 bool accept(String s) {
445 if (peek != s) return false;
446 position++;
447 return true;
448 }
449
450 void expect(String s) {
451 if (!accept(s)) unexpected();
452 }
453
454 Syntax atomTree() {
455 final t = peek;
456 if (t.isEmpty) unexpected();
457 if (accept('(')) {
458 final a = expression();
459 expect(')');
460 return a;
461 }
462 if (!RegExp(r'^[a-zA-Z0-9]').hasMatch(t)) unexpected();
463 position++;
464 if (RegExp(r'^[0-9]').hasMatch(t)) return Syntax('', num(int.parse(t)));
465 if (RegExp(r'^[A-Z]').hasMatch(t)) {
466 if (!names.containsKey(t)) fail("undefined name '$t'");
467 return Syntax('', names[t]!);
468 }
469 if (!functions.contains(t)) return Syntax('', node(Kind.variable, [], t));
470 expect('(');
471 final a = expression();
472 expect(')');
473 return Syntax(t, null, [a]);
474 }
475
476 Syntax unary() {
477 if (accept('-')) return Syntax('neg', null, [unary()]);
478 final a = atomTree();
479 return accept('^') ? Syntax('^', null, [a, unary()]) : a;
480 }
481
482 Syntax term() {
483 var a = unary();
484 while (peek == '*' || peek == '/') {
485 final op = tokens[position++];
486 a = Syntax(op, null, [a, unary()]);
487 }
488 return a;
489 }
490
491 Syntax expression() {
492 var a = term();
493 while (peek == '+' || peek == '-') {
494 final op = tokens[position++];
495 a = Syntax(op, null, [a, term()]);
496 }
497 return a;
498 }
499
500 Expr evaluate(Syntax t) {
501 if (t.value != null) return t.value!;
502 final a = t.args.map(evaluate).toList();
503 return switch (t.op) {
504 '+' => add(a),
505 '-' => add([
506 a[0],
507 mul([minus, a[1]]),
508 ]),
509 '*' => mul(a),
510 '/' => mul([a[0], power(a[1], minus)]),
511 '^' => power(a[0], a[1]),
512 'neg' => mul([minus, a[0]]),
513 _ => fn(t.op, a[0]),
514 };
515 }
516
517 Expr parse() {
518 final t = expression();
519 if (position < tokens.length) unexpected();
520 return evaluate(t);
521 }
522}

524bool variable(String s) =>
525 RegExp(r'^[a-z][a-z0-9]*$').hasMatch(s) && !functions.contains(s);
526bool rationalLiteral(String s) => RegExp(r'^-?[0-9]+(?:/[0-9]+)?$').hasMatch(s);
527Expr literal(String s) {
528 if (!rationalLiteral(s)) fail("invalid number '$s'");
529 final a = s.split('/');
530 return num(int.parse(a[0]), a.length == 1 ? 1 : int.parse(a[1]));
531}

533const signatures = {
534 'SIMPLIFY': 'e',
535 'EXPAND': 'e',
536 'DIFF': 'ev',
537 'DIFFN': 'evn',
538 'GRAD': 'e',
539 'HESSIAN': 'e',
540 'IMPLICIT': 'evv',
541 'SUBST': 'eve',
542 'EVAL': 'ea',
543 'TANGENT': 'evc',
544 'LIMIT': 'evc',
545 'TAYLOR': 'evcn',
546};
547List<String> command(String line, Map<String, Expr> names) {
548 final p = line.indexOf(' '),
549 c = p < 0 ? line : line.substring(0, p),
550 arg = p < 0 ? '' : line.substring(p + 1);
551 if (c == 'LET') {
552 final i = arg.indexOf('=');
553 if (i < 0) fail('wrong number of arguments');
554 final s = arg.substring(0, i).trim();
555 if (!RegExp(r'^[A-Z][A-Z0-9]*$').hasMatch(s)) fail("invalid name '$s'");
556 final f = Parser(arg.substring(i + 1).trim(), names).parse();
557 names[s] = f;
558 return ['$s = ${show(f)}'];
559 }
560 final sig = signatures[c];
561 if (sig == null) fail('unknown command $c');
562 final fields = p < 0
563 ? <String>[]
564 : arg.split(';').map((s) => s.trim()).toList();
565 if (c == 'EVAL'
566 ? !{1, 2}.contains(fields.length)
567 : fields.length != sig.length)
568 fail('wrong number of arguments');
569 final forms = List.filled(fields.length, zero);
570 var n = 0;
571 for (var i = 0; i < fields.length; i++) {
572 final s = fields[i];
573 switch (sig[i]) {
574 case 'e':
575 forms[i] = Parser(s, names).parse();
576 case 'v':
577 if (!variable(s)) fail("invalid variable '$s'");
578 case 'n':
579 if (!RegExp(r'^[1-8]$').hasMatch(s)) fail("invalid order '$s'");
580 n = int.parse(s);
581 case 'c':
582 forms[i] = literal(s);
583 }
584 }
585 var f = forms[0];
586 final v = fields.length > 1 ? fields[1] : '';
587 final out = <String>[];
588 if (c == 'SIMPLIFY') return ['simplified: ${show(f)}'];
589 if (c == 'EXPAND') return ['expanded: ${show(expand(f))}'];
590 if (c == 'DIFF') return ['d/d$v = ${show(derivative(f, v))}'];
591 if (c == 'DIFFN') {
592 for (var k = 1; k <= n; k++) {
593 f = derivative(f, v);
594 out.add('d^$k/d$v^$k = ${show(f)}');
595 }
596 return out;
597 }
598 if (c == 'GRAD' || c == 'HESSIAN') {
599 final vs = variables(f);
600 if (vs.isEmpty) return ['no variables'];
601 for (var i = 0; i < vs.length; i++) {
602 final d = derivative(f, vs[i]);
603 if (c == 'GRAD') {
604 out.add('d/d${vs[i]} = ${show(d)}');
605 } else {
606 for (var j = i; j < vs.length; j++) {
607 out.add('d2/d${vs[i]}d${vs[j]} = ${show(derivative(d, vs[j]))}');
608 }
609 }
610 }
611 return out;
612 }
613 if (c == 'IMPLICIT') {
614 final w = fields[2];
615 if (v == w) fail('same variable');
616 return [
617 'd$w/d$v = ${show(mul([minus, derivative(f, v), power(derivative(f, w), minus)]))}',
618 ];
619 }
620 if (c == 'SUBST')
621 return [
622 'substituted: ${show(substitute(f, {v: forms[2]}))}',
623 ];
624 if (c == 'EVAL') {
625 final values = <String, Expr>{};
626 for (final word in v.split(' ').where((s) => s.isNotEmpty)) {
627 final i = word.indexOf('='),
628 s = i < 0 ? word : word.substring(0, i),
629 t = i < 0 ? '' : word.substring(i + 1);
630 if (i < 0 || !variable(s) || !rationalLiteral(t))
631 fail("invalid assignment '$word'");
632 final q = literal(t);
633 if (values.containsKey(s)) fail('duplicate variable $s');
634 values[s] = q;
635 }
636 for (final s in variables(f)) {
637 if (!values.containsKey(s)) fail('no value for $s');
638 }
639 final r = substitute(f, values);
640 return [
641 r.kind == Kind.number
642 ? 'exact: ${show(r)}'
643 : 'approx: ${numeric(r).toStringAsFixed(6)}',
644 ];
645 }
646 final x = forms[2], point = {v: forms[2]};
647 if (c == 'TANGENT') {
648 final m = substitute(derivative(f, v), point),
649 y = substitute(f, point),
650 z = node(Kind.variable, [], v);
651 final t =
652 'tangent: y = ${show(add([
653 mul([m, z]),
654 y,
655 mul([number(-x.value), m]),
656 ]))}';
657 if (equal(m, zero)) return [t, 'normal: $v = ${show(x)}'];
658 final k = power(m, minus);
659 return [
660 t,
661 'normal: y = ${show(add([
662 mul([minus, k, z]),
663 y,
664 mul([x, k]),
665 ]))}',
666 ];
667 }
668 if (c == 'LIMIT') {
669 final top = <Expr>[], bottom = <Expr>[];
670 for (final t in f.kind == Kind.product ? f.args : [f]) {
671 if (negative(t)) {
672 bottom.add(power(t.args[0], number(-t.args[1].value)));
673 } else {
674 top.add(t);
675 }
676 }
677 var g = mul(top), h = mul(bottom);
678 for (var i = 0; i <= 5; i++) {
679 final u = substitute(g, point), b = substitute(h, point);
680 if (!equal(b, zero))
681 return [
682 'limit: ${show(mul([u, power(b, minus)]))}',
683 ];
684 if (!equal(u, zero)) return ['limit: unbounded'];
685 if (i < 5) {
686 g = derivative(g, v);
687 h = derivative(h, v);
688 }
689 }
690 return ['limit: unknown'];
691 }
692 final ds = [f], ps = <Expr>[];
693 for (var i = 0; i < n; i++) {
694 ds.add(derivative(ds.last, v));
695 }
696 var factorial = 1;
697 for (var k = 0; k <= n; k++) {
698 if (k > 0) factorial *= k;
699 final co = mul([substitute(ds[k], point), num(1, factorial)]);
700 ps.add(
701 mul([
702 co,
703 power(add([node(Kind.variable, [], v), number(-x.value)]), num(k)),
704 ]),
705 );
706 }
707 return ['taylor: ${show(expand(add(ps)))}'];
708}
