7readonly struct Rational : IComparable<Rational>, IEquatable<Rational> {
8 public readonly long P, Q;
9 public Rational(long p, long q = 1) {
10 if (q == 0)
11 throw new InvalidOperationException("division by zero");
12 if (q < 0) {
13 p = -p;
14 q = -q;
15 }
16 long a = Math.Abs(p), b = q;
17 while (b != 0) {
18 long t = a % b;
19 a = b;
20 b = t;
21 }
22 P = p / a;
23 Q = q / a;
24 }
25 public int CompareTo(Rational b) => (P * b.Q).CompareTo(b.P * Q);
26 public bool Equals(Rational b) => P == b.P && Q == b.Q;
27 public override bool Equals(object? b) => b is Rational r && Equals(r);
28 public override int GetHashCode() => HashCode.Combine(P, Q);
29 public static Rational operator +(Rational a, Rational b) => new(a.P * b.Q + b.P * a.Q,
30 a.Q *b.Q);
31 public static Rational operator *(Rational a, Rational b) => new(a.P * b.P, a.Q *b.Q);
32 public static Rational operator -(Rational a) => new(-a.P, a.Q);
33 public static bool operator ==(Rational a, Rational b) => a.Equals(b);
34 public static bool operator !=(Rational a, Rational b) => !a.Equals(b);
35}
36enum Kind { Variable, Sum, Product, Power, Function, Number }
37sealed record Expr(Kind Kind, Rational Value, string Name, Expr[] Args);
38sealed record Syntax(string Op, Expr? Value, Syntax[] Args);
39static class Program {
40 static readonly HashSet<string> Functions = new() { "sin", "cos", "tan", "exp", "ln", "sqrt" };
41 static Expr Num(long p, long q = 1) => new(Kind.Number, new Rational(p, q), "", []);
42 static Expr Num(Rational q) => new(Kind.Number, q, "", []);
43 static Expr Node(Kind k, Expr[] a, string s = "") => new(k, new Rational(0), s, a);
44 static readonly Expr Zero = Num(0), One = Num(1), Minus = Num(-1), Half = Num(1, 2);
45 static int Compare(Expr a, Expr b) {
46 if (a.Kind != b.Kind)
47 return a.Kind.CompareTo(b.Kind);
48 if (a.Kind == Kind.Number)
49 return a.Value.CompareTo(b.Value);
50 int c = StringComparer.Ordinal.Compare(a.Name, b.Name);
51 if (c != 0)
52 return c;
53 for (int i = 0; i < Math.Min(a.Args.Length, b.Args.Length); i++) {
54 c = Compare(a.Args[i], b.Args[i]);
55 if (c != 0)
56 return c;
57 }
58 return a.Args.Length.CompareTo(b.Args.Length);
59 }
60 static bool Equal(Expr a, Expr b) => Compare(a, b) == 0;
61 static bool Integer(Expr f) => f.Kind == Kind.Number && f.Value.Q == 1;
62 static bool Negative(Expr f) => f.Kind == Kind.Power && f.Args[1].Kind == Kind.Number
63 && f.Args[1].Value.P < 0;
64 static Expr Base(Expr f) => f.Kind == Kind.Power ? f.Args[0] : f;
65 static Expr Exponent(Expr f) => f.Kind == Kind.Power ? f.Args[1] : One;
66 static (Rational, Expr[]) Coefficient(Expr f) {
67 if (f.Kind == Kind.Number)
68 return (f.Value, []);
69 if (f.Kind == Kind.Product)
70 return f.Args[0].Kind == Kind.Number ? (f.Args[0].Value, f.Args[1..])
71 : (new Rational(1), f.Args);
72 return (new Rational(1), [f]);
73 }
74 static Expr Product(Rational c, Expr[] a) {
75 if (a.Length == 0)
76 return Num(c);
77 if (c != new Rational(1))
78 a = [Num(c), ..a];
79 return a.Length == 1 ? a[0] : Node(Kind.Product, a);
80 }
81 static int ExponentCompare(Expr a, Expr b) {
82 if (a.Kind == Kind.Number && b.Kind != Kind.Number)
83 return -1;
84 if (a.Kind != Kind.Number && b.Kind == Kind.Number)
85 return 1;
86 return Compare(a, b);
87 }
88 static int TermCompare(Expr a, Expr b) {
89 Expr[] x = Coefficient(a).Item2, y = Coefficient(b).Item2;
90 List<Expr> bases = [];
91 foreach (Expr f in x.Concat(y)) {
92 Expr v = Base(f);
93 if (!bases.Any(t => Equal(t, v)))
94 bases.Add(v);
95 }
96 bases.Sort(Compare);
97 foreach (Expr f in bases) {
98 Expr? s = x.FirstOrDefault(t => Equal(Base(t), f)),
99 t = y.FirstOrDefault(g => Equal(Base(g), f));
100 int c = ExponentCompare(t == null ? Zero : Exponent(t), s == null ? Zero : Exponent(s));
101 if (c != 0)
102 return c;
103 }
104 return 0;
105 }
106 static Expr Add(params Expr[] operands) {
107 List<Expr> terms = [];
108 foreach (Expr f in operands) {
109 if (f.Kind == Kind.Sum)
110 terms.AddRange(f.Args);
111 else if (f.Kind == Kind.Product && f.Args.Length == 2 &&
112 f.Args[0].Kind == Kind.Number && f.Args[1].Kind == Kind.Sum)
113 foreach (Expr t in f.Args[1].Args)
114 terms.Add(Mul(f.Args[0], t));
115 else
116 terms.Add(f);
117 }
118 List<Expr[]> keys = [];
119 List<Rational> values = [];
120 foreach (Expr f in terms) {
121 var (c, m) = Coefficient(f);
122 int i = keys.FindIndex(a => a.Length == m.Length &&
123 a.Zip(m).All(p => Equal(p.First, p.Second)));
124 if (i < 0) {
125 keys.Add(m);
126 values.Add(c);
127 } else
128 values[i] += c;
129 }
130 List<Expr> built = [];
131 for (int i = 0; i < keys.Count; i++)
132 if (values[i].P != 0)
133 built.Add(Product(values[i], keys[i]));
134 built.Sort(TermCompare);
135 return built.Count == 0 ? Zero
136 : built.Count == 1 ? built[0]
137 : Node(Kind.Sum, built.ToArray());
138 }
139 static Expr Mul(params Expr[] operands) {
140 Rational c = new(1);
141 List<Expr> bases = [];
142 List<List<Expr>> groups = [];
143 foreach (Expr f in operands.SelectMany(t => t.Kind == Kind.Product ? t.Args : [t])) {
144 if (f.Kind == Kind.Number)
145 c *= f.Value;
146 else {
147 Expr b = Base(f);
148 int i = bases.FindIndex(t => Equal(t, b));
149 if (i < 0) {
150 bases.Add(b);
151 groups.Add([]);
152 i = groups.Count - 1;
153 }
154 groups[i].Add(f);
155 }
156 }
157 if (c.P == 0)
158 return Zero;
159 List<Expr> r = [];
160 for (int i = 0; i < groups.Count; i++)
161 r.Add(groups[i].Count == 1
162 ? groups[i][0]
163 : Power(bases[i], Add(groups[i].Select(Exponent).ToArray())));
164 bool regroup = r.Any(f => f.Kind is Kind.Number or Kind.Product);
165 for (int i = 0; i < r.Count; i++)
166 for (int j = 0; j < i; j++)
167 if (Equal(Base(r[i]), Base(r[j])))
168 regroup = true;
169 if (regroup)
170 return Mul([Num(c), ..r]);
171 r.Sort((a, b) => Compare(Base(a), Base(b)));
172 return Product(c, r.ToArray());
173 }
174 static long? Root(long v, long d) {
175 if (v == 1)
176 return 1;
177 if (d > 20)
178 return null;
179 long r = (long)Math.Round(Math.Pow(v, 1.0 / d));
180 for (long s = Math.Max(0, r - 1); s <= r + 1; s++) {
181 long a = 1;
182 for (int i = 0; i < d && a <= v; i++)
183 a *= s;
184 if (a == v)
185 return s;
186 }
187 return null;
188 }
189 static Expr RationalPower(Rational b, long n) {
190 if (n < 0) {
191 b = new Rational(b.Q, b.P);
192 n = -n;
193 }
194 Rational r = new(1);
195 while (n > 0) {
196 if (n % 2 != 0)
197 r *= b;
198 n /= 2;
199 if (n > 0)
200 b *= b;
201 }
202 return Num(r);
203 }
204 static Expr Power(Expr b, Expr x) {
205 if (Equal(x, Zero))
206 return One;
207 if (Equal(x, One))
208 return b;
209 if (b.Kind == Kind.Number) {
210 if (Equal(b, One))
211 return One;
212 if (x.Kind != Kind.Number)
213 return Node(Kind.Power, [b, x]);
214 if (b.Value.P == 0) {
215 if (x.Value.P > 0)
216 return Zero;
217 throw new InvalidOperationException("division by zero");
218 }
219 if (Integer(x))
220 return RationalPower(b.Value, x.Value.P);
221 if (b.Value.P > 0) {
222 long? s = Root(b.Value.P, x.Value.Q), t = Root(b.Value.Q, x.Value.Q);
223 if (s.HasValue && t.HasValue)
224 return RationalPower(new Rational(s.Value, t.Value), x.Value.P);
225 }
226 } else if (Integer(x)) {
227 if (b.Kind == Kind.Power)
228 return Power(b.Args[0], Mul(b.Args[1], x));
229 if (b.Kind == Kind.Product)
230 return Mul(b.Args.Select(f => Power(f, x)).ToArray());
231 }
232 return Node(Kind.Power, [b, x]);
233 }
234 static Expr Fn(string s, Expr a) {
235 if (s == "sqrt")
236 return Power(a, Half);
237 if (Equal(a, Zero) && (s is "sin" or "tan"))
238 return Zero;
239 if (Equal(a, Zero) && (s is "cos" or "exp"))
240 return One;
241 if (Equal(a, One) && s == "ln")
242 return Zero;
243 if ((s is "sin" or "cos" or "tan") && Coefficient(a).Item1.P < 0) {
244 Expr r = Fn(s, Mul(Minus, a));
245 return s == "cos" ? r : Mul(Minus, r);
246 }
247 if (a.Kind == Kind.Function &&
248 ((s == "exp" && a.Name == "ln") || (s == "ln" && a.Name == "exp")))
249 return a.Args[0];
250 return Node(Kind.Function, [a], s);
251 }
252 static SortedSet<string> Variables(Expr f) {
253 SortedSet<string> r = new(StringComparer.Ordinal);
254 if (f.Kind == Kind.Variable)
255 r.Add(f.Name);
256 foreach (Expr a in f.Args)
257 r.UnionWith(Variables(a));
258 return r;
259 }
260 static Expr Derivative(Expr f, string v) {
261 if (f.Kind == Kind.Number)
262 return Zero;
263 if (f.Kind == Kind.Variable)
264 return f.Name == v ? One : Zero;
265 if (f.Kind == Kind.Sum)
266 return Add(f.Args.Select(t => Derivative(t, v)).ToArray());
267 if (f.Kind == Kind.Product)
268 return Add(
269 f.Args
270 .Select((_, i) => Mul(
271 f.Args.Select((t, j) => i == j ? Derivative(t, v) : t).ToArray()))
272 .ToArray());
273 if (f.Kind == Kind.Power) {
274 Expr b = f.Args[0], x = f.Args[1];
275 if (!Variables(x).Contains(v))
276 return Mul(x, Power(b, Add(x, Minus)), Derivative(b, v));
277 if (!Variables(b).Contains(v))
278 return Mul(f, Fn("ln", b), Derivative(x, v));
279 return Mul(f, Add(Mul(Derivative(x, v), Fn("ln", b)),
280 Mul(x, Derivative(b, v), Power(b, Minus))));
281 }
282 Expr a = f.Args[0],
283 h = f.Name switch { "sin" => Fn("cos", a), "cos" => Mul(Minus, Fn("sin", a)),
284 "tan" => Power(Fn("cos", a), Num(-2)), "exp" => f,
285 _ => Power(a, Minus) };
286 return Mul(h, Derivative(a, v));
287 }
288 static Expr Rebuild(Expr f, Expr[] a) => f.Kind
289 switch { Kind.Sum => Add(a), Kind.Product => Mul(a),
290 Kind.Power => Power(a[0], a[1]),
291 _ => Fn(f.Name, a[0]) };
292 static Expr Substitute(Expr f, Dictionary<string, Expr> m) {
293 if (f.Kind == Kind.Variable)
294 return m.GetValueOrDefault(f.Name, f);
295 if (f.Kind == Kind.Number)
296 return f;
297 return Rebuild(f, f.Args.Select(t => Substitute(t, m)).ToArray());
298 }
299 static Expr[] Terms(Expr f) => f.Kind == Kind.Sum ? f.Args : [f];
300 static Expr Distribute(Expr a, Expr b) =>
301 Add(Terms(a).SelectMany(x => Terms(b).Select(y => Mul(x, y))).ToArray());
302 static Expr Expand(Expr f) {
303 if (f.Kind is Kind.Variable or Kind.Number)
304 return f;
305 Expr[] a = f.Args.Select(Expand).ToArray();
306 if (f.Kind == Kind.Product)
307 return a.Aggregate(Distribute);
308 if (f.Kind == Kind.Power && a[0].Kind == Kind.Sum && Integer(a[1]) && a[1].Value.P > 0) {
309 Expr r = a[0];
310 for (int i = 1; i < a[1].Value.P; i++)
311 r = Distribute(r, a[0]);
312 return r;
313 }
314 return Rebuild(f, a);
315 }
316 static double Numeric(Expr f) {
317 if (f.Kind == Kind.Number)
318 return (double)f.Value.P / f.Value.Q;
319 if (f.Kind == Kind.Sum)
320 return f.Args.Aggregate(0.0, (r, t) => r + Numeric(t));
321 if (f.Kind == Kind.Product)
322 return f.Args.Aggregate(1.0, (r, t) => r * Numeric(t));
323 if (f.Kind == Kind.Power) {
324 double u = Numeric(f.Args[0]), z = Numeric(f.Args[1]);
325 bool integral =
326 f.Args[1].Kind == Kind.Number ? Integer(f.Args[1]) : z == Math.Truncate(z);
327 if (u < 0 && !integral)
328 throw new InvalidOperationException("domain error");
329 return Equal(f.Args[1], Half) ? Math.Sqrt(u) : Math.Pow(u, z);
330 }
331 double a = Numeric(f.Args[0]);
332 if (f.Name == "ln" && a <= 0)
333 throw new InvalidOperationException("domain error");
334 return f.Name switch { "sin" => Math.Sin(a), "cos" => Math.Cos(a), "tan" => Math.Tan(a),
335 "exp" => Math.Exp(a),
336 _ => Math.Log(a) };
337 }
338 static bool Atomic(Expr f) => f.Kind is Kind.Variable or Kind.Function ||
339 (Integer(f) && f.Value.P >= 0) ||
340 (f.Kind == Kind.Power && Equal(f.Args[1], Half));
341 static string Atom(Expr f) => Atomic(f) ? Show(f) : "(" + Show(f) + ")";
342 static string ShowPower(Expr b, Expr x) => Equal(x, Half) ? "sqrt(" + Show(b) + ")"
343 : Atom(b) + "^" + Atom(x);
344 static string Factor(Expr f) => f.Kind == Kind.Sum ? "(" + Show(f) + ")"
345 : f.Kind == Kind.Power ? ShowPower(f.Args[0], f.Args[1])
346 : Show(f);
347 static string Quotient(Rational c, Expr[] a) {
348 List<string> top = [], bottom = [];
349 if (Math.Abs(c.P) != 1)
350 top.Add(Math.Abs(c.P).ToString());
351 if (c.Q != 1)
352 bottom.Add(c.Q.ToString());
353 foreach (Expr f in a)
354 if (Negative(f)) {
355 Expr r = Num(-f.Args[1].Value);
356 bottom.Add(Equal(r, One) ? Factor(f.Args[0]) : ShowPower(f.Args[0], r));
357 } else
358 top.Add(Factor(f));
359 string s = top.Count == 0 ? "1" : string.Join("*", top);
360 if (bottom.Count == 1)
361 s += "/" + bottom[0];
362 else if (bottom.Count > 1)
363 s += "/(" + string.Join("*", bottom) + ")";
364 return c.P < 0 ? "-" + s : s;
365 }
366 static string Show(Expr f) {
367 if (f.Kind == Kind.Number)
368 return f.Value.P + (f.Value.Q == 1 ? "" : "/" + f.Value.Q);
369 if (f.Kind == Kind.Variable)
370 return f.Name;
371 if (f.Kind == Kind.Function)
372 return f.Name + "(" + Show(f.Args[0]) + ")";
373 if (f.Kind == Kind.Power)
374 return Negative(f) ? Quotient(new Rational(1), [f]) : ShowPower(f.Args[0], f.Args[1]);
375 if (f.Kind == Kind.Product) {
376 var (c, a) = Coefficient(f);
377 return Quotient(c, a);
378 }
379 return f.Args.Select(Show).Aggregate(
380 (a, b) => a + (b.StartsWith('-') ? " - " + b[1..] : " + " + b));
381 }
382 sealed class Parser {
383 readonly List<string> tokens = [];
384 readonly Dictionary<string, Expr> names;
385 int position;
386 string Peek => position < tokens.Count ? tokens[position] : "";
387 void
388 Unexpected() => throw new InvalidOperationException(position == tokens.Count
389 ? "unexpected end of expression"
390 : "unexpected '" + Peek + "'");
391 bool Accept(string s) {
392 if (Peek != s)
393 return false;
394 position++;
395 return true;
396 }
397 void Expect(string s) {
398 if (!Accept(s))
399 Unexpected();
400 }
401 public Parser(string text, Dictionary<string, Expr> n) {
402 names = n;
403 foreach (Match m in Regex.Matches(
404 text, @"[0-9]+|[a-z][a-z0-9]*|[A-Z][A-Z0-9]*|[-+*/^()]| +|.")) {
405 string t = m.Value;
406 if (t[0] == ' ')
407 continue;
408 if (!Regex.IsMatch(t, @"\A[a-zA-Z0-9+*/^()-]+\z"))
409 throw new InvalidOperationException("invalid character '" + t + "'");
410 tokens.Add(t);
411 }
412 }
413 Syntax AtomTree() {
414 string t = Peek;
415 if (t == "")
416 Unexpected();
417 if (Accept("(")) {
418 Syntax a = Expression();
419 Expect(")");
420 return a;
421 }
422 if (!char.IsAsciiLetterOrDigit(t[0]))
423 Unexpected();
424 position++;
425 if (char.IsAsciiDigit(t[0]))
426 return new("", Num(long.Parse(t)), []);
427 if (char.IsAsciiLetterUpper(t[0])) {if(!names.TryGetValue(t,out Expr? f))
428 throw new InvalidOperationException("undefined name '" + t + "'");
429 return new("", f, []);
430 }
431 if (!Functions.Contains(t))
432 return new("", Node(Kind.Variable, [], t), []);
433 Expect("(");
434 Syntax inner = Expression();
435 Expect(")");
436 return new(t, null, [inner]);
437 }
438 Syntax Unary() {
439 if (Accept("-"))
440 return new("neg", null, [Unary()]);
441 Syntax a = AtomTree();
442 return Accept("^") ? new("^", null, [a, Unary()]) : a;
443 }
444 Syntax Term() {
445 Syntax a = Unary();
446 while (Peek is "*" or "/") {
447 string op = tokens[position++];
448 a = new(op, null, [a, Unary()]);
449 }
450 return a;
451 }
452 Syntax Expression() {
453 Syntax a = Term();
454 while (Peek is "+" or "-") {
455 string op = tokens[position++];
456 a = new(op, null, [a, Term()]);
457 }
458 return a;
459 }
460 static Expr Evaluate(Syntax t) {
461 if (t.Value != null)
462 return t.Value;
463 Expr[] a = t.Args.Select(Evaluate).ToArray();
464 return t.Op switch { "+" => Add(a),
465 "-" => Add(a[0], Mul(Minus, a[1])),
466 "*" => Mul(a),
467 "/" => Mul(a[0], Power(a[1], Minus)),
468 "^" => Power(a[0], a[1]),
469 "neg" => Mul(Minus, a[0]),
470 _ => Fn(t.Op, a[0]) };
471 }
472 public Expr Parse() {
473 Syntax t = Expression();
474 if (position < tokens.Count)
475 Unexpected();
476 return Evaluate(t);
477 }
478 }
479 static bool Variable(string s) => Regex.IsMatch(s, @"\A[a-z][a-z0-9]*\z") &&
480 !Functions.Contains(s);
481 static bool RationalLiteral(string s) => Regex.IsMatch(s, @"\A-?[0-9]+(?:/[0-9]+)?\z");
482 static Expr Literal(string s) {
483 if (!RationalLiteral(s))
484 throw new InvalidOperationException("invalid number '" + s + "'");
485 string[] a = s.Split('/');
486 return Num(long.Parse(a[0]), a.Length == 1 ? 1 : long.Parse(a[1]));
487 }
488 static readonly Dictionary<string, string> Signatures =
489 new() { { "SIMPLIFY", "e" }, { "EXPAND", "e" }, { "DIFF", "ev" },
490 { "DIFFN", "evn" }, { "GRAD", "e" }, { "HESSIAN", "e" },
491 { "IMPLICIT", "evv" }, { "SUBST", "eve" }, { "EVAL", "ea" },
492 { "TANGENT", "evc" }, { "LIMIT", "evc" }, { "TAYLOR", "evcn" } };
493 static string[] Command(string line, Dictionary<string, Expr> names) {
494 int p = line.IndexOf(' ');
495 string c = p < 0 ? line : line[..p], arg = p < 0 ? "" : line[(p + 1)..];
496 if (c == "LET") {
497 int i = arg.IndexOf('=');
498 if (i < 0)
499 throw new InvalidOperationException("wrong number of arguments");
500 string s = arg[..i].Trim(' ');
501 if (!Regex.IsMatch(s, @"\A[A-Z][A-Z0-9]*\z"))
502 throw new InvalidOperationException("invalid name '" + s + "'");
503 Expr f = new Parser(arg[(i + 1)..].Trim(' '), names).Parse();
504 names[s] = f;
505 return [s + " = " + Show(f)];
506 }
507 if(!Signatures.TryGetValue(c,out string? sig))
508 throw new InvalidOperationException("unknown command " + c);
509 string[] fields = p < 0 ?[] : arg.Split(';').Select(s => s.Trim(' ')).ToArray();
510 if (c == "EVAL" ? fields.Length is not(1 or 2) : fields.Length != sig.Length)
511 throw new InvalidOperationException("wrong number of arguments");
512 Expr[] forms = Enumerable.Repeat(Zero, fields.Length).ToArray();
513 int n = 0;
514 for (int i = 0; i < fields.Length; i++) {
515 string s = fields[i];
516 switch (sig[i]) {
517 case 'e':
518 forms[i] = new Parser(s, names).Parse();
519 break;
520 case 'v':
521 if (!Variable(s))
522 throw new InvalidOperationException("invalid variable '" + s + "'");
523 break;
524 case 'n':
525 if (!Regex.IsMatch(s, @"\A[1-8]\z"))
526 throw new InvalidOperationException("invalid order '" + s + "'");
527 n = int.Parse(s);
528 break;
529 case 'c':
530 forms[i] = Literal(s);
531 break;
532 }
533 }
534 Expr f0 = forms[0];
535 string v = fields.Length > 1 ? fields[1] : "";
536 List<string> output = [];
537 switch (c) {
538 case "SIMPLIFY":
539 return ["simplified: " + Show(f0)];
540 case "EXPAND":
541 return ["expanded: " + Show(Expand(f0))];
542 case "DIFF":
543 return ["d/d" + v + " = " + Show(Derivative(f0, v))];
544 case "DIFFN":
545 for (int k = 1; k <= n; k++) {
546 f0 = Derivative(f0, v);
547 output.Add($"d^{k}/d{v}^{k} = {Show(f0)}");
548 }
549 return output.ToArray();
550 case "GRAD":
551 case "HESSIAN":
552 string[] vs = Variables(f0).ToArray();
553 if (vs.Length == 0)
554 return ["no variables"];
555 for (int i = 0; i < vs.Length; i++) {
556 Expr d = Derivative(f0, vs[i]);
557 if (c == "GRAD")
558 output.Add($"d/d{vs[i]} = {Show(d)}");
559 else
560 for (int j = i; j < vs.Length; j++)
561 output.Add($"d2/d{vs[i]}d{vs[j]} = {Show(Derivative(d,vs[j]))}");
562 }
563 return output.ToArray();
564 case "IMPLICIT":
565 string w = fields[2];
566 if (v == w)
567 throw new InvalidOperationException("same variable");
568 return
569 [$"d{w}/d{v} = {Show(Mul(Minus,Derivative(f0,v),Power(Derivative(f0,w),Minus)))}"];
570 case "SUBST":
571 return ["substituted: " + Show(Substitute(f0, new() { { v, forms[2] } }))];
572 case "EVAL":
573 Dictionary<string, Expr> values = new();
574 foreach (string word in v.Split(' ', StringSplitOptions.RemoveEmptyEntries)) {
575 int i = word.IndexOf('=');
576 string s = i < 0 ? word : word[..i], t = i < 0 ? "" : word[(i + 1)..];
577 if (i < 0 || !Variable(s) || !RationalLiteral(t))
578 throw new InvalidOperationException("invalid assignment '" + word + "'");
579 Expr q = Literal(t);
580 if (values.ContainsKey(s))
581 throw new InvalidOperationException("duplicate variable " + s);
582 values[s] = q;
583 }
584 foreach (string s in Variables(f0))
585 if (!values.ContainsKey(s))
586 throw new InvalidOperationException("no value for " + s);
587 Expr r = Substitute(f0, values);
588 return [r.Kind == Kind.Number
589 ? "exact: " + Show(r)
590 : "approx: " + Numeric(r).ToString("F6", CultureInfo.InvariantCulture)];
591 }
592 Expr x = forms[2];
593 Dictionary<string, Expr> point = new() { { v, x } };
594 if (c == "TANGENT") {
595 Expr m = Substitute(Derivative(f0, v), point), y = Substitute(f0, point),
596 z = Node(Kind.Variable, [], v);
597 string t = "tangent: y = " + Show(Add(Mul(m, z), y, Mul(Num(-x.Value), m)));
598 if (Equal(m, Zero))
599 return [t, "normal: " + v + " = " + Show(x)];
600 Expr k = Power(m, Minus);
601 return [t, "normal: y = " + Show(Add(Mul(Minus, k, z), y, Mul(x, k)))];
602 }
603 if (c == "LIMIT") {
604 List<Expr> top = [], bottom = [];
605 foreach (Expr t in f0.Kind == Kind.Product ? f0.Args : [f0])
606 if (Negative(t))
607 bottom.Add(Power(t.Args[0], Num(-t.Args[1].Value)));
608 else
609 top.Add(t);
610 Expr g = Mul(top.ToArray()), h = Mul(bottom.ToArray());
611 for (int i = 0; i <= 5; i++) {
612 Expr a = Substitute(g, point), b = Substitute(h, point);
613 if (!Equal(b, Zero))
614 return ["limit: " + Show(Mul(a, Power(b, Minus)))];
615 if (!Equal(a, Zero))
616 return ["limit: unbounded"];
617 if (i < 5) {
618 g = Derivative(g, v);
619 h = Derivative(h, v);
620 }
621 }
622 return ["limit: unknown"];
623 }
624 List<Expr> ds = [f0], ps = [];
625 for (int i = 0; i < n; i++)
626 ds.Add(Derivative(ds[^1], v));
627 long fac = 1;
628 for (int i = 0; i <= n; i++) {
629 if (i > 0)
630 fac *= i;
631 Expr co = Mul(Substitute(ds[i], point), Num(1, fac));
632 ps.Add(Mul(co, Power(Add(Node(Kind.Variable, [], v), Num(-x.Value)), Num(i))));
633 }
634 return ["taylor: " + Show(Expand(Add(ps.ToArray())))];
635 }
