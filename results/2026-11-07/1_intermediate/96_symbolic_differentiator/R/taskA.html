3rational <- function(p, q = 1) {
4 if (q == 0) {
5 fail('division by zero')
6 }
7 if (q < 0) {
8 p <- -p
9 q <- -q
10 }
11 a <- abs(p)
12 b <- q
13 while (b != 0) {
14 t <- a %% b
15 a <- b
16 b <- t
17 }
18 c(p / a, q / a)

142add <- function(operands) {
143 terms <- list()
144 for (f in operands) {
145 if (f$kind == 1) {
146 terms <- c(terms, f$args)
147 } else if (
148 f$kind == 2 &&
149 length(f$args) == 2 &&
150 f$args[[1]]$kind == 5 &&
151 f$args[[2]]$kind == 1

153 terms <- c(
154 terms,
155 lapply(f$args[[2]]$args, function(t) mul(list(f$args[[1]], t)))

158 terms[[length(terms) + 1]] <- f
159 }
160 }
161 keys <- list()

163 for (f in terms) {
164 parts <- coefficient(f)
165 co <- parts[[1]]
166 m <- parts[[2]]
167 i <- 0L
168 for (j in seq_along(keys)) {
169 if (
170 length(m) == length(keys[[j]]) &&
171 all(vapply(
172 seq_along(m),
173 function(k) equal(m[[k]], keys[[j]][[k]]),
174 logical(1)
175 ))
176 ) {
177 i <- j
178 break

180 }

191 built[[length(built) + 1]] <- product(values[[i]], keys[[i]])

193 }
194 built <- sort_forms(built, term_compare)
195 if (!length(built)) {
196 zero

203mul <- function(operands) {
204 co <- c(1, 1)
205 bases <- list()
206 groups <- list()
207 factors <- list()
208 for (f in operands) {
209 factors <- c(factors, if (f$kind == 2) f$args else list(f))
210 }
211 for (f in factors) {
212 if (f$kind == 5) {
213 co <- qmul(co, f$q)
214 } else {
215 b <- base_of(f)
216 i <- find_form(bases, b)
217 if (i == 0) {
218 bases[[length(bases) + 1]] <- b
219 groups[[length(groups) + 1]] <- list(f)
220 } else {
221 groups[[i]][[length(groups[[i]]) + 1]] <- f
222 }
223 }
224 }
225 if (co[1] == 0) {
226 return(zero)
227 }
228 results <- lapply(seq_along(groups), function(i) {
229 if (length(groups[[i]]) == 1) {
230 groups[[i]][[1]]
231 } else {
232 power(bases[[i]], add(lapply(groups[[i]], exponent_of)))
233 }
234 })
235 regroup <- FALSE
236 seen <- list()
237 for (f in results) {
238 b <- base_of(f)
239 if (f$kind %in% c(2, 5) || find_form(seen, b) > 0) {
240 regroup <- TRUE
241 }
242 seen[[length(seen) + 1]] <- b
243 }
244 if (regroup) {
245 return(mul(c(list(num(co)), results)))
246 }
247 product(
248 co,
249 sort_forms(results, function(a, b) compare(base_of(a), base_of(b)))
250 )

749 if (c == 'EVAL') {
750 !length(fields) %in% c(1, 2)
751 } else {
752 length(fields) != nchar(sig)
753 }
754 ) {
755 fail('wrong number of arguments')
756 }
757 a <- list()
758 for (i in seq_along(fields)) {
759 s <- fields[i]
760 kind <- substr(sig, i, i)
761 a[[i]] <- switch(
762 kind,
763 e = parse_form(s, names),
764 v = {
765 if (!variable(s)) {
766 fail(paste0("invalid variable '", s, "'"))
767 }
768 s
769 },
770 n = {
771 if (!grepl('^[1-8]$', s)) {
772 fail(paste0("invalid order '", s, "'"))
773 }
774 as.integer(s)
775 },
776 c = literal(s),
777 s
778 )
779 }
780 f <- a[[1]]

782 x <- if (length(a) >= 3) a[[3]] else zero
783 if (c == 'SIMPLIFY') {
784 return(paste0('simplified: ', show_form(f)))
785 }
786 if (c == 'EXPAND') {
787 return(paste0('expanded: ', show_form(expand(f))))
788 }
789 if (c == 'DIFF') {
790 return(paste0('d/d', v, ' = ', show_form(derivative(f, v))))
791 }
792 if (c == 'DIFFN') {
793 out <- character()
794 for (k in seq_len(x)) {
795 f <- derivative(f, v)
796 out <- c(out, paste0('d^', k, '/d', v, '^', k, ' = ', show_form(f)))
797 }
798 return(out)
799 }
800 if (c %in% c('GRAD', 'HESSIAN')) {
801 vs <- variables(f)
802 if (!length(vs)) {
803 return('no variables')
804 }
805 out <- character()
806 for (i in seq_along(vs)) {
807 d <- derivative(f, vs[i])
808 if (c == 'GRAD') {
809 out <- c(out, paste0('d/d', vs[i], ' = ', show_form(d)))
810 } else {
811 for (j in i:length(vs)) {
812 out <- c(
813 out,
814 paste0(
815 'd2/d',
816 vs[i],
817 'd',
818 vs[j],
819 ' = ',
820 show_form(derivative(d, vs[j]))
821 )
822 )
823 }
824 }
825 }
826 return(out)
827 }
828 if (c == 'IMPLICIT') {
829 if (v == x) {
830 fail('same variable')
831 }
832 return(paste0(
833 'd',
834 x,
835 '/d',
836 v,
837 ' = ',
838 show_form(mul(list(
839 minus,
840 derivative(f, v),
841 power(derivative(f, x), minus)
842 )))
843 ))
844 }

846 if (c == 'SUBST') {
847 return(paste0('substituted: ', show_form(substitute_form(f, point))))
848 }
849 if (c == 'EVAL') {
850 values <- list()
851 for (word in strsplit(v, ' ', fixed = TRUE)[[1]]) {
852 if (word == '') {
853 next
854 }
855 pair <- strsplit(word, '=', fixed = TRUE)[[1]]
856 if (
857 length(pair) != 2 || !variable(pair[1]) || !rational_literal(pair[2])
858 ) {
859 fail(paste0("invalid assignment '", word, "'"))
860 }
861 q <- literal(pair[2])
862 if (!is.null(values[[pair[1]]])) {
863 fail(paste('duplicate variable', pair[1]))
864 }
865 values[[pair[1]]] <- q
866 }
867 for (s in variables(f)) {
868 if (is.null(values[[s]])) fail(paste('no value for', s))
869 }
870 r <- substitute_form(f, values)
871 return(
872 if (r$kind == 5) {
873 paste0('exact: ', show_form(r))
874 } else {
875 sprintf('approx: %.6f', numeric_value(r))
876 }
877 )
878 }
879 if (c == 'TANGENT') {
880 m <- substitute_form(derivative(f, v), point)
881 y <- substitute_form(f, point)
882 z <- node(0, list(), v)
883 t <- paste0(
884 'tangent: y = ',
885 show_form(add(list(mul(list(m, z)), y, mul(list(num(qneg(x$q)), m)))))
886 )
887 if (equal(m, zero)) {
888 return(c(t, paste0('normal: ', v, ' = ', show_form(x))))
889 }
890 k <- power(m, minus)
891 return(c(
892 t,
893 paste0(
894 'normal: y = ',
895 show_form(add(list(mul(list(minus, k, z)), y, mul(list(x, k)))))
896 )
897 ))
898 }
899 if (c == 'LIMIT') {
900 top <- list()
901 bottom <- list()
902 for (t in if (f$kind == 2) f$args else list(f)) {
903 if (negative(t)) {
904 bottom[[length(bottom) + 1]] <- power(
905 t$args[[1]],
906 num(qneg(t$args[[2]]$q))
907 )
908 } else {
909 top[[length(top) + 1]] <- t
910 }
911 }
912 g <- mul(top)
913 h <- mul(bottom)
914 for (k in 0:5) {
915 u <- substitute_form(g, point)
916 b <- substitute_form(h, point)
917 if (!equal(b, zero)) {
918 return(paste0('limit: ', show_form(mul(list(u, power(b, minus))))))
919 }
920 if (!equal(u, zero)) {
921 return('limit: unbounded')
922 }
923 if (k < 5) {
924 g <- derivative(g, v)
925 h <- derivative(h, v)
926 }
927 }
928 return('limit: unknown')
929 }
930 n <- a[[4]]
931 ds <- list(f)
932 for (i in seq_len(n)) {
933 ds[[length(ds) + 1]] <- derivative(ds[[length(ds)]], v)
934 }
935 ps <- list()
936 fac <- 1
937 for (k in 0:n) {
938 if (k > 0) {
939 fac <- fac * k
940 }
941 co <- mul(list(substitute_form(ds[[k + 1]], point), num(1, fac)))
942 ps[[length(ps) + 1]] <- mul(list(
943 co,
944 power(add(list(node(0, list(), v), num(qneg(x$q)))), num(k))
945 ))
946 }
947 paste0('taylor: ', show_form(expand(add(ps))))
948}
