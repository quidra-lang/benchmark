3type Totals = Record<string, bigint>;
4type Garnishment = {rate: bigint; balance: bigint};
5type Calculation = {values: Totals; balances: bigint[]};
6type SavedStub = {values: Totals; department: string};
7const RATE_SETTINGS = ["STATE", "SUPP", "SSRATE", "MEDRATE", "ADDMED", "UIRATE", "MATCH", "MATCHMAX", "GARNCAP"];
8const INITIAL: Record<string, bigint> = {
9 STATE: 500n, SUPP: 2200n, SSRATE: 620n, MEDRATE: 145n, ADDMED: 90n,
10 UIRATE: 270n, MATCH: 5000n, MATCHMAX: 600n, GARNCAP: 2500n,
11 SSBASE: 16860000n, ADDBASE: 20000000n, UIBASE: 700000n,
12 RETIRECAP: 2300000n, ALLOWANCE: 15000n, NIGHT: 100n
13};
14const FIELDS = ["R", "O", "D", "N", "P", "accrued", "salary", "leave", "deduction",
15 "sales", "commission", "earnings", "bonus", "gross", "retirement", "health", "federal",
16 "supplemental", "state", "social", "medicare", "garnishment", "net", "employer_social",
17 "employer_medicare", "unemployment", "match", "employer", "W", "Y", "F", "J"];
18const WEEKDAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];
19const cents = (s: string): bigint => BigInt(s.replace(".", ""));
20const money = (v: bigint): string => v / 100n + "." + String(v % 100n).padStart(2, "0");
21const rateText = (v: bigint): string => money(v) + "%";
22const duration = (v: number | bigint): string => BigInt(v) / 60n + ":" + String(BigInt(v) % 60n).padStart(2, "0");
23const clock = (v: number): string => String(Math.floor(v / 60)).padStart(2, "0") + ":" + String(v % 60).padStart(2, "0");
24const weekday = (d: number): string => WEEKDAYS[(d - 1) % 7];
25const periodOf = (d: number): number => Math.floor((d - 1) / 14) + 1;
26const compare = (a: bigint, b: bigint): number => a < b ? -1 : a > b ? 1 : 0;
27const min = (...values: bigint[]): bigint => values.reduce((a, b) => a < b ? a : b);
28const max = (a: bigint, b: bigint): bigint => a > b ? a : b;
29const emptyTotals = (): Totals => Object.fromEntries(FIELDS.map(f => [f, 0n]));
30class CommandError extends Error {}

32function sumStubs(stubs: Totals[]): Totals {
33 return Object.fromEntries(FIELDS.map(field => [field, stubs.reduce((total, stub) => total + stub[field], 0n)]));
34}

36function progressive(table: Map<bigint, bigint>, amount: bigint): bigint {
37 const pairs = [...table].sort((a, b) => compare(a[0], b[0]));
38 let numerator = 0n;
39 for (let i = 0; i < pairs.length; i++) {
40 const [lower, rate] = pairs[i];
41 if (lower >= amount) break;
42 const upper = i + 1 < pairs.length ? pairs[i + 1][0] : amount;
43 numerator += rate * (min(amount, upper) - lower);
44 }
45 return numerator / 10000n;
46}

94 employee(text: string, active = false, kind: string | null = null): Employee {
95 const e = this.employees.get(Number(text));
96 if (!e) throw new CommandError("unknown employee " + text);
97 if (active && !e.active) throw new CommandError("terminated employee " + text);
98 if (kind !== null && e.kind !== kind) throw new CommandError("not " + kind.toLowerCase() + " " + text);
99 return e;
100 }

102 dayCheck(text: string): number {
103 const day = Number(text);
104 if (day < 1 || day > 364) throw new CommandError("invalid day " + text);
105 const period = periodOf(day);
106 if (period <= this.closed) throw new CommandError("period closed " + period);
107 return day;
108 }

110 periodCheck(text: string): number {
111 const p = Number(text);
112 if (p < 1 || p > 26) throw new CommandError("invalid period " + text);
113 return p;
114 }
115
116 rateCheck(text: string): bigint {
117 const value = cents(text);
118 if (value > 10000n) throw new CommandError("invalid rate " + text);
119 return value;
120 }
121
122 amountCheck(text: string): bigint {
123 const value = cents(text);
124 if (value === 0n) throw new CommandError("invalid amount");
125 return value;
126 }

128 timeCheck(text: string): number {
129 const [hour, minute] = text.split(":").map(Number);
130 if (hour > 23 || minute > 59) throw new CommandError("invalid time " + text);
131 return hour * 60 + minute;
132 }

134 daily(e: Employee, day: number): number[] {
135 const shifts = e.shifts.get(day) ?? [];
136 const worked = shifts.reduce((v, [s, t]) => v + t - s, 0);
137 const night = shifts.reduce((v, [s, t]) => v + Math.max(0, Math.min(t, 360) - s) + Math.max(0, t - Math.max(s, 1320)), 0);
138 const pto = e.pto.get(day) ?? 0;
139 if (day % 7 === 0 || this.holidays.has(day)) return [worked, 0, 0, worked, night, pto];
140 return [worked, Math.min(worked, 480), Math.min(Math.max(worked - 480, 0), 240), Math.max(worked - 720, 0), night, pto];
141 }

143 weekly(e: Employee, week: number): {values: number[]; overtime: number} {
144 const values = [0, 0, 0, 0, 0];
145 for (let day = 7 * week - 6; day <= 7 * week; day++) {
146 const daily = this.daily(e, day);
147 for (let i = 0; i < 5; i++) values[i] += daily[i + 1];
148 }
149 const overtime = Math.max(0, values[0] - 2400);
150 values[0] -= overtime;
151 values[1] += overtime;
152 return {values, overtime};
153 }

155 breakdown(e: Employee, p: number): number[] {
156 const a = this.weekly(e, 2 * p - 1).values;
157 const b = this.weekly(e, 2 * p).values;
158 return a.map((v, i) => v + b[i]);
159 }

161 calculate(e: Employee, p: number): Calculation | null {
162 const s = emptyTotals();
163 const c = this.settings;
164 const ytd = e.ytd;
165 if (e.kind === "HOURLY") {
166 const values = this.breakdown(e, p);
167 for (let i = 0; i < 5; i++) s[["R", "O", "D", "N", "P"][i]] = BigInt(values[i]);
168 const weighted = 2n * (s.R + s.P) + 3n * s.O + 4n * s.D;
169 s.earnings = (weighted * e.base + 2n * s.N * c.NIGHT) / 120n;
170 } else if (e.kind === "SALARIED") {
171 if (e.active) {
172 s.salary = e.base / 26n;
173 s.leave = BigInt([...e.leave].filter(d => periodOf(d) === p && !this.holidays.has(d)).length);
174 s.deduction = s.leave * (e.base / 260n);
175 s.earnings = s.salary - s.deduction;
176 }
177 } else {
178 s.sales = e.sales.get(p) ?? 0n;
179 s.commission = progressive(this.tiers, s.sales);
180 s.earnings = s.commission + (e.active ? e.base : 0n);
181 }
182 s.bonus = e.bonus;
183 const gross = s.earnings + s.bonus;
184 if (gross === 0n) return null;
185 s.gross = gross;
186 let remaining = gross;
187 const deduct = (field: string, value: bigint): void => {
188 s[field] = min(value, remaining);
189 remaining -= s[field];
190 };
191 const retirementRoom = max(0n, c.RETIRECAP - ytd.retirement);
192 deduct("retirement", min(gross * e.retirement / 10000n, retirementRoom));
193 deduct("health", e.health);
194 s.W = remaining;
195 const baseTaxable = max(0n, s.earnings - s.retirement - s.health);
196 const allowanceTaxable = max(0n, baseTaxable - BigInt(e.allowances) * c.ALLOWANCE);
197 deduct("federal", progressive(this.brackets, allowanceTaxable) + e.extra);
198 deduct("supplemental", (s.W - baseTaxable) * c.SUPP / 10000n);
199 deduct("state", s.W * c.STATE / 10000n);
200 s.F = gross - s.health;
201 s.Y = min(s.F, max(0n, c.SSBASE - ytd.Y));
202 const additional = max(0n, min(s.F, ytd.F + s.F - c.ADDBASE));
203 deduct("social", s.Y * c.SSRATE / 10000n);
204 deduct("medicare", s.F * c.MEDRATE / 10000n + additional * c.ADDMED / 10000n);
205 const disposable = remaining;
206 const cap = disposable * c.GARNCAP / 10000n;
207 const balances: bigint[] = [];
208 let garnishment = 0n;
209 for (const order of e.garnishments) {
210 const taken = min(disposable * order.rate / 10000n, order.balance, cap - garnishment);
211 garnishment += taken;
212 balances.push(order.balance - taken);
213 }
214 s.garnishment = garnishment;
215 s.garnishment_balance = balances.reduce((v, b) => v + b, 0n);
216 s.net = remaining - garnishment;
217 s.employer_social = s.Y * c.SSRATE / 10000n;
218 s.employer_medicare = s.F * c.MEDRATE / 10000n;
219 s.J = min(s.F, max(0n, c.UIBASE - ytd.J));
220 s.unemployment = s.J * c.UIRATE / 10000n;
221 s.match = min(s.retirement, gross * c.MATCHMAX / 10000n) * c.MATCH / 10000n;
222 s.employer = s.employer_social + s.employer_medicare + s.unemployment + s.match;
223 if (e.kind === "HOURLY") s.accrued = min(BigInt(960 - e.ptoBalance), (s.R + s.O + s.D) / 20n);
224 s.pto_balance = BigInt(e.ptoBalance) + s.accrued;
225 return {values: s, balances};
226 }

228 hoursParts(values: (number | bigint)[]): string[] {
229 return values.flatMap((v, i) => [["reg", "ot", "dbl", "night", "pto"][i], duration(v)]);
230 }

232 stubBlock(e: Employee, s: Totals, indent: number): void {
233 if (e.kind === "HOURLY") {
234 this.emit(indent, "hours", ...this.hoursParts(["R", "O", "D", "N", "P"].map(f => s[f])));
235 this.emit(indent, "pto", "accrued", duration(s.accrued), "balance", duration(s.pto_balance));
236 } else if (e.kind === "SALARIED") {
237 this.emit(indent, "salary", money(s.salary), "leave", s.leave, "deduction", money(s.deduction));
238 } else {
239 this.emit(indent, "sales", money(s.sales), "commission", money(s.commission));
240 }
241 for (const f of ["earnings", "bonus", "gross", "retirement", "health", "federal", "supplemental", "state", "social", "medicare"]) {
242 this.emit(indent, f === "social" ? "social-security" : f, money(s[f]));
243 }
244 this.emit(indent, "garnishment", money(s.garnishment), "balance", money(s.garnishment_balance));
245 this.emit(indent, "net", money(s.net));
246 this.emit(indent, "employer", "social-security", money(s.employer_social), "medicare", money(s.employer_medicare),
247 "unemployment", money(s.unemployment), "match", money(s.match));
248 }

252 this.stubBlock(e, s, 4);

255 showHours(e: Employee, p: number): void {
256 this.emit(0, "HOURS", e.id, "period", p);
257 let found = false;
258 for (let day = 14 * p - 13; day <= 14 * p; day++) {
259 const shifts = [...(e.shifts.get(day) ?? [])].sort((a, b) => a[0] - b[0]);
260 if (!shifts.length && !e.pto.has(day)) continue;
261 const [worked, ...values] = this.daily(e, day);
262 this.emit(2, "day", day, weekday(day), ...(this.holidays.has(day) ? ["holiday"] : []),
263 ...shifts.map(([s, t]) => clock(s) + "-" + clock(t)), "total", duration(worked), ...this.hoursParts(values));
264 found = true;
265 }
266 if (!found) this.emit(2, "(none)");
267 for (const week of [2 * p - 1, 2 * p]) {
268 const {values, overtime} = this.weekly(e, week);
269 this.emit(2, "week", week, ...this.hoursParts(values), "weekly-ot", duration(overtime));
270 }
271 this.emit(2, "total", ...this.hoursParts(this.breakdown(e, p)));
272 }

274 run(p: number): void {
275 this.closed = p;
276 this.emit(0, "RUN", p, "days", (14 * p - 13) + "-" + 14 * p);
277 let paid = 0;
278 let skipped = 0;
279 const stubs: Totals[] = [];
280 for (const e of [...this.employees.values()].sort((a, b) => a.id - b.id)) {
281 const calculated = this.calculate(e, p);
282 if (calculated === null) {
283 if (e.active) skipped++;
284 continue;
285 }
286 const {values: s, balances} = calculated;
287 this.printStub(e, p, s, e.department);
288 e.bonus = 0n;
289 e.ptoBalance = Number(s.pto_balance);
290 for (let i = 0; i < balances.length; i++) e.garnishments[i].balance = balances[i];
291 for (const field of FIELDS) e.ytd[field] += s[field];
292 e.periods++;
293 e.stubs.set(p, {values: s, department: e.department});
294 stubs.push(s);
295 paid++;
296 }
297 const t = sumStubs(stubs);
298 this.emit(2, "TOTAL", "paid", paid, "skipped", skipped, "gross", money(t.gross),
299 "deductions", money(t.gross - t.net), "net", money(t.net), "employer", money(t.employer));
300 }

302 journal(p: number): void {
303 const stubs = [...this.employees.values()].filter(e => e.stubs.has(p)).map(e => e.stubs.get(p)!.values);
304 const t = sumStubs(stubs);
305 const rows: [string, string, bigint][] = [
306 ["debit", "wages", t.gross],
307 ["debit", "employer-tax", t.employer_social + t.employer_medicare + t.unemployment],
308 ["debit", "retirement-match", t.match],
309 ["credit", "retirement", t.retirement + t.match],
310 ["credit", "health", t.health],
311 ["credit", "federal", t.federal + t.supplemental],
312 ["credit", "state", t.state],
313 ["credit", "fica", t.social + t.medicare + t.employer_social + t.employer_medicare],
314 ["credit", "unemployment", t.unemployment],
315 ["credit", "garnishment", t.garnishment],
316 ["credit", "net-pay", t.net]
317 ];
318 this.emit(0, "JOURNAL", p, "stubs", stubs.length);
319 for (const [side, label, value] of rows) this.emit(2, side, label, money(value));
320 const debit = rows.filter(r => r[0] === "debit").reduce((v, r) => v + r[2], 0n);
321 const credit = rows.filter(r => r[0] === "credit").reduce((v, r) => v + r[2], 0n);
322 this.emit(2, "total", "debit", money(debit), "credit", money(credit));
323 }

332 command(cmd: string, a: string[]): void {
333 switch (cmd) {
334 case "SET": {
335 const name = a[0];
336 if (!Object.hasOwn(this.settings, name)) throw new CommandError("unknown setting " + name);
337 const value = RATE_SETTINGS.includes(name) ? this.rateCheck(a[1]) : cents(a[1]);
338 const formatter = RATE_SETTINGS.includes(name) ? rateText : money;
339 const old = this.settings[name];
340 this.settings[name] = value;
341 this.emit(0, "SET", name, formatter(old), "->", formatter(value));
342 break;
343 }
344 case "BRACKET":
345 case "TIER": {
346 const rate = this.rateCheck(a[1]);
347 const table = cmd === "BRACKET" ? this.brackets : this.tiers;
348 const lower = cents(a[0]);
349 const action = table.has(lower) ? "replaced" : "added";
350 table.set(lower, rate);
351 this.emit(0, cmd, money(lower), rateText(rate), action);
352 break;
353 }
354 case "EMP": {
355 const id = Number(a[0]);
356 if (this.employees.has(id)) throw new CommandError("duplicate employee " + a[0]);
357 const base = this.amountCheck(a[3]);
358 this.employees.set(id, new Employee(id, a[1], a[2], base));
359 this.emit(0, "EMP", id, a[1], a[2], money(base));
360 break;
361 }
362 case "RATE":
363 case "DEPT":
364 case "BENEFIT":
365 case "W4":
366 case "GARNISH":
367 case "TERM":
368 case "BONUS": {
369 const e = this.employee(a[0], true);
370 if (cmd === "RATE") {
371 const base = this.amountCheck(a[1]);
372 const old = e.base;
373 e.base = base;
374 this.emit(0, "RATE", e.id, money(old), "->", money(base));
375 } else if (cmd === "DEPT") {
376 if (a[1] === e.department) throw new CommandError("same department " + a[1]);
377 const old = e.department;
378 e.department = a[1];
379 this.emit(0, "DEPT", e.id, old, "->", a[1]);
380 } else if (cmd === "BENEFIT") {
381 const rate = this.rateCheck(a[1]);
382 e.retirement = rate;
383 e.health = cents(a[2]);
384 this.emit(0, "BENEFIT", e.id, "retirement", rateText(rate), "health", money(e.health));
385 } else if (cmd === "W4") {
386 e.allowances = Number(a[1]);
387 e.extra = cents(a[2]);
388 this.emit(0, "W4", e.id, "allowances", e.allowances, "extra", money(e.extra));
389 } else if (cmd === "GARNISH") {
390 const rate = this.rateCheck(a[1]);
391 const balance = this.amountCheck(a[2]);
392 e.garnishments.push({rate, balance});
393 this.emit(0, "GARNISH", e.id, "order", e.garnishments.length, rateText(rate), "balance", money(balance));
394 } else if (cmd === "TERM") {
395 e.active = false;
396 this.emit(0, "TERM", e.id);
397 } else {
398 const amount = this.amountCheck(a[1]);
399 e.bonus += amount;
400 this.emit(0, "BONUS", e.id, money(amount), "pending", money(e.bonus));
401 }
402 break;
403 }
404 case "HOLIDAY": {
405 const day = this.dayCheck(a[0]);
406 if (this.holidays.has(day)) throw new CommandError("duplicate holiday " + a[0]);
407 this.holidays.add(day);
408 this.emit(0, "HOLIDAY", day, weekday(day));
409 break;
410 }
411 case "TIME":
412 case "UNTIME":
413 case "PTO":
414 case "LEAVE":
415 case "SALES": {
416 const kind = cmd === "LEAVE" ? "SALARIED" : cmd === "SALES" ? "COMMISSION" : "HOURLY";
417 const e = this.employee(a[0], true, kind);
418 const day = this.dayCheck(a[1]);
419 if (cmd === "TIME") {
420 const start = this.timeCheck(a[2]);
421 const end = this.timeCheck(a[3]);
422 if (start >= end) throw new CommandError("invalid shift " + a[2] + "-" + a[3]);
423 const shifts = [...(e.shifts.get(day) ?? [])].sort((a, b) => a[0] - b[0]);
424 for (const [s, t] of shifts) {
425 if (start < t && s < end) throw new CommandError("overlapping shift " + clock(s) + "-" + clock(t));
426 }
427 if (shifts.reduce((v, [s, t]) => v + t - s, 0) + end - start > 960) throw new CommandError("daily limit " + a[0] + " " + a[1]);
428 if (!e.shifts.has(day)) e.shifts.set(day, []);
429 e.shifts.get(day)!.push([start, end]);
430 this.emit(0, "TIME", e.id, "day", day, clock(start) + "-" + clock(end), duration(end - start));
431 } else if (cmd === "UNTIME") {
432 const start = this.timeCheck(a[2]);
433 const shifts = e.shifts.get(day) ?? [];
434 const i = shifts.findIndex(s => s[0] === start);
435 if (i < 0) throw new CommandError("no shift " + a[0] + " " + a[1] + " " + a[2]);
436 const [shift] = shifts.splice(i, 1);
437 this.emit(0, "UNTIME", e.id, "day", day, clock(shift[0]) + "-" + clock(shift[1]));
438 } else if (cmd === "PTO") {
439 const [hour, minute] = a[2].split(":").map(Number);
440 const length = hour * 60 + minute;
441 if (minute > 59 || length < 1 || length > 480) throw new CommandError("invalid duration " + a[2]);
442 if (e.pto.has(day)) throw new CommandError("duplicate pto " + a[0] + " " + a[1]);
443 if (length > e.ptoBalance) throw new CommandError("insufficient pto " + a[0] + " " + duration(e.ptoBalance));
444 e.pto.set(day, length);
445 e.ptoBalance -= length;
446 this.emit(0, "PTO", e.id, "day", day, duration(length), "balance", duration(e.ptoBalance));
447 } else if (cmd === "LEAVE") {
448 if ([0, 6].includes(day % 7) || this.holidays.has(day)) throw new CommandError("not workday " + a[1]);
449 if (e.leave.has(day)) throw new CommandError("duplicate leave " + a[0] + " " + a[1]);
450 e.leave.add(day);
451 this.emit(0, "LEAVE", e.id, "day", day, weekday(day));
452 } else {
453 const amount = this.amountCheck(a[2]);
454 const p = periodOf(day);
455 e.sales.set(p, (e.sales.get(p) ?? 0n) + amount);
456 this.emit(0, "SALES", e.id, "day", day, money(amount), "period", p, "total", money(e.sales.get(p)!));
457 }
458 break;
459 }
460 case "HOURS": {
461 const e = this.employee(a[0], false, "HOURLY");
462 this.showHours(e, this.periodCheck(a[1]));
463 break;
464 }
465 case "RUN": {
466 const p = Number(a[0]);
467 if (p !== this.closed + 1 || p > 26) throw new CommandError("invalid period " + a[0]);
468 this.run(p);
469 break;
470 }
471 case "PREVIEW": {
472 const e = this.employee(a[0]);
473 const p = this.closed + 1;
474 const calculated = this.calculate(e, p);
475 this.emit(0, "PREVIEW", e.id, "period", p);
476 if (calculated === null) this.emit(2, "no pay");
477 else this.printStub(e, p, calculated.values, e.department);
478 break;
479 }
480 case "REPRINT": {
481 const e = this.employee(a[0]);
482 const p = this.periodCheck(a[1]);
483 const saved = e.stubs.get(p);
484 if (!saved) throw new CommandError("no stub " + a[0] + " " + a[1]);
485 this.emit(0, "REPRINT", e.id, p);
486 this.printStub(e, p, saved.values, saved.department);
487 break;
488 }
489 case "YTD": {
490 const e = this.employee(a[0]);
491 this.emit(0, "YTD", e.id, e.department, e.kind, e.active ? "active" : "terminated", "periods", e.periods);
492 const s: Totals = {...e.ytd, pto_balance: BigInt(e.ptoBalance), garnishment_balance: e.garnishmentBalance()};
493 this.stubBlock(e, s, 2);
494 this.emit(2, "wages", "federal", money(s.W), "social-security", money(s.Y), "medicare", money(s.F), "unemployment", money(s.J));
495 this.emit(2, "retirement-room", money(max(0n, this.settings.RETIRECAP - s.retirement)), "pending-bonus", money(e.bonus));
496 break;
497 }
498 case "JOURNAL": {
499 const p = Number(a[0]);
500 if (p < 1 || p > this.closed) throw new CommandError("no run " + a[0]);
501 this.journal(p);
502 break;
503 }
504 case "REPORT": {
505 const saved = [...this.employees.values()].flatMap(e => [...e.stubs.values()]);
506 const departments = Map.groupBy(saved, stub => stub.department);
507 this.emit(0, "REPORT", "period", this.closed);
508 for (const department of [...departments.keys()].sort()) {
509 this.reportLine(department, departments.get(department)!.map(stub => stub.values));
510 }
511 this.reportLine("total", saved.map(stub => stub.values));
512 break;
513 }
514 case "TOP": {
515 const k = Number(a[0]);
516 const employees = [...this.employees.values()].filter(e => e.periods > 0).sort((a, b) => compare(b.ytd.gross, a.ytd.gross) || a.id - b.id);
517 this.emit(0, "TOP", k);
518 for (let i = 0; i < Math.min(k, employees.length); i++) {
519 const e = employees[i];
520 this.emit(2, i + 1, e.id, e.department, "gross", money(e.ytd.gross), "net", money(e.ytd.net));
521 }
522 if (!employees.length) this.emit(2, "(none)");
523 break;
524 }
525 case "EMPLOYEES": {
526 this.emit(0, "EMPLOYEES");
527 for (const e of [...this.employees.values()].sort((a, b) => a.id - b.id)) {
528 this.emit(2, e.id, e.department, e.kind, money(e.base), e.active ? "active" : "terminated", "retirement", rateText(e.retirement),
529 "health", money(e.health), "allowances", e.allowances, "extra", money(e.extra), "bonus", money(e.bonus));
530 for (let i = 0; i < e.garnishments.length; i++) {
531 const order = e.garnishments[i];
532 this.emit(4, "garnish", i + 1, rateText(order.rate), "balance", money(order.balance));
533 }
534 }
535 if (!this.employees.size) this.emit(2, "(none)");
536 break;
537 }
538 case "CONFIG":
539 this.emit(0, "CONFIG");
540 for (const [name, value] of Object.entries(this.settings)) this.emit(2, name, RATE_SETTINGS.includes(name) ? rateText(value) : money(value));
541 for (const [label, table] of [["bracket", this.brackets], ["tier", this.tiers]] as [string, Map<bigint, bigint>][]) {
542 for (const [lower, rate] of [...table].sort((a, b) => compare(a[0], b[0]))) this.emit(2, label, money(lower), rateText(rate));
543 }
544 break;
545 }
546 }

548 execute(line: string): void {
549 const [cmd, ...args] = line.split(" ");
550 try {
551 this.command(cmd, args);
552 } catch (error) {
553 if (!(error instanceof CommandError)) throw error;
554 this.emit(0, "ERROR", error.message);
555 }
556 }
