78function progressive(array $table, int $amount): int
79{
80 ksort($table, SORT_NUMERIC);
81 $bounds = array_keys($table);
82 $numerator = 0;
83 foreach ($bounds as $i => $lower) {
84 if ($lower >= $amount) {
85 break;
86 }
87 $upper = $bounds[$i + 1] ?? $amount;
88 $numerator += $table[$lower] * (min($amount, $upper) - $lower);
89 }
90 return intdiv($numerator, 10000);
91}

214 private function daily(Employee $employee, int $day): array
215 {
216 $shifts = $employee->shifts[$day] ?? [];
217 $worked = 0;
218 $night = 0;
219 foreach ($shifts as [$start, $end]) {
220 $worked += $end - $start;
221 $night += max(0, min($end, 360) - $start) +
222 max(0, $end - max($start, 1320));
223 }
224 $pto = $employee->pto[$day] ?? 0;
225 if ($day % 7 === 0 || isset($this->holidays[$day])) {
226 [$regular, $overtime, $double] = [0, 0, $worked];
227 } else {
228 $regular = min($worked, 480);
229 $overtime = min(max($worked - 480, 0), 240);
230 $double = max($worked - 720, 0);
231 }
232 return [$worked, $regular, $overtime, $double, $night, $pto];
233 }

235 private function weekly(Employee $employee, int $week): array
236 {
237 $totals = [0, 0, 0, 0, 0];
238 for ($day = 7 * $week - 6; $day <= 7 * $week; ++$day) {
239 $daily = $this->daily($employee, $day);
240 for ($i = 0; $i < 5; ++$i) {
241 $totals[$i] += $daily[$i + 1];
242 }
243 }
244 $weeklyOvertime = max(0, $totals[0] - 2400);
245 $totals[0] -= $weeklyOvertime;
246 $totals[1] += $weeklyOvertime;
247 return [$totals, $weeklyOvertime];
248 }

250 private function breakdown(Employee $employee, int $period): array
251 {
252 [$first] = $this->weekly($employee, 2 * $period - 1);
253 [$second] = $this->weekly($employee, 2 * $period);
254 $result = [];
255 for ($i = 0; $i < 5; ++$i) {
256 $result[] = $first[$i] + $second[$i];
257 }
258 return $result;
259 }

261 private function calculate(Employee $employee, int $period): ?array
262 {
263 $stub = emptyTotals();
264 $settings = $this->settings;
265 $ytd = $employee->ytd;
266 if ($employee->kind === 'HOURLY') {
267 $amounts = $this->breakdown($employee, $period);
268 foreach (['R', 'O', 'D', 'N', 'P'] as $i => $field) {
269 $stub[$field] = $amounts[$i];
270 }
271 [$regular, $overtime, $double, $night, $pto] = $amounts;
272 $weighted = 2 * ($regular + $pto) + 3 * $overtime + 4 * $double;
273 $stub['earnings'] = intdiv(
274 $weighted * $employee->base + 2 * $night * $settings['NIGHT'], 120
275 );
276 } elseif ($employee->kind === 'SALARIED') {
277 if ($employee->active) {
278 $stub['salary'] = intdiv($employee->base, 26);
279 foreach ($employee->leave as $day => $present) {
280 if (periodOf($day) === $period && !isset($this->holidays[$day])) {
281 ++$stub['leave'];
282 }
283 }
284 $stub['deduction'] = $stub['leave'] * intdiv($employee->base, 260);
285 $stub['earnings'] = $stub['salary'] - $stub['deduction'];
286 }
287 } else {
288 $stub['sales'] = $employee->sales[$period] ?? 0;
289 $stub['commission'] = progressive($this->tiers, $stub['sales']);
290 $stub['earnings'] = $stub['commission'] + ($employee->active ? $employee->base : 0);
291 }
292 $stub['bonus'] = $employee->bonus;
293 $gross = $stub['earnings'] + $stub['bonus'];
294 if ($gross === 0) {
295 return null;
296 }
297 $stub['gross'] = $gross;
298 $remaining = $gross;
299 $deduct = static function (string $field, int $amount) use (&$remaining, &$stub): void {
300 $stub[$field] = min($amount, $remaining);
301 $remaining -= $stub[$field];
302 };
303 $room = max(0, $settings['RETIRECAP'] - $ytd['retirement']);
304 $deduct('retirement', min(intdiv($gross * $employee->retirement, 10000), $room));
305 $deduct('health', $employee->health);
306 $stub['W'] = $remaining;
307 $baseTaxable = max(0, $stub['earnings'] - $stub['retirement'] - $stub['health']);
308 $allowanceTaxable = max(0, $baseTaxable - $employee->allowances * $settings['ALLOWANCE']);
309 $deduct('federal', progressive($this->brackets, $allowanceTaxable) + $employee->extra);
310 $deduct('supplemental', intdiv(($stub['W'] - $baseTaxable) * $settings['SUPP'], 10000));
311 $deduct('state', intdiv($stub['W'] * $settings['STATE'], 10000));
312 $medicareWages = $gross - $stub['health'];
313 $socialWages = min($medicareWages, max(0, $settings['SSBASE'] - $ytd['Y']));
314 $additional = max(
315 0,
316 min($medicareWages, $ytd['F'] + $medicareWages - $settings['ADDBASE'])
317 );
318 $stub['F'] = $medicareWages;
319 $stub['Y'] = $socialWages;
320 $deduct('social', intdiv($socialWages * $settings['SSRATE'], 10000));
321 $deduct('medicare', intdiv($medicareWages * $settings['MEDRATE'], 10000) +
322 intdiv($additional * $settings['ADDMED'], 10000));
323 $disposable = $remaining;
324 $cap = intdiv($disposable * $settings['GARNCAP'], 10000);
325 $balances = [];
326 $garnishment = 0;
327 foreach ($employee->garnishments as [$rate, $balance]) {
328 $taken = min(intdiv($disposable * $rate, 10000), $balance, $cap - $garnishment);
329 $garnishment += $taken;
330 $balances[] = $balance - $taken;
331 }
332 $stub['garnishment'] = $garnishment;
333 $stub['garnishment_balance'] = array_sum($balances);
334 $stub['net'] = $remaining - $garnishment;
335 $stub['employer_social'] = intdiv($socialWages * $settings['SSRATE'], 10000);
336 $stub['employer_medicare'] = intdiv($medicareWages * $settings['MEDRATE'], 10000);
337 $stub['J'] = min($medicareWages, max(0, $settings['UIBASE'] - $ytd['J']));
338 $stub['unemployment'] = intdiv($stub['J'] * $settings['UIRATE'], 10000);
339 $eligibleMatch = min($stub['retirement'], intdiv($gross * $settings['MATCHMAX'], 10000));
340 $stub['match'] = intdiv($eligibleMatch * $settings['MATCH'], 10000);
341 $stub['employer'] = $stub['employer_social'] + $stub['employer_medicare'] +
342 $stub['unemployment'] + $stub['match'];
343 if ($employee->kind === 'HOURLY') {
344 $stub['accrued'] = min(
345 960 - $employee->ptoBalance,
346 intdiv($stub['R'] + $stub['O'] + $stub['D'], 20)
347 );
348 }
349 $stub['pto_balance'] = $employee->ptoBalance + $stub['accrued'];
350 return [$stub, $balances];
351 }

410 private function showHours(Employee $employee, int $period): void
411 {
412 $this->emit(['HOURS', $employee->id, 'period', $period]);
413 $found = false;
414 for ($day = 14 * $period - 13; $day <= 14 * $period; ++$day) {
415 $shifts = $employee->shifts[$day] ?? [];
416 usort($shifts, static fn(array $a, array $b): int => $a <=> $b);
417 if (!$shifts && !isset($employee->pto[$day])) {
418 continue;
419 }
420 [$worked, $regular, $overtime, $double, $night, $pto] = $this->daily($employee, $day);
421 $parts = ['day', $day, weekday($day)];
422 if (isset($this->holidays[$day])) {
423 $parts[] = 'holiday';
424 }
425 foreach ($shifts as [$start, $end]) {
426 $parts[] = clockText($start) . '-' . clockText($end);
427 }
428 $this->emit(
429 array_merge(
430 $parts,
431 ['total', duration($worked)],
432 $this->hoursParts([$regular, $overtime, $double, $night, $pto])
433 ),
434 2
435 );
436 $found = true;
437 }
438 if (!$found) {
439 $this->emit(['(none)'], 2);
440 }
441 foreach ([2 * $period - 1, 2 * $period] as $week) {
442 [$values, $overtime] = $this->weekly($employee, $week);
443 $this->emit(
444 array_merge(['week', $week], $this->hoursParts($values),
445 ['weekly-ot', duration($overtime)]), 2
446 );
447 }
448 $this->emit(array_merge(['total'], $this->hoursParts($this->breakdown($employee, $period))), 2);
449 }

451 private function run(int $period): void
452 {
453 $this->closed = $period;
454 $this->emit(['RUN', $period, 'days',
455 (14 * $period - 13) . '-' . (14 * $period)]);
456 $paid = 0;
457 $skipped = 0;
458 $stubs = [];
459 $ids = array_keys($this->employees);
460 sort($ids, SORT_NUMERIC);
461 foreach ($ids as $id) {
462 $employee = $this->employees[$id];
463 $calculated = $this->calculate($employee, $period);
464 if ($calculated === null) {
465 if ($employee->active) {
466 ++$skipped;
467 }
468 continue;
469 }
470 [$stub, $balances] = $calculated;
471 $this->printStub($employee, $period, $stub, $employee->department);
472 $employee->bonus = 0;
473 $employee->ptoBalance = $stub['pto_balance'];
474 foreach ($balances as $i => $balance) {
475 $employee->garnishments[$i][1] = $balance;
476 }
477 foreach (FIELDS as $field) {
478 $employee->ytd[$field] += $stub[$field];
479 }
480 ++$employee->periods;
481 $employee->stubs[$period] = ['stub' => $stub, 'department' => $employee->department];
482 $stubs[] = $stub;
483 ++$paid;
484 }
485 $total = sumStubs($stubs);
486 $this->emit(
487 ['TOTAL', 'paid', $paid, 'skipped', $skipped,
488 'gross', money($total['gross']),
489 'deductions', money($total['gross'] - $total['net']),
490 'net', money($total['net']),
491 'employer', money($total['employer'])], 2
492 );
493 }

496 {
497 $stubs = [];
498 foreach ($this->employees as $employee) {
499 if (isset($employee->stubs[$period])) {
500 $stubs[] = $employee->stubs[$period]['stub'];

508 ['debit', 'retirement-match', $total['match']],

511 ['credit', 'federal', $total['federal'] + $total['supplemental']],
512 ['credit', 'state', $total['state']],
513 ['credit', 'fica', $total['social'] + $total['medicare'] +
514 $total['employer_social'] + $total['employer_medicare']],
515 ['credit', 'unemployment', $total['unemployment']],
516 ['credit', 'garnishment', $total['garnishment']],
517 ['credit', 'net-pay', $total['net']],
518 ];
519 $this->emit(['JOURNAL', $period, 'stubs', count($stubs)]);
520 $debit = 0;
521 $credit = 0;
522 foreach ($rows as [$side, $label, $amount]) {
523 $this->emit([$side, $label, money($amount)], 2);

525 $debit += $amount;
526 } else {
527 $credit += $amount;
528 }
529 }
530 $this->emit(['total', 'debit', money($debit), 'credit', money($credit)], 2);

548 private function command(string $command, array $a): void
549 {
550 if ($command === 'SET') {
551 $name = $a[0];
552 if (!array_key_exists($name, $this->settings)) {
553 throw new CommandError('unknown setting ' . $name);
554 }
555 $rate = in_array($name, RATE_SETTINGS, true);
556 $value = $rate ? $this->rateCheck($a[1]) : cents($a[1]);
557 $format = $rate ? 'rateText' : 'money';
558 $old = $this->settings[$name];
559 $this->settings[$name] = $value;
560 $this->emit(['SET', $name, $format($old), '->', $format($value)]);
561 } elseif ($command === 'BRACKET' || $command === 'TIER') {
562 $rate = $this->rateCheck($a[1]);
563 $lower = cents($a[0]);
564 $table = $command === 'BRACKET' ? 'brackets' : 'tiers';
565 $action = array_key_exists($lower, $this->$table) ? 'replaced' : 'added';
566 $this->{$table}[$lower] = $rate;
567 $this->emit([$command, money($lower), rateText($rate), $action]);
568 } elseif ($command === 'EMP') {
569 $id = (int) $a[0];
570 if (isset($this->employees[$id])) {
571 throw new CommandError('duplicate employee ' . $a[0]);
572 }
573 $base = $this->amountCheck($a[3]);
574 $this->employees[$id] = new Employee($id, $a[1], $a[2], $base);
575 $this->emit(['EMP', $id, $a[1], $a[2], money($base)]);
576 } elseif (in_array($command, ['RATE', 'DEPT', 'BENEFIT', 'W4', 'GARNISH', 'TERM', 'BONUS'], true)) {
577 $employee = $this->employee($a[0], true);
578 if ($command === 'RATE') {
579 $base = $this->amountCheck($a[1]);
580 $old = $employee->base;
581 $employee->base = $base;
582 $this->emit(['RATE', $employee->id, money($old), '->', money($base)]);
583 } elseif ($command === 'DEPT') {
584 if ($a[1] === $employee->department) {
585 throw new CommandError('same department ' . $a[1]);
586 }
587 $old = $employee->department;
588 $employee->department = $a[1];
589 $this->emit(['DEPT', $employee->id, $old, '->', $a[1]]);
590 } elseif ($command === 'BENEFIT') {
591 $employee->retirement = $this->rateCheck($a[1]);
592 $employee->health = cents($a[2]);
593 $this->emit([
594 'BENEFIT', $employee->id, 'retirement', rateText($employee->retirement),
595 'health', money($employee->health),
596 ]);
597 } elseif ($command === 'W4') {
598 $employee->allowances = (int) $a[1];
599 $employee->extra = cents($a[2]);
600 $this->emit([
601 'W4', $employee->id, 'allowances', $employee->allowances,
602 'extra', money($employee->extra),
603 ]);
604 } elseif ($command === 'GARNISH') {
605 $rate = $this->rateCheck($a[1]);
606 $amount = $this->amountCheck($a[2]);
607 $employee->garnishments[] = [$rate, $amount];
608 $this->emit([
609 'GARNISH', $employee->id, 'order', count($employee->garnishments),
610 rateText($rate), 'balance', money($amount),
611 ]);
612 } elseif ($command === 'TERM') {
613 $employee->active = false;
614 $this->emit(['TERM', $employee->id]);
615 } else {
616 $amount = $this->amountCheck($a[1]);
617 $employee->bonus += $amount;
618 $this->emit([
619 'BONUS', $employee->id, money($amount),
620 'pending', money($employee->bonus),
621 ]);
622 }
623 } elseif ($command === 'HOLIDAY') {
624 $day = $this->dayCheck($a[0]);
625 if (isset($this->holidays[$day])) {
626 throw new CommandError('duplicate holiday ' . $a[0]);
627 }
628 $this->holidays[$day] = true;
629 $this->emit(['HOLIDAY', $day, weekday($day)]);
630 } elseif (in_array($command, ['TIME', 'UNTIME', 'PTO', 'LEAVE', 'SALES'], true)) {
631 $kind = $command === 'LEAVE' ? 'SALARIED' :
632 ($command === 'SALES' ? 'COMMISSION' : 'HOURLY');
633 $employee = $this->employee($a[0], true, $kind);
634 $day = $this->dayCheck($a[1]);
635 if ($command === 'TIME') {
636 $start = $this->timeCheck($a[2]);
637 $end = $this->timeCheck($a[3]);
638 if ($start >= $end) {
639 throw new CommandError('invalid shift ' . $a[2] . '-' . $a[3]);
640 }
641 $shifts = $employee->shifts[$day] ?? [];
642 usort($shifts, static fn(array $x, array $y): int => $x <=> $y);
643 foreach ($shifts as [$oldStart, $oldEnd]) {
644 if ($start < $oldEnd && $oldStart < $end) {
645 throw new CommandError(
646 'overlapping shift ' . clockText($oldStart) . '-' . clockText($oldEnd)
647 );
648 }
649 }
650 $minutes = 0;
651 foreach ($shifts as [$oldStart, $oldEnd]) {
652 $minutes += $oldEnd - $oldStart;
653 }
654 if ($minutes + $end - $start > 960) {
655 throw new CommandError('daily limit ' . $a[0] . ' ' . $a[1]);
656 }
657 $employee->shifts[$day][] = [$start, $end];
658 $this->emit([
659 'TIME', $employee->id, 'day', $day,
660 clockText($start) . '-' . clockText($end), duration($end - $start),
661 ]);
662 } elseif ($command === 'UNTIME') {
663 $start = $this->timeCheck($a[2]);
664 $found = null;
665 foreach (($employee->shifts[$day] ?? []) as $i => $shift) {
666 if ($shift[0] === $start) {
667 $found = [$i, $shift];
668 break;
669 }
670 }
671 if ($found === null) {
672 throw new CommandError('no shift ' . $a[0] . ' ' . $a[1] . ' ' . $a[2]);
673 }
674 [$i, [$begin, $end]] = $found;
675 array_splice($employee->shifts[$day], $i, 1);
676 $this->emit([
677 'UNTIME', $employee->id, 'day', $day,
678 clockText($begin) . '-' . clockText($end),
679 ]);
680 } elseif ($command === 'PTO') {
681 [$hour, $minute] = array_map('intval', explode(':', $a[2]));
682 $length = 60 * $hour + $minute;
683 if ($minute > 59 || $length < 1 || $length > 480) {
684 throw new CommandError('invalid duration ' . $a[2]);
685 }
686 if (isset($employee->pto[$day])) {
687 throw new CommandError('duplicate pto ' . $a[0] . ' ' . $a[1]);
688 }
689 if ($length > $employee->ptoBalance) {
690 throw new CommandError(
691 'insufficient pto ' . $a[0] . ' ' . duration($employee->ptoBalance)
692 );
693 }
694 $employee->pto[$day] = $length;
695 $employee->ptoBalance -= $length;
696 $this->emit([
697 'PTO', $employee->id, 'day', $day, duration($length),
698 'balance', duration($employee->ptoBalance),
699 ]);
700 } elseif ($command === 'LEAVE') {
701 if ($day % 7 === 0 || $day % 7 === 6 || isset($this->holidays[$day])) {
702 throw new CommandError('not workday ' . $a[1]);
703 }
704 if (isset($employee->leave[$day])) {
705 throw new CommandError('duplicate leave ' . $a[0] . ' ' . $a[1]);
706 }
707 $employee->leave[$day] = true;
708 $this->emit(['LEAVE', $employee->id, 'day', $day, weekday($day)]);
709 } else {
710 $amount = $this->amountCheck($a[2]);
711 $period = periodOf($day);
712 $employee->sales[$period] = ($employee->sales[$period] ?? 0) + $amount;
713 $this->emit([
714 'SALES', $employee->id, 'day', $day,
715 money($amount), 'period', $period, 'total', money($employee->sales[$period]),
716 ]);
717 }
718 } elseif ($command === 'HOURS') {
719 $employee = $this->employee($a[0], false, 'HOURLY');
720 $period = $this->periodCheck($a[1]);
721 $this->showHours($employee, $period);
722 } elseif ($command === 'RUN') {
723 $period = (int) $a[0];
724 if ($period !== $this->closed + 1 || $period > 26) {
725 throw new CommandError('invalid period ' . $a[0]);
726 }
727 $this->run($period);
728 } elseif ($command === 'PREVIEW') {
729 $employee = $this->employee($a[0]);
730 $period = $this->closed + 1;
731 $calculated = $this->calculate($employee, $period);
732 $this->emit(['PREVIEW', $employee->id, 'period', $period]);
733 if ($calculated === null) {
734 $this->emit(['no pay'], 2);
735 } else {
736 $this->printStub($employee, $period, $calculated[0], $employee->department);
737 }
738 } elseif ($command === 'REPRINT') {
739 $employee = $this->employee($a[0]);
740 $period = $this->periodCheck($a[1]);
741 if (!isset($employee->stubs[$period])) {
742 throw new CommandError('no stub ' . $a[0] . ' ' . $a[1]);
743 }
744 $saved = $employee->stubs[$period];
745 $this->emit(['REPRINT', $employee->id, $period]);
746 $this->printStub($employee, $period, $saved['stub'], $saved['department']);
747 } elseif ($command === 'YTD') {
748 $employee = $this->employee($a[0]);
749 $this->emit([
750 'YTD', $employee->id, $employee->department, $employee->kind,
751 $employee->active ? 'active' : 'terminated',
752 'periods', $employee->periods,
753 ]);
754 $totals = $employee->ytd;
755 $totals['pto_balance'] = $employee->ptoBalance;
756 $totals['garnishment_balance'] = $employee->garnishmentBalance();
757 $this->stubBlock($employee, $totals, 2);
758 $this->emit([
759 'wages', 'federal', money($totals['W']),
760 'social-security', money($totals['Y']),
761 'medicare', money($totals['F']),
762 'unemployment', money($totals['J']),
763 ], 2);
764 $this->emit([
765 'retirement-room', money(max(0, $this->settings['RETIRECAP'] - $totals['retirement'])),
766 'pending-bonus', money($employee->bonus),
767 ], 2);
768 } elseif ($command === 'JOURNAL') {
769 $period = (int) $a[0];
770 if ($period < 1 || $period > $this->closed) {
771 throw new CommandError('no run ' . $a[0]);
772 }
773 $this->journal($period);
774 } elseif ($command === 'REPORT') {
775 $departments = [];
776 $allStubs = [];
777 foreach ($this->employees as $employee) {
778 foreach ($employee->stubs as $saved) {
779 $departments[$saved['department']][] = $saved['stub'];
780 $allStubs[] = $saved['stub'];
781 }
782 }
783 $this->emit(['REPORT', 'period', $this->closed]);
784 ksort($departments, SORT_STRING);
785 foreach ($departments as $department => $stubs) {
786 $this->reportLine($department, $stubs);
787 }
788 $this->reportLine('total', $allStubs);
789 } elseif ($command === 'TOP') {
790 $count = (int) $a[0];
791 $employees = array_values(array_filter(
792 $this->employees,
793 static fn(Employee $employee): bool => $employee->periods > 0
794 ));
795 usort($employees, static fn(Employee $x, Employee $y): int =>
796 ($y->ytd['gross'] <=> $x->ytd['gross']) ?: ($x->id <=> $y->id)
797 );
798 $this->emit(['TOP', $count]);
799 foreach (array_slice($employees, 0, $count) as $rank => $employee) {
800 $this->emit([
801 $rank + 1, $employee->id, $employee->department,
802 'gross', money($employee->ytd['gross']),
803 'net', money($employee->ytd['net']),
804 ], 2);
805 }
806 if (!$employees) {
807 $this->emit(['(none)'], 2);
808 }
809 } elseif ($command === 'EMPLOYEES') {
810 $this->emit(['EMPLOYEES']);
811 $ids = array_keys($this->employees);
812 sort($ids, SORT_NUMERIC);
813 foreach ($ids as $id) {
814 $employee = $this->employees[$id];
815 $this->emit([
816 $id, $employee->department, $employee->kind, money($employee->base),
817 $employee->active ? 'active' : 'terminated',
818 'retirement', rateText($employee->retirement),
819 'health', money($employee->health),
820 'allowances', $employee->allowances,
821 'extra', money($employee->extra),
822 'bonus', money($employee->bonus),
823 ], 2);
824 foreach ($employee->garnishments as $i => [$rate, $balance]) {
825 $this->emit([
826 'garnish', $i + 1, rateText($rate), 'balance', money($balance),
827 ], 4);
828 }
829 }
830 if (!$ids) {
831 $this->emit(['(none)'], 2);
832 }
833 } elseif ($command === 'CONFIG') {
834 $this->emit(['CONFIG']);
835 foreach ($this->settings as $name => $value) {
836 $this->emit([
837 $name, in_array($name, RATE_SETTINGS, true) ? rateText($value) : money($value),
838 ], 2);
839 }
840 foreach (['bracket' => $this->brackets, 'tier' => $this->tiers] as $label => $table) {
841 ksort($table, SORT_NUMERIC);
842 foreach ($table as $lower => $rate) {
843 $this->emit([$label, money((int) $lower), rateText($rate)], 2);
844 }
845 }
846 }
847 }

849 public function execute(string $line): void
850 {
851 $tokens = explode(' ', $line);
852 $command = array_shift($tokens);
853 try {
854 $this->command($command, $tokens);
855 } catch (CommandError $error) {
856 $this->emit(['ERROR', $error->getMessage()]);
857 }
858 }
