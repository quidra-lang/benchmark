85Amount progressive(const map<Amount, Amount>& table, Amount amount) {
86 Amount numerator = 0;
87 for (auto it = table.begin(); it != table.end() && it->first < amount; ++it) {
88 auto next = std::next(it);
89 Amount upper = next == table.end() ? amount : next->first;
90 numerator += it->second * (min(amount, upper) - it->first);
91 }
92 return numerator / 10000;
93}

172 Employee& employee(const string& text, bool active = false, const string& kind = "") {
173 int identifier = stoi(text);
174 auto it = employees.find(identifier);
175 if (it == employees.end()) fail("unknown employee " + text);
176 Employee& person = it->second;
177 if (active && !person.active) fail("terminated employee " + text);
178 if (!kind.empty() && person.kind != kind) {
179 string lowered = kind;
180 transform(lowered.begin(), lowered.end(), lowered.begin(),
181 [](unsigned char c) { return static_cast<char>(tolower(c)); });
182 fail("not " + lowered + " " + text);
183 }
184 return person;
185 }

219 Six daily(const Employee& person, int day) const {
220 Amount worked = 0, night = 0;
221 auto it = person.shifts.find(day);
222 if (it != person.shifts.end()) {
223 for (const Shift& shift : it->second) {
224 worked += shift.end - shift.start;
225 night += max(0, min(shift.end, 360) - shift.start);
226 night += max(0, shift.end - max(shift.start, 1320));
227 }
228 }
229 auto pto_it = person.pto.find(day);
230 Amount pto = pto_it == person.pto.end() ? 0 : pto_it->second;
231 if (day % 7 == 0 || holidays.count(day))
232 return {worked, 0, 0, worked, night, pto};
233 return {worked, min<Amount>(worked, 480),
234 min<Amount>(max<Amount>(0, worked - 480), 240),
235 max<Amount>(0, worked - 720), night, pto};
236 }

238 pair<Five, Amount> weekly(const Employee& person, int week) const {
239 Five result{};
240 for (int day = 7 * week - 6; day <= 7 * week; ++day) {
241 Six values = daily(person, day);
242 for (int i = 0; i < 5; ++i) result[i] += values[i + 1];
243 }
244 Amount weekly_overtime = max<Amount>(0, result[0] - 2400);
245 result[0] -= weekly_overtime;
246 result[1] += weekly_overtime;
247 return {result, weekly_overtime};
248 }

250 Five breakdown(const Employee& person, int period) const {
251 Five first = weekly(person, 2 * period - 1).first;
252 Five second = weekly(person, 2 * period).first;
253 for (int i = 0; i < 5; ++i) first[i] += second[i];
254 return first;
255 }

257 optional<Stub> calculate(const Employee& person, int period) const {
258 Stub stub;
259 if (person.kind == "HOURLY") {
260 Five hours = breakdown(person, period);
261 for (int i = 0; i < 5; ++i) stub[fields[i]] = hours[i];
262 Amount weighted = 2 * (hours[0] + hours[4]) + 3 * hours[1] + 4 * hours[2];
263 stub["earnings"] =
264 (weighted * person.base + 2 * hours[3] * settings.at("NIGHT")) / 120;
265 } else if (person.kind == "SALARIED") {
266 if (person.active) {
267 stub["salary"] = person.base / 26;
268 for (int day : person.leave)
269 if (period_of(day) == period && !holidays.count(day))
270 stub["leave"]++;
271 stub["deduction"] = stub.get("leave") * (person.base / 260);
272 stub["earnings"] = stub.get("salary") - stub.get("deduction");
273 }
274 } else {
275 auto sale = person.sales.find(period);
276 stub["sales"] = sale == person.sales.end() ? 0 : sale->second;
277 stub["commission"] = progressive(tiers, stub.get("sales"));
278 stub["earnings"] = stub.get("commission") + (person.active ? person.base : 0);
279 }
280
281 stub["bonus"] = person.bonus;
282 Amount gross = stub.get("earnings") + stub.get("bonus");
283 if (gross == 0) return nullopt;
284 stub["gross"] = gross;
285 Amount remaining = gross;
286
287 auto deduct = [&](const string& field, Amount amount) {
288 Amount actual = min(amount, remaining);
289 stub[field] = actual;
290 remaining -= actual;
291 };
292
293 Amount retirement_room =
294 max<Amount>(0, settings.at("RETIRECAP") - person.ytd.at("retirement"));
295 deduct("retirement", min(gross * person.retirement / 10000, retirement_room));
296 deduct("health", person.health);
297 stub["W"] = remaining;
298 Amount base_taxable = max<Amount>(0,
299 stub.get("earnings") - stub.get("retirement") - stub.get("health"));
300 Amount taxable = max<Amount>(0,
301 base_taxable - person.allowances * settings.at("ALLOWANCE"));
302 deduct("federal", progressive(brackets, taxable) + person.extra);
303 deduct("supplemental", (stub.get("W") - base_taxable) * settings.at("SUPP") / 10000);
304 deduct("state", stub.get("W") * settings.at("STATE") / 10000);
305
306 Amount medicare_wages = gross - stub.get("health");
307 Amount social_wages = min(medicare_wages,
308 max<Amount>(0, settings.at("SSBASE") - person.ytd.at("Y")));
309 Amount additional = max<Amount>(0, min(medicare_wages,
310 person.ytd.at("F") + medicare_wages - settings.at("ADDBASE")));
311 stub["F"] = medicare_wages;
312 stub["Y"] = social_wages;
313 deduct("social", social_wages * settings.at("SSRATE") / 10000);
314 deduct("medicare", medicare_wages * settings.at("MEDRATE") / 10000 +
315 additional * settings.at("ADDMED") / 10000);
316
317 Amount disposable = remaining;
318 Amount cap = disposable * settings.at("GARNCAP") / 10000;
319 Amount garnished = 0;
320 for (const Garnishment& order : person.garnishments) {
321 Amount taken = min({disposable * order.rate / 10000,
322 order.balance, cap - garnished});
323 garnished += taken;
324 stub.balances.push_back(order.balance - taken);
325 }
326 stub["garnishment"] = garnished;
327 stub.garnishment_balance =
328 accumulate(stub.balances.begin(), stub.balances.end(), 0LL);
329 stub["net"] = remaining - garnished;
330 stub["employer_social"] = social_wages * settings.at("SSRATE") / 10000;
331 stub["employer_medicare"] = medicare_wages * settings.at("MEDRATE") / 10000;
332 stub["J"] = min(medicare_wages,
333 max<Amount>(0, settings.at("UIBASE") - person.ytd.at("J")));
334 stub["unemployment"] = stub.get("J") * settings.at("UIRATE") / 10000;
335 Amount eligible_match = min(stub.get("retirement"),
336 gross * settings.at("MATCHMAX") / 10000);
337 stub["match"] = eligible_match * settings.at("MATCH") / 10000;
338 stub["employer"] = stub.get("employer_social") + stub.get("employer_medicare") +
339 stub.get("unemployment") + stub.get("match");
340 if (person.kind == "HOURLY")
341 stub["accrued"] = min<Amount>(960 - person.pto_balance,
342 (stub.get("R") + stub.get("O") + stub.get("D")) / 20);
343 stub.pto_balance = person.pto_balance + stub.get("accrued");
344 return stub;
345 }

357 void stub_block(const Employee& person, const Totals& stub, Amount pto_balance,
358 Amount garnish_balance, int indentation) {
359 if (person.kind == "HOURLY") {
360 Five values{};
361 for (int i = 0; i < 5; ++i) values[i] = stub.at(fields[i]);
362 vector<string> parts = {"hours"};
363 vector<string> hourly = hours_parts(values);
364 parts.insert(parts.end(), hourly.begin(), hourly.end());
365 emit_parts(indentation, parts);
366 emit(indentation, "pto", "accrued", duration(stub.at("accrued")),
367 "balance", duration(pto_balance));
368 } else if (person.kind == "SALARIED") {
369 emit(indentation, "salary", money(stub.at("salary")), "leave",
370 stub.at("leave"), "deduction", money(stub.at("deduction")));
371 } else {
372 emit(indentation, "sales", money(stub.at("sales")),
373 "commission", money(stub.at("commission")));
374 }
375 for (const string& field : {
376 "earnings", "bonus", "gross", "retirement", "health", "federal",
377 "supplemental", "state", "social", "medicare"
378 }) {
379 emit(indentation, field == "social" ? "social-security" : field,
380 money(stub.at(field)));
381 }
382 emit(indentation, "garnishment", money(stub.at("garnishment")),
383 "balance", money(garnish_balance));
384 emit(indentation, "net", money(stub.at("net")));
385 emit(indentation, "employer", "social-security",
386 money(stub.at("employer_social")),
387 "medicare", money(stub.at("employer_medicare")),
388 "unemployment", money(stub.at("unemployment")),
389 "match", money(stub.at("match")));
390 }

398 void show_hours(const Employee& person, int period) {
399 emit(0, "HOURS", person.id, "period", period);
400 bool found = false;
401 for (int day = 14 * period - 13; day <= 14 * period; ++day) {
402 auto it = person.shifts.find(day);
403 vector<Shift> shifts = it == person.shifts.end() ? vector<Shift>{} : it->second;
404 if (shifts.empty() && !person.pto.count(day)) continue;
405 sort(shifts.begin(), shifts.end(), [](const Shift& a, const Shift& b) {
406 return a.start < b.start;
407 });
408 Six values = daily(person, day);
409 vector<string> parts = {"day", to_string(day), weekday(day)};
410 if (holidays.count(day)) parts.push_back("holiday");
411 for (const Shift& shift : shifts)
412 parts.push_back(clock_text(shift.start) + "-" + clock_text(shift.end));
413 parts.push_back("total");
414 parts.push_back(duration(values[0]));
415 Five current = {values[1], values[2], values[3], values[4], values[5]};
416 vector<string> hours = hours_parts(current);
417 parts.insert(parts.end(), hours.begin(), hours.end());
418 emit_parts(2, parts);
419 found = true;
420 }
421 if (!found) emit(2, "(none)");
422 for (int week : {2 * period - 1, 2 * period}) {
423 auto [values, overtime] = weekly(person, week);
424 vector<string> parts = {"week", to_string(week)};
425 vector<string> hours = hours_parts(values);
426 parts.insert(parts.end(), hours.begin(), hours.end());
427 parts.push_back("weekly-ot");
428 parts.push_back(duration(overtime));
429 emit_parts(2, parts);
430 }
431 vector<string> parts = {"total"};
432 vector<string> hours = hours_parts(breakdown(person, period));
433 parts.insert(parts.end(), hours.begin(), hours.end());
434 emit_parts(2, parts);
435 }

437 void run(int period) {
438 closed = period;
439 emit(0, "RUN", period, "days",
440 to_string(14 * period - 13) + "-" + to_string(14 * period));
441 int paid = 0, skipped = 0;
442 vector<const Totals*> created;
443 for (auto& [id, person] : employees) {
444 optional<Stub> calculated = calculate(person, period);
445 if (!calculated) {
446 if (person.active) ++skipped;
447 continue;
448 }
449 Stub stub = move(*calculated);
450 print_stub(person, period, stub, person.department);
451 person.bonus = 0;
452 person.pto_balance = stub.pto_balance;
453 for (size_t i = 0; i < person.garnishments.size(); ++i)
454 person.garnishments[i].balance = stub.balances[i];
455 for (const string& field : fields)
456 person.ytd[field] += stub.get(field);
457 ++person.periods;
458 person.stubs.emplace(period, StubRecord{move(stub), person.department});
459 created.push_back(&person.stubs.at(period).stub.values);
460 ++paid;
461 }
462 Totals total = sum_stubs(created);
463 emit(2, "TOTAL", "paid", paid, "skipped", skipped, "gross",
464 money(total["gross"]), "deductions", money(total["gross"] - total["net"]),
465 "net", money(total["net"]), "employer", money(total["employer"]));
466 }

468 void journal(int period) {
469 vector<const Totals*> stubs;
470 for (const auto& [id, person] : employees) {
471 auto it = person.stubs.find(period);
472 if (it != person.stubs.end())
473 stubs.push_back(&it->second.stub.values);
474 }
475 Totals total = sum_stubs(stubs);
476 struct Row { string side, label; Amount amount; };
477 vector<Row> rows = {
478 {"debit", "wages", total["gross"]},
479 {"debit", "employer-tax",
480 total["employer_social"] + total["employer_medicare"] + total["unemployment"]},
481 {"debit", "retirement-match", total["match"]},
482 {"credit", "retirement", total["retirement"] + total["match"]},
483 {"credit", "health", total["health"]},
484 {"credit", "federal", total["federal"] + total["supplemental"]},
485 {"credit", "state", total["state"]},
486 {"credit", "fica", total["social"] + total["medicare"] +
487 total["employer_social"] + total["employer_medicare"]},
488 {"credit", "unemployment", total["unemployment"]},
489 {"credit", "garnishment", total["garnishment"]},
490 {"credit", "net-pay", total["net"]}
491 };
492 emit(0, "JOURNAL", period, "stubs", stubs.size());
493 Amount debit = 0, credit = 0;
494 for (const Row& row : rows) {
495 emit(2, row.side, row.label, money(row.amount));
496 if (row.side == "debit") debit += row.amount;
497 else credit += row.amount;
498 }
499 emit(2, "total", "debit", money(debit), "credit", money(credit));

512 void command(const string& cmd, const vector<string>& a) {
513 if (cmd == "SET") {
514 auto it = settings.find(a[0]);
515 if (it == settings.end()) fail("unknown setting " + a[0]);
516 bool rate = find(rate_settings.begin(), rate_settings.end(), a[0])
517 != rate_settings.end();
518 Amount value = rate ? rate_check(a[1]) : cents(a[1]);
519 Amount old = it->second;
520 it->second = value;
521 emit(0, "SET", a[0], rate ? rate_text(old) : money(old),
522 "->", rate ? rate_text(value) : money(value));
523 } else if (cmd == "BRACKET" || cmd == "TIER") {
524 Amount rate = rate_check(a[1]);
525 auto& table = cmd == "BRACKET" ? brackets : tiers;
526 Amount lower = cents(a[0]);
527 bool replaced = table.count(lower);
528 table[lower] = rate;
529 emit(0, cmd, money(lower), rate_text(rate),
530 replaced ? "replaced" : "added");
531 } else if (cmd == "EMP") {
532 int id = stoi(a[0]);
533 if (employees.count(id)) fail("duplicate employee " + a[0]);
534 Amount base = amount_check(a[3]);
535 Employee person;
536 person.id = id;
537 person.department = a[1];
538 person.kind = a[2];
539 person.base = base;
540 employees.emplace(id, move(person));
541 emit(0, "EMP", id, a[1], a[2], money(base));
542 } else if (cmd == "RATE" || cmd == "DEPT" || cmd == "BENEFIT" ||
543 cmd == "W4" || cmd == "GARNISH" || cmd == "TERM" || cmd == "BONUS") {
544 Employee& person = employee(a[0], true);
545 if (cmd == "RATE") {
546 Amount base = amount_check(a[1]), old = person.base;
547 person.base = base;
548 emit(0, "RATE", person.id, money(old), "->", money(base));
549 } else if (cmd == "DEPT") {
550 if (a[1] == person.department) fail("same department " + a[1]);
551 string old = person.department;
552 person.department = a[1];
553 emit(0, "DEPT", person.id, old, "->", a[1]);
554 } else if (cmd == "BENEFIT") {
555 Amount rate = rate_check(a[1]);
556 person.retirement = rate;
557 person.health = cents(a[2]);
558 emit(0, "BENEFIT", person.id, "retirement", rate_text(rate),
559 "health", money(person.health));
560 } else if (cmd == "W4") {
561 person.allowances = stoll(a[1]);
562 person.extra = cents(a[2]);
563 emit(0, "W4", person.id, "allowances", person.allowances,
564 "extra", money(person.extra));
565 } else if (cmd == "GARNISH") {
566 Amount rate = rate_check(a[1]);
567 Amount amount = amount_check(a[2]);
568 person.garnishments.push_back({rate, amount});
569 emit(0, "GARNISH", person.id, "order", person.garnishments.size(),
570 rate_text(rate), "balance", money(amount));
571 } else if (cmd == "TERM") {
572 person.active = false;
573 emit(0, "TERM", person.id);
574 } else {
575 Amount amount = amount_check(a[1]);
576 person.bonus += amount;
577 emit(0, "BONUS", person.id, money(amount), "pending", money(person.bonus));
578 }
579 } else if (cmd == "HOLIDAY") {
580 int day = day_check(a[0]);
581 if (holidays.count(day)) fail("duplicate holiday " + a[0]);
582 holidays.insert(day);
583 emit(0, "HOLIDAY", day, weekday(day));
584 } else if (cmd == "TIME" || cmd == "UNTIME" || cmd == "PTO" ||
585 cmd == "LEAVE" || cmd == "SALES") {
586 string kind = cmd == "LEAVE" ? "SALARIED" :
587 cmd == "SALES" ? "COMMISSION" : "HOURLY";
588 Employee& person = employee(a[0], true, kind);
589 int day = day_check(a[1]);
590 if (cmd == "TIME") {
591 int start = time_check(a[2]), end = time_check(a[3]);
592 if (start >= end) fail("invalid shift " + a[2] + "-" + a[3]);
593 auto shift_it = person.shifts.find(day);
594 vector<Shift> shifts = shift_it == person.shifts.end() ?
595 vector<Shift>{} : shift_it->second;
596 sort(shifts.begin(), shifts.end(), [](const Shift& a, const Shift& b) {
597 return a.start < b.start;
598 });
599 for (const Shift& shift : shifts)
600 if (start < shift.end && shift.start < end)
601 fail("overlapping shift " + clock_text(shift.start) + "-" +
602 clock_text(shift.end));
603 int worked = end - start;
604 for (const Shift& shift : shifts) worked += shift.end - shift.start;
605 if (worked > 960) fail("daily limit " + a[0] + " " + a[1]);
606 person.shifts[day].push_back({start, end});
607 emit(0, "TIME", person.id, "day", day,
608 clock_text(start) + "-" + clock_text(end), duration(end - start));
609 } else if (cmd == "UNTIME") {
610 int start = time_check(a[2]);
611 auto it = person.shifts.find(day);
612 if (it == person.shifts.end()) fail("no shift " + a[0] + " " + a[1] + " " + a[2]);
613 auto& shifts = it->second;
614 auto found = find_if(shifts.begin(), shifts.end(), [start](const Shift& shift) {
615 return shift.start == start;
616 });
617 if (found == shifts.end()) fail("no shift " + a[0] + " " + a[1] + " " + a[2]);
618 Shift removed = *found;
619 shifts.erase(found);
620 emit(0, "UNTIME", person.id, "day", day,
621 clock_text(removed.start) + "-" + clock_text(removed.end));
622 } else if (cmd == "PTO") {
623 int hour = stoi(a[2].substr(0, 2));
624 int minute = stoi(a[2].substr(3, 2));
625 int length = hour * 60 + minute;
626 if (minute > 59 || length < 1 || length > 480)
627 fail("invalid duration " + a[2]);
628 if (person.pto.count(day)) fail("duplicate pto " + a[0] + " " + a[1]);
629 if (length > person.pto_balance)
630 fail("insufficient pto " + a[0] + " " + duration(person.pto_balance));
631 person.pto[day] = length;
632 person.pto_balance -= length;
633 emit(0, "PTO", person.id, "day", day, duration(length),
634 "balance", duration(person.pto_balance));
635 } else if (cmd == "LEAVE") {
636 if (day % 7 == 0 || day % 7 == 6 || holidays.count(day))
637 fail("not workday " + a[1]);
638 if (person.leave.count(day)) fail("duplicate leave " + a[0] + " " + a[1]);
639 person.leave.insert(day);
640 emit(0, "LEAVE", person.id, "day", day, weekday(day));
641 } else {
642 Amount amount = amount_check(a[2]);
643 int period = period_of(day);
644 person.sales[period] += amount;
645 emit(0, "SALES", person.id, "day", day, money(amount),
646 "period", period, "total", money(person.sales[period]));
647 }
648 } else if (cmd == "HOURS") {
649 Employee& person = employee(a[0], false, "HOURLY");
650 int period = period_check(a[1]);
651 show_hours(person, period);
652 } else if (cmd == "RUN") {
653 int period = stoi(a[0]);
654 if (period != closed + 1 || period > 26) fail("invalid period " + a[0]);
655 run(period);
656 } else if (cmd == "PREVIEW") {
657 Employee& person = employee(a[0]);
658 int period = closed + 1;
659 optional<Stub> stub = calculate(person, period);
660 emit(0, "PREVIEW", person.id, "period", period);
661 if (!stub) emit(2, "no pay");
662 else print_stub(person, period, *stub, person.department);
663 } else if (cmd == "REPRINT") {
664 Employee& person = employee(a[0]);
665 int period = period_check(a[1]);
666 auto it = person.stubs.find(period);
667 if (it == person.stubs.end()) fail("no stub " + a[0] + " " + a[1]);
668 emit(0, "REPRINT", person.id, period);
669 print_stub(person, period, it->second.stub, it->second.department);
670 } else if (cmd == "YTD") {
671 Employee& person = employee(a[0]);
672 emit(0, "YTD", person.id, person.department, person.kind,
673 person.active ? "active" : "terminated", "periods", person.periods);
674 stub_block(person, person.ytd, person.pto_balance, person.garnish_balance(), 2);
675 emit(2, "wages", "federal", money(person.ytd["W"]),
676 "social-security", money(person.ytd["Y"]),
677 "medicare", money(person.ytd["F"]),
678 "unemployment", money(person.ytd["J"]));
679 emit(2, "retirement-room",
680 money(max<Amount>(0, settings["RETIRECAP"] - person.ytd["retirement"])),
681 "pending-bonus", money(person.bonus));
682 } else if (cmd == "JOURNAL") {
683 int period = stoi(a[0]);
684 if (period < 1 || period > closed) fail("no run " + a[0]);
685 journal(period);
686 } else if (cmd == "REPORT") {
687 map<string, vector<const Totals*>> departments;
688 vector<const Totals*> all;
689 for (const auto& [id, person] : employees) {
690 for (const auto& [period, record] : person.stubs) {
691 departments[record.department].push_back(&record.stub.values);
692 all.push_back(&record.stub.values);
693 }
694 }
695 emit(0, "REPORT", "period", closed);
696 for (const auto& [department, stubs] : departments)
697 report_line(department, stubs);
698 report_line("total", all);
699 } else if (cmd == "TOP") {
700 int count = stoi(a[0]);
701 vector<const Employee*> ranking;
702 for (const auto& [id, person] : employees)
703 if (person.periods) ranking.push_back(&person);
704 sort(ranking.begin(), ranking.end(), [](const Employee* a, const Employee* b) {
705 return a->ytd.at("gross") != b->ytd.at("gross") ?
706 a->ytd.at("gross") > b->ytd.at("gross") : a->id < b->id;
707 });
708 emit(0, "TOP", count);
709 for (int i = 0; i < count && i < static_cast<int>(ranking.size()); ++i) {
710 const Employee& person = *ranking[i];
711 emit(2, i + 1, person.id, person.department, "gross",
712 money(person.ytd.at("gross")), "net", money(person.ytd.at("net")));
713 }
714 if (ranking.empty()) emit(2, "(none)");
715 } else if (cmd == "EMPLOYEES") {
716 emit(0, "EMPLOYEES");
717 for (const auto& [id, person] : employees) {
718 emit(2, id, person.department, person.kind, money(person.base),
719 person.active ? "active" : "terminated",
720 "retirement", rate_text(person.retirement),
721 "health", money(person.health),
722 "allowances", person.allowances, "extra", money(person.extra),
723 "bonus", money(person.bonus));
724 for (size_t i = 0; i < person.garnishments.size(); ++i) {
725 const Garnishment& order = person.garnishments[i];
726 emit(4, "garnish", i + 1, rate_text(order.rate),
727 "balance", money(order.balance));
728 }
729 }
730 if (employees.empty()) emit(2, "(none)");
731 } else if (cmd == "CONFIG") {
732 emit(0, "CONFIG");
733 for (const auto& [name, default_value] : initial_settings) {
734 Amount value = settings[name];
735 bool rate = find(rate_settings.begin(), rate_settings.end(), name)
736 != rate_settings.end();
737 emit(2, name, rate ? rate_text(value) : money(value));
738 }
739 for (const auto& [lower, rate] : brackets)
740 emit(2, "bracket", money(lower), rate_text(rate));
741 for (const auto& [lower, rate] : tiers)
742 emit(2, "tier", money(lower), rate_text(rate));
743 }
744 }
745
746public:
747 Payroll() {
748 for (const auto& [name, value] : initial_settings) settings[name] = value;
749 }

751 void execute(const string& line) {
752 istringstream input(line);
753 string cmd;
754 input >> cmd;
755 vector<string> args;
756 string token;
757 while (input >> token) args.push_back(token);
758 try {
759 command(cmd, args);
760 } catch (const runtime_error& error) {
761 emit(0, "ERROR", error.what());
762 }
763 }
764
765 void print() const {
766 for (const string& line : output) cout << line << '\n';
767 }
768};
