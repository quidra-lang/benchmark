90 private static long progressive(TreeMap<Long, Long> table, long amount) {
91 List<Map.Entry<Long, Long>> pairs = new ArrayList<>(table.entrySet());
92 long numerator = 0;
93 for (int i = 0; i < pairs.size(); ++i) {
94 long lower = pairs.get(i).getKey();
95 if (lower >= amount) break;
96 long upper = i + 1 < pairs.size() ? pairs.get(i + 1).getKey() : amount;
97 numerator += pairs.get(i).getValue() * (Math.min(amount, upper) - lower);
98 }
99 return numerator / 10000;
100 }

230 private Daily daily(Employee person, int day) {
231 List<Shift> shifts = person.shifts.getOrDefault(day, Collections.emptyList());
232 long worked = 0, night = 0;
233 for (Shift shift : shifts) {
234 worked += shift.end() - shift.start();
235 night += Math.max(0, Math.min(shift.end(), 360) - shift.start());
236 night += Math.max(0, shift.end() - Math.max(shift.start(), 1320));
237 }
238 long pto = person.pto.getOrDefault(day, 0);
239 if (day % 7 == 0 || holidays.contains(day))
240 return new Daily(worked, 0, 0, worked, night, pto);
241 return new Daily(worked, Math.min(worked, 480),
242 Math.min(Math.max(worked - 480, 0), 240),
243 Math.max(worked - 720, 0), night, pto);
244 }

246 private Week weekly(Employee person, int week) {
247 long[] result = new long[5];
248 for (int day = 7 * week - 6; day <= 7 * week; ++day) {
249 long[] values = daily(person, day).classified();
250 for (int i = 0; i < 5; ++i) result[i] += values[i];
251 }
252 long overtime = Math.max(0, result[0] - 2400);
253 result[0] -= overtime;
254 result[1] += overtime;
255 return new Week(result, overtime);
256 }

258 private long[] breakdown(Employee person, int period) {
259 long[] first = weekly(person, 2 * period - 1).values();
260 long[] second = weekly(person, 2 * period).values();
261 for (int i = 0; i < 5; ++i) first[i] += second[i];
262 return first;
263 }

265 private Stub calculate(Employee person, int period) {
266 Stub stub = new Stub();
267 if (person.kind.equals("HOURLY")) {
268 long[] hours = breakdown(person, period);
269 for (int i = 0; i < 5; ++i) stub.set(FIELDS.get(i), hours[i]);
270 long weighted = 2 * (hours[0] + hours[4]) + 3 * hours[1] + 4 * hours[2];
271 stub.set("earnings", (weighted * person.base +
272 2 * hours[3] * settings.get("NIGHT")) / 120);
273 } else if (person.kind.equals("SALARIED")) {
274 if (person.active) {
275 stub.set("salary", person.base / 26);
276 long leave = person.leave.stream().filter(day ->
277 periodOf(day) == period && !holidays.contains(day)).count();
278 stub.set("leave", leave);
279 stub.set("deduction", leave * (person.base / 260));
280 stub.set("earnings", stub.get("salary") - stub.get("deduction"));
281 }
282 } else {
283 stub.set("sales", person.sales.getOrDefault(period, 0L));
284 stub.set("commission", progressive(tiers, stub.get("sales")));
285 stub.set("earnings", stub.get("commission") + (person.active ? person.base : 0));
286 }
287
288 stub.set("bonus", person.bonus);
289 long gross = stub.get("earnings") + stub.get("bonus");
290 if (gross == 0) return null;
291 stub.set("gross", gross);
292
293 long remaining = gross;
294 long retirementRoom = Math.max(0, settings.get("RETIRECAP") - person.ytd.get("retirement"));
295 remaining = deduct(stub, "retirement",
296 Math.min(gross * person.retirement / 10000, retirementRoom), remaining);
297 remaining = deduct(stub, "health", person.health, remaining);
298 stub.set("W", remaining);
299 long baseTaxable = Math.max(0,
300 stub.get("earnings") - stub.get("retirement") - stub.get("health"));
301 long allowanceTaxable = Math.max(0, baseTaxable - person.allowances * settings.get("ALLOWANCE"));
302 remaining = deduct(stub, "federal",
303 progressive(brackets, allowanceTaxable) + person.extra, remaining);
304 remaining = deduct(stub, "supplemental",
305 (stub.get("W") - baseTaxable) * settings.get("SUPP") / 10000, remaining);
306 remaining = deduct(stub, "state",
307 stub.get("W") * settings.get("STATE") / 10000, remaining);
308
309 long medicareWages = gross - stub.get("health");
310 long socialWages = Math.min(medicareWages,
311 Math.max(0, settings.get("SSBASE") - person.ytd.get("Y")));
312 long additional = Math.max(0, Math.min(medicareWages,
313 person.ytd.get("F") + medicareWages - settings.get("ADDBASE")));
314 stub.set("F", medicareWages);
315 stub.set("Y", socialWages);
316 remaining = deduct(stub, "social",
317 socialWages * settings.get("SSRATE") / 10000, remaining);
318 remaining = deduct(stub, "medicare",
319 medicareWages * settings.get("MEDRATE") / 10000 +
320 additional * settings.get("ADDMED") / 10000, remaining);
321
322 long disposable = remaining;
323 long cap = disposable * settings.get("GARNCAP") / 10000;
324 long garnished = 0;
325 for (Garnishment order : person.garnishments) {
326 long taken = Math.min(disposable * order.rate / 10000,
327 Math.min(order.balance, cap - garnished));
328 garnished += taken;
329 stub.balances.add(order.balance - taken);
330 }
331 stub.set("garnishment", garnished);
332 stub.garnishmentBalance = stub.balances.stream().mapToLong(Long::longValue).sum();
333 stub.set("net", remaining - garnished);
334 stub.set("employer_social", socialWages * settings.get("SSRATE") / 10000);
335 stub.set("employer_medicare", medicareWages * settings.get("MEDRATE") / 10000);
336 stub.set("J", Math.min(medicareWages,
337 Math.max(0, settings.get("UIBASE") - person.ytd.get("J"))));
338 stub.set("unemployment", stub.get("J") * settings.get("UIRATE") / 10000);
339 long eligibleMatch = Math.min(stub.get("retirement"),
340 gross * settings.get("MATCHMAX") / 10000);
341 stub.set("match", eligibleMatch * settings.get("MATCH") / 10000);
342 stub.set("employer", stub.get("employer_social") + stub.get("employer_medicare") +
343 stub.get("unemployment") + stub.get("match"));
344 if (person.kind.equals("HOURLY"))
345 stub.set("accrued", Math.min(960 - person.ptoBalance,
346 (stub.get("R") + stub.get("O") + stub.get("D")) / 20));
347 stub.ptoBalance = person.ptoBalance + stub.get("accrued");
348 return stub;
349 }

352 long deduction = Math.min(amount, remaining);
353 stub.set(field, deduction);
354 return remaining - deduction;

405 private void showHours(Employee person, int period) {
406 emit(0, "HOURS", person.id, "period", period);
407 boolean found = false;
408 for (int day = 14 * period - 13; day <= 14 * period; ++day) {
409 List<Shift> shifts = new ArrayList<>(person.shifts.getOrDefault(day, List.of()));
410 if (shifts.isEmpty() && !person.pto.containsKey(day)) continue;
411 shifts.sort(Comparator.comparingInt(Shift::start));
412 Daily daily = daily(person, day);
413 List<Object> parts = new ArrayList<>(List.of("day", day, weekday(day)));
414 if (holidays.contains(day)) parts.add("holiday");
415 for (Shift shift : shifts)
416 parts.add(clock(shift.start()) + "-" + clock(shift.end()));
417 parts.add("total");
418 parts.add(duration(daily.worked()));
419 parts.addAll(hoursParts(daily.classified()));
420 emitParts(2, parts);
421 found = true;
422 }
423 if (!found) emit(2, "(none)");
424 for (int week : new int[] {2 * period - 1, 2 * period}) {
425 Week breakdown = weekly(person, week);
426 List<Object> parts = new ArrayList<>(List.of("week", week));
427 parts.addAll(hoursParts(breakdown.values()));
428 parts.add("weekly-ot");
429 parts.add(duration(breakdown.overtime()));
430 emitParts(2, parts);
431 }
432 List<Object> parts = new ArrayList<>();
433 parts.add("total");
434 parts.addAll(hoursParts(breakdown(person, period)));
435 emitParts(2, parts);
436 }

438 private void run(int period) {
439 closed = period;
440 emit(0, "RUN", period, "days", (14 * period - 13) + "-" + (14 * period));
441 int paid = 0, skipped = 0;
442 List<Map<String, Long>> stubs = new ArrayList<>();
443 for (Employee person : employees.values()) {
444 Stub stub = calculate(person, period);
445 if (stub == null) {
446 if (person.active) ++skipped;
447 continue;
448 }
449 printStub(person, period, stub, person.department);
450 person.bonus = 0;
451 person.ptoBalance = stub.ptoBalance;
452 for (int i = 0; i < person.garnishments.size(); ++i)
453 person.garnishments.get(i).balance = stub.balances.get(i);
454 for (String field : FIELDS)
455 person.ytd.merge(field, stub.get(field), Long::sum);
456 ++person.periods;
457 person.stubs.put(period, new StubRecord(stub, person.department));
458 stubs.add(stub.values);
459 ++paid;
460 }
461 Map<String, Long> totals = sumStubs(stubs);
462 emit(2, "TOTAL", "paid", paid, "skipped", skipped,
463 "gross", money(totals.get("gross")),
464 "deductions", money(totals.get("gross") - totals.get("net")),
465 "net", money(totals.get("net")), "employer", money(totals.get("employer")));
466 }

510 private void command(String command, String[] a) {
511 switch (command) {
512 case "SET": {
513 String name = a[0];
514 if (!settings.containsKey(name))
515 throw new PayrollError("unknown setting " + name);
516 boolean rate = RATE_SETTINGS.contains(name);
517 long value = rate ? checkRate(a[1]) : cents(a[1]);
518 long previous = settings.get(name);
519 settings.put(name, value);
520 emit(0, "SET", name, rate ? rateText(previous) : money(previous), "->",
521 rate ? rateText(value) : money(value));
522 break;
523 }
524 case "BRACKET":
525 case "TIER": {
526 long rate = checkRate(a[1]);
527 TreeMap<Long, Long> table = command.equals("BRACKET") ? brackets : tiers;
528 long lower = cents(a[0]);
529 String action = table.containsKey(lower) ? "replaced" : "added";
530 table.put(lower, rate);
531 emit(0, command, money(lower), rateText(rate), action);
532 break;
533 }
534 case "EMP": {
535 int id = Integer.parseInt(a[0]);
536 if (employees.containsKey(id))
537 throw new PayrollError("duplicate employee " + a[0]);
538 long base = checkAmount(a[3]);
539 employees.put(id, new Employee(id, a[1], a[2], base));
540 emit(0, "EMP", id, a[1], a[2], money(base));
541 break;
542 }
543 case "RATE":
544 case "DEPT":
545 case "BENEFIT":
546 case "W4":
547 case "GARNISH":
548 case "TERM":
549 case "BONUS": {
550 Employee person = employee(a[0], true, null);
551 switch (command) {
552 case "RATE": {
553 long amount = checkAmount(a[1]);
554 long previous = person.base;
555 person.base = amount;
556 emit(0, "RATE", person.id, money(previous), "->", money(amount));
557 break;
558 }
559 case "DEPT": {
560 if (person.department.equals(a[1]))
561 throw new PayrollError("same department " + a[1]);
562 String previous = person.department;
563 person.department = a[1];
564 emit(0, "DEPT", person.id, previous, "->", a[1]);
565 break;
566 }
567 case "BENEFIT": {
568 long rate = checkRate(a[1]);
569 person.retirement = rate;
570 person.health = cents(a[2]);
571 emit(0, "BENEFIT", person.id, "retirement", rateText(rate),
572 "health", money(person.health));
573 break;
574 }
575 case "W4":
576 person.allowances = Long.parseLong(a[1]);
577 person.extra = cents(a[2]);
578 emit(0, "W4", person.id, "allowances", person.allowances,
579 "extra", money(person.extra));
580 break;
581 case "GARNISH": {
582 long rate = checkRate(a[1]);
583 long balance = checkAmount(a[2]);
584 person.garnishments.add(new Garnishment(rate, balance));
585 emit(0, "GARNISH", person.id, "order", person.garnishments.size(),
586 rateText(rate), "balance", money(balance));
587 break;
588 }
589 case "TERM":
590 person.active = false;
591 emit(0, "TERM", person.id);
592 break;
593 default: {
594 long bonus = checkAmount(a[1]);
595 person.bonus += bonus;
596 emit(0, "BONUS", person.id, money(bonus), "pending", money(person.bonus));
597 break;
598 }
599 }
600 break;
601 }
602 case "HOLIDAY": {
603 int day = checkDay(a[0]);
604 if (holidays.contains(day)) throw new PayrollError("duplicate holiday " + a[0]);
605 holidays.add(day);
606 emit(0, "HOLIDAY", day, weekday(day));
607 break;
608 }
609 case "TIME":
610 case "UNTIME":
611 case "PTO":
612 case "LEAVE":
613 case "SALES": {
614 String kind = command.equals("LEAVE") ? "SALARIED" :
615 command.equals("SALES") ? "COMMISSION" : "HOURLY";
616 Employee person = employee(a[0], true, kind);
617 int day = checkDay(a[1]);
618 switch (command) {
619 case "TIME": {
620 int start = checkTime(a[2]), end = checkTime(a[3]);
621 if (start >= end)
622 throw new PayrollError("invalid shift " + a[2] + "-" + a[3]);
623 List<Shift> shifts = new ArrayList<>(person.shifts.getOrDefault(day, List.of()));
624 shifts.sort(Comparator.comparingInt(Shift::start));
625 for (Shift shift : shifts) {
626 if (start < shift.end() && shift.start() < end) {
627 throw new PayrollError("overlapping shift " +
628 clock(shift.start()) + "-" + clock(shift.end()));
629 }
630 }
631 int worked = end - start;
632 for (Shift shift : shifts) worked += shift.end() - shift.start();
633 if (worked > 960)
634 throw new PayrollError("daily limit " + a[0] + " " + a[1]);
635 person.shifts.computeIfAbsent(day, key -> new ArrayList<>())
636 .add(new Shift(start, end));
637 emit(0, "TIME", person.id, "day", day,
638 clock(start) + "-" + clock(end), duration(end - start));
639 break;
640 }
641 case "UNTIME": {
642 int start = checkTime(a[2]);
643 List<Shift> shifts = person.shifts.getOrDefault(day, List.of());
644 Shift found = null;
645 for (Shift shift : shifts) {
646 if (shift.start() == start) { found = shift; break; }
647 }
648 if (found == null)
649 throw new PayrollError("no shift " + a[0] + " " + a[1] + " " + a[2]);
650 shifts.remove(found);
651 emit(0, "UNTIME", person.id, "day", day,
652 clock(found.start()) + "-" + clock(found.end()));
653 break;
654 }
655 case "PTO": {
656 int hour = Integer.parseInt(a[2].substring(0, 2));
657 int minute = Integer.parseInt(a[2].substring(3));
658 int length = hour * 60 + minute;
659 if (minute > 59 || length < 1 || length > 480)
660 throw new PayrollError("invalid duration " + a[2]);
661 if (person.pto.containsKey(day))
662 throw new PayrollError("duplicate pto " + a[0] + " " + a[1]);
663 if (length > person.ptoBalance)
664 throw new PayrollError("insufficient pto " + a[0] + " " + duration(person.ptoBalance));
665 person.pto.put(day, length);
666 person.ptoBalance -= length;
667 emit(0, "PTO", person.id, "day", day, duration(length),
668 "balance", duration(person.ptoBalance));
669 break;
670 }
671 case "LEAVE": {
672 if (day % 7 == 0 || day % 7 == 6 || holidays.contains(day))
673 throw new PayrollError("not workday " + a[1]);
674 if (person.leave.contains(day))
675 throw new PayrollError("duplicate leave " + a[0] + " " + a[1]);
676 person.leave.add(day);
677 emit(0, "LEAVE", person.id, "day", day, weekday(day));
678 break;
679 }
680 default: {
681 long amount = checkAmount(a[2]);
682 int period = periodOf(day);
683 person.sales.merge(period, amount, Long::sum);
684 emit(0, "SALES", person.id, "day", day, money(amount),
685 "period", period, "total", money(person.sales.get(period)));
686 }
687 }
688 break;
689 }
690 case "HOURS": {
691 Employee person = employee(a[0], false, "HOURLY");
692 showHours(person, checkPeriod(a[1]));
693 break;
694 }
695 case "RUN": {
696 int period = Integer.parseInt(a[0]);
697 if (period != closed + 1 || period > 26)
698 throw new PayrollError("invalid period " + a[0]);
699 run(period);
700 break;
701 }
702 case "PREVIEW": {
703 Employee person = employee(a[0]);
704 int period = closed + 1;
705 Stub stub = calculate(person, period);
706 emit(0, "PREVIEW", person.id, "period", period);
707 if (stub == null) emit(2, "no pay");
708 else printStub(person, period, stub, person.department);
709 break;
710 }
711 case "REPRINT": {
712 Employee person = employee(a[0]);
713 int period = checkPeriod(a[1]);
714 StubRecord record = person.stubs.get(period);
715 if (record == null)
716 throw new PayrollError("no stub " + a[0] + " " + a[1]);
717 emit(0, "REPRINT", person.id, period);
718 printStub(person, period, record.stub(), record.department());
719 break;
720 }
721 case "YTD": {
722 Employee person = employee(a[0]);
723 emit(0, "YTD", person.id, person.department, person.kind,
724 person.active ? "active" : "terminated", "periods", person.periods);
725 stubBlock(person, person.ytd, person.ptoBalance, person.garnishBalance(), 2);
726 emit(2, "wages", "federal", money(person.ytd.get("W")),
727 "social-security", money(person.ytd.get("Y")),
728 "medicare", money(person.ytd.get("F")),
729 "unemployment", money(person.ytd.get("J")));
730 emit(2, "retirement-room",
731 money(Math.max(0, settings.get("RETIRECAP") - person.ytd.get("retirement"))),
732 "pending-bonus", money(person.bonus));
733 break;
734 }
735 case "JOURNAL": {
736 int period = Integer.parseInt(a[0]);
737 if (period < 1 || period > closed)
738 throw new PayrollError("no run " + a[0]);
739 journal(period);
740 break;
741 }
742 case "REPORT": {
743 TreeMap<String, List<Map<String, Long>>> departments = new TreeMap<>();
744 List<Map<String, Long>> all = new ArrayList<>();
745 for (Employee person : employees.values()) {
746 for (StubRecord record : person.stubs.values()) {
747 departments.computeIfAbsent(record.department(), key -> new ArrayList<>())
748 .add(record.stub().values);
749 all.add(record.stub().values);
750 }
751 }
752 emit(0, "REPORT", "period", closed);
753 for (Map.Entry<String, List<Map<String, Long>>> entry : departments.entrySet())
754 reportLine(entry.getKey(), entry.getValue());
755 reportLine("total", all);
756 break;
757 }
758 case "TOP": {
759 int count = Integer.parseInt(a[0]);
760 List<Employee> ranking = new ArrayList<>();
761 for (Employee person : employees.values())
762 if (person.periods > 0) ranking.add(person);
763 ranking.sort(Comparator.comparingLong(
764 (Employee person) -> person.ytd.get("gross")).reversed()
765 .thenComparingInt(person -> person.id));
766 emit(0, "TOP", count);
767 for (int i = 0; i < Math.min(count, ranking.size()); ++i) {
768 Employee person = ranking.get(i);
769 emit(2, i + 1, person.id, person.department, "gross",
770 money(person.ytd.get("gross")), "net", money(person.ytd.get("net")));
771 }
772 if (ranking.isEmpty()) emit(2, "(none)");
773 break;
774 }
775 case "EMPLOYEES": {
776 emit(0, "EMPLOYEES");
777 for (Employee person : employees.values()) {
778 emit(2, person.id, person.department, person.kind, money(person.base),
779 person.active ? "active" : "terminated",
780 "retirement", rateText(person.retirement),
781 "health", money(person.health),
782 "allowances", person.allowances,
783 "extra", money(person.extra), "bonus", money(person.bonus));
784 for (int i = 0; i < person.garnishments.size(); ++i) {
785 Garnishment order = person.garnishments.get(i);
786 emit(4, "garnish", i + 1, rateText(order.rate),
787 "balance", money(order.balance));
788 }
789 }
790 if (employees.isEmpty()) emit(2, "(none)");
791 break;
792 }
793 case "CONFIG": {
794 emit(0, "CONFIG");
795 for (Map.Entry<String, Long> entry : settings.entrySet()) {
796 boolean rate = RATE_SETTINGS.contains(entry.getKey());
797 emit(2, entry.getKey(), rate ? rateText(entry.getValue()) : money(entry.getValue()));
798 }
799 for (Map.Entry<Long, Long> entry : brackets.entrySet())
800 emit(2, "bracket", money(entry.getKey()), rateText(entry.getValue()));
801 for (Map.Entry<Long, Long> entry : tiers.entrySet())
802 emit(2, "tier", money(entry.getKey()), rateText(entry.getValue()));
803 break;
804 }
805 default:
806 break;
807 }
808 }

810 void execute(String line) {
811 String[] parts = line.split(" ");
812 try {
813 command(parts[0], Arrays.copyOfRange(parts, 1, parts.length));
814 } catch (PayrollError error) {
815 emit(0, "ERROR", error.getMessage());
816 }
817 }

820 for (String line : output) System.out.println(line);
821 }
