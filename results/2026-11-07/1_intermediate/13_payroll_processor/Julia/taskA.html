35cents(text::String) = begin
36 whole, fraction = split(text, '.')
37 parse(Int64, whole) * 100 + parse(Int64, fraction)

40money(value::Integer) = @sprintf("%d.%02d", div(value, 100), mod(value, 100))
41rate_text(value::Integer) = money(value) * "%"
42duration(value::Integer) = @sprintf("%d:%02d", div(value, 60), mod(value, 60))
43clock(value::Integer) = @sprintf("%02d:%02d", div(value, 60), mod(value, 60))
44weekday(day::Int) = WEEKDAYS[mod(day - 1, 7) + 1]
45period_of(day::Int) = div(day - 1, 14) + 1
46empty_totals() = Dict(field => Int64(0) for field in FIELDS)

48function sum_stubs(stubs)
49 total = empty_totals()
50 for stub in stubs, field in FIELDS
51 total[field] += stub[field]
52 end
53 total
54end

56function progressive(brackets::Dict{Int64, Int64}, amount::Int64)
57 limits = sort(collect(keys(brackets)))
58 numerator = Int64(0)
59 for (i, lower) in enumerate(limits)
60 lower >= amount && break
61 upper = i < length(limits) ? limits[i+1] : amount
62 numerator += brackets[lower] * (min(amount, upper) - lower)
63 end
64 div(numerator, 10000)
65end

123function employee_of(payroll::Payroll, text::String; active=false, kind=nothing)
124 id = parse(Int, text)
125 haskey(payroll.employees, id) || fail("unknown employee " * text)
126 employee = payroll.employees[id]
127 active && !employee.active && fail("terminated employee " * text)
128 kind !== nothing && employee.kind != kind && fail("not " * lowercase(kind) * " " * text)
129 employee
130end

164function daily(payroll::Payroll, employee::Employee, day::Int)
165 worked = 0
166 night = 0
167 for (start_time, end_time) in get(employee.shifts, day, Tuple{Int, Int}[])
168 worked += end_time - start_time
169 night += max(0, min(end_time, 360) - start_time)
170 night += max(0, end_time - max(start_time, 1320))
171 end
172 pto = get(employee.pto, day, 0)
173 if day % 7 == 0 || day in payroll.holidays
174 return worked, Int64[0, 0, worked, night, pto]
175 end
176 regular = min(worked, 480)
177 overtime = min(max(worked - 480, 0), 240)
178 double = max(worked - 720, 0)
179 worked, Int64[regular, overtime, double, night, pto]
180end

182function weekly(payroll::Payroll, employee::Employee, week::Int)
183 totals = zeros(Int64, 5)
184 for day in (7 * week - 6):(7 * week)
185 _, values = daily(payroll, employee, day)
186 totals .+= values
187 end
188 extra = max(0, totals[1] - 2400)
189 totals[1] -= extra
190 totals[2] += extra
191 totals, extra
192end

194function breakdown(payroll::Payroll, employee::Employee, period::Int)
195 first, _ = weekly(payroll, employee, 2 * period - 1)
196 second, _ = weekly(payroll, employee, 2 * period)
197 first .+ second
198end

200function calculate(payroll::Payroll, employee::Employee, period::Int)
201 stub = empty_totals()
202 settings = payroll.settings
203 ytd = employee.ytd
204
205 if employee.kind == "HOURLY"
206 values = breakdown(payroll, employee, period)
207 for (field, value) in zip(("R", "O", "D", "N", "P"), values)
208 stub[field] = value
209 end
210 weighted = 2 * (stub["R"] + stub["P"]) + 3 * stub["O"] + 4 * stub["D"]
211 stub["earnings"] = div(weighted * employee.base + 2 * stub["N"] * settings["NIGHT"], 120)
212 elseif employee.kind == "SALARIED"
213 if employee.active
214 stub["salary"] = div(employee.base, 26)
215 stub["leave"] = count(day -> period_of(day) == period &&
216 !(day in payroll.holidays), employee.leave)
217 stub["deduction"] = stub["leave"] * div(employee.base, 260)
218 stub["earnings"] = stub["salary"] - stub["deduction"]
219 end
220 else
221 stub["sales"] = get(employee.sales, period, Int64(0))
222 stub["commission"] = progressive(payroll.tiers, stub["sales"])
223 stub["earnings"] = stub["commission"] + (employee.active ? employee.base : 0)
224 end
225
226 stub["bonus"] = employee.bonus
227 gross = stub["earnings"] + stub["bonus"]
228 gross == 0 && return nothing
229 stub["gross"] = gross
230 remaining = gross
231
232 function deduct!(field::String, amount::Int64)
233 stub[field] = min(amount, remaining)
234 remaining -= stub[field]
235 end
236
237 retirement_room = max(0, settings["RETIRECAP"] - ytd["retirement"])
238 deduct!("retirement", min(div(gross * employee.retirement, 10000), retirement_room))
239 deduct!("health", employee.health)
240 stub["W"] = remaining
241
242 base_taxable = max(0, stub["earnings"] - stub["retirement"] - stub["health"])
243 allowance_taxable = max(0, base_taxable - employee.allowances * settings["ALLOWANCE"])
244 deduct!("federal", progressive(payroll.brackets, allowance_taxable) + employee.extra)
245 deduct!("supplemental", div((stub["W"] - base_taxable) * settings["SUPP"], 10000))
246 deduct!("state", div(stub["W"] * settings["STATE"], 10000))
247
248 medicare_wages = gross - stub["health"]
249 social_wages = min(medicare_wages, max(0, settings["SSBASE"] - ytd["Y"]))
250 additional = max(0, min(medicare_wages, ytd["F"] + medicare_wages - settings["ADDBASE"]))
251 stub["F"] = medicare_wages
252 stub["Y"] = social_wages
253 deduct!("social", div(social_wages * settings["SSRATE"], 10000))
254 deduct!("medicare",
255 div(medicare_wages * settings["MEDRATE"], 10000) +
256 div(additional * settings["ADDMED"], 10000))
257
258 disposable = remaining
259 cap = div(disposable * settings["GARNCAP"], 10000)
260 balances = Int64[]
261 garnishment = Int64(0)
262 for order in employee.garnishments
263 taken = min(div(disposable * order.rate, 10000), order.balance, cap - garnishment)
264 garnishment += taken
265 push!(balances, order.balance - taken)
266 end
267 stub["garnishment"] = garnishment
268 stub["garnishment_balance"] = sum(balances; init=Int64(0))
269 stub["net"] = remaining - garnishment
270
271 stub["employer_social"] = div(social_wages * settings["SSRATE"], 10000)
272 stub["employer_medicare"] = div(medicare_wages * settings["MEDRATE"], 10000)
273 stub["J"] = min(medicare_wages, max(0, settings["UIBASE"] - ytd["J"]))
274 stub["unemployment"] = div(stub["J"] * settings["UIRATE"], 10000)
275 match_eligible = min(stub["retirement"], div(gross * settings["MATCHMAX"], 10000))
276 stub["match"] = div(match_eligible * settings["MATCH"], 10000)
277 stub["employer"] = sum(stub[field] for field in (
278 "employer_social", "employer_medicare", "unemployment", "match"
279 ))
280
281 if employee.kind == "HOURLY"
282 stub["accrued"] = min(960 - employee.pto_balance,
283 div(stub["R"] + stub["O"] + stub["D"], 20))
284 end
285 stub["pto_balance"] = employee.pto_balance + stub["accrued"]
286 stub, balances
287end

289function hours_parts(values)
290 parts = Any[]
291 for (label, value) in zip(("reg", "ot", "dbl", "night", "pto"), values)
292 append!(parts, (label, duration(value)))

294 parts

297function stub_block!(payroll::Payroll, employee::Employee,
298 stub::Dict{String, Int64}, indent::Int)
299 if employee.kind == "HOURLY"
300 values = [stub[field] for field in ("R", "O", "D", "N", "P")]
301 emit!(payroll, indent, "hours", hours_parts(values)...)
302 emit!(payroll, indent, "pto", "accrued", duration(stub["accrued"]),
303 "balance", duration(stub["pto_balance"]))
304 elseif employee.kind == "SALARIED"
305 emit!(payroll, indent, "salary", money(stub["salary"]), "leave", stub["leave"],
306 "deduction", money(stub["deduction"]))
307 else
308 emit!(payroll, indent, "sales", money(stub["sales"]),
309 "commission", money(stub["commission"]))
310 end
311
312 for field in ("earnings", "bonus", "gross", "retirement", "health",
313 "federal", "supplemental", "state", "social", "medicare")
314 label = field == "social" ? "social-security" : field
315 emit!(payroll, indent, label, money(stub[field]))
316 end
317 emit!(payroll, indent, "garnishment", money(stub["garnishment"]),
318 "balance", money(stub["garnishment_balance"]))
319 emit!(payroll, indent, "net", money(stub["net"]))
320 emit!(payroll, indent, "employer", "social-security", money(stub["employer_social"]),
321 "medicare", money(stub["employer_medicare"]),
322 "unemployment", money(stub["unemployment"]),
323 "match", money(stub["match"]))
324end

327 stub::Dict{String, Int64}, department::String)
328 emit!(payroll, 2, "STUB", employee.id, department, employee.kind, "period", period)

332function show_hours!(payroll::Payroll, employee::Employee, period::Int)
333 emit!(payroll, 0, "HOURS", employee.id, "period", period)
334 found = false
335 for day in (14 * period - 13):(14 * period)
336 shifts = sort(get(employee.shifts, day, Tuple{Int, Int}[]))
337 isempty(shifts) && !haskey(employee.pto, day) && continue
338 worked, values = daily(payroll, employee, day)
339 parts = Any["day", day, weekday(day)]
340 day in payroll.holidays && push!(parts, "holiday")
341 for (first, last) in shifts
342 push!(parts, clock(first) * "-" * clock(last))
343 end
344 append!(parts, ("total", duration(worked)))
345 append!(parts, hours_parts(values))
346 emit!(payroll, 2, parts...)
347 found = true
348 end
349 !found && emit!(payroll, 2, "(none)")
350 for week in (2 * period - 1, 2 * period)
351 values, overtime = weekly(payroll, employee, week)
352 emit!(payroll, 2, "week", week, hours_parts(values)...,
353 "weekly-ot", duration(overtime))
354 end
355 emit!(payroll, 2, "total", hours_parts(breakdown(payroll, employee, period))...)
356end

358function run!(payroll::Payroll, period::Int)
359 payroll.closed = period
360 emit!(payroll, 0, "RUN", period, "days", "$(14*period-13)-$(14*period)")
361 paid, skipped = 0, 0
362 stubs = Dict{String, Int64}[]
363 for id in sort(collect(keys(payroll.employees)))
364 employee = payroll.employees[id]
365 result = calculate(payroll, employee, period)
366 if result === nothing
367 skipped += employee.active
368 continue
369 end
370 stub, balances = result
371 print_stub!(payroll, employee, period, stub, employee.department)
372 employee.bonus = 0
373 employee.pto_balance = stub["pto_balance"]
374 for (order, balance) in zip(employee.garnishments, balances)
375 order.balance = balance
376 end
377 for field in FIELDS
378 employee.ytd[field] += stub[field]
379 end
380 employee.periods += 1
381 employee.stubs[period] = (stub, employee.department)
382 push!(stubs, stub)
383 paid += 1
384 end
385 total = sum_stubs(stubs)
386 emit!(payroll, 2, "TOTAL", "paid", paid, "skipped", skipped,
387 "gross", money(total["gross"]),
388 "deductions", money(total["gross"] - total["net"]),
389 "net", money(total["net"]), "employer", money(total["employer"]))
390end

392function journal!(payroll::Payroll, period::Int)
393 stubs = [employee.stubs[period][1] for employee in values(payroll.employees)
394 if haskey(employee.stubs, period)]
395 totals = sum_stubs(stubs)
396 rows = [
397 ("debit", "wages", totals["gross"]),
398 ("debit", "employer-tax", totals["employer_social"] +
399 totals["employer_medicare"] + totals["unemployment"]),
400 ("debit", "retirement-match", totals["match"]),
401 ("credit", "retirement", totals["retirement"] + totals["match"]),
402 ("credit", "health", totals["health"]),
403 ("credit", "federal", totals["federal"] + totals["supplemental"]),
404 ("credit", "state", totals["state"]),
405 ("credit", "fica", totals["social"] + totals["medicare"] +
406 totals["employer_social"] + totals["employer_medicare"]),
407 ("credit", "unemployment", totals["unemployment"]),
408 ("credit", "garnishment", totals["garnishment"]),
409 ("credit", "net-pay", totals["net"])
410 ]
411 emit!(payroll, 0, "JOURNAL", period, "stubs", length(stubs))
412 for (side, label, amount) in rows
413 emit!(payroll, 2, side, label, money(amount))
414 end
415 emit!(payroll, 2, "total",
416 "debit", money(sum(amount for (side, _, amount) in rows if side == "debit")),
417 "credit", money(sum(amount for (side, _, amount) in rows if side == "credit")))
418end

433function command!(payroll::Payroll, command::String, a::Vector{String})
434 if command == "SET"
435 name = a[1]
436 haskey(payroll.settings, name) || fail("unknown setting " * name)
437 value = name in RATE_SETTINGS ? rate_check(a[2]) : cents(a[2])
438 format = name in RATE_SETTINGS ? rate_text : money
439 previous = payroll.settings[name]
440 payroll.settings[name] = value
441 emit!(payroll, 0, "SET", name, format(previous), "->", format(value))
442
443 elseif command in ("BRACKET", "TIER")
444 rate = rate_check(a[2])
445 table = command == "BRACKET" ? payroll.brackets : payroll.tiers
446 lower = cents(a[1])
447 action = haskey(table, lower) ? "replaced" : "added"
448 table[lower] = rate
449 emit!(payroll, 0, command, money(lower), rate_text(rate), action)
450
451 elseif command == "EMP"
452 id = parse(Int, a[1])
453 haskey(payroll.employees, id) && fail("duplicate employee " * a[1])
454 base = amount_check(a[4])
455 payroll.employees[id] = Employee(id, a[2], a[3], base)
456 emit!(payroll, 0, "EMP", id, a[2], a[3], money(base))
457
458 elseif command in ("RATE", "DEPT", "BENEFIT", "W4", "GARNISH", "TERM", "BONUS")
459 employee = employee_of(payroll, a[1]; active=true)
460 if command == "RATE"
461 base = amount_check(a[2])
462 previous = employee.base
463 employee.base = base
464 emit!(payroll, 0, "RATE", employee.id, money(previous), "->", money(base))
465 elseif command == "DEPT"
466 a[2] == employee.department && fail("same department " * a[2])
467 previous = employee.department
468 employee.department = a[2]
469 emit!(payroll, 0, "DEPT", employee.id, previous, "->", a[2])
470 elseif command == "BENEFIT"
471 rate = rate_check(a[2])
472 employee.retirement = rate
473 employee.health = cents(a[3])
474 emit!(payroll, 0, "BENEFIT", employee.id, "retirement",
475 rate_text(rate), "health", money(employee.health))
476 elseif command == "W4"
477 employee.allowances = parse(Int, a[2])
478 employee.extra = cents(a[3])
479 emit!(payroll, 0, "W4", employee.id, "allowances", employee.allowances,
480 "extra", money(employee.extra))
481 elseif command == "GARNISH"
482 rate = rate_check(a[2])
483 amount = amount_check(a[3])
484 push!(employee.garnishments, Garnishment(rate, amount))
485 emit!(payroll, 0, "GARNISH", employee.id, "order", length(employee.garnishments),
486 rate_text(rate), "balance", money(amount))
487 elseif command == "TERM"
488 employee.active = false
489 emit!(payroll, 0, "TERM", employee.id)
490 else
491 amount = amount_check(a[2])
492 employee.bonus += amount
493 emit!(payroll, 0, "BONUS", employee.id, money(amount),
494 "pending", money(employee.bonus))
495 end
496
497 elseif command == "HOLIDAY"
498 day = day_check(payroll, a[1])
499 day in payroll.holidays && fail("duplicate holiday " * a[1])
500 push!(payroll.holidays, day)
501 emit!(payroll, 0, "HOLIDAY", day, weekday(day))
502
503 elseif command in ("TIME", "UNTIME", "PTO", "LEAVE", "SALES")
504 kind = command == "LEAVE" ? "SALARIED" :
505 command == "SALES" ? "COMMISSION" : "HOURLY"
506 employee = employee_of(payroll, a[1]; active=true, kind=kind)
507 day = day_check(payroll, a[2])
508 if command == "TIME"
509 first = time_check(a[3])
510 last = time_check(a[4])
511 first < last || fail("invalid shift $(a[3])-$(a[4])")
512 shifts = sort(get(employee.shifts, day, Tuple{Int, Int}[]))
513 for (previous_first, previous_last) in shifts
514 if first < previous_last && previous_first < last
515 fail("overlapping shift $(clock(previous_first))-$(clock(previous_last))")
516 end
517 end
518 sum(last - first for (first, last) in shifts; init=0) + last - first <= 960 ||
519 fail("daily limit $(a[1]) $(a[2])")
520 push!(get!(employee.shifts, day, Tuple{Int, Int}[]), (first, last))
521 emit!(payroll, 0, "TIME", employee.id, "day", day,
522 "$(clock(first))-$(clock(last))", duration(last - first))
523
524 elseif command == "UNTIME"
525 start_time = time_check(a[3])
526 shifts = get(employee.shifts, day, Tuple{Int, Int}[])
527 position = findfirst(shift -> shift[1] == start_time, shifts)
528 position === nothing && fail("no shift $(a[1]) $(a[2]) $(a[3])")
529 first, last = shifts[position]
530 deleteat!(shifts, position)
531 emit!(payroll, 0, "UNTIME", employee.id, "day", day,
532 "$(clock(first))-$(clock(last))")
533
534 elseif command == "PTO"
535 hours, minutes = parse.(Int, split(a[3], ':'))
536 duration_minutes = 60 * hours + minutes
537 minutes <= 59 && 1 <= duration_minutes <= 480 ||
538 fail("invalid duration " * a[3])
539 haskey(employee.pto, day) && fail("duplicate pto $(a[1]) $(a[2])")
540 duration_minutes <= employee.pto_balance ||
541 fail("insufficient pto $(a[1]) $(duration(employee.pto_balance))")
542 employee.pto[day] = duration_minutes
543 employee.pto_balance -= duration_minutes
544 emit!(payroll, 0, "PTO", employee.id, "day", day,
545 duration(duration_minutes), "balance", duration(employee.pto_balance))
546
547 elseif command == "LEAVE"
548 if day % 7 in (0, 6) || day in payroll.holidays
549 fail("not workday " * a[2])
550 end
551 day in employee.leave && fail("duplicate leave $(a[1]) $(a[2])")
552 push!(employee.leave, day)
553 emit!(payroll, 0, "LEAVE", employee.id, "day", day, weekday(day))
554
555 else
556 amount = amount_check(a[3])
557 period = period_of(day)
558 employee.sales[period] = get(employee.sales, period, Int64(0)) + amount
559 emit!(payroll, 0, "SALES", employee.id, "day", day, money(amount),
560 "period", period, "total", money(employee.sales[period]))
561 end
562
563 elseif command == "HOURS"
564 employee = employee_of(payroll, a[1]; kind="HOURLY")
565 show_hours!(payroll, employee, period_check(a[2]))
566
567 elseif command == "RUN"
568 period = parse(Int, a[1])
569 period == payroll.closed + 1 && period <= 26 ||
570 fail("invalid period " * a[1])
571 run!(payroll, period)
572
573 elseif command == "PREVIEW"
574 employee = employee_of(payroll, a[1])
575 period = payroll.closed + 1
576 result = calculate(payroll, employee, period)
577 emit!(payroll, 0, "PREVIEW", employee.id, "period", period)
578 if result === nothing
579 emit!(payroll, 2, "no pay")
580 else
581 print_stub!(payroll, employee, period, result[1], employee.department)
582 end
583
584 elseif command == "REPRINT"
585 employee = employee_of(payroll, a[1])
586 period = period_check(a[2])
587 haskey(employee.stubs, period) || fail("no stub $(a[1]) $(a[2])")
588 stub, department = employee.stubs[period]
589 emit!(payroll, 0, "REPRINT", employee.id, period)
590 print_stub!(payroll, employee, period, stub, department)
591
592 elseif command == "YTD"
593 employee = employee_of(payroll, a[1])
594 emit!(payroll, 0, "YTD", employee.id, employee.department, employee.kind,
595 employee.active ? "active" : "terminated", "periods", employee.periods)
596 totals = copy(employee.ytd)
597 totals["pto_balance"] = employee.pto_balance
598 totals["garnishment_balance"] = garnishment_balance(employee)
599 stub_block!(payroll, employee, totals, 2)
600 emit!(payroll, 2, "wages", "federal", money(totals["W"]),
601 "social-security", money(totals["Y"]),
602 "medicare", money(totals["F"]), "unemployment", money(totals["J"]))
603 emit!(payroll, 2, "retirement-room",
604 money(max(0, payroll.settings["RETIRECAP"] - totals["retirement"])),
605 "pending-bonus", money(employee.bonus))
606
607 elseif command == "JOURNAL"
608 period = parse(Int, a[1])
609 1 <= period <= payroll.closed || fail("no run " * a[1])
610 journal!(payroll, period)
611
612 elseif command == "REPORT"
613 departments = Dict{String, Vector{Dict{String, Int64}}}()
614 all_stubs = Dict{String, Int64}[]
615 for employee in values(payroll.employees)
616 for (stub, department) in values(employee.stubs)
617 push!(get!(departments, department, Dict{String, Int64}[]), stub)
618 push!(all_stubs, stub)
619 end
620 end
621 emit!(payroll, 0, "REPORT", "period", payroll.closed)
622 for department in sort(collect(keys(departments)))
623 report_line!(payroll, department, departments[department])
624 end
625 report_line!(payroll, "total", all_stubs)
626
627 elseif command == "TOP"
628 count = parse(Int, a[1])
629 ranked = sort([employee for employee in values(payroll.employees)
630 if employee.periods > 0];
631 by=employee -> (-employee.ytd["gross"], employee.id))
632 emit!(payroll, 0, "TOP", count)
633 for (position, employee) in enumerate(ranked[1:min(count, length(ranked))])
634 emit!(payroll, 2, position, employee.id, employee.department,
635 "gross", money(employee.ytd["gross"]), "net", money(employee.ytd["net"]))
636 end
637 isempty(ranked) && emit!(payroll, 2, "(none)")
638
639 elseif command == "EMPLOYEES"
640 emit!(payroll, 0, "EMPLOYEES")
641 for id in sort(collect(keys(payroll.employees)))
642 employee = payroll.employees[id]
643 emit!(payroll, 2, id, employee.department, employee.kind, money(employee.base),
644 employee.active ? "active" : "terminated",
645 "retirement", rate_text(employee.retirement),
646 "health", money(employee.health), "allowances", employee.allowances,
647 "extra", money(employee.extra), "bonus", money(employee.bonus))
648 for (position, order) in enumerate(employee.garnishments)
649 emit!(payroll, 4, "garnish", position, rate_text(order.rate),
650 "balance", money(order.balance))
651 end
652 end
653 isempty(payroll.employees) && emit!(payroll, 2, "(none)")
654
655 elseif command == "CONFIG"
656 emit!(payroll, 0, "CONFIG")
657 for name in SETTING_ORDER
658 value = payroll.settings[name]
659 emit!(payroll, 2, name, name in RATE_SETTINGS ? rate_text(value) : money(value))
660 end
661 for (label, table) in (("bracket", payroll.brackets), ("tier", payroll.tiers))
662 for lower in sort(collect(keys(table)))
663 emit!(payroll, 2, label, money(lower), rate_text(table[lower]))
664 end
665 end
666 end
667end

669function execute!(payroll::Payroll, line::String)
670 words = String.(split(line))
671 command = first(words)
672 try
673 command!(payroll, command, words[2:end])
674 catch error
675 if error isa CommandError
676 emit!(payroll, 0, "ERROR", error.message)
677 else
678 rethrow()
679 end
680 end
681end

683function main()
684 count = parse(Int, readline(stdin))
685 payroll = Payroll()
686 for _ in 1:count
687 execute!(payroll, readline(stdin))
688 end
689 println(join(payroll.output, "\n"))
690end
