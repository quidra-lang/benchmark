23private fun cents(text: String): Long {
24 val parts = text.split('.')
25 return parts[0].toLong() * 100 + parts[1].toLong()
26}

28private fun money(value: Long): String =
29 (value / 100).toString() + "." + (value % 100).toString().padStart(2, '0')

31private fun rateText(value: Long) = money(value) + "%"
32private fun duration(value: Long) = (value / 60).toString() + ":" +
33 (value % 60).toString().padStart(2, '0')

35private fun clock(value: Int) = (value / 60).toString().padStart(2, '0') + ":" +
36 (value % 60).toString().padStart(2, '0')

38private fun weekday(day: Int) = weekdays[(day - 1) % 7]
39private fun periodOf(day: Int) = (day - 1) / 14 + 1
40private fun emptyTotals(): MutableMap<String, Long> =

43private fun progressive(table: Map<Long, Long>, amount: Long): Long {
44 val entries = table.entries.sortedBy { it.key }
45 var numerator = 0L
46 for ((index, entry) in entries.withIndex()) {
47 if (entry.key >= amount) break
48 val upper = if (index + 1 < entries.size) entries[index + 1].key else amount
49 numerator += entry.value * (minOf(amount, upper) - entry.key)
50 }
51 return numerator / 10000
52}

160 private fun daily(person: Employee, day: Int): Daily {
161 val shifts = person.shifts[day].orEmpty()
162 val worked = shifts.sumOf { (it.end - it.start).toLong() }
163 val night = shifts.sumOf {
164 maxOf(0, minOf(it.end, 360) - it.start).toLong() +
165 maxOf(0, it.end - maxOf(it.start, 1320))
166 }
167 val pto = person.pto[day] ?: 0L
168 if (day % 7 == 0 || day in holidays)
169 return Daily(worked, 0, 0, worked, night, pto)
170 return Daily(
171 worked, minOf(worked, 480L), minOf(maxOf(worked - 480L, 0L), 240L),
172 maxOf(worked - 720L, 0L), night, pto
173 )
174 }

176 private fun weekly(person: Employee, week: Int): Weekly {
177 val totals = MutableList(5) { 0L }
178 for (day in (7 * week - 6)..(7 * week)) {
179 val values = daily(person, day).let {
180 listOf(it.regular, it.overtime, it.double, it.night, it.pto)
181 }
182 for (i in totals.indices) totals[i] += values[i]
183 }
184 val excess = maxOf(0L, totals[0] - 2400)
185 totals[0] -= excess
186 totals[1] += excess
187 return Weekly(totals, excess)
188 }

190 private fun breakdown(person: Employee, period: Int): List<Long> {
191 val left = weekly(person, 2 * period - 1).values
192 val right = weekly(person, 2 * period).values
193 return left.indices.map { left[it] + right[it] }
194 }

196 private fun calculate(person: Employee, period: Int): Calculation? {
197 val stub = Stub()
198 val ytd = person.ytd
199 when (person.kind) {
200 "HOURLY" -> {
201 val values = breakdown(person, period)
202 for ((index, field) in listOf("R", "O", "D", "N", "P").withIndex())
203 stub[field] = values[index]
204 val weighted = 2 * (stub["R"] + stub["P"]) + 3 * stub["O"] + 4 * stub["D"]
205 stub["earnings"] = (weighted * person.base +
206 2 * stub["N"] * settings.getValue("NIGHT")) / 120
207 }
208 "SALARIED" -> if (person.active) {
209 stub["salary"] = person.base / 26
210 stub["leave"] = person.leave.count {
211 periodOf(it) == period && it !in holidays
212 }.toLong()
213 stub["deduction"] = stub["leave"] * (person.base / 260)
214 stub["earnings"] = stub["salary"] - stub["deduction"]
215 }
216 "COMMISSION" -> {
217 stub["sales"] = person.sales[period] ?: 0L
218 stub["commission"] = progressive(tiers, stub["sales"])
219 stub["earnings"] = stub["commission"] +
220 (if (person.active) person.base else 0L)
221 }
222 }
223 stub["bonus"] = person.bonus
224 val gross = stub["earnings"] + stub["bonus"]
225 if (gross == 0L) return null
226 stub["gross"] = gross
227 var remaining = gross
228
229 fun deduct(field: String, amount: Long) {
230 stub[field] = minOf(amount, remaining)
231 remaining -= stub[field]
232 }
233
234 val room = maxOf(0L, settings.getValue("RETIRECAP") - ytd.getValue("retirement"))
235 deduct("retirement", minOf(gross * person.retirement / 10000, room))
236 deduct("health", person.health)
237 stub["W"] = remaining
238 val baseTaxable = maxOf(0L, stub["earnings"] - stub["retirement"] - stub["health"])
239 val allowanceTaxable = maxOf(
240 0L, baseTaxable - person.allowances * settings.getValue("ALLOWANCE")
241 )
242 deduct("federal", progressive(brackets, allowanceTaxable) + person.extra)
243 deduct("supplemental", (stub["W"] - baseTaxable) * settings.getValue("SUPP") / 10000)
244 deduct("state", stub["W"] * settings.getValue("STATE") / 10000)
245 val medicareWages = gross - stub["health"]
246 val socialWages = minOf(
247 medicareWages, maxOf(0L, settings.getValue("SSBASE") - ytd.getValue("Y"))
248 )
249 val additional = maxOf(
250 0L, minOf(medicareWages, ytd.getValue("F") + medicareWages - settings.getValue("ADDBASE"))
251 )
252 stub["F"] = medicareWages
253 stub["Y"] = socialWages
254 deduct("social", socialWages * settings.getValue("SSRATE") / 10000)
255 deduct(
256 "medicare", medicareWages * settings.getValue("MEDRATE") / 10000 +
257 additional * settings.getValue("ADDMED") / 10000
258 )
259
260 val disposable = remaining
261 val cap = disposable * settings.getValue("GARNCAP") / 10000
262 var garnishment = 0L
263 val balances = mutableListOf<Long>()
264 for (order in person.garnishments) {
265 val taken = minOf(disposable * order.rate / 10000, order.balance, cap - garnishment)
266 garnishment += taken
267 balances.add(order.balance - taken)
268 }
269 stub["garnishment"] = garnishment
270 stub.garnishmentBalance = balances.sum()
271 stub["net"] = remaining - garnishment
272 stub["employer_social"] = socialWages * settings.getValue("SSRATE") / 10000
273 stub["employer_medicare"] = medicareWages * settings.getValue("MEDRATE") / 10000
274 stub["J"] = minOf(medicareWages, maxOf(0L, settings.getValue("UIBASE") - ytd.getValue("J")))
275 stub["unemployment"] = stub["J"] * settings.getValue("UIRATE") / 10000
276 val eligibleMatch = minOf(stub["retirement"], gross * settings.getValue("MATCHMAX") / 10000)
277 stub["match"] = eligibleMatch * settings.getValue("MATCH") / 10000
278 stub["employer"] = listOf(
279 "employer_social", "employer_medicare", "unemployment", "match"
280 ).sumOf { stub[it] }
281 if (person.kind == "HOURLY") {
282 stub["accrued"] = minOf(
283 960L - person.ptoBalance, (stub["R"] + stub["O"] + stub["D"]) / 20
284 )
285 }
286 stub.ptoBalance = person.ptoBalance + stub["accrued"]
287 return Calculation(stub, balances)
288 }

294 private fun stubBlock(person: Employee, stub: Stub, indent: Int) {
295 when (person.kind) {
296 "HOURLY" -> {
297 emit("hours", hoursParts(listOf("R", "O", "D", "N", "P").map { stub[it] }), indent = indent)
298 emit("pto", "accrued", duration(stub["accrued"]),
299 "balance", duration(stub.ptoBalance), indent = indent)
300 }
301 "SALARIED" -> emit(
302 "salary", money(stub["salary"]), "leave", stub["leave"],
303 "deduction", money(stub["deduction"]), indent = indent
304 )
305 else -> emit(
306 "sales", money(stub["sales"]), "commission", money(stub["commission"]), indent = indent
307 )
308 }
309 for (field in listOf(
310 "earnings", "bonus", "gross", "retirement", "health", "federal",
311 "supplemental", "state", "social", "medicare"
312 )) {
313 emit(if (field == "social") "social-security" else field, money(stub[field]), indent = indent)
314 }
315 emit("garnishment", money(stub["garnishment"]),
316 "balance", money(stub.garnishmentBalance), indent = indent)
317 emit("net", money(stub["net"]), indent = indent)
318 emit(
319 "employer", "social-security", money(stub["employer_social"]),
320 "medicare", money(stub["employer_medicare"]),
321 "unemployment", money(stub["unemployment"]),
322 "match", money(stub["match"]), indent = indent
323 )

331 private fun showHours(person: Employee, period: Int) {
332 emit("HOURS", person.id, "period", period)
333 var found = false
334 for (day in (14 * period - 13)..(14 * period)) {
335 val shifts = person.shifts[day].orEmpty().sortedBy { it.start }
336 if (shifts.isEmpty() && day !in person.pto) continue
337 val daily = daily(person, day)
338 val parts = mutableListOf("day", day.toString(), weekday(day))
339 if (day in holidays) parts.add("holiday")
340 for (shift in shifts)
341 parts.add(clock(shift.start) + "-" + clock(shift.end))
342 val amounts = listOf(daily.regular, daily.overtime, daily.double, daily.night, daily.pto)
343 emit(parts.joinToString(" "), "total", duration(daily.worked),
344 hoursParts(amounts), indent = 2)
345 found = true
346 }
347 if (!found) emit("(none)", indent = 2)
348 for (week in listOf(2 * period - 1, 2 * period)) {
349 val values = weekly(person, week)
350 emit("week", week, hoursParts(values.values),
351 "weekly-ot", duration(values.excess), indent = 2)
352 }
353 emit("total", hoursParts(breakdown(person, period)), indent = 2)
354 }

356 private fun run(period: Int) {
357 closed = period
358 emit("RUN", period, "days", (14 * period - 13).toString() + "-" + 14 * period)
359 var paid = 0
360 var skipped = 0
361 val stubs = mutableListOf<Stub>()
362 for (person in employees.values) {
363 val result = calculate(person, period)
364 if (result == null) {
365 if (person.active) skipped++
366 continue
367 }
368 val stub = result.stub
369 printStub(person, period, stub, person.department)
370 person.bonus = 0L
371 person.ptoBalance = stub.ptoBalance
372 for ((index, balance) in result.balances.withIndex())
373 person.garnishments[index].balance = balance
374 for (field in fields)
375 person.ytd[field] = person.ytd.getValue(field) + stub[field]
376 person.periods++
377 person.stubs[period] = StoredStub(stub, person.department)
378 stubs.add(stub)
379 paid++
380 }
381 val total = sumStubs(stubs)
382 emit(
383 "TOTAL", "paid", paid, "skipped", skipped, "gross", money(total["gross"]),
384 "deductions", money(total["gross"] - total["net"]), "net", money(total["net"]),
385 "employer", money(total["employer"]), indent = 2
386 )
387 }

389 private fun journal(period: Int) {
390 val stubs = employees.values.mapNotNull { it.stubs[period]?.stub }
391 val total = sumStubs(stubs)
392 val rows = listOf(
393 Triple("debit", "wages", total["gross"]),
394 Triple("debit", "employer-tax",
395 total["employer_social"] + total["employer_medicare"] + total["unemployment"]),
396 Triple("debit", "retirement-match", total["match"]),
397 Triple("credit", "retirement", total["retirement"] + total["match"]),
398 Triple("credit", "health", total["health"]),
399 Triple("credit", "federal", total["federal"] + total["supplemental"]),
400 Triple("credit", "state", total["state"]),
401 Triple("credit", "fica", total["social"] + total["medicare"] +
402 total["employer_social"] + total["employer_medicare"]),
403 Triple("credit", "unemployment", total["unemployment"]),
404 Triple("credit", "garnishment", total["garnishment"]),
405 Triple("credit", "net-pay", total["net"])
406 )
407 emit("JOURNAL", period, "stubs", stubs.size)
408 for ((side, label, amount) in rows) emit(side, label, money(amount), indent = 2)
409 val debits = rows.filter { it.first == "debit" }.sumOf { it.third }
410 val credits = rows.filter { it.first == "credit" }.sumOf { it.third }
411 emit("total", "debit", money(debits), "credit", money(credits), indent = 2)
412 }

426 private fun command(name: String, a: List<String>) {
427 when (name) {
428 "SET" -> {
429 val setting = a[0]
430 if (setting !in settings) fail("unknown setting " + setting)
431 val value = if (setting in rateSettings) rateCheck(a[1]) else cents(a[1])
432 val old = settings.getValue(setting)
433 settings[setting] = value
434 val format: (Long) -> String =
435 if (setting in rateSettings) ::rateText else ::money
436 emit("SET", setting, format(old), "->", format(value))
437 }
438 "BRACKET", "TIER" -> {
439 val rate = rateCheck(a[1])
440 val table = if (name == "BRACKET") brackets else tiers
441 val lower = cents(a[0])
442 val action = if (lower in table) "replaced" else "added"
443 table[lower] = rate
444 emit(name, money(lower), rateText(rate), action)
445 }
446 "EMP" -> {
447 val id = a[0].toInt()
448 if (id in employees) fail("duplicate employee " + a[0])
449 val base = amountCheck(a[3])
450 employees[id] = Employee(id, a[1], a[2], base)
451 emit("EMP", id, a[1], a[2], money(base))
452 }
453 "RATE", "DEPT", "BENEFIT", "W4", "GARNISH", "TERM", "BONUS" -> {
454 val person = employee(a[0], active = true)
455 when (name) {
456 "RATE" -> {
457 val base = amountCheck(a[1])
458 val old = person.base
459 person.base = base
460 emit("RATE", person.id, money(old), "->", money(base))
461 }
462 "DEPT" -> {
463 if (person.department == a[1]) fail("same department " + a[1])
464 val old = person.department
465 person.department = a[1]
466 emit("DEPT", person.id, old, "->", person.department)
467 }
468 "BENEFIT" -> {
469 val rate = rateCheck(a[1])
470 person.retirement = rate
471 person.health = cents(a[2])
472 emit("BENEFIT", person.id, "retirement", rateText(rate),
473 "health", money(person.health))
474 }
475 "W4" -> {
476 person.allowances = a[1].toLong()
477 person.extra = cents(a[2])
478 emit("W4", person.id, "allowances", person.allowances,
479 "extra", money(person.extra))
480 }
481 "GARNISH" -> {
482 val rate = rateCheck(a[1])
483 val amount = amountCheck(a[2])
484 person.garnishments.add(Garnishment(rate, amount))
485 emit("GARNISH", person.id, "order", person.garnishments.size,
486 rateText(rate), "balance", money(amount))
487 }
488 "TERM" -> {
489 person.active = false
490 emit("TERM", person.id)
491 }
492 "BONUS" -> {
493 val amount = amountCheck(a[1])
494 person.bonus += amount
495 emit("BONUS", person.id, money(amount),
496 "pending", money(person.bonus))
497 }
498 }
499 }
500 "HOLIDAY" -> {
501 val day = dayCheck(a[0])
502 if (day in holidays) fail("duplicate holiday " + a[0])
503 holidays.add(day)
504 emit("HOLIDAY", day, weekday(day))
505 }
506 "TIME", "UNTIME", "PTO", "LEAVE", "SALES" -> {
507 val kind = when (name) {
508 "LEAVE" -> "SALARIED"
509 "SALES" -> "COMMISSION"
510 else -> "HOURLY"
511 }
512 val person = employee(a[0], active = true, kind = kind)
513 val day = dayCheck(a[1])
514 when (name) {
515 "TIME" -> {
516 val start = timeCheck(a[2])
517 val end = timeCheck(a[3])
518 if (start >= end)
519 fail("invalid shift " + a[2] + "-" + a[3])
520 val shifts = person.shifts[day].orEmpty()
521 for (shift in shifts.sortedBy { it.start }) {
522 if (start < shift.end && shift.start < end)
523 fail("overlapping shift " + clock(shift.start) +
524 "-" + clock(shift.end))
525 }
526 if (shifts.sumOf { it.end - it.start } + end - start > 960)
527 fail("daily limit " + a[0] + " " + a[1])
528 person.shifts.getOrPut(day) { mutableListOf() }.add(Shift(start, end))
529 emit("TIME", person.id, "day", day,
530 clock(start) + "-" + clock(end), duration((end - start).toLong()))
531 }
532 "UNTIME" -> {
533 val start = timeCheck(a[2])
534 val shifts = person.shifts[day].orEmpty()
535 val found = shifts.find { it.start == start }
536 ?: fail("no shift " + a[0] + " " + a[1] + " " + a[2])
537 person.shifts[day]?.remove(found)
538 emit("UNTIME", person.id, "day", day,
539 clock(found.start) + "-" + clock(found.end))
540 }
541 "PTO" -> {
542 val pieces = a[2].split(':')
543 val hour = pieces[0].toLong()
544 val minute = pieces[1].toLong()
545 val length = hour * 60 + minute
546 if (minute > 59 || length !in 1L..480L)
547 fail("invalid duration " + a[2])
548 if (day in person.pto) fail("duplicate pto " + a[0] + " " + a[1])
549 if (length > person.ptoBalance)
550 fail("insufficient pto " + a[0] + " " + duration(person.ptoBalance))
551 person.pto[day] = length
552 person.ptoBalance -= length
553 emit("PTO", person.id, "day", day, duration(length),
554 "balance", duration(person.ptoBalance))
555 }
556 "LEAVE" -> {
557 if (day % 7 == 0 || day % 7 == 6 || day in holidays)
558 fail("not workday " + a[1])
559 if (day in person.leave)
560 fail("duplicate leave " + a[0] + " " + a[1])
561 person.leave.add(day)
562 emit("LEAVE", person.id, "day", day, weekday(day))
563 }
564 "SALES" -> {
565 val amount = amountCheck(a[2])
566 val period = periodOf(day)
567 person.sales[period] = (person.sales[period] ?: 0L) + amount
568 emit("SALES", person.id, "day", day, money(amount),
569 "period", period, "total", money(person.sales.getValue(period)))
570 }
571 }
572 }
573 "HOURS" -> {
574 val person = employee(a[0], kind = "HOURLY")
575 showHours(person, periodCheck(a[1]))
576 }
577 "RUN" -> {
578 val period = a[0].toInt()
579 if (period != closed + 1 || period > 26)
580 fail("invalid period " + a[0])
581 run(period)
582 }
583 "PREVIEW" -> {
584 val person = employee(a[0])
585 val period = closed + 1
586 val result = calculate(person, period)
587 emit("PREVIEW", person.id, "period", period)
588 if (result == null) emit("no pay", indent = 2)
589 else printStub(person, period, result.stub, person.department)
590 }
591 "REPRINT" -> {
592 val person = employee(a[0])
593 val period = periodCheck(a[1])
594 val record = person.stubs[period] ?: fail("no stub " + a[0] + " " + a[1])
595 emit("REPRINT", person.id, period)
596 printStub(person, period, record.stub, record.department)
597 }
598 "YTD" -> {
599 val person = employee(a[0])
600 emit("YTD", person.id, person.department, person.kind,
601 if (person.active) "active" else "terminated",
602 "periods", person.periods)
603 val totals = Stub()
604 for (field in fields) totals[field] = person.ytd.getValue(field)
605 totals.ptoBalance = person.ptoBalance
606 totals.garnishmentBalance = person.garnishBalance()
607 stubBlock(person, totals, 2)
608 emit("wages", "federal", money(totals["W"]),
609 "social-security", money(totals["Y"]),
610 "medicare", money(totals["F"]),
611 "unemployment", money(totals["J"]), indent = 2)
612 emit("retirement-room",
613 money(maxOf(0L, settings.getValue("RETIRECAP") - totals["retirement"])),
614 "pending-bonus", money(person.bonus), indent = 2)
615 }
616 "JOURNAL" -> {
617 val period = a[0].toInt()
618 if (period !in 1..closed) fail("no run " + a[0])
619 journal(period)
620 }
621 "REPORT" -> {
622 val departments = sortedMapOf<String, MutableList<Stub>>()
623 val all = mutableListOf<Stub>()
624 for (person in employees.values) {
625 for (record in person.stubs.values) {
626 departments.getOrPut(record.department) { mutableListOf() }.add(record.stub)
627 all.add(record.stub)
628 }
629 }
630 emit("REPORT", "period", closed)
631 for ((department, stubs) in departments) reportLine(department, stubs)
632 reportLine("total", all)
633 }
634 "TOP" -> {
635 val count = a[0].toInt()
636 val ranked = employees.values.filter { it.periods > 0 }
637 .sortedWith(compareByDescending<Employee> { it.ytd.getValue("gross") }.thenBy { it.id })
638 emit("TOP", count)
639 for ((index, person) in ranked.take(count).withIndex())
640 emit(index + 1, person.id, person.department,
641 "gross", money(person.ytd.getValue("gross")),
642 "net", money(person.ytd.getValue("net")), indent = 2)
643 if (ranked.isEmpty()) emit("(none)", indent = 2)
644 }
645 "EMPLOYEES" -> {
646 emit("EMPLOYEES")
647 for (person in employees.values) {
648 emit(person.id, person.department, person.kind, money(person.base),
649 if (person.active) "active" else "terminated",
650 "retirement", rateText(person.retirement),
651 "health", money(person.health),
652 "allowances", person.allowances, "extra", money(person.extra),
653 "bonus", money(person.bonus), indent = 2)
654 for ((index, order) in person.garnishments.withIndex())
655 emit("garnish", index + 1, rateText(order.rate),
656 "balance", money(order.balance), indent = 4)
657 }
658 if (employees.isEmpty()) emit("(none)", indent = 2)
659 }
660 "CONFIG" -> {
661 emit("CONFIG")
662 for ((key, value) in settings)
663 emit(key, if (key in rateSettings) rateText(value) else money(value), indent = 2)
664 for ((label, table) in listOf("bracket" to brackets, "tier" to tiers))
665 for ((lower, rate) in table)
666 emit(label, money(lower), rateText(rate), indent = 2)
667 }
668 }
669 }

671 fun execute(line: String) {
672 val tokens = line.split(' ')
673 try {
674 command(tokens[0], tokens.drop(1))
675 } catch (error: CommandError) {
676 emit("ERROR", error.message ?: "")
677 }
678 }
