58progressive <- function(table, amount) {
59 limits <- sorted_ids(table)
60 numerator <- 0
61 for (i in seq_along(limits)) {
62 lower <- limits[i]
63 if (lower >= amount) break
64 upper <- if (i < length(limits)) limits[i + 1] else amount
65 numerator <- numerator + get(as.character(lower), envir = table) *
66 (min(amount, upper) - lower)
67 }
68 floor(numerator / 10000)
69}

162daily <- function(payroll, employee, day) {
163 shifts <- employee$shifts[[as.character(day)]]
164 if (is.null(shifts)) shifts <- list()
165 worked <- 0
166 night <- 0
167 for (shift in shifts) {
168 first <- shift[1]
169 last <- shift[2]
170 worked <- worked + last - first
171 night <- night + max(0, min(last, 360) - first) +
172 max(0, last - max(first, 1320))
173 }
174 pto <- employee$pto[[as.character(day)]]
175 if (is.null(pto)) pto <- 0
176 if (day %% 7 == 0 || day %in% payroll$holidays) {
177 regular <- 0
178 overtime <- 0
179 double <- worked
180 } else {
181 regular <- min(worked, 480)
182 overtime <- min(max(worked - 480, 0), 240)
183 double <- max(worked - 720, 0)
184 }
185 list(worked = worked, values = c(regular, overtime, double, night, pto))
186}

188weekly <- function(payroll, employee, week) {
189 totals <- rep(0, 5)
190 for (day in (7 * week - 6):(7 * week)) {
191 totals <- totals + daily(payroll, employee, day)$values
192 }
193 weekly_ot <- max(0, totals[1] - 2400)
194 totals[1] <- totals[1] - weekly_ot
195 totals[2] <- totals[2] + weekly_ot
196 list(values = totals, overtime = weekly_ot)
197}

204calculate <- function(payroll, employee, period) {
205 stub <- empty_totals()
206 settings <- payroll$settings
207 ytd <- employee$ytd
208 if (employee$kind == "HOURLY") {
209 values <- breakdown(payroll, employee, period)
210 for (i in seq_along(c("R", "O", "D", "N", "P"))) {
211 field <- c("R", "O", "D", "N", "P")[i]
212 stub[[field]] <- values[i]
213 }
214 weighted <- 2 * (stub$R + stub$P) + 3 * stub$O + 4 * stub$D
215 stub$earnings <- floor((weighted * employee$base +
216 2 * stub$N * settings$NIGHT) / 120)
217 } else if (employee$kind == "SALARIED") {
218 if (employee$active) {
219 stub$salary <- floor(employee$base / 26)
220 stub$leave <- sum(vapply(employee$leave, function(day)
221 period_of(day) == period && !(day %in% payroll$holidays), logical(1)))
222 stub$deduction <- stub$leave * floor(employee$base / 260)
223 stub$earnings <- stub$salary - stub$deduction
224 }
225 } else {
226 sales <- employee$sales[[as.character(period)]]
227 if (is.null(sales)) sales <- 0
228 stub$sales <- sales
229 stub$commission <- progressive(payroll$tiers, stub$sales)
230 stub$earnings <- stub$commission + (if (employee$active) employee$base else 0)
231 }
232
233 stub$bonus <- employee$bonus
234 gross <- stub$earnings + stub$bonus
235 if (gross == 0) return(NULL)
236 stub$gross <- gross
237 remaining <- gross
238
239 deduct <- function(field, amount) {
240 taken <- min(amount, remaining)
241 stub[[field]] <<- taken
242 remaining <<- remaining - taken
243 }
244
245 retirement_room <- max(0, settings$RETIRECAP - ytd$retirement)
246 deduct("retirement", min(floor(gross * employee$retirement / 10000), retirement_room))
247 deduct("health", employee$health)
248 stub$W <- remaining
249 base_taxable <- max(0, stub$earnings - stub$retirement - stub$health)
250 taxable <- max(0, base_taxable - employee$allowances * settings$ALLOWANCE)
251 deduct("federal", progressive(payroll$brackets, taxable) + employee$extra)
252 deduct("supplemental", floor((stub$W - base_taxable) * settings$SUPP / 10000))
253 deduct("state", floor(stub$W * settings$STATE / 10000))
254
255 medicare_wages <- gross - stub$health
256 social_wages <- min(medicare_wages, max(0, settings$SSBASE - ytd$Y))
257 additional <- max(0, min(medicare_wages,
258 ytd$F + medicare_wages - settings$ADDBASE))
259 stub$F <- medicare_wages
260 stub$Y <- social_wages
261 deduct("social", floor(social_wages * settings$SSRATE / 10000))
262 deduct("medicare", floor(medicare_wages * settings$MEDRATE / 10000) +
263 floor(additional * settings$ADDMED / 10000))
264
265 disposable <- remaining
266 cap <- floor(disposable * settings$GARNCAP / 10000)
267 balances <- numeric()
268 garnishment <- 0
269 for (order in employee$garnishments) {
270 taken <- min(floor(disposable * order[1] / 10000),
271 order[2], cap - garnishment)
272 garnishment <- garnishment + taken
273 balances <- c(balances, order[2] - taken)
274 }
275 stub$garnishment <- garnishment
276 stub$garnishment_balance <- sum(balances)
277 stub$net <- remaining - garnishment
278 stub$employer_social <- floor(social_wages * settings$SSRATE / 10000)
279 stub$employer_medicare <- floor(medicare_wages * settings$MEDRATE / 10000)
280 stub$J <- min(medicare_wages, max(0, settings$UIBASE - ytd$J))
281 stub$unemployment <- floor(stub$J * settings$UIRATE / 10000)
282 eligible_match <- min(stub$retirement,
283 floor(gross * settings$MATCHMAX / 10000))
284 stub$match <- floor(eligible_match * settings$MATCH / 10000)
285 stub$employer <- stub$employer_social + stub$employer_medicare +
286 stub$unemployment + stub$match
287 if (employee$kind == "HOURLY") {
288 stub$accrued <- min(960 - employee$pto_balance,
289 floor((stub$R + stub$O + stub$D) / 20))
290 }
291 stub$pto_balance <- employee$pto_balance + stub$accrued
292 list(stub = stub, balances = balances)
293}

300stub_block <- function(payroll, employee, stub, indent) {
301 if (employee$kind == "HOURLY") {
302 values <- vapply(c("R", "O", "D", "N", "P"),
303 function(field) stub[[field]], numeric(1))

305 emit(payroll, indent, "pto", "accrued", duration(stub$accrued),

307 } else if (employee$kind == "SALARIED") {
308 emit(payroll, indent, "salary", money(stub$salary), "leave", stub$leave,
309 "deduction", money(stub$deduction))
310 } else {
311 emit(payroll, indent, "sales", money(stub$sales),
312 "commission", money(stub$commission))
313 }
314 for (field in c(
315 "earnings", "bonus", "gross", "retirement", "health",
316 "federal", "supplemental", "state", "social", "medicare"
317 )) {
318 label <- if (field == "social") "social-security" else field
319 emit(payroll, indent, label, money(stub[[field]]))
320 }
321 emit(payroll, indent, "garnishment", money(stub$garnishment),
322 "balance", money(stub$garnishment_balance))
323 emit(payroll, indent, "net", money(stub$net))
324 emit(payroll, indent, "employer", "social-security", money(stub$employer_social),
325 "medicare", money(stub$employer_medicare),
326 "unemployment", money(stub$unemployment), "match", money(stub$match))
327}

334show_hours <- function(payroll, employee, period) {
335 emit(payroll, 0, "HOURS", employee$id, "period", period)
336 found <- FALSE
337 for (day in (14 * period - 13):(14 * period)) {
338 shifts <- employee$shifts[[as.character(day)]]
339 if (is.null(shifts)) shifts <- list()
340 pto <- employee$pto[[as.character(day)]]
341 if (length(shifts) == 0 && is.null(pto)) next
342 if (length(shifts)) {
343 shifts <- shifts[order(vapply(shifts, function(x) x[1], numeric(1)))]
344 }
345 result <- daily(payroll, employee, day)
346 parts <- c("day", as.character(day), weekday(day))
347 if (day %in% payroll$holidays) parts <- c(parts, "holiday")
348 for (shift in shifts) {
349 parts <- c(parts, paste0(clock(shift[1]), "-", clock(shift[2])))
350 }
351 parts <- c(parts, "total", duration(result$worked),
352 hours_parts(result$values))
353 emit(payroll, 2, parts)
354 found <- TRUE
355 }
356 if (!found) emit(payroll, 2, "(none)")
357 for (week in c(2 * period - 1, 2 * period)) {
358 result <- weekly(payroll, employee, week)
359 emit(payroll, 2, "week", week, hours_parts(result$values),
360 "weekly-ot", duration(result$overtime))
361 }
362 emit(payroll, 2, "total", hours_parts(breakdown(payroll, employee, period)))
363}

365run <- function(payroll, period) {
366 payroll$closed <- period
367 emit(payroll, 0, "RUN", period, "days",
368 paste0(14 * period - 13, "-", 14 * period))
369 paid <- 0
370 skipped <- 0
371 stubs <- list()
372 for (id in sorted_ids(payroll$employees)) {
373 employee <- get(as.character(id), envir = payroll$employees)
374 calculated <- calculate(payroll, employee, period)
375 if (is.null(calculated)) {
376 if (employee$active) skipped <- skipped + 1
377 next
378 }
379 stub <- calculated$stub
380 print_stub(payroll, employee, period, stub, employee$department)
381 employee$bonus <- 0
382 employee$pto_balance <- stub$pto_balance
383 for (i in seq_along(calculated$balances)) {
384 employee$garnishments[[i]][2] <- calculated$balances[i]
385 }
386 for (field in fields) {
387 employee$ytd[[field]] <- employee$ytd[[field]] + stub[[field]]
388 }
389 employee$periods <- employee$periods + 1
390 employee$stubs[[as.character(period)]] <- list(
391 stub = stub, department = employee$department
392 )
393 stubs[[length(stubs) + 1]] <- stub
394 paid <- paid + 1
395 }
396 totals <- sum_stubs(stubs)
397 emit(payroll, 2, "TOTAL", "paid", paid, "skipped", skipped,
398 "gross", money(totals$gross),
399 "deductions", money(totals$gross - totals$net),
400 "net", money(totals$net), "employer", money(totals$employer))
401}

403journal <- function(payroll, period) {
404 stubs <- list()
405 for (id in sorted_ids(payroll$employees)) {
406 employee <- get(as.character(id), envir = payroll$employees)
407 entry <- employee$stubs[[as.character(period)]]

411 rows <- list(
412 c("debit", "wages", total$gross),
413 c("debit", "employer-tax", total$employer_social + total$employer_medicare + total$unemployment),
414 c("debit", "retirement-match", total$match),
415 c("credit", "retirement", total$retirement + total$match),
416 c("credit", "health", total$health),
417 c("credit", "federal", total$federal + total$supplemental),
418 c("credit", "state", total$state),
419 c("credit", "fica", total$social + total$medicare + total$employer_social + total$employer_medicare),
420 c("credit", "unemployment", total$unemployment),
421 c("credit", "garnishment", total$garnishment),
422 c("credit", "net-pay", total$net)

450command <- function(payroll, operation, a) {
451 if (operation == "SET") {
452 name <- a[1]
453 if (is.null(payroll$settings[[name]])) fail(paste("unknown setting", name))
454 value <- if (name %in% rate_settings) rate_check(a[2]) else cents(a[2])
455 format <- if (name %in% rate_settings) rate_text else money
456 previous <- payroll$settings[[name]]
457 payroll$settings[[name]] <- value
458 emit(payroll, 0, "SET", name, format(previous), "->", format(value))
459 } else if (operation %in% c("BRACKET", "TIER")) {
460 rate <- rate_check(a[2])
461 table <- if (operation == "BRACKET") payroll$brackets else payroll$tiers
462 lower <- cents(a[1])
463 key <- as.character(lower)
464 action <- if (exists(key, envir = table, inherits = FALSE)) "replaced" else "added"
465 assign(key, rate, envir = table)
466 emit(payroll, 0, operation, money(lower), rate_text(rate), action)
467 } else if (operation == "EMP") {
468 id <- number(a[1])
469 key <- as.character(id)
470 if (exists(key, envir = payroll$employees, inherits = FALSE))
471 fail(paste("duplicate employee", a[1]))
472 base <- amount_check(a[4])
473 assign(key, new_employee(id, a[2], a[3], base), envir = payroll$employees)
474 emit(payroll, 0, "EMP", id, a[2], a[3], money(base))
475 } else if (operation %in% c("RATE", "DEPT", "BENEFIT", "W4", "GARNISH", "TERM", "BONUS")) {
476 employee <- employee_of(payroll, a[1], active = TRUE)
477 if (operation == "RATE") {
478 base <- amount_check(a[2])
479 previous <- employee$base
480 employee$base <- base
481 emit(payroll, 0, "RATE", employee$id, money(previous), "->", money(base))
482 } else if (operation == "DEPT") {
483 if (a[2] == employee$department) fail(paste("same department", a[2]))
484 previous <- employee$department
485 employee$department <- a[2]
486 emit(payroll, 0, "DEPT", employee$id, previous, "->", a[2])
487 } else if (operation == "BENEFIT") {
488 rate <- rate_check(a[2])
489 employee$retirement <- rate
490 employee$health <- cents(a[3])
491 emit(payroll, 0, "BENEFIT", employee$id, "retirement",
492 rate_text(rate), "health", money(employee$health))
493 } else if (operation == "W4") {
494 employee$allowances <- number(a[2])
495 employee$extra <- cents(a[3])
496 emit(payroll, 0, "W4", employee$id, "allowances", employee$allowances,
497 "extra", money(employee$extra))
498 } else if (operation == "GARNISH") {
499 rate <- rate_check(a[2])
500 amount <- amount_check(a[3])
501 employee$garnishments[[length(employee$garnishments) + 1]] <- c(rate, amount)
502 emit(payroll, 0, "GARNISH", employee$id, "order", length(employee$garnishments),
503 rate_text(rate), "balance", money(amount))
504 } else if (operation == "TERM") {
505 employee$active <- FALSE
506 emit(payroll, 0, "TERM", employee$id)
507 } else {
508 amount <- amount_check(a[2])
509 employee$bonus <- employee$bonus + amount
510 emit(payroll, 0, "BONUS", employee$id, money(amount),
511 "pending", money(employee$bonus))
512 }
513 } else if (operation == "HOLIDAY") {
514 day <- day_check(payroll, a[1])
515 if (day %in% payroll$holidays) fail(paste("duplicate holiday", a[1]))
516 payroll$holidays <- c(payroll$holidays, day)
517 emit(payroll, 0, "HOLIDAY", day, weekday(day))
518 } else if (operation %in% c("TIME", "UNTIME", "PTO", "LEAVE", "SALES")) {
519 kind <- if (operation == "LEAVE") "SALARIED" else if (operation == "SALES") "COMMISSION" else "HOURLY"
520 employee <- employee_of(payroll, a[1], active = TRUE, kind = kind)
521 day <- day_check(payroll, a[2])
522 key <- as.character(day)
523 if (operation == "TIME") {
524 start_time <- time_check(a[3])
525 end_time <- time_check(a[4])
526 if (start_time >= end_time)
527 fail(paste0("invalid shift ", a[3], "-", a[4]))
528 shifts <- employee$shifts[[key]]
529 if (is.null(shifts)) shifts <- list()
530 if (length(shifts)) {
531 shifts <- shifts[order(vapply(shifts, function(s) s[1], numeric(1)))]
532 }
533 for (shift in shifts) {
534 if (start_time < shift[2] && shift[1] < end_time) {
535 fail(paste("overlapping shift", paste0(clock(shift[1]), "-", clock(shift[2]))))
536 }
537 }
538 worked <- sum(vapply(shifts, function(s) s[2] - s[1], numeric(1)))
539 if (worked + end_time - start_time > 960)
540 fail(paste("daily limit", a[1], a[2]))
541 shifts[[length(shifts) + 1]] <- c(start_time, end_time)
542 employee$shifts[[key]] <- shifts
543 emit(payroll, 0, "TIME", employee$id, "day", day,
544 paste0(clock(start_time), "-", clock(end_time)),
545 duration(end_time - start_time))
546 } else if (operation == "UNTIME") {
547 start_time <- time_check(a[3])
548 shifts <- employee$shifts[[key]]
549 if (is.null(shifts)) shifts <- list()
550 position <- which(vapply(shifts, function(s) s[1] == start_time, logical(1)))
551 if (!length(position)) fail(paste("no shift", a[1], a[2], a[3]))
552 shift <- shifts[[position[1]]]
553 shifts <- shifts[-position[1]]
554 employee$shifts[[key]] <- shifts
555 emit(payroll, 0, "UNTIME", employee$id, "day", day,
556 paste0(clock(shift[1]), "-", clock(shift[2])))
557 } else if (operation == "PTO") {
558 parts <- as.numeric(strsplit(a[3], ":", fixed = TRUE)[[1]])
559 length_minutes <- parts[1] * 60 + parts[2]
560 if (parts[2] > 59 || length_minutes < 1 || length_minutes > 480)
561 fail(paste("invalid duration", a[3]))
562 if (!is.null(employee$pto[[key]])) fail(paste("duplicate pto", a[1], a[2]))
563 if (length_minutes > employee$pto_balance)
564 fail(paste("insufficient pto", a[1], duration(employee$pto_balance)))
565 employee$pto[[key]] <- length_minutes
566 employee$pto_balance <- employee$pto_balance - length_minutes
567 emit(payroll, 0, "PTO", employee$id, "day", day,
568 duration(length_minutes), "balance", duration(employee$pto_balance))
569 } else if (operation == "LEAVE") {
570 if (day %% 7 %in% c(0, 6) || day %in% payroll$holidays)
571 fail(paste("not workday", a[2]))
572 if (day %in% employee$leave) fail(paste("duplicate leave", a[1], a[2]))
573 employee$leave <- c(employee$leave, day)
574 emit(payroll, 0, "LEAVE", employee$id, "day", day, weekday(day))
575 } else {
576 amount <- amount_check(a[3])
577 period <- period_of(day)
578 previous <- employee$sales[[as.character(period)]]
579 if (is.null(previous)) previous <- 0
580 employee$sales[[as.character(period)]] <- previous + amount
581 emit(payroll, 0, "SALES", employee$id, "day", day, money(amount),
582 "period", period, "total", money(previous + amount))
583 }
584 } else if (operation == "HOURS") {
585 employee <- employee_of(payroll, a[1], kind = "HOURLY")
586 show_hours(payroll, employee, period_check(a[2]))
587 } else if (operation == "RUN") {
588 period <- number(a[1])
589 if (period != payroll$closed + 1 || period > 26)
590 fail(paste("invalid period", a[1]))
591 run(payroll, period)
592 } else if (operation == "PREVIEW") {
593 employee <- employee_of(payroll, a[1])
594 period <- payroll$closed + 1
595 calculated <- calculate(payroll, employee, period)
596 emit(payroll, 0, "PREVIEW", employee$id, "period", period)
597 if (is.null(calculated)) {
598 emit(payroll, 2, "no pay")
599 } else {
600 print_stub(payroll, employee, period, calculated$stub, employee$department)
601 }
602 } else if (operation == "REPRINT") {
603 employee <- employee_of(payroll, a[1])
604 period <- period_check(a[2])
605 entry <- employee$stubs[[as.character(period)]]
606 if (is.null(entry)) fail(paste("no stub", a[1], a[2]))
607 emit(payroll, 0, "REPRINT", employee$id, period)
608 print_stub(payroll, employee, period, entry$stub, entry$department)
609 } else if (operation == "YTD") {
610 employee <- employee_of(payroll, a[1])
611 emit(payroll, 0, "YTD", employee$id, employee$department, employee$kind,
612 if (employee$active) "active" else "terminated",
613 "periods", employee$periods)
614 totals <- employee$ytd
615 totals$pto_balance <- employee$pto_balance
616 totals$garnishment_balance <- garnishment_balance(employee)
617 stub_block(payroll, employee, totals, 2)
618 emit(payroll, 2, "wages", "federal", money(totals$W),
619 "social-security", money(totals$Y),
620 "medicare", money(totals$F), "unemployment", money(totals$J))
621 emit(payroll, 2, "retirement-room",
622 money(max(0, payroll$settings$RETIRECAP - totals$retirement)),
623 "pending-bonus", money(employee$bonus))
624 } else if (operation == "JOURNAL") {
625 period <- number(a[1])
626 if (period < 1 || period > payroll$closed) fail(paste("no run", a[1]))
627 journal(payroll, period)
628 } else if (operation == "REPORT") {
629 departments <- list()
630 all_stubs <- list()
631 for (id in sorted_ids(payroll$employees)) {
632 employee <- get(as.character(id), envir = payroll$employees)
633 for (entry in employee$stubs) {
634 department <- entry$department
635 if (is.null(departments[[department]])) departments[[department]] <- list()
636 departments[[department]][[length(departments[[department]]) + 1]] <- entry$stub
637 all_stubs[[length(all_stubs) + 1]] <- entry$stub
638 }
639 }
640 emit(payroll, 0, "REPORT", "period", payroll$closed)
641 for (department in sort(names(departments))) {
642 report_line(payroll, department, departments[[department]])
643 }
644 report_line(payroll, "total", all_stubs)
645 } else if (operation == "TOP") {
646 count <- number(a[1])
647 eligible <- list()
648 for (id in sorted_ids(payroll$employees)) {
649 employee <- get(as.character(id), envir = payroll$employees)
650 if (employee$periods > 0) eligible[[length(eligible) + 1]] <- employee
651 }
652 if (length(eligible)) {
653 order <- order(
654 -vapply(eligible, function(e) e$ytd$gross, numeric(1)),
655 vapply(eligible, function(e) e$id, numeric(1))
656 )
657 eligible <- eligible[order]
658 }
659 emit(payroll, 0, "TOP", count)
660 if (length(eligible)) {
661 for (i in seq_len(min(count, length(eligible)))) {
662 employee <- eligible[[i]]
663 emit(payroll, 2, i, employee$id, employee$department,
664 "gross", money(employee$ytd$gross), "net", money(employee$ytd$net))
665 }
666 } else {
667 emit(payroll, 2, "(none)")
668 }
669 } else if (operation == "EMPLOYEES") {
670 emit(payroll, 0, "EMPLOYEES")
671 ids <- sorted_ids(payroll$employees)
672 for (id in ids) {
673 employee <- get(as.character(id), envir = payroll$employees)
674 emit(payroll, 2, id, employee$department, employee$kind, money(employee$base),
675 if (employee$active) "active" else "terminated",
676 "retirement", rate_text(employee$retirement), "health", money(employee$health),
677 "allowances", employee$allowances, "extra", money(employee$extra),
678 "bonus", money(employee$bonus))
679 for (i in seq_along(employee$garnishments)) {
680 order <- employee$garnishments[[i]]
681 emit(payroll, 4, "garnish", i, rate_text(order[1]), "balance", money(order[2]))
682 }
683 }
684 if (!length(ids)) emit(payroll, 2, "(none)")
685 } else if (operation == "CONFIG") {
686 emit(payroll, 0, "CONFIG")
687 for (name in setting_order) {
688 value <- payroll$settings[[name]]
689 emit(payroll, 2, name,
690 if (name %in% rate_settings) rate_text(value) else money(value))
691 }
692 for (label in c("bracket", "tier")) {
693 table <- if (label == "bracket") payroll$brackets else payroll$tiers
694 for (lower in sorted_ids(table)) {
695 rate <- get(as.character(lower), envir = table)
696 emit(payroll, 2, label, money(lower), rate_text(rate))
697 }
698 }
699 }
700}

707 operation <- words[1]
