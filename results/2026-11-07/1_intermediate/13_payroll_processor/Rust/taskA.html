22fn cents(text: &str) -> i64 {
23 let (whole, fraction) = text.split_once('.').unwrap();
24 whole.parse::<i64>().unwrap() * 100 + fraction.parse::<i64>().unwrap()
25}

63fn sum_stubs<'a>(stubs: impl IntoIterator<Item = &'a Stub>) -> Stub {
64 let mut totals = new_stub();
65 for stub in stubs {
66 for field in FIELDS {
67 *totals.get_mut(field).unwrap() += value(stub, field);
68 }
69 }
70 totals
71}

73fn progressive(table: &BTreeMap<i64, i64>, amount: i64) -> i64 {
74 let pairs: Vec<(i64, i64)> = table.iter().map(|(a, b)| (*a, *b)).collect();
75 let mut numerator = 0;
76 for (i, (lower, rate)) in pairs.iter().enumerate() {
77 if *lower >= amount {
78 break;
79 }
80 let upper = pairs.get(i + 1).map_or(amount, |next| next.0);
81 numerator += *rate * (amount.min(upper) - *lower);
82 }
83 numerator / 10000
84}

86fn deduct(stub: &mut Stub, remaining: &mut i64, field: &str, amount: i64) {
87 let actual = amount.min(*remaining);
88 set(stub, field, actual);
89 *remaining -= actual;
90}

194 fn day_check(&self, text: &str) -> Result<i32, String> {
195 let day = text.parse::<i32>().unwrap();
196 if !(1..=364).contains(&day) {
197 return Err(format!("invalid day {text}"));
198 }
199 let period = period_of(day);
200 if period <= self.closed {
201 return Err(format!("period closed {period}"));
202 }
203 Ok(day)

233 fn time_check(text: &str) -> Result<i32, String> {
234 let (hour, minute) = text.split_once(':').unwrap();
235 let hour = hour.parse::<i32>().unwrap();
236 let minute = minute.parse::<i32>().unwrap();
237 if hour > 23 || minute > 59 {
238 Err(format!("invalid time {text}"))
239 } else {
240 Ok(hour * 60 + minute)
241 }

244 fn daily(&self, employee: &Employee, day: i32) -> [i64; 6] {
245 let shifts = employee.shifts.get(&day);
246 let mut worked = 0i64;
247 let mut night = 0i64;
248 if let Some(shifts) = shifts {
249 for shift in shifts {
250 worked += (shift.end - shift.start) as i64;
251 night += (shift.end.min(360) - shift.start).max(0) as i64;
252 night += (shift.end - shift.start.max(1320)).max(0) as i64;
253 }
254 }
255 let pto = *employee.pto.get(&day).unwrap_or(&0);
256 let special = day % 7 == 0 || self.holidays.contains(&day);
257 let (regular, overtime, double) = if special {
258 (0, 0, worked)
259 } else {
260 (worked.min(480), (worked - 480).max(0).min(240), (worked - 720).max(0))
261 };
262 [worked, regular, overtime, double, night, pto]
263 }

265 fn weekly(&self, employee: &Employee, week: i32) -> ([i64; 5], i64) {
266 let mut total = [0; 5];
267 for day in (7 * week - 6)..=(7 * week) {
268 let daily = self.daily(employee, day);
269 for i in 0..5 {
270 total[i] += daily[i + 1];
271 }
272 }
273 let extra = (total[0] - 2400).max(0);
274 total[0] -= extra;
275 total[1] += extra;
276 (total, extra)
277 }

279 fn breakdown(&self, employee: &Employee, period: i32) -> [i64; 5] {
280 let (first, _) = self.weekly(employee, 2 * period - 1);
281 let (second, _) = self.weekly(employee, 2 * period);
282 let mut result = [0; 5];
283 for i in 0..5 {
284 result[i] = first[i] + second[i];
285 }
286 result
287 }

289 fn calculate(&self, employee: &Employee, period: i32) -> Option<Calculation> {
290 let mut stub = new_stub();
291 let ytd = &employee.ytd;
292 match employee.kind.as_str() {
293 "HOURLY" => {
294 let [regular, overtime, double, night, pto] =
295 self.breakdown(employee, period);
296 for (field, amount) in ["R", "O", "D", "N", "P"]
297 .iter().zip([regular, overtime, double, night, pto]) {
298 set(&mut stub, field, amount);
299 }
300 let weighted = 2 * (regular + pto) + 3 * overtime + 4 * double;
301 set(&mut stub, "earnings",
302 (weighted * employee.base + 2 * night * self.setting("NIGHT")) / 120);
303 }
304 "SALARIED" => {
305 if employee.active {
306 let salary = employee.base / 26;
307 let leave = employee.leave.iter()
308 .filter(|day| period_of(**day) == period &&
309 !self.holidays.contains(*day)).count() as i64;
310 let deduction = leave * (employee.base / 260);
311 set(&mut stub, "salary", salary);
312 set(&mut stub, "leave", leave);
313 set(&mut stub, "deduction", deduction);
314 set(&mut stub, "earnings", salary - deduction);
315 }
316 }
317 "COMMISSION" => {
318 let sales = *employee.sales.get(&period).unwrap_or(&0);
319 let commission = progressive(&self.tiers, sales);
320 set(&mut stub, "sales", sales);
321 set(&mut stub, "commission", commission);
322 set(&mut stub, "earnings",
323 commission + if employee.active { employee.base } else { 0 });
324 }
325 _ => unreachable!(),
326 }
327 set(&mut stub, "bonus", employee.bonus);
328 let gross = value(&stub, "earnings") + employee.bonus;
329 if gross == 0 {
330 return None;
331 }
332 set(&mut stub, "gross", gross);
333 let mut remaining = gross;
334
335 let room = (self.setting("RETIRECAP") - value(ytd, "retirement")).max(0);
336 deduct(&mut stub, &mut remaining, "retirement",
337 (gross * employee.retirement / 10000).min(room));
338 deduct(&mut stub, &mut remaining, "health", employee.health);
339 set(&mut stub, "W", remaining);
340 let base_taxable =
341 (value(&stub, "earnings") - value(&stub, "retirement") - value(&stub, "health")).max(0);
342 let taxable = (base_taxable - employee.allowances * self.setting("ALLOWANCE")).max(0);
343 deduct(&mut stub, &mut remaining, "federal",
344 progressive(&self.brackets, taxable) + employee.extra);
345 let wage_base = value(&stub, "W");
346 deduct(&mut stub, &mut remaining, "supplemental",
347 (wage_base - base_taxable) * self.setting("SUPP") / 10000);
348 deduct(&mut stub, &mut remaining, "state",
349 wage_base * self.setting("STATE") / 10000);
350
351 let medicare_wages = gross - value(&stub, "health");
352 let social_wages = medicare_wages.min((self.setting("SSBASE") - value(ytd, "Y")).max(0));
353 let additional = (value(ytd, "F") + medicare_wages - self.setting("ADDBASE"))
354 .min(medicare_wages).max(0);
355 set(&mut stub, "F", medicare_wages);
356 set(&mut stub, "Y", social_wages);
357 deduct(&mut stub, &mut remaining, "social",
358 social_wages * self.setting("SSRATE") / 10000);
359 deduct(&mut stub, &mut remaining, "medicare",
360 medicare_wages * self.setting("MEDRATE") / 10000 +
361 additional * self.setting("ADDMED") / 10000);
362
363 let disposable = remaining;
364 let cap = disposable * self.setting("GARNCAP") / 10000;
365 let mut garnishment = 0;
366 let mut balances = Vec::new();
367 for order in &employee.garnishments {
368 let taken = (disposable * order.rate / 10000)
369 .min(order.balance).min(cap - garnishment);
370 garnishment += taken;
371 balances.push(order.balance - taken);
372 }
373 set(&mut stub, "garnishment", garnishment);
374 set(&mut stub, "garnishment_balance", balances.iter().sum());
375 set(&mut stub, "net", remaining - garnishment);
376 set(&mut stub, "employer_social",
377 social_wages * self.setting("SSRATE") / 10000);
378 set(&mut stub, "employer_medicare",
379 medicare_wages * self.setting("MEDRATE") / 10000);
380 let j = medicare_wages.min((self.setting("UIBASE") - value(ytd, "J")).max(0));
381 set(&mut stub, "J", j);
382 set(&mut stub, "unemployment", j * self.setting("UIRATE") / 10000);
383 let eligible_match = value(&stub, "retirement")
384 .min(gross * self.setting("MATCHMAX") / 10000);
385 set(&mut stub, "match", eligible_match * self.setting("MATCH") / 10000);
386 let employer_total: i64 =
387 ["employer_social", "employer_medicare", "unemployment", "match"]
388 .iter().map(|f| value(&stub, f)).sum();
389 set(&mut stub, "employer", employer_total);
390 if employee.kind == "HOURLY" {
391 let minutes = value(&stub, "R") + value(&stub, "O") + value(&stub, "D");
392 set(&mut stub, "accrued", (960 - employee.pto_balance).min(minutes / 20));
393 }
394 let post_pto = employee.pto_balance + value(&stub, "accrued");
395 set(&mut stub, "pto_balance", post_pto);
396 Some(Calculation { stub, balances })
397 }

405 fn stub_block(&mut self, person: &Employee, stub: &Stub, indent: usize) {
406 match person.kind.as_str() {
407 "HOURLY" => {
408 let times = ["R", "O", "D", "N", "P"].map(|field| value(stub, field));
409 self.emit(format!("hours {}", Self::hours_parts(&times)), indent);
410 self.emit(format!("pto accrued {} balance {}",
411 duration(value(stub, "accrued")), duration(value(stub, "pto_balance"))), indent);
412 }
413 "SALARIED" => self.emit(format!("salary {} leave {} deduction {}",
414 money(value(stub, "salary")), value(stub, "leave"),
415 money(value(stub, "deduction"))), indent),
416 "COMMISSION" => self.emit(format!("sales {} commission {}",
417 money(value(stub, "sales")), money(value(stub, "commission"))), indent),
418 _ => unreachable!(),
419 }
420 for field in ["earnings", "bonus", "gross", "retirement", "health",
421 "federal", "supplemental", "state", "social", "medicare"] {
422 let label = if field == "social" { "social-security" } else { field };
423 self.emit(format!("{label} {}", money(value(stub, field))), indent);
424 }
425 self.emit(format!("garnishment {} balance {}",
426 money(value(stub, "garnishment")),
427 money(value(stub, "garnishment_balance"))), indent);
428 self.emit(format!("net {}", money(value(stub, "net"))), indent);
429 self.emit(format!("employer social-security {} medicare {} unemployment {} match {}",
430 money(value(stub, "employer_social")),
431 money(value(stub, "employer_medicare")),
432 money(value(stub, "unemployment")),
433 money(value(stub, "match"))), indent);
434 }

443 fn show_hours(&mut self, person: &Employee, period: i32) {
444 self.emit(format!("HOURS {} period {period}", person.id), 0);
445 let mut found = false;
446 for day in (14 * period - 13)..=(14 * period) {
447 let mut shifts = person.shifts.get(&day).cloned().unwrap_or_default();
448 shifts.sort_by_key(|shift| shift.start);
449 if shifts.is_empty() && !person.pto.contains_key(&day) {
450 continue;
451 }
452 let daily = self.daily(person, day);
453 let mut text = format!("day {day} {}", weekday(day));
454 if self.holidays.contains(&day) {
455 text += " holiday";
456 }
457 for shift in shifts {
458 text += &format!(" {}-{}", clock(shift.start), clock(shift.end));
459 }
460 let values = [daily[1], daily[2], daily[3], daily[4], daily[5]];
461 text += &format!(" total {} {}", duration(daily[0]), Self::hours_parts(&values));
462 self.emit(text, 2);
463 found = true;
464 }
465 if !found {
466 self.emit("(none)".into(), 2);
467 }
468 for week in [2 * period - 1, 2 * period] {
469 let (values, overtime) = self.weekly(person, week);
470 self.emit(format!("week {week} {} weekly-ot {}",
471 Self::hours_parts(&values), duration(overtime)), 2);
472 }
473 self.emit(format!("total {}", Self::hours_parts(&self.breakdown(person, period))), 2);
474 }

476 fn run(&mut self, period: i32) {
477 self.closed = period;
478 self.emit(format!("RUN {period} days {}-{}", 14 * period - 13, 14 * period), 0);
479 let mut paid = 0;
480 let mut skipped = 0;
481 let mut stubs = Vec::new();
482 let ids: Vec<i32> = self.employees.keys().copied().collect();
483 for id in ids {
484 let person = self.employees[&id].clone();
485 let calculated = self.calculate(&person, period);
486 let Some(calculation) = calculated else {
487 if person.active {
488 skipped += 1;
489 }
490 continue;
491 };
492 self.print_stub(&person, period, &calculation.stub, &person.department);
493 let employee = self.employees.get_mut(&id).unwrap();
494 employee.bonus = 0;
495 employee.pto_balance = value(&calculation.stub, "pto_balance");
496 for (order, balance) in employee.garnishments.iter_mut().zip(&calculation.balances) {
497 order.balance = *balance;
498 }
499 for field in FIELDS {
500 *employee.ytd.get_mut(field).unwrap() += value(&calculation.stub, field);
501 }
502 employee.periods += 1;
503 employee.stubs.insert(period, (calculation.stub.clone(), employee.department.clone()));
504 stubs.push(calculation.stub);
505 paid += 1;
506 }
507 let total = sum_stubs(stubs.iter());
508 self.emit(format!(
509 "TOTAL paid {paid} skipped {skipped} gross {} deductions {} net {} employer {}",
510 money(value(&total, "gross")),
511 money(value(&total, "gross") - value(&total, "net")),
512 money(value(&total, "net")),
513 money(value(&total, "employer"))
514 ), 2);
515 }

517 fn journal(&mut self, period: i32) {
518 let stubs: Vec<Stub> = self.employees.values()
519 .filter_map(|person| person.stubs.get(&period).map(|record| record.0.clone()))
520 .collect();
521 let total = sum_stubs(stubs.iter());
522 let rows = [
523 ("debit", "wages", value(&total, "gross")),
524 ("debit", "employer-tax",
525 value(&total, "employer_social") + value(&total, "employer_medicare")
526 + value(&total, "unemployment")),
527 ("debit", "retirement-match", value(&total, "match")),
528 ("credit", "retirement", value(&total, "retirement") + value(&total, "match")),
529 ("credit", "health", value(&total, "health")),
530 ("credit", "federal", value(&total, "federal") + value(&total, "supplemental")),
531 ("credit", "state", value(&total, "state")),
532 ("credit", "fica", value(&total, "social") + value(&total, "medicare")
533 + value(&total, "employer_social") + value(&total, "employer_medicare")),
534 ("credit", "unemployment", value(&total, "unemployment")),
535 ("credit", "garnishment", value(&total, "garnishment")),
536 ("credit", "net-pay", value(&total, "net")),
537 ];
538 self.emit(format!("JOURNAL {period} stubs {}", stubs.len()), 0);
539 let mut debits = 0;
540 let mut credits = 0;
541 for (side, label, amount) in rows {
542 self.emit(format!("{side} {label} {}", money(amount)), 2);
543 if side == "debit" {
544 debits += amount;
545 } else {
546 credits += amount;
547 }
548 }
549 self.emit(format!("total debit {} credit {}", money(debits), money(credits)), 2);
550 }

565 fn command(&mut self, parts: &[&str]) -> Result<(), String> {
566 let cmd = parts[0];
567 let a = &parts[1..];
568 match cmd {
569 "SET" => {
570 if !self.settings.contains_key(a[0]) {
571 return Err(format!("unknown setting {}", a[0]));
572 }
573 let rate = SETTINGS.iter().take(9).any(|(name, _)| *name == a[0]);
574 let new = if rate { Self::rate_check(a[1])? } else { cents(a[1]) };
575 let before = self.settings[a[0]];
576 self.settings.insert(a[0].to_string(), new);
577 self.emit(format!("SET {} {} -> {}", a[0],
578 if rate { rate_text(before) } else { money(before) },
579 if rate { rate_text(new) } else { money(new) }), 0);
580 }
581 "BRACKET" | "TIER" => {
582 let rate = Self::rate_check(a[1])?;
583 let lower = cents(a[0]);
584 let table = if cmd == "BRACKET" { &mut self.brackets } else { &mut self.tiers };
585 let action = if table.insert(lower, rate).is_some() { "replaced" } else { "added" };
586 self.emit(format!("{cmd} {} {} {action}", money(lower), rate_text(rate)), 0);
587 }
588 "EMP" => {
589 let id = a[0].parse::<i32>().unwrap();
590 if self.employees.contains_key(&id) {
591 return Err(format!("duplicate employee {}", a[0]));
592 }
593 let base = Self::amount_check(a[3])?;
594 self.employees.insert(id, Employee::new(id, a[1], a[2], base));
595 self.emit(format!("EMP {id} {} {} {}", a[1], a[2], money(base)), 0);
596 }
597 "RATE" | "DEPT" | "BENEFIT" | "W4" | "GARNISH" | "TERM" | "BONUS" => {
598 let old = self.employee(a[0], true, None)?.clone();
599 let id = old.id;
600 match cmd {
601 "RATE" => {
602 let base = Self::amount_check(a[1])?;
603 self.employees.get_mut(&id).unwrap().base = base;
604 self.emit(format!("RATE {id} {} -> {}", money(old.base), money(base)), 0);
605 }
606 "DEPT" => {
607 if a[1] == old.department {
608 return Err(format!("same department {}", a[1]));
609 }
610 self.employees.get_mut(&id).unwrap().department = a[1].into();
611 self.emit(format!("DEPT {id} {} -> {}", old.department, a[1]), 0);
612 }
613 "BENEFIT" => {
614 let rate = Self::rate_check(a[1])?;
615 let health = cents(a[2]);
616 let employee = self.employees.get_mut(&id).unwrap();
617 employee.retirement = rate;
618 employee.health = health;
619 self.emit(format!("BENEFIT {id} retirement {} health {}",
620 rate_text(rate), money(health)), 0);
621 }
622 "W4" => {
623 let count = a[1].parse::<i64>().unwrap();
624 let extra = cents(a[2]);
625 let employee = self.employees.get_mut(&id).unwrap();
626 employee.allowances = count;
627 employee.extra = extra;
628 self.emit(format!("W4 {id} allowances {count} extra {}", money(extra)), 0);
629 }
630 "GARNISH" => {
631 let rate = Self::rate_check(a[1])?;
632 let balance = Self::amount_check(a[2])?;
633 let employee = self.employees.get_mut(&id).unwrap();
634 employee.garnishments.push(Garnishment { rate, balance });
635 let position = employee.garnishments.len();
636 self.emit(format!("GARNISH {id} order {position} {} balance {}",
637 rate_text(rate), money(balance)), 0);
638 }
639 "TERM" => {
640 self.employees.get_mut(&id).unwrap().active = false;
641 self.emit(format!("TERM {id}"), 0);
642 }
643 "BONUS" => {
644 let amount = Self::amount_check(a[1])?;
645 let employee = self.employees.get_mut(&id).unwrap();
646 employee.bonus += amount;
647 let pending = employee.bonus;
648 self.emit(format!("BONUS {id} {} pending {}", money(amount),
649 money(pending)), 0);
650 }
651 _ => unreachable!(),
652 }
653 }
654 "HOLIDAY" => {
655 let day = self.day_check(a[0])?;
656 if !self.holidays.insert(day) {
657 return Err(format!("duplicate holiday {}", a[0]));
658 }
659 self.emit(format!("HOLIDAY {day} {}", weekday(day)), 0);
660 }
661 "TIME" | "UNTIME" | "PTO" | "LEAVE" | "SALES" => {
662 let kind = match cmd {
663 "LEAVE" => "SALARIED",
664 "SALES" => "COMMISSION",
665 _ => "HOURLY",
666 };
667 let person = self.employee(a[0], true, Some(kind))?.clone();
668 let id = person.id;
669 let day = self.day_check(a[1])?;
670 match cmd {
671 "TIME" => {
672 let start = Self::time_check(a[2])?;
673 let end = Self::time_check(a[3])?;
674 if start >= end {
675 return Err(format!("invalid shift {}-{}", a[2], a[3]));
676 }
677 let mut shifts = person.shifts.get(&day).cloned().unwrap_or_default();
678 shifts.sort_by_key(|shift| shift.start);
679 for shift in &shifts {
680 if start < shift.end && shift.start < end {
681 return Err(format!("overlapping shift {}-{}",
682 clock(shift.start), clock(shift.end)));
683 }
684 }
685 let worked: i32 = shifts.iter().map(|shift| shift.end - shift.start).sum();
686 if worked + end - start > 960 {
687 return Err(format!("daily limit {} {}", a[0], a[1]));
688 }
689 self.employees.get_mut(&id).unwrap().shifts.entry(day).or_default()
690 .push(Shift { start, end });
691 self.emit(format!("TIME {id} day {day} {}-{} {}",
692 clock(start), clock(end),
693 duration((end - start) as i64)), 0);
694 }
695 "UNTIME" => {
696 let start = Self::time_check(a[2])?;
697 let shifts = person.shifts.get(&day).cloned().unwrap_or_default();
698 let found = shifts.iter().find(|shift| shift.start == start)
699 .copied().ok_or_else(|| {
700 format!("no shift {} {} {}", a[0], a[1], a[2])
701 })?;
702 let live = self.employees.get_mut(&id).unwrap()
703 .shifts.get_mut(&day).unwrap();
704 let position = live.iter().position(|shift| shift.start == start).unwrap();
705 live.remove(position);
706 self.emit(format!("UNTIME {id} day {day} {}-{}",
707 clock(found.start), clock(found.end)), 0);
708 }
709 "PTO" => {
710 let (hours, minutes) = a[2].split_once(':').unwrap();
711 let hours = hours.parse::<i64>().unwrap();
712 let minutes = minutes.parse::<i64>().unwrap();
713 let length = hours * 60 + minutes;
714 if minutes > 59 || !(1..=480).contains(&length) {
715 return Err(format!("invalid duration {}", a[2]));
716 }
717 if person.pto.contains_key(&day) {
718 return Err(format!("duplicate pto {} {}", a[0], a[1]));
719 }
720 if length > person.pto_balance {
721 return Err(format!("insufficient pto {} {}",
722 a[0], duration(person.pto_balance)));
723 }
724 let employee = self.employees.get_mut(&id).unwrap();
725 employee.pto.insert(day, length);
726 employee.pto_balance -= length;
727 let balance = employee.pto_balance;
728 self.emit(format!("PTO {id} day {day} {} balance {}",
729 duration(length), duration(balance)), 0);
730 }
731 "LEAVE" => {
732 if day % 7 == 0 || day % 7 == 6 || self.holidays.contains(&day) {
733 return Err(format!("not workday {}", a[1]));
734 }
735 if person.leave.contains(&day) {
736 return Err(format!("duplicate leave {} {}", a[0], a[1]));
737 }
738 self.employees.get_mut(&id).unwrap().leave.insert(day);
739 self.emit(format!("LEAVE {id} day {day} {}", weekday(day)), 0);
740 }
741 "SALES" => {
742 let amount = Self::amount_check(a[2])?;
743 let period = period_of(day);
744 let employee = self.employees.get_mut(&id).unwrap();
745 *employee.sales.entry(period).or_insert(0) += amount;
746 let total = employee.sales[&period];
747 self.emit(format!("SALES {id} day {day} {} period {period} total {}",
748 money(amount), money(total)), 0);
749 }
750 _ => unreachable!(),
751 }
752 }
753 "HOURS" => {
754 let person = self.employee(a[0], false, Some("HOURLY"))?.clone();
755 let period = self.period_check(a[1])?;
756 self.show_hours(&person, period);
757 }
758 "RUN" => {
759 let period = a[0].parse::<i32>().unwrap();
760 if period != self.closed + 1 || period > 26 {
761 return Err(format!("invalid period {}", a[0]));
762 }
763 self.run(period);
764 }
765 "PREVIEW" => {
766 let person = self.employee(a[0], false, None)?.clone();
767 let period = self.closed + 1;
768 let calculated = self.calculate(&person, period);
769 self.emit(format!("PREVIEW {} period {period}", person.id), 0);
770 if let Some(result) = calculated {
771 self.print_stub(&person, period, &result.stub, &person.department);
772 } else {
773 self.emit("no pay".into(), 2);
774 }
775 }
776 "REPRINT" => {
777 let person = self.employee(a[0], false, None)?.clone();
778 let period = self.period_check(a[1])?;
779 let (stub, department) = person.stubs.get(&period)
780 .ok_or_else(|| format!("no stub {} {}", a[0], a[1]))?;
781 self.emit(format!("REPRINT {} {period}", person.id), 0);
782 self.print_stub(&person, period, stub, department);
783 }
784 "YTD" => {
785 let person = self.employee(a[0], false, None)?.clone();
786 self.emit(format!("YTD {} {} {} {} periods {}", person.id,
787 person.department, person.kind,
788 if person.active { "active" } else { "terminated" }, person.periods), 0);
789 let mut totals = person.ytd.clone();
790 set(&mut totals, "pto_balance", person.pto_balance);
791 set(&mut totals, "garnishment_balance", person.garnish_balance());
792 self.stub_block(&person, &totals, 2);
793 self.emit(format!("wages federal {} social-security {} medicare {} unemployment {}",
794 money(value(&totals, "W")), money(value(&totals, "Y")),
795 money(value(&totals, "F")), money(value(&totals, "J"))), 2);
796 self.emit(format!("retirement-room {} pending-bonus {}",
797 money((self.setting("RETIRECAP") - value(&totals, "retirement")).max(0)),
798 money(person.bonus)), 2);
799 }
800 "JOURNAL" => {
801 let period = a[0].parse::<i32>().unwrap();
802 if period < 1 || period > self.closed {
803 return Err(format!("no run {}", a[0]));
804 }
805 self.journal(period);
806 }
807 "REPORT" => {
808 let mut departments: BTreeMap<String, Vec<Stub>> = BTreeMap::new();
809 let mut stubs = Vec::new();
810 for person in self.employees.values() {
811 for (stub, department) in person.stubs.values() {
812 departments.entry(department.clone()).or_default().push(stub.clone());
813 stubs.push(stub.clone());
814 }
815 }
816 self.emit(format!("REPORT period {}", self.closed), 0);
817 for (department, entries) in &departments {
818 self.report_line(department, entries);
819 }
820 self.report_line("total", &stubs);
821 }
822 "TOP" => {
823 let limit = a[0].parse::<usize>().unwrap();
824 let mut people: Vec<Employee> = self.employees.values()
825 .filter(|e| e.periods > 0).cloned().collect();
826 people.sort_by(|a, b| {
827 value(&b.ytd, "gross").cmp(&value(&a.ytd, "gross"))
828 .then(a.id.cmp(&b.id))
829 });
830 self.emit(format!("TOP {limit}"), 0);
831 for (rank, person) in people.iter().take(limit).enumerate() {
832 self.emit(format!("{} {} {} gross {} net {}", rank + 1,
833 person.id, person.department,
834 money(value(&person.ytd, "gross")),
835 money(value(&person.ytd, "net"))), 2);
836 }
837 if people.is_empty() {
838 self.emit("(none)".into(), 2);
839 }
840 }
841 "EMPLOYEES" => {
842 self.emit("EMPLOYEES".into(), 0);
843 let people: Vec<Employee> = self.employees.values().cloned().collect();
844 for person in &people {
845 self.emit(format!(
846 "{} {} {} {} {} retirement {} health {} allowances {} extra {} bonus {}",
847 person.id, person.department, person.kind, money(person.base),
848 if person.active { "active" } else { "terminated" },
849 rate_text(person.retirement), money(person.health),
850 person.allowances, money(person.extra), money(person.bonus)
851 ), 2);
852 for (index, order) in person.garnishments.iter().enumerate() {
853 self.emit(format!("garnish {} {} balance {}",
854 index + 1, rate_text(order.rate), money(order.balance)), 4);
855 }
856 }
857 if people.is_empty() {
858 self.emit("(none)".into(), 2);
859 }
860 }
861 "CONFIG" => {
862 self.emit("CONFIG".into(), 0);
863 for (index, (name, _)) in SETTINGS.iter().enumerate() {
864 let amount = self.setting(name);
865 self.emit(format!("{name} {}", if index < 9 { rate_text(amount) }
866 else { money(amount) }), 2);
867 }
868 let brackets: Vec<(i64, i64)> =
869 self.brackets.iter().map(|(a, b)| (*a, *b)).collect();
870 let tiers: Vec<(i64, i64)> =
871 self.tiers.iter().map(|(a, b)| (*a, *b)).collect();
872 for (lower, rate) in brackets {
873 self.emit(format!("bracket {} {}", money(lower), rate_text(rate)), 2);
874 }
875 for (lower, rate) in tiers {
876 self.emit(format!("tier {} {}", money(lower), rate_text(rate)), 2);
877 }
878 }
879 _ => unreachable!("command names are guaranteed valid"),
880 }
881 Ok(())
882 }
883
884 fn execute(&mut self, line: &str) {
885 let parts: Vec<&str> = line.split_ascii_whitespace().collect();
886 if let Err(error) = self.command(&parts) {
887 self.emit(format!("ERROR {error}"), 0);
888 }
889 }
890}

892fn main() {
893 let mut input = String::new();
894 io::stdin().read_to_string(&mut input).unwrap();
895 let mut lines = input.lines();
896 let count: usize = lines.next().unwrap().parse().unwrap();
897 let mut payroll = Payroll::new();
898 for line in lines.take(count) {
899 payroll.execute(line);
900 }
901 if !payroll.output.is_empty() {
902 println!("{}", payroll.output.join("\n"));
903 }
904}
