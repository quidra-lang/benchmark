94 keys := make([]int, 0, len(values))

111func progressive(brackets map[int]int, amount int) int {
112 limits := sortedKeys(brackets)
113 numerator := 0
114 for i, lower := range limits {
115 if lower >= amount {
116 break
117 }
118 upper := amount
119 if i+1 < len(limits) {
120 upper = limits[i+1]
121 }
122 if upper > amount {
123 upper = amount
124 }
125 numerator += brackets[lower] * (upper - lower)
126 }
127 return numerator / 10000
128}

214func (p *Payroll) employee(text string, active bool, kind string) (*Employee, error) {
215 employee := p.employees[number(text)]
216 if employee == nil {
217 return nil, fmt.Errorf("unknown employee %s", text)
218 }
219 if active && !employee.active {
220 return nil, fmt.Errorf("terminated employee %s", text)
221 }
222 if kind != "" && employee.kind != kind {
223 return nil, fmt.Errorf("not %s %s", strings.ToLower(kind), text)
224 }
225 return employee, nil
226}

228func (p *Payroll) dayCheck(text string) (int, error) {
229 day := number(text)
230 if day < 1 || day > 364 {
231 return 0, fmt.Errorf("invalid day %s", text)
232 }
233 if period := periodOf(day); period <= p.closed {
234 return 0, fmt.Errorf("period closed %d", period)
235 }
236 return day, nil
237}

263func timeCheck(text string) (int, error) {
264 parts := strings.Split(text, ":")
265 hour, minute := number(parts[0]), number(parts[1])

274func (p *Payroll) daily(employee *Employee, day int) (int, Hours) {
275 var values Hours
276 worked := 0
277 for _, shift := range employee.shifts[day] {
278 worked += shift.end - shift.start
279 nightEnd := shift.end
280 if nightEnd > 360 {
281 nightEnd = 360
282 }
283 if nightEnd > shift.start {
284 values[3] += nightEnd - shift.start
285 }
286 nightStart := shift.start
287 if nightStart < 1320 {
288 nightStart = 1320
289 }
290 if shift.end > nightStart {
291 values[3] += shift.end - nightStart
292 }
293 }
294 values[4] = employee.pto[day]
295 if day%7 == 0 || p.holidays[day] {
296 values[2] = worked
297 } else {
298 values[0] = min(worked, 480)
299 values[1] = min(max(worked-480, 0), 240)
300 values[2] = max(worked-720, 0)
301 }
302 return worked, values
303}

305func (p *Payroll) weekly(employee *Employee, week int) (Hours, int) {
306 var totals Hours
307 for day := 7*week - 6; day <= 7*week; day++ {
308 _, values := p.daily(employee, day)
309 for i := range totals {
310 totals[i] += values[i]
311 }
312 }
313 weeklyOvertime := max(0, totals[0]-2400)
314 totals[0] -= weeklyOvertime
315 totals[1] += weeklyOvertime
316 return totals, weeklyOvertime
317}

319func (p *Payroll) breakdown(employee *Employee, period int) Hours {
320 first, _ := p.weekly(employee, 2*period-1)
321 second, _ := p.weekly(employee, 2*period)
322 for i := range first {
323 first[i] += second[i]
324 }
325 return first
326}

328func (p *Payroll) calculate(employee *Employee, period int) (map[string]int, []int) {
329 stub := emptyTotals()
330 settings := p.settings
331 ytd := employee.ytd
332 switch employee.kind {
333 case "HOURLY":
334 values := p.breakdown(employee, period)
335 for i, field := range []string{"R", "O", "D", "N", "P"} {
336 stub[field] = values[i]
337 }
338 weighted := 2*(stub["R"]+stub["P"]) + 3*stub["O"] + 4*stub["D"]
339 stub["earnings"] = (weighted*employee.base + 2*stub["N"]*settings["NIGHT"]) / 120
340 case "SALARIED":
341 if employee.active {
342 stub["salary"] = employee.base / 26
343 for day := range employee.leave {
344 if periodOf(day) == period && !p.holidays[day] {
345 stub["leave"]++
346 }
347 }
348 stub["deduction"] = stub["leave"] * (employee.base / 260)
349 stub["earnings"] = stub["salary"] - stub["deduction"]
350 }
351 case "COMMISSION":
352 stub["sales"] = employee.sales[period]
353 stub["commission"] = progressive(p.tiers, stub["sales"])
354 stub["earnings"] = stub["commission"]
355 if employee.active {
356 stub["earnings"] += employee.base
357 }
358 }
359 stub["bonus"] = employee.bonus
360 gross := stub["earnings"] + stub["bonus"]
361 if gross == 0 {
362 return nil, nil
363 }
364 stub["gross"] = gross
365 remaining := gross
366 deduct := func(field string, amount int) {
367 stub[field] = min(amount, remaining)
368 remaining -= stub[field]
369 }
370 retirementRoom := max(0, settings["RETIRECAP"]-ytd["retirement"])
371 deduct("retirement", min(gross*employee.retirement/10000, retirementRoom))
372 deduct("health", employee.health)
373 stub["W"] = remaining
374 baseTaxable := max(0, stub["earnings"]-stub["retirement"]-stub["health"])
375 allowanceTaxable := max(0, baseTaxable-employee.allowances*settings["ALLOWANCE"])
376 deduct("federal", progressive(p.brackets, allowanceTaxable)+employee.extra)
377 deduct("supplemental", (stub["W"]-baseTaxable)*settings["SUPP"]/10000)
378 deduct("state", stub["W"]*settings["STATE"]/10000)
379
380 medicareWages := gross - stub["health"]
381 socialWages := min(medicareWages, max(0, settings["SSBASE"]-ytd["Y"]))
382 additional := max(0, min(medicareWages, ytd["F"]+medicareWages-settings["ADDBASE"]))
383 stub["F"], stub["Y"] = medicareWages, socialWages
384 deduct("social", socialWages*settings["SSRATE"]/10000)
385 deduct("medicare", medicareWages*settings["MEDRATE"]/10000+
386 additional*settings["ADDMED"]/10000)
387 disposable := remaining
388 cap := disposable * settings["GARNCAP"] / 10000
389 balances := make([]int, len(employee.garnishments))
390 garnishment := 0
391 for i, order := range employee.garnishments {
392 taken := min(disposable*order.rate/10000, order.balance, cap-garnishment)
393 garnishment += taken
394 balances[i] = order.balance - taken
395 stub["garnishment_balance"] += balances[i]
396 }
397 stub["garnishment"] = garnishment
398 stub["net"] = remaining - garnishment
399 stub["employer_social"] = socialWages * settings["SSRATE"] / 10000
400 stub["employer_medicare"] = medicareWages * settings["MEDRATE"] / 10000
401 stub["J"] = min(medicareWages, max(0, settings["UIBASE"]-ytd["J"]))
402 stub["unemployment"] = stub["J"] * settings["UIRATE"] / 10000
403 eligibleMatch := min(stub["retirement"], gross*settings["MATCHMAX"]/10000)
404 stub["match"] = eligibleMatch * settings["MATCH"] / 10000
405 stub["employer"] = stub["employer_social"] + stub["employer_medicare"] +
406 stub["unemployment"] + stub["match"]
407 if employee.kind == "HOURLY" {
408 stub["accrued"] = min(960-employee.ptoBalance,
409 (stub["R"]+stub["O"]+stub["D"])/20)
410 }
411 stub["pto_balance"] = employee.ptoBalance + stub["accrued"]
412 return stub, balances
413}

423func (p *Payroll) stubBlock(employee *Employee, stub map[string]int, indent int) {
424 switch employee.kind {
425 case "HOURLY":
426 values := Hours{
427 stub["R"], stub["O"], stub["D"], stub["N"], stub["P"],
428 }
429 p.emit(indent, append([]any{"hours"}, hoursParts(values)...)...)
430 p.emit(indent, "pto", "accrued", duration(stub["accrued"]),
431 "balance", duration(stub["pto_balance"]))
432 case "SALARIED":
433 p.emit(indent, "salary", money(stub["salary"]), "leave", stub["leave"],
434 "deduction", money(stub["deduction"]))
435 case "COMMISSION":
436 p.emit(indent, "sales", money(stub["sales"]),
437 "commission", money(stub["commission"]))
438 }
439 for _, field := range []string{
440 "earnings", "bonus", "gross", "retirement", "health",
441 "federal", "supplemental", "state", "social", "medicare",
442 } {
443 label := field
444 if field == "social" {
445 label = "social-security"
446 }
447 p.emit(indent, label, money(stub[field]))
448 }
449 p.emit(indent, "garnishment", money(stub["garnishment"]),
450 "balance", money(stub["garnishment_balance"]))
451 p.emit(indent, "net", money(stub["net"]))
452 p.emit(indent, "employer", "social-security", money(stub["employer_social"]),
453 "medicare", money(stub["employer_medicare"]),
454 "unemployment", money(stub["unemployment"]), "match", money(stub["match"]))
455}

462func (p *Payroll) showHours(employee *Employee, period int) {
463 p.emit(0, "HOURS", employee.id, "period", period)
464 found := false
465 for day := 14*period - 13; day <= 14*period; day++ {
466 shifts := append([]Shift(nil), employee.shifts[day]...)
467 if len(shifts) == 0 && employee.pto[day] == 0 {
468 continue
469 }
470 sort.Slice(shifts, func(i, j int) bool {
471 return shifts[i].start < shifts[j].start
472 })
473 worked, values := p.daily(employee, day)
474 parts := []any{"day", day, weekday(day)}
475 if p.holidays[day] {
476 parts = append(parts, "holiday")
477 }
478 for _, shift := range shifts {
479 parts = append(parts, clock(shift.start)+"-"+clock(shift.end))
480 }
481 parts = append(parts, "total", duration(worked))
482 parts = append(parts, hoursParts(values)...)
483 p.emit(2, parts...)
484 found = true
485 }
486 if !found {
487 p.emit(2, "(none)")
488 }
489 for _, week := range []int{2*period - 1, 2 * period} {
490 values, overtime := p.weekly(employee, week)
491 parts := []any{"week", week}
492 parts = append(parts, hoursParts(values)...)
493 parts = append(parts, "weekly-ot", duration(overtime))
494 p.emit(2, parts...)
495 }
496 p.emit(2, append([]any{"total"}, hoursParts(p.breakdown(employee, period))...)...)
497}

499func (p *Payroll) run(period int) {
500 p.closed = period
501 p.emit(0, "RUN", period, "days", fmt.Sprintf("%d-%d", 14*period-13, 14*period))
502 paid, skipped := 0, 0
503 var stubs []map[string]int
504 for _, id := range sortedKeys(p.employees) {
505 employee := p.employees[id]
506 stub, balances := p.calculate(employee, period)
507 if stub == nil {
508 if employee.active {
509 skipped++
510 }
511 continue
512 }
513 p.printStub(employee, period, stub, employee.department)
514 employee.bonus = 0
515 employee.ptoBalance = stub["pto_balance"]
516 for i, balance := range balances {
517 employee.garnishments[i].balance = balance
518 }
519 for _, field := range fields {
520 employee.ytd[field] += stub[field]
521 }
522 employee.periods++
523 employee.stubs[period] = StubRecord{totals: stub, department: employee.department}
524 stubs = append(stubs, stub)
525 paid++
526 }
527 totals := sumStubs(stubs)
528 p.emit(2, "TOTAL", "paid", paid, "skipped", skipped,
529 "gross", money(totals["gross"]),
530 "deductions", money(totals["gross"]-totals["net"]),
531 "net", money(totals["net"]), "employer", money(totals["employer"]))
532}

534func (p *Payroll) journal(period int) {
535 var stubs []map[string]int
536 for _, employee := range p.employees {
537 if entry, ok := employee.stubs[period]; ok {
538 stubs = append(stubs, entry.totals)
539 }
540 }
541 total := sumStubs(stubs)
542 type JournalRow struct {
543 side string
544 label string
545 amount int
546 }
547 rows := []JournalRow{
548 {"debit", "wages", total["gross"]},
549 {"debit", "employer-tax", total["employer_social"] + total["employer_medicare"] + total["unemployment"]},
550 {"debit", "retirement-match", total["match"]},
551 {"credit", "retirement", total["retirement"] + total["match"]},
552 {"credit", "health", total["health"]},
553 {"credit", "federal", total["federal"] + total["supplemental"]},
554 {"credit", "state", total["state"]},
555 {"credit", "fica", total["social"] + total["medicare"] + total["employer_social"] + total["employer_medicare"]},
556 {"credit", "unemployment", total["unemployment"]},
557 {"credit", "garnishment", total["garnishment"]},
558 {"credit", "net-pay", total["net"]},
559 }
560 p.emit(0, "JOURNAL", period, "stubs", len(stubs))
561 debit, credit := 0, 0
562 for _, row := range rows {
563 p.emit(2, row.side, row.label, money(row.amount))
564 if row.side == "debit" {
565 debit += row.amount
566 } else {
567 credit += row.amount
568 }
569 }
570 p.emit(2, "total", "debit", money(debit), "credit", money(credit))
571}

583func (p *Payroll) command(command string, a []string) error {
584 switch command {
585 case "SET":
586 name := a[0]
587 if _, ok := p.settings[name]; !ok {
588 return fmt.Errorf("unknown setting %s", name)
589 }
590 var value int
591 if rateSettings[name] {
592 var err error
593 value, err = rateCheck(a[1])
594 if err != nil {
595 return err
596 }
597 } else {
598 value = cents(a[1])
599 }
600 formatter := money
601 if rateSettings[name] {
602 formatter = rateText
603 }
604 old := p.settings[name]
605 p.settings[name] = value
606 p.emit(0, "SET", name, formatter(old), "->", formatter(value))
607 case "BRACKET", "TIER":
608 rate, err := rateCheck(a[1])
609 if err != nil {
610 return err
611 }
612 brackets := p.brackets
613 if command == "TIER" {
614 brackets = p.tiers
615 }
616 lower := cents(a[0])
617 action := "added"
618 if _, ok := brackets[lower]; ok {
619 action = "replaced"
620 }
621 brackets[lower] = rate
622 p.emit(0, command, money(lower), rateText(rate), action)
623 case "EMP":
624 id := number(a[0])
625 if p.employees[id] != nil {
626 return fmt.Errorf("duplicate employee %s", a[0])
627 }
628 base, err := amountCheck(a[3])
629 if err != nil {
630 return err
631 }
632 p.employees[id] = newEmployee(id, a[1], a[2], base)
633 p.emit(0, "EMP", id, a[1], a[2], money(base))
634 case "RATE", "DEPT", "BENEFIT", "W4", "GARNISH", "TERM", "BONUS":
635 employee, err := p.employee(a[0], true, "")
636 if err != nil {
637 return err
638 }
639 switch command {
640 case "RATE":
641 base, err := amountCheck(a[1])
642 if err != nil {
643 return err
644 }
645 old := employee.base
646 employee.base = base
647 p.emit(0, "RATE", employee.id, money(old), "->", money(base))
648 case "DEPT":
649 if a[1] == employee.department {
650 return fmt.Errorf("same department %s", a[1])
651 }
652 old := employee.department
653 employee.department = a[1]
654 p.emit(0, "DEPT", employee.id, old, "->", a[1])
655 case "BENEFIT":
656 rate, err := rateCheck(a[1])
657 if err != nil {
658 return err
659 }
660 employee.retirement = rate
661 employee.health = cents(a[2])
662 p.emit(0, "BENEFIT", employee.id, "retirement", rateText(rate),
663 "health", money(employee.health))
664 case "W4":
665 employee.allowances = number(a[1])
666 employee.extra = cents(a[2])
667 p.emit(0, "W4", employee.id, "allowances", employee.allowances,
668 "extra", money(employee.extra))
669 case "GARNISH":
670 rate, err := rateCheck(a[1])
671 if err != nil {
672 return err
673 }
674 amount, err := amountCheck(a[2])
675 if err != nil {
676 return err
677 }
678 employee.garnishments = append(employee.garnishments, Garnishment{rate, amount})
679 p.emit(0, "GARNISH", employee.id, "order", len(employee.garnishments),
680 rateText(rate), "balance", money(amount))
681 case "TERM":
682 employee.active = false
683 p.emit(0, "TERM", employee.id)
684 case "BONUS":
685 amount, err := amountCheck(a[1])
686 if err != nil {
687 return err
688 }
689 employee.bonus += amount
690 p.emit(0, "BONUS", employee.id, money(amount),
691 "pending", money(employee.bonus))
692 }
693 case "HOLIDAY":
694 day, err := p.dayCheck(a[0])
695 if err != nil {
696 return err
697 }
698 if p.holidays[day] {
699 return fmt.Errorf("duplicate holiday %s", a[0])
700 }
701 p.holidays[day] = true
702 p.emit(0, "HOLIDAY", day, weekday(day))
703 case "TIME", "UNTIME", "PTO", "LEAVE", "SALES":
704 kind := "HOURLY"
705 if command == "LEAVE" {
706 kind = "SALARIED"
707 } else if command == "SALES" {
708 kind = "COMMISSION"
709 }
710 employee, err := p.employee(a[0], true, kind)
711 if err != nil {
712 return err
713 }
714 day, err := p.dayCheck(a[1])
715 if err != nil {
716 return err
717 }
718 switch command {
719 case "TIME":
720 start, err := timeCheck(a[2])
721 if err != nil {
722 return err
723 }
724 end, err := timeCheck(a[3])
725 if err != nil {
726 return err
727 }
728 if start >= end {
729 return fmt.Errorf("invalid shift %s-%s", a[2], a[3])
730 }
731 shifts := append([]Shift(nil), employee.shifts[day]...)
732 sort.Slice(shifts, func(i, j int) bool {
733 return shifts[i].start < shifts[j].start
734 })
735 total := 0
736 for _, shift := range shifts {
737 if start < shift.end && shift.start < end {
738 return fmt.Errorf("overlapping shift %s-%s", clock(shift.start), clock(shift.end))
739 }
740 total += shift.end - shift.start
741 }
742 if total+end-start > 960 {
743 return fmt.Errorf("daily limit %s %s", a[0], a[1])
744 }
745 employee.shifts[day] = append(employee.shifts[day], Shift{start, end})
746 p.emit(0, "TIME", employee.id, "day", day,
747 clock(start)+"-"+clock(end), duration(end-start))
748 case "UNTIME":
749 start, err := timeCheck(a[2])
750 if err != nil {
751 return err
752 }
753 shifts := employee.shifts[day]
754 position := -1
755 for i, shift := range shifts {
756 if shift.start == start {
757 position = i
758 break
759 }
760 }
761 if position < 0 {
762 return fmt.Errorf("no shift %s %s %s", a[0], a[1], a[2])
763 }
764 shift := shifts[position]
765 employee.shifts[day] = append(shifts[:position], shifts[position+1:]...)
766 p.emit(0, "UNTIME", employee.id, "day", day,
767 clock(shift.start)+"-"+clock(shift.end))
768 case "PTO":
769 parts := strings.Split(a[2], ":")
770 hour, minute := number(parts[0]), number(parts[1])
771 length := hour*60 + minute
772 if minute > 59 || length < 1 || length > 480 {
773 return fmt.Errorf("invalid duration %s", a[2])
774 }
775 if _, ok := employee.pto[day]; ok {
776 return fmt.Errorf("duplicate pto %s %s", a[0], a[1])
777 }
778 if length > employee.ptoBalance {
779 return fmt.Errorf("insufficient pto %s %s", a[0], duration(employee.ptoBalance))
780 }
781 employee.pto[day] = length
782 employee.ptoBalance -= length
783 p.emit(0, "PTO", employee.id, "day", day, duration(length),
784 "balance", duration(employee.ptoBalance))
785 case "LEAVE":
786 if day%7 == 0 || day%7 == 6 || p.holidays[day] {
787 return fmt.Errorf("not workday %s", a[1])
788 }
789 if employee.leave[day] {
790 return fmt.Errorf("duplicate leave %s %s", a[0], a[1])
791 }
792 employee.leave[day] = true
793 p.emit(0, "LEAVE", employee.id, "day", day, weekday(day))
794 case "SALES":
795 amount, err := amountCheck(a[2])
796 if err != nil {
797 return err
798 }
799 period := periodOf(day)
800 employee.sales[period] += amount
801 p.emit(0, "SALES", employee.id, "day", day, money(amount),
802 "period", period, "total", money(employee.sales[period]))
803 }
804 case "HOURS":
805 employee, err := p.employee(a[0], false, "HOURLY")
806 if err != nil {
807 return err
808 }
809 period, err := periodCheck(a[1])
810 if err != nil {
811 return err
812 }
813 p.showHours(employee, period)
814 case "RUN":
815 period := number(a[0])
816 if period != p.closed+1 || period > 26 {
817 return fmt.Errorf("invalid period %s", a[0])
818 }
819 p.run(period)
820 case "PREVIEW":
821 employee, err := p.employee(a[0], false, "")
822 if err != nil {
823 return err
824 }
825 period := p.closed + 1
826 stub, _ := p.calculate(employee, period)
827 p.emit(0, "PREVIEW", employee.id, "period", period)
828 if stub == nil {
829 p.emit(2, "no pay")
830 } else {
831 p.printStub(employee, period, stub, employee.department)
832 }
833 case "REPRINT":
834 employee, err := p.employee(a[0], false, "")
835 if err != nil {
836 return err
837 }
838 period, err := periodCheck(a[1])
839 if err != nil {
840 return err
841 }
842 entry, ok := employee.stubs[period]
843 if !ok {
844 return fmt.Errorf("no stub %s %s", a[0], a[1])
845 }
846 p.emit(0, "REPRINT", employee.id, period)
847 p.printStub(employee, period, entry.totals, entry.department)
848 case "YTD":
849 employee, err := p.employee(a[0], false, "")
850 if err != nil {
851 return err
852 }
853 status := "terminated"
854 if employee.active {
855 status = "active"
856 }
857 p.emit(0, "YTD", employee.id, employee.department, employee.kind,
858 status, "periods", employee.periods)
859 totals := emptyTotals()
860 for field, value := range employee.ytd {
861 totals[field] = value
862 }
863 totals["pto_balance"] = employee.ptoBalance
864 totals["garnishment_balance"] = employee.garnishmentBalance()
865 p.stubBlock(employee, totals, 2)
866 p.emit(2, "wages", "federal", money(totals["W"]), "social-security",
867 money(totals["Y"]), "medicare", money(totals["F"]),
868 "unemployment", money(totals["J"]))
869 p.emit(2, "retirement-room",
870 money(max(0, p.settings["RETIRECAP"]-totals["retirement"])),
871 "pending-bonus", money(employee.bonus))
872 case "JOURNAL":
873 period := number(a[0])
874 if period < 1 || period > p.closed {
875 return fmt.Errorf("no run %s", a[0])
876 }
877 p.journal(period)
878 case "REPORT":
879 departments := make(map[string][]map[string]int)
880 var allStubs []map[string]int
881 for _, employee := range p.employees {
882 for _, entry := range employee.stubs {
883 departments[entry.department] = append(departments[entry.department], entry.totals)
884 allStubs = append(allStubs, entry.totals)
885 }
886 }
887 p.emit(0, "REPORT", "period", p.closed)
888 for _, department := range sortedStrings(departments) {
889 p.reportLine(department, departments[department])
890 }
891 p.reportLine("total", allStubs)
892 case "TOP":
893 count := number(a[0])
894 var employees []*Employee
895 for _, employee := range p.employees {
896 if employee.periods > 0 {
897 employees = append(employees, employee)
898 }
899 }
900 sort.Slice(employees, func(i, j int) bool {
901 if employees[i].ytd["gross"] != employees[j].ytd["gross"] {
902 return employees[i].ytd["gross"] > employees[j].ytd["gross"]
903 }
904 return employees[i].id < employees[j].id
905 })
906 p.emit(0, "TOP", count)
907 for i := 0; i < min(count, len(employees)); i++ {
908 employee := employees[i]
909 p.emit(2, i+1, employee.id, employee.department,
910 "gross", money(employee.ytd["gross"]), "net", money(employee.ytd["net"]))
911 }
912 if len(employees) == 0 {
913 p.emit(2, "(none)")
914 }
915 case "EMPLOYEES":
916 p.emit(0, "EMPLOYEES")
917 ids := sortedKeys(p.employees)
918 for _, id := range ids {
919 employee := p.employees[id]
920 status := "terminated"
921 if employee.active {
922 status = "active"
923 }
924 p.emit(2, id, employee.department, employee.kind, money(employee.base),
925 status, "retirement", rateText(employee.retirement),
926 "health", money(employee.health), "allowances", employee.allowances,
927 "extra", money(employee.extra), "bonus", money(employee.bonus))
928 for i, order := range employee.garnishments {
929 p.emit(4, "garnish", i+1, rateText(order.rate),
930 "balance", money(order.balance))
931 }
932 }
933 if len(ids) == 0 {
934 p.emit(2, "(none)")
935 }
936 case "CONFIG":
937 p.emit(0, "CONFIG")
938 for _, name := range settingOrder {
939 value := p.settings[name]
940 formatter := money
941 if rateSettings[name] {
942 formatter = rateText
943 }
944 p.emit(2, name, formatter(value))
945 }
946 for _, item := range []struct {
947 label string
948 values map[int]int
949 }{{"bracket", p.brackets}, {"tier", p.tiers}} {
950 for _, lower := range sortedKeys(item.values) {
951 p.emit(2, item.label, money(lower), rateText(item.values[lower]))
952 }
953 }
954 }
955 return nil
956}
