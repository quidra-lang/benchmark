27local function cents(text)
28 local whole, fraction = text:match("^(%d+)%.(%d%d)$")
29 return tonumber(whole) * 100 + tonumber(fraction)
30end

72local function sorted_keys(map)

79local function progressive(brackets, amount)
80 local limits = sorted_keys(brackets)
81 local numerator = 0
82 for i, lower in ipairs(limits) do
83 if lower >= amount then break end
84 local upper = limits[i + 1] or amount
85 numerator = numerator + brackets[lower] * (math.min(amount, upper) - lower)
86 end
87 return numerator // 10000
88end

125local function employee_of(payroll, text, active, kind)
126 local id = tonumber(text)
127 local employee = payroll.employees[id]
128 if not employee then fail("unknown employee " .. text) end
129 if active and not employee.active then fail("terminated employee " .. text) end
130 if kind and employee.kind ~= kind then fail("not " .. kind:lower() .. " " .. text) end
131 return employee
132end

134local function day_check(payroll, text)
135 local day = tonumber(text)
136 if day < 1 or day > 364 then fail("invalid day " .. text) end
137 local period = period_of(day)
138 if period <= payroll.closed then fail("period closed " .. period) end
139 return day

146end

160local function time_check(text)
161 local hours, minutes = text:match("^(%d%d):(%d%d)$")
162 hours, minutes = tonumber(hours), tonumber(minutes)
163 if hours > 23 or minutes > 59 then fail("invalid time " .. text) end
164 return hours * 60 + minutes
165end

167local function daily(payroll, employee, day)
168 local shifts = employee.shifts[day] or {}
169 local worked, night = 0, 0
170 for _, shift in ipairs(shifts) do
171 local first, last = shift[1], shift[2]
172 worked = worked + last - first
173 night = night + math.max(0, math.min(last, 360) - first) +
174 math.max(0, last - math.max(first, 1320))
175 end
176 local pto = employee.pto[day] or 0
177 local regular, overtime, double
178 if day % 7 == 0 or payroll.holidays[day] then
179 regular, overtime, double = 0, 0, worked
180 else
181 regular = math.min(worked, 480)
182 overtime = math.min(math.max(worked - 480, 0), 240)
183 double = math.max(worked - 720, 0)
184 end
185 return worked, regular, overtime, double, night, pto
186end

188local function weekly(payroll, employee, week)
189 local totals = {0, 0, 0, 0, 0}
190 for day = 7 * week - 6, 7 * week do
191 local _, regular, overtime, double, night, pto = daily(payroll, employee, day)
192 local parts = {regular, overtime, double, night, pto}
193 for i = 1, 5 do totals[i] = totals[i] + parts[i] end
194 end
195 local weekly_ot = math.max(0, totals[1] - 2400)
196 totals[1] = totals[1] - weekly_ot
197 totals[2] = totals[2] + weekly_ot
198 return totals, weekly_ot
199end

201local function breakdown(payroll, employee, period)
202 local first = weekly(payroll, employee, 2 * period - 1)
203 local second = weekly(payroll, employee, 2 * period)
204 local totals = {}
205 for i = 1, 5 do totals[i] = first[i] + second[i] end
206 return totals
207end

209local function calculate(payroll, employee, period)
210 local stub = empty_totals()
211 local settings, ytd = payroll.settings, employee.ytd
212 if employee.kind == "HOURLY" then
213 local values = breakdown(payroll, employee, period)
214 for i, field in ipairs({"R", "O", "D", "N", "P"}) do stub[field] = values[i] end
215 local weighted = 2 * (stub.R + stub.P) + 3 * stub.O + 4 * stub.D
216 stub.earnings = (weighted * employee.base + 2 * stub.N * settings.NIGHT) // 120
217 elseif employee.kind == "SALARIED" then
218 if employee.active then
219 stub.salary = employee.base // 26
220 for day in pairs(employee.leave) do
221 if period_of(day) == period and not payroll.holidays[day] then
222 stub.leave = stub.leave + 1
223 end
224 end
225 stub.deduction = stub.leave * (employee.base // 260)
226 stub.earnings = stub.salary - stub.deduction
227 end
228 else
229 stub.sales = employee.sales[period] or 0
230 stub.commission = progressive(payroll.tiers, stub.sales)
231 stub.earnings = stub.commission + (employee.active and employee.base or 0)
232 end
233
234 stub.bonus = employee.bonus
235 local gross = stub.earnings + stub.bonus
236 if gross == 0 then return nil end
237 stub.gross = gross
238 local remaining = gross
239
240 local function deduct(field, amount)
241 stub[field] = math.min(amount, remaining)
242 remaining = remaining - stub[field]
243 end
244
245 local retirement_room = math.max(0, settings.RETIRECAP - ytd.retirement)
246 deduct("retirement", math.min(gross * employee.retirement // 10000, retirement_room))
247 deduct("health", employee.health)
248 stub.W = remaining
249 local taxable = math.max(0, stub.earnings - stub.retirement - stub.health)
250 local allowance_taxable = math.max(0, taxable - employee.allowances * settings.ALLOWANCE)
251 deduct("federal", progressive(payroll.brackets, allowance_taxable) + employee.extra)
252 deduct("supplemental", (stub.W - taxable) * settings.SUPP // 10000)
253 deduct("state", stub.W * settings.STATE // 10000)
254
255 local medicare_wages = gross - stub.health
256 local social_wages = math.min(medicare_wages, math.max(0, settings.SSBASE - ytd.Y))
257 local additional = math.max(0, math.min(medicare_wages,
258 ytd.F + medicare_wages - settings.ADDBASE))
259 stub.F, stub.Y = medicare_wages, social_wages
260 deduct("social", social_wages * settings.SSRATE // 10000)
261 deduct("medicare", medicare_wages * settings.MEDRATE // 10000 +
262 additional * settings.ADDMED // 10000)
263
264 local disposable = remaining
265 local cap = disposable * settings.GARNCAP // 10000
266 local balances = {}
267 local garnishment = 0
268 for _, order in ipairs(employee.garnishments) do
269 local rate, balance = order[1], order[2]
270 local taken = math.min(disposable * rate // 10000, balance, cap - garnishment)
271 garnishment = garnishment + taken
272 balances[#balances + 1] = balance - taken
273 end
274 stub.garnishment = garnishment
275 stub.garnishment_balance = 0
276 for _, balance in ipairs(balances) do
277 stub.garnishment_balance = stub.garnishment_balance + balance
278 end
279 stub.net = remaining - garnishment
280 stub.employer_social = social_wages * settings.SSRATE // 10000
281 stub.employer_medicare = medicare_wages * settings.MEDRATE // 10000
282 stub.J = math.min(medicare_wages, math.max(0, settings.UIBASE - ytd.J))
283 stub.unemployment = stub.J * settings.UIRATE // 10000
284 local match_eligible = math.min(stub.retirement, gross * settings.MATCHMAX // 10000)
285 stub.match = match_eligible * settings.MATCH // 10000
286 stub.employer = stub.employer_social + stub.employer_medicare +
287 stub.unemployment + stub.match
288 if employee.kind == "HOURLY" then
289 stub.accrued = math.min(960 - employee.pto_balance,
290 (stub.R + stub.O + stub.D) // 20)
291 end
292 stub.pto_balance = employee.pto_balance + stub.accrued
293 return stub, balances
294end

296local function hours_parts(values)
297 local parts = {}
298 local names = {"reg", "ot", "dbl", "night", "pto"}
299 for i = 1, 5 do
300 parts[#parts + 1] = names[i]
301 parts[#parts + 1] = duration(values[i])
302 end
303 return parts
304end

306local function stub_block(payroll, employee, stub, indent)
307 if employee.kind == "HOURLY" then
308 local values = {}
309 for _, field in ipairs({"R", "O", "D", "N", "P"}) do
310 values[#values + 1] = stub[field]
311 end
312 emit(payroll, indent, "hours", table.unpack(hours_parts(values)))
313 emit(payroll, indent, "pto", "accrued", duration(stub.accrued),
314 "balance", duration(stub.pto_balance))
315 elseif employee.kind == "SALARIED" then
316 emit(payroll, indent, "salary", money(stub.salary), "leave", stub.leave,
317 "deduction", money(stub.deduction))
318 else
319 emit(payroll, indent, "sales", money(stub.sales),
320 "commission", money(stub.commission))
321 end
322 for _, field in ipairs({
323 "earnings", "bonus", "gross", "retirement", "health",
324 "federal", "supplemental", "state", "social", "medicare"
325 }) do
326 emit(payroll, indent, field == "social" and "social-security" or field,
327 money(stub[field]))
328 end
329 emit(payroll, indent, "garnishment", money(stub.garnishment),
330 "balance", money(stub.garnishment_balance))
331 emit(payroll, indent, "net", money(stub.net))
332 emit(payroll, indent, "employer", "social-security", money(stub.employer_social),
333 "medicare", money(stub.employer_medicare),
334 "unemployment", money(stub.unemployment), "match", money(stub.match))
335end

342local function show_hours(payroll, employee, period)
343 emit(payroll, 0, "HOURS", employee.id, "period", period)
344 local found = false
345 for day = 14 * period - 13, 14 * period do
346 local shifts = employee.shifts[day] or {}
347 if #shifts > 0 or employee.pto[day] then
348 local sorted = {}
349 for _, shift in ipairs(shifts) do sorted[#sorted + 1] = shift end
350 table.sort(sorted, function(a, b) return a[1] < b[1] end)
351 local worked, regular, overtime, double, night, pto = daily(payroll, employee, day)
352 local parts = {"day", day, weekday(day)}
353 if payroll.holidays[day] then parts[#parts + 1] = "holiday" end
354 for _, shift in ipairs(sorted) do
355 parts[#parts + 1] = clock(shift[1]) .. "-" .. clock(shift[2])
356 end
357 parts[#parts + 1] = "total"
358 parts[#parts + 1] = duration(worked)
359 local more = hours_parts({regular, overtime, double, night, pto})
360 for _, part in ipairs(more) do parts[#parts + 1] = part end
361 emit(payroll, 2, table.unpack(parts))
362 found = true
363 end
364 end
365 if not found then emit(payroll, 2, "(none)") end
366 for _, week in ipairs({2 * period - 1, 2 * period}) do
367 local values, overtime = weekly(payroll, employee, week)
368 local parts = {"week", week}
369 for _, item in ipairs(hours_parts(values)) do parts[#parts + 1] = item end
370 parts[#parts + 1] = "weekly-ot"
371 parts[#parts + 1] = duration(overtime)
372 emit(payroll, 2, table.unpack(parts))
373 end
374 emit(payroll, 2, "total", table.unpack(hours_parts(breakdown(payroll, employee, period))))
375end

377local function run(payroll, period)
378 payroll.closed = period
379 emit(payroll, 0, "RUN", period, "days", (14 * period - 13) .. "-" .. (14 * period))
380 local paid, skipped, stubs = 0, 0, {}
381 for _, id in ipairs(sorted_keys(payroll.employees)) do
382 local employee = payroll.employees[id]
383 local stub, balances = calculate(payroll, employee, period)
384 if not stub then
385 if employee.active then skipped = skipped + 1 end
386 else
387 print_stub(payroll, employee, period, stub, employee.department)
388 employee.bonus = 0
389 employee.pto_balance = stub.pto_balance
390 for i, balance in ipairs(balances) do employee.garnishments[i][2] = balance end
391 for _, field in ipairs(fields) do
392 employee.ytd[field] = employee.ytd[field] + stub[field]
393 end
394 employee.periods = employee.periods + 1
395 employee.stubs[period] = {stub = stub, department = employee.department}
396 stubs[#stubs + 1] = stub
397 paid = paid + 1
398 end
399 end
400 local total = sum_stubs(stubs)
401 emit(payroll, 2, "TOTAL", "paid", paid, "skipped", skipped,
402 "gross", money(total.gross), "deductions", money(total.gross - total.net),
403 "net", money(total.net), "employer", money(total.employer))
404end

406local function journal(payroll, period)
407 local stubs = {}
408 for _, employee in pairs(payroll.employees) do
409 if employee.stubs[period] then
410 stubs[#stubs + 1] = employee.stubs[period].stub
411 end
412 end
413 local total = sum_stubs(stubs)
414 local rows = {
415 {"debit", "wages", total.gross},
416 {"debit", "employer-tax", total.employer_social + total.employer_medicare + total.unemployment},
417 {"debit", "retirement-match", total.match},
418 {"credit", "retirement", total.retirement + total.match},
419 {"credit", "health", total.health},
420 {"credit", "federal", total.federal + total.supplemental},
421 {"credit", "state", total.state},
422 {"credit", "fica", total.social + total.medicare + total.employer_social + total.employer_medicare},
423 {"credit", "unemployment", total.unemployment},
424 {"credit", "garnishment", total.garnishment},
425 {"credit", "net-pay", total.net}
426 }
427 emit(payroll, 0, "JOURNAL", period, "stubs", #stubs)
428 local debit, credit = 0, 0
429 for _, row in ipairs(rows) do
430 emit(payroll, 2, row[1], row[2], money(row[3]))
431 if row[1] == "debit" then debit = debit + row[3]
432 else credit = credit + row[3] end
433 end
434 emit(payroll, 2, "total", "debit", money(debit), "credit", money(credit))
435end

447local function execute_command(payroll, command, a)
448 if command == "SET" then
449 local name = a[1]
450 if not payroll.settings[name] then fail("unknown setting " .. name) end
451 local value = rate_settings[name] and rate_check(a[2]) or cents(a[2])
452 local format = rate_settings[name] and rate_text or money
453 local previous = payroll.settings[name]
454 payroll.settings[name] = value
455 emit(payroll, 0, "SET", name, format(previous), "->", format(value))
456 elseif command == "BRACKET" or command == "TIER" then
457 local rate = rate_check(a[2])
458 local tab = command == "BRACKET" and payroll.brackets or payroll.tiers
459 local lower = cents(a[1])
460 local action = tab[lower] and "replaced" or "added"
461 tab[lower] = rate
462 emit(payroll, 0, command, money(lower), rate_text(rate), action)
463 elseif command == "EMP" then
464 local id = tonumber(a[1])
465 if payroll.employees[id] then fail("duplicate employee " .. a[1]) end
466 local base = amount_check(a[4])
467 payroll.employees[id] = new_employee(id, a[2], a[3], base)
468 emit(payroll, 0, "EMP", id, a[2], a[3], money(base))
469 elseif command == "RATE" or command == "DEPT" or command == "BENEFIT" or
470 command == "W4" or command == "GARNISH" or command == "TERM" or
471 command == "BONUS" then
472 local employee = employee_of(payroll, a[1], true)
473 if command == "RATE" then
474 local base = amount_check(a[2])
475 local previous = employee.base
476 employee.base = base
477 emit(payroll, 0, "RATE", employee.id, money(previous), "->", money(base))
478 elseif command == "DEPT" then
479 if a[2] == employee.department then fail("same department " .. a[2]) end
480 local previous = employee.department
481 employee.department = a[2]
482 emit(payroll, 0, "DEPT", employee.id, previous, "->", a[2])
483 elseif command == "BENEFIT" then
484 employee.retirement = rate_check(a[2])
485 employee.health = cents(a[3])
486 emit(payroll, 0, "BENEFIT", employee.id, "retirement",
487 rate_text(employee.retirement), "health", money(employee.health))
488 elseif command == "W4" then
489 employee.allowances = tonumber(a[2])
490 employee.extra = cents(a[3])
491 emit(payroll, 0, "W4", employee.id, "allowances", employee.allowances,
492 "extra", money(employee.extra))
493 elseif command == "GARNISH" then
494 local rate = rate_check(a[2])
495 local balance = amount_check(a[3])
496 employee.garnishments[#employee.garnishments + 1] = {rate, balance}
497 emit(payroll, 0, "GARNISH", employee.id, "order", #employee.garnishments,
498 rate_text(rate), "balance", money(balance))
499 elseif command == "TERM" then
500 employee.active = false
501 emit(payroll, 0, "TERM", employee.id)
502 else
503 local amount = amount_check(a[2])
504 employee.bonus = employee.bonus + amount
505 emit(payroll, 0, "BONUS", employee.id, money(amount),
506 "pending", money(employee.bonus))
507 end
508 elseif command == "HOLIDAY" then
509 local day = day_check(payroll, a[1])
510 if payroll.holidays[day] then fail("duplicate holiday " .. a[1]) end
511 payroll.holidays[day] = true
512 emit(payroll, 0, "HOLIDAY", day, weekday(day))
513 elseif command == "TIME" or command == "UNTIME" or command == "PTO" or
514 command == "LEAVE" or command == "SALES" then
515 local kind = command == "LEAVE" and "SALARIED" or
516 command == "SALES" and "COMMISSION" or "HOURLY"
517 local employee = employee_of(payroll, a[1], true, kind)
518 local day = day_check(payroll, a[2])
519 if command == "TIME" then
520 local start_time, end_time = time_check(a[3]), time_check(a[4])
521 if start_time >= end_time then fail("invalid shift " .. a[3] .. "-" .. a[4]) end
522 local shifts = employee.shifts[day] or {}
523 table.sort(shifts, function(x, y) return x[1] < y[1] end)
524 local total = 0
525 for _, shift in ipairs(shifts) do
526 if start_time < shift[2] and shift[1] < end_time then
527 fail("overlapping shift " .. clock(shift[1]) .. "-" .. clock(shift[2]))
528 end
529 total = total + shift[2] - shift[1]
530 end
531 if total + end_time - start_time > 960 then
532 fail("daily limit " .. a[1] .. " " .. a[2])
533 end
534 shifts[#shifts + 1] = {start_time, end_time}
535 employee.shifts[day] = shifts
536 emit(payroll, 0, "TIME", employee.id, "day", day,
537 clock(start_time) .. "-" .. clock(end_time), duration(end_time - start_time))
538 elseif command == "UNTIME" then
539 local start_time = time_check(a[3])
540 local shifts = employee.shifts[day] or {}
541 local found
542 for i, shift in ipairs(shifts) do
543 if shift[1] == start_time then found = i; break end
544 end
545 if not found then fail("no shift " .. a[1] .. " " .. a[2] .. " " .. a[3]) end
546 local shift = table.remove(shifts, found)
547 emit(payroll, 0, "UNTIME", employee.id, "day", day,
548 clock(shift[1]) .. "-" .. clock(shift[2]))
549 elseif command == "PTO" then
550 local hour, minute = a[3]:match("^(%d%d):(%d%d)$")
551 hour, minute = tonumber(hour), tonumber(minute)
552 local length = hour * 60 + minute
553 if minute > 59 or length < 1 or length > 480 then
554 fail("invalid duration " .. a[3])
555 end
556 if employee.pto[day] then fail("duplicate pto " .. a[1] .. " " .. a[2]) end
557 if length > employee.pto_balance then
558 fail("insufficient pto " .. a[1] .. " " .. duration(employee.pto_balance))
559 end
560 employee.pto[day] = length
561 employee.pto_balance = employee.pto_balance - length
562 emit(payroll, 0, "PTO", employee.id, "day", day, duration(length),
563 "balance", duration(employee.pto_balance))
564 elseif command == "LEAVE" then
565 if day % 7 == 0 or day % 7 == 6 or payroll.holidays[day] then
566 fail("not workday " .. a[2])
567 end
568 if employee.leave[day] then fail("duplicate leave " .. a[1] .. " " .. a[2]) end
569 employee.leave[day] = true
570 emit(payroll, 0, "LEAVE", employee.id, "day", day, weekday(day))
571 else
572 local amount = amount_check(a[3])
573 local period = period_of(day)
574 employee.sales[period] = (employee.sales[period] or 0) + amount
575 emit(payroll, 0, "SALES", employee.id, "day", day, money(amount),
576 "period", period, "total", money(employee.sales[period]))
577 end
578 elseif command == "HOURS" then
579 local employee = employee_of(payroll, a[1], false, "HOURLY")
580 show_hours(payroll, employee, period_check(a[2]))
581 elseif command == "RUN" then
582 local period = tonumber(a[1])
583 if period ~= payroll.closed + 1 or period > 26 then
584 fail("invalid period " .. a[1])
585 end
586 run(payroll, period)
587 elseif command == "PREVIEW" then
588 local employee = employee_of(payroll, a[1])
589 local period = payroll.closed + 1
590 local stub = calculate(payroll, employee, period)
591 emit(payroll, 0, "PREVIEW", employee.id, "period", period)
592 if not stub then
593 emit(payroll, 2, "no pay")
594 else
595 print_stub(payroll, employee, period, stub, employee.department)
596 end
597 elseif command == "REPRINT" then
598 local employee = employee_of(payroll, a[1])
599 local period = period_check(a[2])
600 if not employee.stubs[period] then fail("no stub " .. a[1] .. " " .. a[2]) end
601 local entry = employee.stubs[period]
602 emit(payroll, 0, "REPRINT", employee.id, period)
603 print_stub(payroll, employee, period, entry.stub, entry.department)
604 elseif command == "YTD" then
605 local employee = employee_of(payroll, a[1])
606 emit(payroll, 0, "YTD", employee.id, employee.department, employee.kind,
607 employee.active and "active" or "terminated", "periods", employee.periods)
608 local totals = {}
609 for key, value in pairs(employee.ytd) do totals[key] = value end
610 totals.pto_balance = employee.pto_balance
611 totals.garnishment_balance = garnish_balance(employee)
612 stub_block(payroll, employee, totals, 2)
613 emit(payroll, 2, "wages", "federal", money(totals.W),
614 "social-security", money(totals.Y),
615 "medicare", money(totals.F), "unemployment", money(totals.J))
616 emit(payroll, 2, "retirement-room",
617 money(math.max(0, payroll.settings.RETIRECAP - totals.retirement)),
618 "pending-bonus", money(employee.bonus))
619 elseif command == "JOURNAL" then
620 local period = tonumber(a[1])
621 if period < 1 or period > payroll.closed then fail("no run " .. a[1]) end
622 journal(payroll, period)
623 elseif command == "REPORT" then
624 local departments, all_stubs = {}, {}
625 for _, employee in pairs(payroll.employees) do
626 for _, entry in pairs(employee.stubs) do
627 local name = entry.department
628 if not departments[name] then departments[name] = {} end
629 departments[name][#departments[name] + 1] = entry.stub
630 all_stubs[#all_stubs + 1] = entry.stub
631 end
632 end
633 emit(payroll, 0, "REPORT", "period", payroll.closed)
634 for _, name in ipairs(sorted_keys(departments)) do
635 report_line(payroll, name, departments[name])
636 end
637 report_line(payroll, "total", all_stubs)
638 elseif command == "TOP" then
639 local k = tonumber(a[1])
640 local employees = {}
641 for _, employee in pairs(payroll.employees) do
642 if employee.periods > 0 then employees[#employees + 1] = employee end
643 end
644 table.sort(employees, function(x, y)
645 if x.ytd.gross ~= y.ytd.gross then return x.ytd.gross > y.ytd.gross end
646 return x.id < y.id
647 end)
648 emit(payroll, 0, "TOP", k)
649 for i = 1, math.min(k, #employees) do
650 local employee = employees[i]
651 emit(payroll, 2, i, employee.id, employee.department,
652 "gross", money(employee.ytd.gross), "net", money(employee.ytd.net))
653 end
654 if #employees == 0 then emit(payroll, 2, "(none)") end
655 elseif command == "EMPLOYEES" then
656 emit(payroll, 0, "EMPLOYEES")
657 local ids = sorted_keys(payroll.employees)
658 for _, id in ipairs(ids) do
659 local employee = payroll.employees[id]
660 emit(payroll, 2, id, employee.department, employee.kind,
661 money(employee.base), employee.active and "active" or "terminated",
662 "retirement", rate_text(employee.retirement), "health", money(employee.health),
663 "allowances", employee.allowances, "extra", money(employee.extra),
664 "bonus", money(employee.bonus))
665 for i, order in ipairs(employee.garnishments) do
666 emit(payroll, 4, "garnish", i, rate_text(order[1]), "balance", money(order[2]))
667 end
668 end
669 if #ids == 0 then emit(payroll, 2, "(none)") end
670 elseif command == "CONFIG" then
671 emit(payroll, 0, "CONFIG")
672 for _, name in ipairs(setting_order) do
673 local value = payroll.settings[name]
674 emit(payroll, 2, name, rate_settings[name] and rate_text(value) or money(value))
675 end
676 for _, entry in ipairs({
677 {"bracket", payroll.brackets}, {"tier", payroll.tiers}
678 }) do
679 for _, lower in ipairs(sorted_keys(entry[2])) do
680 emit(payroll, 2, entry[1], money(lower), rate_text(entry[2][lower]))
681 end
682 end
683 end
684end

686local payroll = new_payroll()
687local count = tonumber(io.read("*l"))
688for _ = 1, count do
689 local line = io.read("*l")
690 local words = {}
691 for token in line:gmatch("%S+") do words[#words + 1] = token end
692 local command = words[1]
693 local args = {}
694 for i = 2, #words do args[#args + 1] = words[i] end
695 local ok, reason = pcall(execute_command, payroll, command, args)
696 if not ok then
697 if type(reason) == "table" and reason.payroll_error then
698 emit(payroll, 0, "ERROR", reason.payroll_error)
699 else
700 error(reason)
701 end
702 end
703end
704io.write(table.concat(payroll.output, "\n"), "\n")
