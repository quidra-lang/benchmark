28func cents(_ text: String) -> Int {
29 let parts = text.split(separator: ".")
30 return Int(parts[0])! * 100 + Int(parts[1])!
31}
32
33func money(_ value: Int) -> String {
34 "\(value / 100)." + String(format: "%02d", value % 100)
35}
36
37func rateText(_ value: Int) -> String { money(value) + "%" }

39func duration(_ minutes: Int) -> String {
40 "\(minutes / 60):" + String(format: "%02d", minutes % 60)
41}
42
43func clock(_ minutes: Int) -> String {
44 String(format: "%02d:%02d", minutes / 60, minutes % 60)

47func weekday(_ day: Int) -> String { weekdays[(day - 1) % 7] }
48func periodOf(_ day: Int) -> Int { (day - 1) / 14 + 1 }

50func emptyTotals() -> [String: Int] {
51 Dictionary(uniqueKeysWithValues: fields.map { ($0, 0) })

54func sumStubs(_ stubs: [[String: Int]]) -> [String: Int] {
55 var totals = emptyTotals()
56 for stub in stubs {
57 for field in fields {
58 totals[field, default: 0] += stub[field, default: 0]
59 }
60 }
61 return totals
62}

64func progressive(_ table: [Int: Int], _ amount: Int) -> Int {
65 let pairs = table.sorted { $0.key < $1.key }
66 var numerator = 0
67 for i in pairs.indices {
68 let lower = pairs[i].key
69 if lower >= amount { break }
70 let upper = i + 1 < pairs.count ? pairs[i + 1].key : amount
71 numerator += pairs[i].value * (min(amount, upper) - lower)
72 }
73 return numerator / 10000
74}

168 private func employee(_ text: String, active: Bool = false,
169 kind: String? = nil) throws -> Employee {
170 guard let person = employees[Int(text)!] else {
171 throw PayrollError("unknown employee \(text)")
172 }
173 if active && !person.active { throw PayrollError("terminated employee \(text)") }
174 if let kind, person.kind != kind {
175 throw PayrollError("not \(kind.lowercased()) \(text)")
176 }
177 return person
178 }

180 private func checkDay(_ text: String) throws -> Int {
181 let day = Int(text)!
182 if !(1...364).contains(day) { throw PayrollError("invalid day \(text)") }
183 let period = periodOf(day)
184 if period <= closed { throw PayrollError("period closed \(period)") }
185 return day
186 }

188 private func checkPeriod(_ text: String) throws -> Int {
189 let period = Int(text)!
190 if !(1...26).contains(period) { throw PayrollError("invalid period \(text)") }
191 return period
192 }

195 let rate = cents(text)

197 return rate

201 let amount = cents(text)

203 return amount

206 private func checkTime(_ text: String) throws -> Int {
207 let parts = text.split(separator: ":")
208 let hour = Int(parts[0])!, minute = Int(parts[1])!
209 if hour > 23 || minute > 59 { throw PayrollError("invalid time \(text)") }
210 return hour * 60 + minute
211 }

213 private func daily(_ employee: Employee, _ day: Int) -> Daily {
214 let shifts = employee.shifts[day] ?? []
215 let worked = shifts.reduce(0) { $0 + $1.end - $1.start }
216 let night = shifts.reduce(0) { count, shift in
217 count + max(0, min(shift.end, 360) - shift.start) +
218 max(0, shift.end - max(shift.start, 1320))
219 }
220 let pto = employee.pto[day] ?? 0
221 if day % 7 == 0 || holidays.contains(day) {
222 return Daily(worked: worked, regular: 0, overtime: 0,
223 double: worked, night: night, pto: pto)
224 }
225 return Daily(worked: worked, regular: min(worked, 480),
226 overtime: min(max(0, worked - 480), 240),
227 double: max(0, worked - 720), night: night, pto: pto)
228 }

230 private func weekly(_ employee: Employee, _ week: Int) -> ([Int], Int) {
231 var totals = [Int](repeating: 0, count: 5)
232 for day in (7 * week - 6)...(7 * week) {
233 let values = daily(employee, day).classified
234 for i in totals.indices { totals[i] += values[i] }
235 }
236 let overtime = max(0, totals[0] - 2400)
237 totals[0] -= overtime
238 totals[1] += overtime
239 return (totals, overtime)
240 }

242 private func breakdown(_ employee: Employee, _ period: Int) -> [Int] {
243 let first = weekly(employee, 2 * period - 1).0
244 let second = weekly(employee, 2 * period).0
245 return zip(first, second).map { $0.0 + $0.1 }
246 }

248 private func calculate(_ employee: Employee, _ period: Int) -> Stub? {
249 var stub = Stub()
250 if employee.kind == "HOURLY" {
251 let hours = breakdown(employee, period)
252 for i in 0..<5 { stub[fields[i]] = hours[i] }
253 let weighted = 2 * (hours[0] + hours[4]) + 3 * hours[1] + 4 * hours[2]
254 stub["earnings"] = (weighted * employee.base +
255 2 * hours[3] * settings["NIGHT"]!) / 120
256 } else if employee.kind == "SALARIED" {
257 if employee.active {
258 stub["salary"] = employee.base / 26
259 stub["leave"] = employee.leave.filter {
260 periodOf($0) == period && !holidays.contains($0)
261 }.count
262 stub["deduction"] = stub["leave"] * (employee.base / 260)
263 stub["earnings"] = stub["salary"] - stub["deduction"]
264 }
265 } else {
266 stub["sales"] = employee.sales[period] ?? 0
267 stub["commission"] = progressive(tiers, stub["sales"])
268 stub["earnings"] = stub["commission"] + (employee.active ? employee.base : 0)
269 }
270
271 stub["bonus"] = employee.bonus
272 let gross = stub["earnings"] + stub["bonus"]
273 if gross == 0 { return nil }
274 stub["gross"] = gross
275 var remaining = gross
276
277 func deduct(_ field: String, _ amount: Int) {
278 let value = min(amount, remaining)
279 stub[field] = value
280 remaining -= value
281 }
282
283 let retirementRoom = max(0, settings["RETIRECAP"]! - employee.ytd["retirement"]!)
284 deduct("retirement", min(gross * employee.retirement / 10000, retirementRoom))
285 deduct("health", employee.health)
286 stub["W"] = remaining
287 let baseTaxable = max(0, stub["earnings"] - stub["retirement"] - stub["health"])
288 let allowanceTaxable = max(0, baseTaxable - employee.allowances * settings["ALLOWANCE"]!)
289 deduct("federal", progressive(brackets, allowanceTaxable) + employee.extra)
290 deduct("supplemental", (stub["W"] - baseTaxable) * settings["SUPP"]! / 10000)
291 deduct("state", stub["W"] * settings["STATE"]! / 10000)
292
293 let medicareWages = gross - stub["health"]
294 let socialWages = min(medicareWages,
295 max(0, settings["SSBASE"]! - employee.ytd["Y"]!))
296 let additional = max(0, min(medicareWages,
297 employee.ytd["F"]! + medicareWages - settings["ADDBASE"]!))
298 stub["F"] = medicareWages
299 stub["Y"] = socialWages
300 deduct("social", socialWages * settings["SSRATE"]! / 10000)
301 deduct("medicare", medicareWages * settings["MEDRATE"]! / 10000 +
302 additional * settings["ADDMED"]! / 10000)
303
304 let disposable = remaining
305 let cap = disposable * settings["GARNCAP"]! / 10000
306 var garnished = 0
307 for order in employee.garnishments {
308 let taken = min(disposable * order.rate / 10000,
309 order.balance, cap - garnished)
310 garnished += taken
311 stub.balances.append(order.balance - taken)
312 }
313 stub["garnishment"] = garnished
314 stub.garnishmentBalance = stub.balances.reduce(0, +)
315 stub["net"] = remaining - garnished
316 stub["employer_social"] = socialWages * settings["SSRATE"]! / 10000
317 stub["employer_medicare"] = medicareWages * settings["MEDRATE"]! / 10000
318 stub["J"] = min(medicareWages, max(0, settings["UIBASE"]! - employee.ytd["J"]!))
319 stub["unemployment"] = stub["J"] * settings["UIRATE"]! / 10000
320 let eligibleMatch = min(stub["retirement"], gross * settings["MATCHMAX"]! / 10000)
321 stub["match"] = eligibleMatch * settings["MATCH"]! / 10000
322 stub["employer"] = stub["employer_social"] + stub["employer_medicare"] +
323 stub["unemployment"] + stub["match"]
324 if employee.kind == "HOURLY" {
325 stub["accrued"] = min(960 - employee.ptoBalance,
326 (stub["R"] + stub["O"] + stub["D"]) / 20)
327 }
328 stub.ptoBalance = employee.ptoBalance + stub["accrued"]
329 return stub
330 }

332 private func hoursParts(_ values: [Int]) -> [Any] {
333 let labels = ["reg", "ot", "dbl", "night", "pto"]
334 var parts: [Any] = []
335 for (label, minutes) in zip(labels, values) {
336 parts += [label, duration(minutes)]
337 }
338 return parts
339 }

341 private func stubBlock(_ employee: Employee, _ values: [String: Int],
342 ptoBalance: Int, garnishmentBalance: Int, indent: Int) {
343 func value(_ field: String) -> Int { values[field, default: 0] }
344 if employee.kind == "HOURLY" {
345 putParts(indent, ["hours"] + hoursParts(["R", "O", "D", "N", "P"].map(value)))
346 put(indent, "pto", "accrued", duration(value("accrued")),
347 "balance", duration(ptoBalance))
348 } else if employee.kind == "SALARIED" {
349 put(indent, "salary", money(value("salary")), "leave", value("leave"),
350 "deduction", money(value("deduction")))
351 } else {
352 put(indent, "sales", money(value("sales")),
353 "commission", money(value("commission")))
354 }
355 for field in [
356 "earnings", "bonus", "gross", "retirement", "health", "federal",
357 "supplemental", "state", "social", "medicare"
358 ] {
359 put(indent, field == "social" ? "social-security" : field, money(value(field)))
360 }
361 put(indent, "garnishment", money(value("garnishment")),
362 "balance", money(garnishmentBalance))
363 put(indent, "net", money(value("net")))
364 put(indent, "employer", "social-security", money(value("employer_social")),
365 "medicare", money(value("employer_medicare")),
366 "unemployment", money(value("unemployment")),
367 "match", money(value("match")))
368 }

377 private func showHours(_ employee: Employee, _ period: Int) {
378 put(0, "HOURS", employee.id, "period", period)
379 var found = false
380 for day in (14 * period - 13)...(14 * period) {
381 let shifts = (employee.shifts[day] ?? []).sorted { $0.start < $1.start }
382 if shifts.isEmpty && employee.pto[day] == nil { continue }
383 let daily = daily(employee, day)
384 var parts: [Any] = ["day", day, weekday(day)]
385 if holidays.contains(day) { parts.append("holiday") }
386 for shift in shifts {
387 parts.append("\(clock(shift.start))-\(clock(shift.end))")
388 }
389 parts += ["total", duration(daily.worked)]
390 parts += hoursParts(daily.classified)
391 putParts(2, parts)
392 found = true
393 }
394 if !found { put(2, "(none)") }
395 for week in [2 * period - 1, 2 * period] {
396 let (values, overtime) = weekly(employee, week)
397 putParts(2, ["week", week] + hoursParts(values) +
398 ["weekly-ot", duration(overtime)])
399 }
400 putParts(2, ["total"] + hoursParts(breakdown(employee, period)))
401 }

403 private func run(_ period: Int) {
404 closed = period
405 put(0, "RUN", period, "days", "\(14 * period - 13)-\(14 * period)")
406 var paid = 0, skipped = 0
407 var stubs: [[String: Int]] = []
408 for id in employees.keys.sorted() {
409 let employee = employees[id]!
410 guard let stub = calculate(employee, period) else {
411 if employee.active { skipped += 1 }
412 continue
413 }
414 printStub(employee, period, stub, employee.department)
415 employee.bonus = 0
416 employee.ptoBalance = stub.ptoBalance
417 for i in employee.garnishments.indices {
418 employee.garnishments[i].balance = stub.balances[i]
419 }
420 for field in fields {
421 employee.ytd[field, default: 0] += stub[field]
422 }
423 employee.periods += 1
424 employee.stubs[period] = StubRecord(stub: stub, department: employee.department)
425 stubs.append(stub.values)
426 paid += 1
427 }
428 let totals = sumStubs(stubs)
429 let gross = totals["gross"]!
430 let net = totals["net"]!
431 put(2, "TOTAL", "paid", paid, "skipped", skipped, "gross", money(gross),
432 "deductions", money(gross - net), "net", money(net),
433 "employer", money(totals["employer"]!))
434 }

475 private func command(_ cmd: String, _ a: [String]) throws {
476 switch cmd {
477 case "SET":
478 guard let previous = settings[a[0]] else {
479 throw PayrollError("unknown setting \(a[0])")
480 }
481 let rate = rateSettings.contains(a[0])
482 let value = try rate ? checkRate(a[1]) : cents(a[1])
483 settings[a[0]] = value
484 put(0, "SET", a[0], rate ? rateText(previous) : money(previous),
485 "->", rate ? rateText(value) : money(value))
486
487 case "BRACKET", "TIER":
488 let rate = try checkRate(a[1])
489 let lower = cents(a[0])
490 let replaced = cmd == "BRACKET" ? brackets[lower] != nil : tiers[lower] != nil
491 if cmd == "BRACKET" { brackets[lower] = rate }
492 else { tiers[lower] = rate }
493 put(0, cmd, money(lower), rateText(rate), replaced ? "replaced" : "added")
494
495 case "EMP":
496 let id = Int(a[0])!
497 if employees[id] != nil { throw PayrollError("duplicate employee \(a[0])") }
498 let amount = try checkAmount(a[3])
499 employees[id] = Employee(id: id, department: a[1], kind: a[2], base: amount)
500 put(0, "EMP", id, a[1], a[2], money(amount))
501
502 case "RATE", "DEPT", "BENEFIT", "W4", "GARNISH", "TERM", "BONUS":
503 let person = try employee(a[0], active: true)
504 switch cmd {
505 case "RATE":
506 let amount = try checkAmount(a[1])
507 let previous = person.base
508 person.base = amount
509 put(0, "RATE", person.id, money(previous), "->", money(amount))
510 case "DEPT":
511 if person.department == a[1] { throw PayrollError("same department \(a[1])") }
512 let old = person.department
513 person.department = a[1]
514 put(0, "DEPT", person.id, old, "->", a[1])
515 case "BENEFIT":
516 let rate = try checkRate(a[1])
517 person.retirement = rate
518 person.health = cents(a[2])
519 put(0, "BENEFIT", person.id, "retirement", rateText(rate),
520 "health", money(person.health))
521 case "W4":
522 person.allowances = Int(a[1])!
523 person.extra = cents(a[2])
524 put(0, "W4", person.id, "allowances", person.allowances,
525 "extra", money(person.extra))
526 case "GARNISH":
527 let rate = try checkRate(a[1])
528 let amount = try checkAmount(a[2])
529 person.garnishments.append(Garnishment(rate: rate, balance: amount))
530 put(0, "GARNISH", person.id, "order", person.garnishments.count,
531 rateText(rate), "balance", money(amount))
532 case "TERM":
533 person.active = false
534 put(0, "TERM", person.id)
535 default:
536 let amount = try checkAmount(a[1])
537 person.bonus += amount
538 put(0, "BONUS", person.id, money(amount), "pending", money(person.bonus))
539 }
540
541 case "HOLIDAY":
542 let day = try checkDay(a[0])
543 if holidays.contains(day) { throw PayrollError("duplicate holiday \(a[0])") }
544 holidays.insert(day)
545 put(0, "HOLIDAY", day, weekday(day))
546
547 case "TIME", "UNTIME", "PTO", "LEAVE", "SALES":
548 let kind = cmd == "LEAVE" ? "SALARIED" :
549 cmd == "SALES" ? "COMMISSION" : "HOURLY"
550 let person = try employee(a[0], active: true, kind: kind)
551 let day = try checkDay(a[1])
552 switch cmd {
553 case "TIME":
554 let start = try checkTime(a[2])
555 let end = try checkTime(a[3])
556 if start >= end { throw PayrollError("invalid shift \(a[2])-\(a[3])") }
557 let shifts = (person.shifts[day] ?? []).sorted { $0.start < $1.start }
558 for shift in shifts {
559 if start < shift.end && shift.start < end {
560 throw PayrollError("overlapping shift \(clock(shift.start))-\(clock(shift.end))")
561 }
562 }
563 if shifts.reduce(0, { $0 + $1.end - $1.start }) + end - start > 960 {
564 throw PayrollError("daily limit \(a[0]) \(a[1])")
565 }
566 person.shifts[day, default: []].append(Shift(start: start, end: end))
567 put(0, "TIME", person.id, "day", day,
568 "\(clock(start))-\(clock(end))", duration(end - start))
569 case "UNTIME":
570 let start = try checkTime(a[2])
571 let oldShifts = person.shifts[day] ?? []
572 guard let index = oldShifts.firstIndex(where: { $0.start == start }) else {
573 throw PayrollError("no shift \(a[0]) \(a[1]) \(a[2])")
574 }
575 let removed = oldShifts[index]
576 person.shifts[day]!.remove(at: index)
577 put(0, "UNTIME", person.id, "day", day,
578 "\(clock(removed.start))-\(clock(removed.end))")
579 case "PTO":
580 let parts = a[2].split(separator: ":")
581 let hour = Int(parts[0])!, minute = Int(parts[1])!
582 let length = hour * 60 + minute
583 if minute > 59 || length < 1 || length > 480 {
584 throw PayrollError("invalid duration \(a[2])")
585 }
586 if person.pto[day] != nil { throw PayrollError("duplicate pto \(a[0]) \(a[1])") }
587 if length > person.ptoBalance {
588 throw PayrollError("insufficient pto \(a[0]) \(duration(person.ptoBalance))")
589 }
590 person.pto[day] = length
591 person.ptoBalance -= length
592 put(0, "PTO", person.id, "day", day, duration(length),
593 "balance", duration(person.ptoBalance))
594 case "LEAVE":
595 if day % 7 == 0 || day % 7 == 6 || holidays.contains(day) {
596 throw PayrollError("not workday \(a[1])")
597 }
598 if person.leave.contains(day) {
599 throw PayrollError("duplicate leave \(a[0]) \(a[1])")
600 }
601 person.leave.insert(day)
602 put(0, "LEAVE", person.id, "day", day, weekday(day))
603 default:
604 let amount = try checkAmount(a[2])
605 let period = periodOf(day)
606 person.sales[period, default: 0] += amount
607 put(0, "SALES", person.id, "day", day, money(amount),
608 "period", period, "total", money(person.sales[period]!))
609 }
610
611 case "HOURS":
612 let person = try employee(a[0], kind: "HOURLY")
613 showHours(person, try checkPeriod(a[1]))
614
615 case "RUN":
616 let period = Int(a[0])!
617 if period != closed + 1 || period > 26 {
618 throw PayrollError("invalid period \(a[0])")
619 }
620 run(period)
621
622 case "PREVIEW":
623 let person = try employee(a[0])
624 let period = closed + 1
625 let stub = calculate(person, period)
626 put(0, "PREVIEW", person.id, "period", period)
627 if let stub { printStub(person, period, stub, person.department) }
628 else { put(2, "no pay") }
629
630 case "REPRINT":
631 let person = try employee(a[0])
632 let period = try checkPeriod(a[1])
633 guard let record = person.stubs[period] else {
634 throw PayrollError("no stub \(a[0]) \(a[1])")
635 }
636 put(0, "REPRINT", person.id, period)
637 printStub(person, period, record.stub, record.department)
638
639 case "YTD":
640 let person = try employee(a[0])
641 put(0, "YTD", person.id, person.department, person.kind,
642 person.active ? "active" : "terminated", "periods", person.periods)
643 stubBlock(person, person.ytd, ptoBalance: person.ptoBalance,
644 garnishmentBalance: person.garnishmentBalance, indent: 2)
645 put(2, "wages", "federal", money(person.ytd["W"]!),
646 "social-security", money(person.ytd["Y"]!),
647 "medicare", money(person.ytd["F"]!),
648 "unemployment", money(person.ytd["J"]!))
649 put(2, "retirement-room",
650 money(max(0, settings["RETIRECAP"]! - person.ytd["retirement"]!)),
651 "pending-bonus", money(person.bonus))
652
653 case "JOURNAL":
654 let period = Int(a[0])!
655 if period < 1 || period > closed { throw PayrollError("no run \(a[0])") }
656 journal(period)
657
658 case "REPORT":
659 var departments: [String: [[String: Int]]] = [:]
660 var all: [[String: Int]] = []
661 for person in employees.values {
662 for record in person.stubs.values {
663 departments[record.department, default: []].append(record.stub.values)
664 all.append(record.stub.values)
665 }
666 }
667 put(0, "REPORT", "period", closed)
668 for department in departments.keys.sorted() {
669 reportLine(department, departments[department]!)
670 }
671 reportLine("total", all)
672
673 case "TOP":
674 let count = Int(a[0])!
675 let ranking = employees.values.filter { $0.periods > 0 }
676 .sorted {
677 let left = $0.ytd["gross"]!, right = $1.ytd["gross"]!
678 return left == right ? $0.id < $1.id : left > right
679 }
680 put(0, "TOP", count)
681 for (index, person) in ranking.prefix(count).enumerated() {
682 put(2, index + 1, person.id, person.department, "gross",
683 money(person.ytd["gross"]!), "net", money(person.ytd["net"]!))
684 }
685 if ranking.isEmpty { put(2, "(none)") }
686
687 case "EMPLOYEES":
688 put(0, "EMPLOYEES")
689 for id in employees.keys.sorted() {
690 let person = employees[id]!
691 put(2, person.id, person.department, person.kind, money(person.base),
692 person.active ? "active" : "terminated",
693 "retirement", rateText(person.retirement),
694 "health", money(person.health), "allowances", person.allowances,
695 "extra", money(person.extra), "bonus", money(person.bonus))
696 for (index, order) in person.garnishments.enumerated() {
697 put(4, "garnish", index + 1, rateText(order.rate),
698 "balance", money(order.balance))
699 }
700 }
701 if employees.isEmpty { put(2, "(none)") }
702
703 case "CONFIG":
704 put(0, "CONFIG")
705 for (name, _) in initialSettings {
706 let value = settings[name]!
707 put(2, name, rateSettings.contains(name) ? rateText(value) : money(value))
708 }
709 for (lower, rate) in brackets.sorted(by: { $0.key < $1.key }) {
710 put(2, "bracket", money(lower), rateText(rate))
711 }
712 for (lower, rate) in tiers.sorted(by: { $0.key < $1.key }) {
713 put(2, "tier", money(lower), rateText(rate))
714 }
715 default:
716 break
717 }
718 }

720 func execute(_ line: String) {
721 let words = line.split(separator: " ").map(String.init)
722 guard let command = words.first else { return }
723 do {
724 try self.command(command, Array(words.dropFirst()))
725 } catch let error as PayrollError {
726 put(0, "ERROR", error.message)
727 } catch {
728 fatalError("Unexpected payroll error: \(error)")
729 }
730 }
731}

734let payroll = Payroll()
735for _ in 0..<count {
736 payroll.execute(readLine()!)
737}
738for line in payroll.output { print(line) }
