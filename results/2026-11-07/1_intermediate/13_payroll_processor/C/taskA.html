216static void set_bracket(Table *table, Amount lower, Amount rate, int *replaced) {
217 int i = 0;
218 while (i < table->count && table->rows[i].bound < lower) ++i;
219 if (i < table->count && table->rows[i].bound == lower) {
220 *replaced = 1;
221 table->rows[i].rate = rate;
222 } else {
223 *replaced = 0;
224 memmove(table->rows + i + 1, table->rows + i,
225 (table->count - i) * sizeof(Bracket));
226 table->rows[i].bound = lower;
227 table->rows[i].rate = rate;
228 ++table->count;
229 }
230}

232static Amount progressive(const Table *table, Amount value) {
233 Amount numerator = 0;
234 for (int i = 0; i < table->count; ++i) {
235 Amount lower = table->rows[i].bound;
236 if (lower >= value) break;
237 Amount upper = i + 1 < table->count ?
238 minimum(value, table->rows[i + 1].bound) : value;
239 numerator += table->rows[i].rate * (upper - lower);
240 }
241 return numerator / 10000;
242}

244static Daily daily(const Employee *employee, int day) {
245 Daily result = {0};
246 if (day < 1 || day > 364) return result;
247 for (const Shift *shift = employee->shifts[day]; shift; shift = shift->next) {
248 int start = shift->start, end = shift->end;
249 result.worked += end - start;
250 result.values[3] += maximum(0, minimum(end, 360) - start);
251 result.values[3] += maximum(0, end - maximum(start, 1320));
252 }
253 result.values[4] = employee->pto[day];
254 if (day % 7 == 0 || holidays[day]) {
255 result.values[2] = result.worked;
256 } else {
257 result.values[0] = minimum(result.worked, 480);
258 result.values[1] = minimum(maximum(result.worked - 480, 0), 240);
259 result.values[2] = maximum(result.worked - 720, 0);
260 }
261 return result;
262}

264static Week weekly(const Employee *employee, int week) {
265 Week result = {0};
266 for (int day = 7 * week - 6; day <= 7 * week; ++day) {
267 Daily today = daily(employee, day);
268 for (int i = 0; i < 5; ++i) result.values[i] += today.values[i];
269 }
270 result.weekly_ot = maximum(0, result.values[0] - 2400);
271 result.values[0] -= result.weekly_ot;
272 result.values[1] += result.weekly_ot;
273 return result;
274}

287static int calculate(const Employee *employee, int period, Stub *stub, Amount *post_balances) {
288 memset(stub, 0, sizeof(*stub));
289 Amount *v = stub->v;
290 if (employee->kind == HOURLY) {
291 breakdown(employee, period, v);
292 Amount weighted = 2 * (v[REG] + v[PTOMIN]) +
293 3 * v[OT] + 4 * v[DBL];
294 v[EARNINGS] = (weighted * employee->base +
295 2 * v[NIGHTMIN] * settings[NIGHT]) / 120;
296 } else if (employee->kind == SALARIED) {
297 if (employee->active) {
298 v[SALARY] = employee->base / 26;
299 for (int day = 1; day <= 364; ++day)
300 if (employee->leave[day] && period_of(day) == period && !holidays[day])
301 ++v[LEAVE_DAYS];
302 v[LEAVE_DEDUCTION] = v[LEAVE_DAYS] * (employee->base / 260);
303 v[EARNINGS] = v[SALARY] - v[LEAVE_DEDUCTION];
304 }
305 } else {
306 v[SALES_AMOUNT] = employee->sales[period];
307 v[COMMISSION_AMOUNT] = progressive(&tiers, v[SALES_AMOUNT]);
308 v[EARNINGS] = v[COMMISSION_AMOUNT] + (employee->active ? employee->base : 0);
309 }
310 v[BONUS_AMOUNT] = employee->bonus;
311 v[GROSS] = v[EARNINGS] + v[BONUS_AMOUNT];
312 if (v[GROSS] == 0) return 0;
313
314 Amount remaining = v[GROSS];
315 Amount room = maximum(0, settings[RETIRECAP] - employee->ytd[RETIREMENT]);
316 deduct(stub, RETIREMENT, minimum(v[GROSS] * employee->retirement / 10000, room),
317 &remaining);
318 deduct(stub, HEALTH, employee->health, &remaining);
319 v[W] = remaining;
320 Amount base_taxable = maximum(0, v[EARNINGS] - v[RETIREMENT] - v[HEALTH]);
321 Amount allowance_taxable = maximum(0, base_taxable -
322 employee->allowances * settings[ALLOWANCE]);
323 deduct(stub, FEDERAL, progressive(&brackets, allowance_taxable) + employee->extra,
324 &remaining);
325 deduct(stub, SUPPLEMENTAL, (v[W] - base_taxable) * settings[SUPP] / 10000,
326 &remaining);
327 deduct(stub, STATE_TAX, v[W] * settings[STATE] / 10000, &remaining);
328
329 v[F] = v[GROSS] - v[HEALTH];
330 v[Y] = minimum(v[F], maximum(0, settings[SSBASE] - employee->ytd[Y]));
331 Amount additional = maximum(0, minimum(v[F],
332 employee->ytd[F] + v[F] - settings[ADDBASE]));
333 deduct(stub, SOCIAL, v[Y] * settings[SSRATE] / 10000, &remaining);
334 deduct(stub, MEDICARE, v[F] * settings[MEDRATE] / 10000 +
335 additional * settings[ADDMED] / 10000, &remaining);
336
337 Amount disposable = remaining;
338 Amount cap = disposable * settings[GARNCAP] / 10000;
339 for (int i = 0; i < employee->garnishes; ++i) {
340 Garnish order = employee->orders[i];
341 Amount amount = minimum(disposable * order.rate / 10000,
342 minimum(order.balance, cap - v[GARNISHMENT]));
343 v[GARNISHMENT] += amount;
344 post_balances[i] = order.balance - amount;
345 v[GARNISH_BALANCE] += post_balances[i];
346 }
347 v[NET] = remaining - v[GARNISHMENT];
348 v[EMPLOYER_SOCIAL] = v[Y] * settings[SSRATE] / 10000;
349 v[EMPLOYER_MEDICARE] = v[F] * settings[MEDRATE] / 10000;
350 v[J] = minimum(v[F], maximum(0, settings[UIBASE] - employee->ytd[J]));
351 v[UNEMPLOYMENT] = v[J] * settings[UIRATE] / 10000;
352 Amount match_base = minimum(v[RETIREMENT], v[GROSS] * settings[MATCHMAX] / 10000);
353 v[RETIRE_MATCH] = match_base * settings[MATCH] / 10000;
354 v[EMPLOYER] = v[EMPLOYER_SOCIAL] + v[EMPLOYER_MEDICARE] +
355 v[UNEMPLOYMENT] + v[RETIRE_MATCH];
356 if (employee->kind == HOURLY)
357 v[ACCRUED] = minimum(960 - employee->pto_balance,
358 (v[REG] + v[OT] + v[DBL]) / 20);
359 v[PTO_BALANCE] = employee->pto_balance + v[ACCRUED];
360 return 1;
361}

369static void stub_block(const Employee *employee, const Stub *stub, int indent) {
370 const Amount *v = stub->v;
371 if (employee->kind == HOURLY) {
372 printf("%*shours ", indent, "");
373 hours_parts(v);
374 putchar('\n');
375 printf("%*spto accrued %s balance %s\n", indent, "",
376 duration(v[ACCRUED]), duration(v[PTO_BALANCE]));
377 } else if (employee->kind == SALARIED) {
378 printf("%*ssalary %s leave %lld deduction %s\n", indent, "",
379 money(v[SALARY]), v[LEAVE_DAYS], money(v[LEAVE_DEDUCTION]));
380 } else {
381 printf("%*ssales %s commission %s\n", indent, "",
382 money(v[SALES_AMOUNT]), money(v[COMMISSION_AMOUNT]));
383 }
384 for (int i = 0; i < 10; ++i)
385 printf("%*s%s %s\n", indent, "", pay_fields[i], money(v[pay_indexes[i]]));
386 printf("%*sgarnishment %s balance %s\n", indent, "",
387 money(v[GARNISHMENT]), money(v[GARNISH_BALANCE]));
388 printf("%*snet %s\n", indent, "", money(v[NET]));
389 printf("%*semployer social-security %s medicare %s unemployment %s match %s\n",
390 indent, "", money(v[EMPLOYER_SOCIAL]), money(v[EMPLOYER_MEDICARE]),
391 money(v[UNEMPLOYMENT]), money(v[RETIRE_MATCH]));
392}

401static void show_hours(const Employee *employee, int period) {
402 printf("HOURS %d period %d\n", employee->id, period);
403 int found = 0;
404 for (int day = 14 * period - 13; day <= 14 * period; ++day) {

425 printf(" total ");

435static void print_run(int period) {
436 closed_periods = period;
437 printf("RUN %d days %d-%d\n", period, 14 * period - 13, 14 * period);
438 Stub sum = {0};
439 int paid = 0, skipped = 0;
440 for (int id = 1; id <= 9999; ++id) {
441 Employee *employee = employees[id];
442 if (!employee) continue;
443 Stub stub;
444 Amount *balances = calloc((size_t)employee->garnishes + 1, sizeof(Amount));
445 int has_pay = calculate(employee, period, &stub, balances);
446 if (!has_pay) {
447 if (employee->active) ++skipped;
448 free(balances);
449 continue;
450 }
451 print_stub(employee, period, &stub, employee->department);
452 employee->bonus = 0;
453 employee->pto_balance = stub.v[PTO_BALANCE];
454 for (int i = 0; i < employee->garnishes; ++i)
455 employee->orders[i].balance = balances[i];
456 for (int i = 0; i < FIELD_COUNT; ++i)
457 employee->ytd[i] += stub.v[i];
458 ++employee->periods;
459 Stored *saved = &employee->stubs[period];
460 saved->stub = stub;
461 strcpy(saved->department, employee->department);
462 saved->exists = 1;
463 add_stub(&sum, &stub);
464 ++paid;
465 free(balances);
466 }
467 printf(" TOTAL paid %d skipped %d gross %s deductions %s net %s employer %s\n",
468 paid, skipped, money(sum.v[GROSS]), money(sum.v[GROSS] - sum.v[NET]),
469 money(sum.v[NET]), money(sum.v[EMPLOYER]));
470}

529static void report(void) {
530 Department departments[1002] = {0};
531 Stub total = {0};
532 int count = 0, n = 0;
533 for (int id = 1; id <= 9999; ++id) {
534 Employee *employee = employees[id];
535 if (!employee) continue;
536 for (int period = 1; period <= 26; ++period) {
537 Stored *saved = &employee->stubs[period];
538 if (!saved->exists) continue;
539 int index = 0;
540 while (index < n && strcmp(departments[index].name, saved->department))
541 ++index;
542 if (index == n) {
543 strcpy(departments[n].name, saved->department);
544 ++n;
545 }
546 add_stub(&departments[index].sum, &saved->stub);
547 ++departments[index].count;
548 add_stub(&total, &saved->stub);
549 ++count;
550 }
551 }
552 qsort(departments, n, sizeof(departments[0]), compare_department);
553 printf("REPORT period %d\n", closed_periods);
554 for (int i = 0; i < n; ++i)
555 report_line(departments[i].name, &departments[i].sum, departments[i].count);
556 report_line("total", &total, count);
557}

577static int compare_top(const void *a, const void *b) {
578 Employee *left = *(Employee * const *)a;
579 Employee *right = *(Employee * const *)b;
580 if (left->ytd[GROSS] != right->ytd[GROSS])

585 Employee *ranking[1002];

594 printf(" %d %d %s gross %s net %s\n", i + 1, employee->id,
595 employee->department, money(employee->ytd[GROSS]),
596 money(employee->ytd[NET]));
597 }
598 if (!n) puts(" (none)");
599}

634static void execute(char **a) {
635 const char *command = a[0];
636 if (!strcmp(command, "SET")) {
637 int setting = setting_index(a[1]);
638 if (setting < 0) {
639 printf("ERROR unknown setting %s\n", a[1]);
640 return;
641 }
642 Amount value;
643 if (setting <= GARNCAP) {
644 if (!checked_rate(a[2], &value)) return;
645 } else value = cents(a[2]);
646 Amount old = settings[setting];
647 settings[setting] = value;
648 if (setting <= GARNCAP)
649 printf("SET %s %s -> %s\n", a[1], rate_text(old), rate_text(value));
650 else
651 printf("SET %s %s -> %s\n", a[1], money(old), money(value));
652 } else if (!strcmp(command, "BRACKET") || !strcmp(command, "TIER")) {
653 Amount rate;
654 if (!checked_rate(a[2], &rate)) return;
655 Amount lower = cents(a[1]);
656 int replaced;
657 set_bracket(!strcmp(command, "BRACKET") ? &brackets : &tiers,
658 lower, rate, &replaced);
659 printf("%s %s %s %s\n", command, money(lower), rate_text(rate),
660 replaced ? "replaced" : "added");
661 } else if (!strcmp(command, "EMP")) {
662 int id = atoi(a[1]);
663 if (employees[id]) {
664 printf("ERROR duplicate employee %s\n", a[1]);
665 return;
666 }
667 Amount base;
668 if (!checked_amount(a[4], &base)) return;
669 Employee *employee = calloc(1, sizeof(*employee));
670 employee->id = id;
671 employee->active = 1;
672 strcpy(employee->department, a[2]);
673 employee->kind = !strcmp(a[3], "HOURLY") ? HOURLY :
674 !strcmp(a[3], "SALARIED") ? SALARIED : COMMISSION;
675 employee->base = base;
676 employees[id] = employee;
677 printf("EMP %d %s %s %s\n", id, a[2], a[3], money(base));
678 } else if (!strcmp(command, "RATE") || !strcmp(command, "DEPT") ||
679 !strcmp(command, "BENEFIT") || !strcmp(command, "W4") ||
680 !strcmp(command, "GARNISH") || !strcmp(command, "TERM") ||
681 !strcmp(command, "BONUS")) {
682 Employee *employee = require_employee(a[1], 1, -1);
683 if (!employee) return;
684 if (!strcmp(command, "RATE")) {
685 Amount base;
686 if (!checked_amount(a[2], &base)) return;
687 printf("RATE %d %s -> %s\n", employee->id, money(employee->base), money(base));
688 employee->base = base;
689 } else if (!strcmp(command, "DEPT")) {
690 if (!strcmp(employee->department, a[2])) {
691 printf("ERROR same department %s\n", a[2]);
692 return;
693 }
694 printf("DEPT %d %s -> %s\n", employee->id, employee->department, a[2]);
695 strcpy(employee->department, a[2]);
696 } else if (!strcmp(command, "BENEFIT")) {
697 Amount rate;
698 if (!checked_rate(a[2], &rate)) return;
699 employee->retirement = rate;
700 employee->health = cents(a[3]);
701 printf("BENEFIT %d retirement %s health %s\n", employee->id,
702 rate_text(rate), money(employee->health));
703 } else if (!strcmp(command, "W4")) {
704 employee->allowances = atoi(a[2]);
705 employee->extra = cents(a[3]);
706 printf("W4 %d allowances %d extra %s\n", employee->id,
707 employee->allowances, money(employee->extra));
708 } else if (!strcmp(command, "GARNISH")) {
709 Amount rate, amount;
710 if (!checked_rate(a[2], &rate) || !checked_amount(a[3], &amount)) return;
711 if (employee->garnishes == employee->capacity) {
712 employee->capacity = employee->capacity ? employee->capacity * 2 : 4;
713 employee->orders = realloc(employee->orders,
714 sizeof(Garnish) * employee->capacity);
715 }
716 employee->orders[employee->garnishes].rate = rate;
717 employee->orders[employee->garnishes].balance = amount;
718 ++employee->garnishes;
719 printf("GARNISH %d order %d %s balance %s\n", employee->id,
720 employee->garnishes, rate_text(rate), money(amount));
721 } else if (!strcmp(command, "TERM")) {
722 employee->active = 0;
723 printf("TERM %d\n", employee->id);
724 } else {
725 Amount amount;
726 if (!checked_amount(a[2], &amount)) return;
727 employee->bonus += amount;
728 printf("BONUS %d %s pending %s\n", employee->id,
729 money(amount), money(employee->bonus));
730 }
731 } else if (!strcmp(command, "HOLIDAY")) {
732 int day;
733 if (!checked_day(a[1], &day)) return;
734 if (holidays[day]) {
735 printf("ERROR duplicate holiday %s\n", a[1]);
736 return;
737 }
738 holidays[day] = 1;
739 printf("HOLIDAY %d %s\n", day, weekday(day));
740 } else if (!strcmp(command, "TIME") || !strcmp(command, "UNTIME") ||
741 !strcmp(command, "PTO") || !strcmp(command, "LEAVE") ||
742 !strcmp(command, "SALES")) {
743 int kind = !strcmp(command, "LEAVE") ? SALARIED :
744 !strcmp(command, "SALES") ? COMMISSION : HOURLY;
745 Employee *employee = require_employee(a[1], 1, kind);
746 if (!employee) return;
747 int day;
748 if (!checked_day(a[2], &day)) return;
749 if (!strcmp(command, "TIME")) {
750 int start, end;
751 if (!valid_clock(a[3], &start) || !valid_clock(a[4], &end)) return;
752 if (start >= end) {
753 printf("ERROR invalid shift %s-%s\n", a[3], a[4]);
754 return;
755 }
756 Amount worked = 0;
757 for (Shift *shift = employee->shifts[day]; shift; shift = shift->next) {
758 if (start < shift->end && shift->start < end) {
759 printf("ERROR overlapping shift %s-%s\n",
760 clock_text(shift->start), clock_text(shift->end));
761 return;
762 }
763 worked += shift->end - shift->start;
764 }
765 if (worked + end - start > 960) {
766 printf("ERROR daily limit %s %s\n", a[1], a[2]);
767 return;
768 }
769 Shift *shift = malloc(sizeof(*shift));
770 shift->start = start;
771 shift->end = end;
772 Shift **position = &employee->shifts[day];
773 while (*position && (*position)->start < start)
774 position = &(*position)->next;
775 shift->next = *position;
776 *position = shift;
777 printf("TIME %d day %d %s-%s %s\n", employee->id, day,
778 clock_text(start), clock_text(end), duration(end - start));
779 } else if (!strcmp(command, "UNTIME")) {
780 int start;
781 if (!valid_clock(a[3], &start)) return;
782 Shift **position = &employee->shifts[day];
783 while (*position && (*position)->start != start)
784 position = &(*position)->next;
785 if (!*position) {
786 printf("ERROR no shift %s %s %s\n", a[1], a[2], a[3]);
787 return;
788 }
789 Shift *removed = *position;
790 *position = removed->next;
791 printf("UNTIME %d day %d %s-%s\n", employee->id, day,
792 clock_text(removed->start), clock_text(removed->end));
793 free(removed);
794 } else if (!strcmp(command, "PTO")) {
795 int hour = 0, minute = 0;
796 sscanf(a[3], "%d:%d", &hour, &minute);
797 int length = hour * 60 + minute;
798 if (minute > 59 || length < 1 || length > 480) {
799 printf("ERROR invalid duration %s\n", a[3]);
800 return;
801 }
802 if (employee->pto[day]) {
803 printf("ERROR duplicate pto %s %s\n", a[1], a[2]);
804 return;
805 }
806 if (length > employee->pto_balance) {
807 printf("ERROR insufficient pto %s %s\n",
808 a[1], duration(employee->pto_balance));
809 return;
810 }
811 employee->pto[day] = length;
812 employee->pto_balance -= length;
813 printf("PTO %d day %d %s balance %s\n", employee->id, day,
814 duration(length), duration(employee->pto_balance));
815 } else if (!strcmp(command, "LEAVE")) {
816 if (day % 7 == 0 || day % 7 == 6 || holidays[day]) {
817 printf("ERROR not workday %s\n", a[2]);
818 return;
819 }
820 if (employee->leave[day]) {
821 printf("ERROR duplicate leave %s %s\n", a[1], a[2]);
822 return;
823 }
824 employee->leave[day] = 1;
825 printf("LEAVE %d day %d %s\n", employee->id, day, weekday(day));
826 } else {
827 Amount amount;
828 if (!checked_amount(a[3], &amount)) return;
829 int period = period_of(day);
830 employee->sales[period] += amount;
831 printf("SALES %d day %d %s period %d total %s\n", employee->id,
832 day, money(amount), period, money(employee->sales[period]));
833 }
834 } else if (!strcmp(command, "HOURS")) {
835 Employee *employee = require_employee(a[1], 0, HOURLY);
836 if (!employee) return;
837 int period;
838 if (!checked_period(a[2], &period)) return;
839 show_hours(employee, period);
840 } else if (!strcmp(command, "RUN")) {
841 int period = atoi(a[1]);
842 if (period != closed_periods + 1 || period > 26) {
843 printf("ERROR invalid period %s\n", a[1]);
844 return;
845 }
846 print_run(period);
847 } else if (!strcmp(command, "PREVIEW")) {
848 Employee *employee = require_employee(a[1], 0, -1);
849 if (!employee) return;
850 int period = closed_periods + 1;
851 Stub stub;
852 Amount *balances = calloc((size_t)employee->garnishes + 1, sizeof(Amount));
853 int has_pay = calculate(employee, period, &stub, balances);
854 printf("PREVIEW %d period %d\n", employee->id, period);
855 if (has_pay) print_stub(employee, period, &stub, employee->department);
856 else puts(" no pay");
857 free(balances);
858 } else if (!strcmp(command, "REPRINT")) {
859 Employee *employee = require_employee(a[1], 0, -1);
860 if (!employee) return;
861 int period;
862 if (!checked_period(a[2], &period)) return;
863 if (!employee->stubs[period].exists) {
864 printf("ERROR no stub %s %s\n", a[1], a[2]);
865 return;
866 }
867 printf("REPRINT %d %d\n", employee->id, period);
868 Stored *saved = &employee->stubs[period];
869 print_stub(employee, period, &saved->stub, saved->department);
870 } else if (!strcmp(command, "YTD")) {
871 Employee *employee = require_employee(a[1], 0, -1);
872 if (!employee) return;
873 ytd(employee);
874 } else if (!strcmp(command, "JOURNAL")) {
875 int period = atoi(a[1]);
876 if (period < 1 || period > closed_periods) {
877 printf("ERROR no run %s\n", a[1]);
878 return;
879 }
880 journal(period);
881 } else if (!strcmp(command, "REPORT")) {
882 report();
883 } else if (!strcmp(command, "TOP")) {
884 show_top(atoi(a[1]));
885 } else if (!strcmp(command, "EMPLOYEES")) {
886 show_employees();
887 } else if (!strcmp(command, "CONFIG")) {
888 config();
889 }
890}

892int main(void) {
893 int n;
894 if (scanf("%d\n", &n) != 1) return 0;
895 char line[4096];
896 for (int i = 0; i < n && fgets(line, sizeof(line), stdin); ++i) {
897 char *args[64];
898 int count = 0;
899 for (char *token = strtok(line, " \t\r\n");
900 token && count < 64;
901 token = strtok(NULL, " \t\r\n")) {
902 args[count++] = token;
903 }
904 if (count) execute(args);
905 }
906 return 0;
907}
