64int progressive(Map<int, int> table, int x) {
65 final bounds = table.keys.toList()..sort();
66 var total = 0;
67 for (var i = 0; i < bounds.length; i++)
68 if (bounds[i] < x) total += table[bounds[i]]! * ((i + 1 == bounds.length ? x : math.min(x, bounds[i + 1])) - bounds[i]);
69 return total ~/ 10000;
70}

107 List<int> daily(Employee e, int d) {
108 var worked = 0, night = 0;
109 for (final s in e.shifts[d] ?? <Shift>[]) {
110 worked += s.end - s.start;
111 night += math.max(0, math.min(s.end, 360) - s.start) + math.max(0, s.end - math.max(s.start, 1320));
112 }
113 final special = d % 7 == 0 || holidays.contains(d);
114 return [worked, special ? 0 : math.min(worked, 480), special ? 0 : math.min(math.max(worked - 480, 0), 240), special ? worked : math.max(0, worked - 720), night, e.ptos[d] ?? 0];

116 List<int> weekly(Employee e, int week) {
117 final result = List.filled(6, 0);
118 for (var d = week * 7 - 6; d <= week * 7; d++) {
119 final b = daily(e, d);
120 for (var i = 0; i < 5; i++) result[i] += b[i + 1];
121 }
122 result[5] = math.max(0, result[0] - 2400);
123 result[0] -= result[5];
124 result[1] += result[5];
125 return result;
126 }
127 List<int> breakdown(Employee e, int p) {
128 final a = weekly(e, 2 * p - 1), b = weekly(e, 2 * p);
129 return List.generate(5, (i) => a[i] + b[i]);

131 Stub? calculate(Employee e, int p) {
132 final s = Stub();
133 if (e.type == 'HOURLY') {
134 final b = breakdown(e, p);
135 for (var i = 0; i < 5; i++) s[V.values[i]] = b[i];
136 s[V.earnings] = ((2 * (b[0] + b[4]) + 3 * b[1] + 4 * b[2]) * e.base + 2 * b[3] * settings['NIGHT']!) ~/ 120;
137 } else if (e.type == 'SALARIED') {
138 if (e.active) {
139 s[V.salary] = e.base ~/ 26;
140 s[V.leave] = e.leave.where((d) => period(d) == p && !holidays.contains(d)).length;
141 s[V.leaveDeduction] = s[V.leave] * (e.base ~/ 260);
142 s[V.earnings] = s[V.salary] - s[V.leaveDeduction];
143 }
144 } else {
145 s[V.sales] = e.sales[p];
146 s[V.commission] = progressive(commissionTable, s[V.sales]);
147 s[V.earnings] = s[V.commission] + (e.active ? e.base : 0);
148 }
149 s[V.bonus] = e.bonus;
150 s[V.gross] = s[V.earnings] + s[V.bonus];
151 if (s[V.gross] == 0) return null;
152 var left = s[V.gross];
153 void deduct(V key, int value) {
154 s[key] = math.min(left, value);
155 left -= s[key];
156 }
157 deduct(V.retirement, math.min(s[V.gross] * e.retirement ~/ 10000, math.max(0, settings['RETIRECAP']! - e.ytd[V.retirement])));
158 deduct(V.health, e.health);
159 s[V.w] = left;
160 final q = math.max(0, s[V.earnings] - s[V.retirement] - s[V.health]), taxable = math.max(0, q - e.allowances * settings['ALLOWANCE']!);
161 deduct(V.federal, progressive(taxTable, taxable) + e.extra);
162 deduct(V.supplemental, (s[V.w] - q) * settings['SUPP']! ~/ 10000);
163 deduct(V.state, s[V.w] * settings['STATE']! ~/ 10000);
164 s[V.f] = s[V.gross] - s[V.health];
165 s[V.y] = math.min(s[V.f], math.max(0, settings['SSBASE']! - e.ytd[V.y]));
166 final additional = math.max(0, math.min(s[V.f], e.ytd[V.f] + s[V.f] - settings['ADDBASE']!));
167 deduct(V.ss, s[V.y] * settings['SSRATE']! ~/ 10000);
168 deduct(V.medicare, s[V.f] * settings['MEDRATE']! ~/ 10000 + additional * settings['ADDMED']! ~/ 10000);
169 final disposable = left, cap = left * settings['GARNCAP']! ~/ 10000;
170 for (final g in e.garnishments) {
171 final taken = math.min(math.min(disposable * g.rate ~/ 10000, g.balance), cap - s[V.garnishment]);
172 s[V.garnishment] += taken;
173 s.balances.add(g.balance - taken);
174 s.postGarnishment += g.balance - taken;
175 }
176 s[V.net] = left - s[V.garnishment];
177 s[V.employerSS] = s[V.y] * settings['SSRATE']! ~/ 10000;
178 s[V.employerMedicare] = s[V.f] * settings['MEDRATE']! ~/ 10000;
179 s[V.j] = math.min(s[V.f], math.max(0, settings['UIBASE']! - e.ytd[V.j]));
180 s[V.unemployment] = s[V.j] * settings['UIRATE']! ~/ 10000;
181 s[V.match] = math.min(s[V.retirement], s[V.gross] * settings['MATCHMAX']! ~/ 10000) * settings['MATCH']! ~/ 10000;
182 s[V.employerTotal] = s[V.employerSS] + s[V.employerMedicare] + s[V.unemployment] + s[V.match];
183 s[V.accrued] = e.type == 'HOURLY' ? math.min(960 - e.pto, (s[V.r] + s[V.o] + s[V.d]) ~/ 20) : 0;
184 s.postPto = e.pto + s[V.accrued];
185 return s;
186 }

224 void execute(List<String> a, StringBuffer out) {

233 } else if (op == 'BRACKET' || op == 'TIER') {

236 final existed = table.containsKey(bound);

239 } else if (op == 'EMP') {

241 if (employees.containsKey(i)) fail('duplicate employee $id');

245 out.writeln('EMP $id ${a[2]} ${a[3]} ${money(base)}');
246 } else if (op == 'HOLIDAY') {

251 } else if (op == 'RUN') {
252 final p = int.parse(id);
253 if (p != closed + 1 || p > 26) fail('invalid period $id');

268 e.pto = s.postPto;

271 e.stubCount++;

275 } else if (op == 'JOURNAL') {
276 final p = int.parse(id);
277 if (p < 1 || p > closed) fail('no run $id');
278 final total = Stub();

420 hours(e, p, out);
421 } else if (op == 'PREVIEW') {
422 final p = closed + 1;
423 final s = calculate(e, p);
424 out.writeln('PREVIEW $id period $p');
425 if (s == null) {
426 out.writeln(' no pay');
427 } else {
428 stub(e, s, p, e.department, out);
429 }
430 } else if (op == 'REPRINT') {
431 final p = int.parse(a[2]);
432 periodCheck(p, a[2]);
433 final r = e.records[p] ?? fail('no stub $id ${a[2]}');
434 out.writeln('REPRINT $id $p');
435 stub(e, r.stub, p, r.department, out);
436 } else if (op == 'YTD') {
437 out.writeln('YTD $id ${e.department} ${e.type} ${e.active ? 'active' : 'terminated'} periods ${e.stubCount}');
438 block(e, e.ytd, ' ', out, ytd: true);

440 out.writeln(' retirement-room ${money(math.max(0, settings['RETIRECAP']! - e.ytd[V.retirement]))} pending-bonus ${money(e.bonus)}');
441 }
442 }
