82fn progressive(table: []const Band, x: i64) i64 {
83 var total: i64 = 0;
84 for (table, 0..) |b, i| {
85 if (b.bound < x) total += b.rate * ((if (i + 1 == table.len) x else @min(x, table[i + 1].bound)) - b.bound);
86 }
87 return @divTrunc(total, 10000);
88}
89fn deduct(s: *Stub, left: *i64, key: V, value: i64) void {
90 const taken = @min(left.*, value);
91 s.set(key, taken);
92 left.* -= taken;

135 fn employee(self: *Payroll, id: []const u8, active: bool) Error!*Employee {
136 const e = self.employees.get(integer(id)) orelse return self.fail("unknown employee {s}", .{id});

143 fn dayCheck(self: *Payroll, token: []const u8) Error!usize {
144 const day = integer(token);
145 if (day < 1 or day > 364) return self.fail("invalid day {s}", .{token});
146 const d: usize = @intCast(day);
147 if (period(d) <= self.closed) return self.fail("period closed {d}", .{period(d)});
148 return d;
149 }
150 fn periodCheck(self: *Payroll, token: []const u8) Error!usize {
151 const p = integer(token);
152 if (p < 1 or p > 26) return self.fail("invalid period {s}", .{token});
153 return @intCast(p);
154 }
155 fn rateCheck(self: *Payroll, token: []const u8) Error!void {
156 if (amount(token) > 10000) return self.fail("invalid rate {s}", .{token});
157 }
158 fn amountCheck(self: *Payroll, n: i64) Error!void {
159 if (n == 0) return self.fail("invalid amount", .{});

161 fn timeCheck(self: *Payroll, token: []const u8) Error!i64 {
162 const h = integer(token[0..2]);
163 const m = integer(token[3..5]);
164 if (h > 23 or m > 59) return self.fail("invalid time {s}", .{token});
165 return h * 60 + m;
166 }
167 fn daily(self: *Payroll, e: *Employee, day: usize) [6]i64 {
168 var worked: i64 = 0;
169 var night: i64 = 0;
170 for (e.days[day].shifts.items) |s| {
171 worked += s.end - s.start;
172 night += @max(0, @min(s.end, 360) - s.start) + @max(0, s.end - @max(s.start, 1320));
173 }
174 const special = day % 7 == 0 or self.holidays[day];
175 return .{ worked, if (special) 0 else @min(worked, 480), if (special) 0 else @min(@max(worked - 480, 0), 240), if (special) worked else @max(0, worked - 720), night, e.days[day].pto orelse 0 };
176 }
177 fn weekly(self: *Payroll, e: *Employee, week: usize) [6]i64 {
178 var result: [6]i64 = @splat(0);
179 for (week * 7 - 6..week * 7 + 1) |d| {
180 const b = self.daily(e, d);
181 for (0..5) |i| result[i] += b[i + 1];
182 }
183 result[5] = @max(0, result[0] - 2400);
184 result[0] -= result[5];
185 result[1] += result[5];
186 return result;
187 }
188 fn breakdown(self: *Payroll, e: *Employee, p: usize) [5]i64 {
189 const a = self.weekly(e, 2 * p - 1);
190 const b = self.weekly(e, 2 * p);
191 var result: [5]i64 = undefined;
192 for (0..5) |i| result[i] = a[i] + b[i];
193 return result;
194 }
195 fn calculate(self: *Payroll, e: *Employee, p: usize) ?Stub {
196 var s: Stub = .{};
197 switch (e.kind) {
198 .HOURLY => {
199 const b = self.breakdown(e, p);
200 for (0..5) |i| s.values[i] = b[i];
201 s.set(.earnings, @divTrunc((2 * (b[0] + b[4]) + 3 * b[1] + 4 * b[2]) * e.base + 2 * b[3] * self.setting(.NIGHT), 120));
202 },
203 .SALARIED => {
204 if (e.active) {
205 s.set(.salary, @divTrunc(e.base, 26));
206 var count: i64 = 0;
207 for (14 * p - 13..14 * p + 1) |d| {
208 if (e.days[d].leave and !self.holidays[d]) count += 1;
209 }
210 s.set(.leave, count);
211 s.set(.leave_deduction, count * @divTrunc(e.base, 260));
212 s.set(.earnings, s.get(.salary) - s.get(.leave_deduction));
213 }
214 },
215 .COMMISSION => {
216 s.set(.sales, e.sales[p]);
217 s.set(.commission, progressive(self.commission_table.items, s.get(.sales)));
218 s.set(.earnings, s.get(.commission) + (if (e.active) e.base else @as(i64, 0)));
219 },
220 }
221 s.set(.bonus, e.bonus);
222 s.set(.gross, s.get(.earnings) + s.get(.bonus));
223 if (s.get(.gross) == 0) return null;
224 var left = s.get(.gross);
225 deduct(&s, &left, .retirement, @min(@divTrunc(s.get(.gross) * e.retirement, 10000), @max(0, self.setting(.RETIRECAP) - e.ytd.get(.retirement))));
226 deduct(&s, &left, .health, e.health);
227 s.set(.w, left);
228 const q = @max(0, s.get(.earnings) - s.get(.retirement) - s.get(.health));
229 const taxable = @max(0, q - e.allowances * self.setting(.ALLOWANCE));
230 deduct(&s, &left, .federal, progressive(self.tax_table.items, taxable) + e.extra);
231 deduct(&s, &left, .supplemental, @divTrunc((s.get(.w) - q) * self.setting(.SUPP), 10000));
232 deduct(&s, &left, .state, @divTrunc(s.get(.w) * self.setting(.STATE), 10000));
233 s.set(.f, s.get(.gross) - s.get(.health));
234 s.set(.y, @min(s.get(.f), @max(0, self.setting(.SSBASE) - e.ytd.get(.y))));
235 const additional = @max(0, @min(s.get(.f), e.ytd.get(.f) + s.get(.f) - self.setting(.ADDBASE)));
236 deduct(&s, &left, .ss, @divTrunc(s.get(.y) * self.setting(.SSRATE), 10000));
237 deduct(&s, &left, .medicare, @divTrunc(s.get(.f) * self.setting(.MEDRATE), 10000) + @divTrunc(additional * self.setting(.ADDMED), 10000));
238 const disposable = left;
239 const cap = @divTrunc(disposable * self.setting(.GARNCAP), 10000);
240 var garnished: i64 = 0;
241 for (e.garnishments.items, 0..) |g, i| {
242 const take = @min(@divTrunc(disposable * g.rate, 10000), g.balance, cap - garnished);
243 garnished += take;
244 s.balances[i] = g.balance - take;
245 s.post_garnishment += g.balance - take;
246 }
247 s.set(.garnishment, garnished);
248 s.set(.net, left - garnished);
249 s.set(.employer_ss, @divTrunc(s.get(.y) * self.setting(.SSRATE), 10000));
250 s.set(.employer_medicare, @divTrunc(s.get(.f) * self.setting(.MEDRATE), 10000));
251 s.set(.j, @min(s.get(.f), @max(0, self.setting(.UIBASE) - e.ytd.get(.j))));
252 s.set(.unemployment, @divTrunc(s.get(.j) * self.setting(.UIRATE), 10000));
253 s.set(.match, @divTrunc(@min(s.get(.retirement), @divTrunc(s.get(.gross) * self.setting(.MATCHMAX), 10000)) * self.setting(.MATCH), 10000));
254 s.set(.employer_total, s.get(.employer_ss) + s.get(.employer_medicare) + s.get(.unemployment) + s.get(.match));
255 s.set(.accrued, if (e.kind == .HOURLY) @min(960 - e.pto, @divTrunc(s.get(.r) + s.get(.o) + s.get(.d), 20)) else 0);
256 s.post_pto = e.pto + s.get(.accrued);
257 return s;
258 }
259 fn block(self: *Payroll, e: *Employee, s: *const Stub, indent: []const u8, ytd: bool) Error!void {
260 switch (e.kind) {
261 .HOURLY => {
262 try self.out.print("{s}hours reg {s} ot {s} dbl {s} night {s} pto {s}\n", .{ indent, try self.duration(s.get(.r)), try self.duration(s.get(.o)), try self.duration(s.get(.d)), try self.duration(s.get(.n)), try self.duration(s.get(.p)) });
263 try self.out.print("{s}pto accrued {s} balance {s}\n", .{ indent, try self.duration(s.get(.accrued)), try self.duration(if (ytd) e.pto else s.post_pto) });
264 },
265 .SALARIED => try self.out.print("{s}salary {s} leave {d} deduction {s}\n", .{ indent, try self.money(s.get(.salary)), s.get(.leave), try self.money(s.get(.leave_deduction)) }),
266 .COMMISSION => try self.out.print("{s}sales {s} commission {s}\n", .{ indent, try self.money(s.get(.sales)), try self.money(s.get(.commission)) }),
267 }
268 const labels = [_][]const u8{ "earnings", "bonus", "gross", "retirement", "health", "federal", "supplemental", "state", "social-security", "medicare" };
269 const keys = [_]V{ .earnings, .bonus, .gross, .retirement, .health, .federal, .supplemental, .state, .ss, .medicare };
270 for (labels, keys) |label, key| try self.out.print("{s}{s} {s}\n", .{ indent, label, try self.money(s.get(key)) });
271 try self.out.print("{s}garnishment {s} balance {s}\n{s}net {s}\n", .{ indent, try self.money(s.get(.garnishment)), try self.money(if (ytd) e.garnishmentBalance() else s.post_garnishment), indent, try self.money(s.get(.net)) });
272 try self.out.print("{s}employer social-security {s} medicare {s} unemployment {s} match {s}\n", .{ indent, try self.money(s.get(.employer_ss)), try self.money(s.get(.employer_medicare)), try self.money(s.get(.unemployment)), try self.money(s.get(.match)) });
273 }

278 fn hours(self: *Payroll, e: *Employee, p: usize) Error!void {
279 try self.out.print("HOURS {d} period {d}\n", .{ e.id, p });
280 var count: usize = 0;
281 for (14 * p - 13..14 * p + 1) |d| {
282 const shifts = e.days[d].shifts.items;
283 if (shifts.len == 0 and e.days[d].pto == null) continue;
284 count += 1;
285 const b = self.daily(e, d);
286 try self.out.print(" day {d} {s}{s}", .{ d, weekday(d), if (self.holidays[d]) " holiday" else "" });
287 for (shifts) |s| try self.out.print(" {s}-{s}", .{ try self.clock(s.start), try self.clock(s.end) });
288 try self.out.print(" total {s} reg {s} ot {s} dbl {s} night {s} pto {s}\n", .{ try self.duration(b[0]), try self.duration(b[1]), try self.duration(b[2]), try self.duration(b[3]), try self.duration(b[4]), try self.duration(b[5]) });
289 }
290 if (count == 0) try self.out.writeAll(" (none)\n");
291 for ([_]usize{ 2 * p - 1, 2 * p }) |w| {
292 const b = self.weekly(e, w);
293 try self.out.print(" week {d} reg {s} ot {s} dbl {s} night {s} pto {s} weekly-ot {s}\n", .{ w, try self.duration(b[0]), try self.duration(b[1]), try self.duration(b[2]), try self.duration(b[3]), try self.duration(b[4]), try self.duration(b[5]) });
294 }
295 const b = self.breakdown(e, p);
296 try self.out.print(" total reg {s} ot {s} dbl {s} night {s} pto {s}\n", .{ try self.duration(b[0]), try self.duration(b[1]), try self.duration(b[2]), try self.duration(b[3]), try self.duration(b[4]) });
297 }

301 fn execute(self: *Payroll, a: []const []const u8) Error!void {
302 const op = a[0];
303 const id: []const u8 = if (a.len > 1) a[1] else "";
304 if (std.mem.eql(u8, op, "SET")) {
305 const key = std.meta.stringToEnum(Setting, id) orelse return self.fail("unknown setting {s}", .{id});
306 const i = @backingInt(key);
307 if (i < 9) try self.rateCheck(a[2]);
308 const old = self.config[i];
309 const value = amount(a[2]);
310 self.config[i] = value;
311 try self.out.print("SET {s} {s} -> {s}\n", .{ id, if (i < 9) try self.rateText(old) else try self.money(old), if (i < 9) try self.rateText(value) else try self.money(value) });
312 } else if (std.mem.eql(u8, op, "BRACKET") or std.mem.eql(u8, op, "TIER")) {
313 try self.rateCheck(a[2]);
314 const table = if (std.mem.eql(u8, op, "BRACKET")) &self.tax_table else &self.commission_table;
315 const bound = amount(id);
316 const rate = amount(a[2]);
317 var replaced = false;
318 for (table.items) |*b| {
319 if (b.bound == bound) {
320 b.rate = rate;
321 replaced = true;
322 break;
323 }
324 }
325 if (!replaced) {
326 try table.append(self.gpa, .{ .bound = bound, .rate = rate });
327 std.mem.sort(Band, table.items, {}, struct {
328 fn less(_: void, left: Band, right: Band) bool {
329 return left.bound < right.bound;
330 }
331 }.less);
332 }
333 try self.out.print("{s} {s} {s} {s}\n", .{ op, try self.money(bound), try self.rateText(rate), if (replaced) "replaced" else "added" });
334 } else if (std.mem.eql(u8, op, "EMP")) {
335 const number = integer(id);
336 if (self.employees.contains(number)) return self.fail("duplicate employee {s}", .{id});
337 const base = amount(a[4]);
338 try self.amountCheck(base);
339 const e = try self.gpa.create(Employee);
340 const kind: PayType = if (std.mem.eql(u8, a[3], "HOURLY")) .HOURLY else if (std.mem.eql(u8, a[3], "SALARIED")) .SALARIED else .COMMISSION;
341 e.* = .{ .id = number, .department = a[2], .kind = kind, .base = base };
342 try self.employees.put(self.gpa, number, e);
343 try self.out.print("EMP {s} {s} {s} {s}\n", .{ id, a[2], a[3], try self.money(base) });
344 } else if (std.mem.eql(u8, op, "HOLIDAY")) {
345 const d = try self.dayCheck(id);
346 if (self.holidays[d]) return self.fail("duplicate holiday {s}", .{id});
347 self.holidays[d] = true;
348 try self.out.print("HOLIDAY {d} {s}\n", .{ d, weekday(d) });
349 } else if (std.mem.eql(u8, op, "RUN")) {
350 const p: usize = @intCast(integer(id));
351 if (p != self.closed + 1 or p > 26) return self.fail("invalid period {s}", .{id});
352 self.closed = p;
353 try self.out.print("RUN {d} days {d}-{d}\n", .{ p, 14 * p - 13, 14 * p });
354 var paid_count: usize = 0;
355 var skipped: usize = 0;
356 var total: Stub = .{};
357 for (try self.people()) |e| {
358 const s = self.calculate(e, p) orelse {
359 if (e.active) skipped += 1;
360 continue;
361 };
362 try self.stub(e, &s, p, e.department);
363 paid_count += 1;
364 total.add(s);
365 e.bonus = 0;
366 e.pto = s.post_pto;
367 for (e.garnishments.items, 0..) |*g, i| g.balance = s.balances[i];
368 e.ytd.add(s);
369 e.stub_count += 1;
370 e.records[p] = .{ .stub = s, .department = e.department };
371 }
372 try self.out.print(" TOTAL paid {d} skipped {d} gross {s} deductions {s} net {s} employer {s}\n", .{ paid_count, skipped, try self.money(total.get(.gross)), try self.money(total.get(.gross) - total.get(.net)), try self.money(total.get(.net)), try self.money(total.get(.employer_total)) });
373 } else if (std.mem.eql(u8, op, "JOURNAL")) {
374 const p: usize = @intCast(integer(id));
375 if (p < 1 or p > self.closed) return self.fail("no run {s}", .{id});
376 var total: Stub = .{};
377 var count: usize = 0;
378 for (self.employees.values()) |e| {
379 if (e.records[p]) |r| {
380 count += 1;
381 total.add(r.stub);
382 }
383 }
384 try self.out.print("JOURNAL {d} stubs {d}\n", .{ p, count });
385 const amounts = [_]i64{ total.get(.gross), total.get(.employer_ss) + total.get(.employer_medicare) + total.get(.unemployment), total.get(.match), total.get(.retirement) + total.get(.match), total.get(.health), total.get(.federal) + total.get(.supplemental), total.get(.state), total.get(.ss) + total.get(.medicare) + total.get(.employer_ss) + total.get(.employer_medicare), total.get(.unemployment), total.get(.garnishment), total.get(.net) };
386 const labels = [_][]const u8{ "debit wages", "debit employer-tax", "debit retirement-match", "credit retirement", "credit health", "credit federal", "credit state", "credit fica", "credit unemployment", "credit garnishment", "credit net-pay" };
387 var debit: i64 = 0;
388 var credit: i64 = 0;
389 for (amounts, labels, 0..) |value, label, i| {
390 try self.out.print(" {s} {s}\n", .{ label, try self.money(value) });
391 if (i < 3) debit += value else credit += value;
392 }
393 try self.out.print(" total debit {s} credit {s}\n", .{ try self.money(debit), try self.money(credit) });
394 } else if (std.mem.eql(u8, op, "REPORT")) {
395 const Department = struct {
396 name: []const u8,
397 total: Stub = .{},
398 count: usize = 0,
399 };
400 var departments: std.StringArrayHashMapUnmanaged(Department) = .empty;
401 var all: Stub = .{};
402 var count: usize = 0;
403 for (self.employees.values()) |e| {
404 for (e.records) |maybe| {
405 const r = maybe orelse continue;
406 const entry = try departments.getOrPut(self.gpa, r.department);
407 if (!entry.found_existing) entry.value_ptr.* = .{ .name = r.department };
408 const d = entry.value_ptr;
409 d.total.add(r.stub);
410 d.count += 1;
411 all.add(r.stub);
412 count += 1;
413 }
414 }
415 const ordered = try self.gpa.dupe(Department, departments.values());
416 std.mem.sort(Department, ordered, {}, struct {
417 fn less(_: void, left: Department, right: Department) bool {
418 return std.mem.order(u8, left.name, right.name) == .lt;
419 }
420 }.less);
421 try self.out.print("REPORT period {d}\n", .{self.closed});
422 for (ordered) |d| try self.reportLine(d.name, &d.total, d.count);
423 try self.reportLine("total", &all, count);
424 } else if (std.mem.eql(u8, op, "TOP")) {
425 const k: usize = @intCast(integer(id));
426 var selected: std.ArrayList(*Employee) = .empty;
427 for (self.employees.values()) |e| {
428 if (e.stub_count > 0) try selected.append(self.gpa, e);
429 }
430 std.mem.sort(*Employee, selected.items, {}, struct {
431 fn less(_: void, left: *Employee, right: *Employee) bool {
432 const ag = left.ytd.get(.gross);
433 const bg = right.ytd.get(.gross);
434 return if (ag != bg) ag > bg else left.id < right.id;
435 }
436 }.less);
437 try self.out.print("TOP {d}\n", .{k});
438 for (selected.items[0..@min(k, selected.items.len)], 0..) |e, i| try self.out.print(" {d} {d} {s} gross {s} net {s}\n", .{ i + 1, e.id, e.department, try self.money(e.ytd.get(.gross)), try self.money(e.ytd.get(.net)) });
439 if (selected.items.len == 0) try self.out.writeAll(" (none)\n");
440 } else if (std.mem.eql(u8, op, "EMPLOYEES")) {
441 try self.out.writeAll("EMPLOYEES\n");
442 for (try self.people()) |e| {
443 try self.out.print(" {d} {s} {s} {s} {s} retirement {s} health {s} allowances {d} extra {s} bonus {s}\n", .{ e.id, e.department, @tagName(e.kind), try self.money(e.base), if (e.active) "active" else "terminated", try self.rateText(e.retirement), try self.money(e.health), e.allowances, try self.money(e.extra), try self.money(e.bonus) });
444 for (e.garnishments.items, 0..) |g, i| try self.out.print(" garnish {d} {s} balance {s}\n", .{ i + 1, try self.rateText(g.rate), try self.money(g.balance) });
445 }
446 if (self.employees.count() == 0) try self.out.writeAll(" (none)\n");
447 } else if (std.mem.eql(u8, op, "CONFIG")) {
448 try self.out.writeAll("CONFIG\n");
449 for (setting_names, 0..) |name, i| try self.out.print(" {s} {s}\n", .{ name, if (i < 9) try self.rateText(self.config[i]) else try self.money(self.config[i]) });
450 for (self.tax_table.items) |b| try self.out.print(" bracket {s} {s}\n", .{ try self.money(b.bound), try self.rateText(b.rate) });
451 for (self.commission_table.items) |b| try self.out.print(" tier {s} {s}\n", .{ try self.money(b.bound), try self.rateText(b.rate) });
452 } else {
453 const read_only = std.mem.eql(u8, op, "HOURS") or std.mem.eql(u8, op, "PREVIEW") or std.mem.eql(u8, op, "REPRINT") or std.mem.eql(u8, op, "YTD");
454 const e = try self.employee(id, !read_only);
455 if (std.mem.eql(u8, op, "RATE")) {
456 const m = amount(a[2]);
457 try self.amountCheck(m);
458 const old = e.base;
459 e.base = m;
460 try self.out.print("RATE {s} {s} -> {s}\n", .{ id, try self.money(old), try self.money(m) });
461 } else if (std.mem.eql(u8, op, "DEPT")) {
462 if (std.mem.eql(u8, e.department, a[2])) return self.fail("same department {s}", .{a[2]});
463 const old = e.department;
464 e.department = a[2];
465 try self.out.print("DEPT {s} {s} -> {s}\n", .{ id, old, a[2] });
466 } else if (std.mem.eql(u8, op, "BENEFIT")) {
467 try self.rateCheck(a[2]);
468 e.retirement = amount(a[2]);
469 e.health = amount(a[3]);
470 try self.out.print("BENEFIT {s} retirement {s} health {s}\n", .{ id, try self.rateText(e.retirement), try self.money(e.health) });
471 } else if (std.mem.eql(u8, op, "W4")) {
472 e.allowances = integer(a[2]);
473 e.extra = amount(a[3]);
474 try self.out.print("W4 {s} allowances {d} extra {s}\n", .{ id, e.allowances, try self.money(e.extra) });
475 } else if (std.mem.eql(u8, op, "GARNISH")) {
476 try self.rateCheck(a[2]);
477 const m = amount(a[3]);
478 try self.amountCheck(m);
479 const rate = amount(a[2]);
480 try e.garnishments.append(self.gpa, .{ .rate = rate, .balance = m });
481 try self.out.print("GARNISH {s} order {d} {s} balance {s}\n", .{ id, e.garnishments.items.len, try self.rateText(rate), try self.money(m) });
482 } else if (std.mem.eql(u8, op, "TERM")) {
483 e.active = false;
484 try self.out.print("TERM {s}\n", .{id});
485 } else if (std.mem.eql(u8, op, "TIME") or std.mem.eql(u8, op, "UNTIME") or std.mem.eql(u8, op, "PTO") or std.mem.eql(u8, op, "LEAVE") or std.mem.eql(u8, op, "SALES")) {
486 try self.typeCheck(e, if (std.mem.eql(u8, op, "LEAVE")) .SALARIED else if (std.mem.eql(u8, op, "SALES")) .COMMISSION else .HOURLY);
487 const d = try self.dayCheck(a[2]);
488 if (std.mem.eql(u8, op, "TIME") or std.mem.eql(u8, op, "UNTIME")) {
489 const start = try self.timeCheck(a[3]);
490 const shifts = &e.days[d].shifts;
491 if (std.mem.eql(u8, op, "TIME")) {
492 const end = try self.timeCheck(a[4]);
493 if (start >= end) return self.fail("invalid shift {s}-{s}", .{ a[3], a[4] });
494 var worked: i64 = 0;
495 for (shifts.items) |s| {
496 if (start < s.end and s.start < end) return self.fail("overlapping shift {s}-{s}", .{ try self.clock(s.start), try self.clock(s.end) });
497 worked += s.end - s.start;
498 }
499 if (worked + end - start > 960) return self.fail("daily limit {s} {s}", .{ id, a[2] });
500 try shifts.append(self.gpa, .{ .start = start, .end = end });
501 std.mem.sort(Shift, shifts.items, {}, struct {
502 fn less(_: void, left: Shift, right: Shift) bool {
503 return left.start < right.start;
504 }
505 }.less);
506 try self.out.print("TIME {s} day {d} {s}-{s} {s}\n", .{ id, d, try self.clock(start), try self.clock(end), try self.duration(end - start) });
507 } else {
508 var found: ?usize = null;
509 for (shifts.items, 0..) |s, i| {
510 if (s.start == start) {
511 found = i;
512 break;
513 }
514 }
515 const index = found orelse return self.fail("no shift {s} {s} {s}", .{ id, a[2], a[3] });
516 const s = shifts.orderedRemove(index);
517 try self.out.print("UNTIME {s} day {d} {s}-{s}\n", .{ id, d, try self.clock(s.start), try self.clock(s.end) });
518 }
519 } else if (std.mem.eql(u8, op, "PTO")) {
520 const mm = integer(a[3][3..5]);
521 const len = integer(a[3][0..2]) * 60 + mm;
522 if (mm > 59 or len < 1 or len > 480) return self.fail("invalid duration {s}", .{a[3]});
523 if (e.days[d].pto != null) return self.fail("duplicate pto {s} {s}", .{ id, a[2] });
524 if (len > e.pto) return self.fail("insufficient pto {s} {s}", .{ id, try self.duration(e.pto) });
525 e.days[d].pto = len;
526 e.pto -= len;
527 try self.out.print("PTO {s} day {d} {s} balance {s}\n", .{ id, d, try self.duration(len), try self.duration(e.pto) });
528 } else if (std.mem.eql(u8, op, "LEAVE")) {
529 if (d % 7 == 0 or d % 7 == 6 or self.holidays[d]) return self.fail("not workday {s}", .{a[2]});
530 if (e.days[d].leave) return self.fail("duplicate leave {s} {s}", .{ id, a[2] });
531 e.days[d].leave = true;
532 try self.out.print("LEAVE {s} day {d} {s}\n", .{ id, d, weekday(d) });
533 } else {
534 const m = amount(a[3]);
535 try self.amountCheck(m);
536 e.sales[period(d)] += m;
537 try self.out.print("SALES {s} day {d} {s} period {d} total {s}\n", .{ id, d, try self.money(m), period(d), try self.money(e.sales[period(d)]) });
538 }
539 } else if (std.mem.eql(u8, op, "BONUS")) {
540 const m = amount(a[2]);
541 try self.amountCheck(m);
542 e.bonus += m;
543 try self.out.print("BONUS {s} {s} pending {s}\n", .{ id, try self.money(m), try self.money(e.bonus) });
544 } else if (std.mem.eql(u8, op, "HOURS")) {
545 try self.typeCheck(e, .HOURLY);
546 const p = try self.periodCheck(a[2]);
547 try self.hours(e, p);
548 } else if (std.mem.eql(u8, op, "PREVIEW")) {
549 const p = self.closed + 1;
550 try self.out.print("PREVIEW {s} period {d}\n", .{ id, p });
551 if (self.calculate(e, p)) |s| {
552 try self.stub(e, &s, p, e.department);
553 } else {
554 try self.out.writeAll(" no pay\n");
555 }
556 } else if (std.mem.eql(u8, op, "REPRINT")) {
557 const p = try self.periodCheck(a[2]);
558 const r = e.records[p] orelse return self.fail("no stub {s} {s}", .{ id, a[2] });
559 try self.out.print("REPRINT {s} {d}\n", .{ id, p });
560 try self.stub(e, &r.stub, p, r.department);
561 } else if (std.mem.eql(u8, op, "YTD")) {
562 try self.out.print("YTD {s} {s} {s} {s} periods {d}\n", .{ id, e.department, @tagName(e.kind), if (e.active) "active" else "terminated", e.stub_count });
563 try self.block(e, &e.ytd, " ", true);
564 try self.out.print(" wages federal {s} social-security {s} medicare {s} unemployment {s}\n retirement-room {s} pending-bonus {s}\n", .{ try self.money(e.ytd.get(.w)), try self.money(e.ytd.get(.y)), try self.money(e.ytd.get(.f)), try self.money(e.ytd.get(.j)), try self.money(@max(0, self.setting(.RETIRECAP) - e.ytd.get(.retirement))), try self.money(e.bonus) });
565 }
566 }
567 }

569pub fn main(init: std.process.Init) !void {
570 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
571 defer arena.deinit();
572 const gpa = arena.allocator();
573 var ib: [4096]u8 = undefined;
574 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
575 var ob: [8192]u8 = undefined;
576 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
577 var payroll = Payroll{ .gpa = gpa, .out = &output.interface };
578 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
579 for (0..n) |_| {
580 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
581 var a: std.ArrayList([]const u8) = .empty;
582 var it = std.mem.splitScalar(u8, line, ' ');
583 while (it.next()) |part| try a.append(gpa, part);
584 var draft: std.Io.Writer.Allocating = .init(gpa);
585 payroll.out = &draft.writer;
586 payroll.execute(a.items) catch |err| {
587 if (err == error.BadCommand) {
588 try output.interface.print("ERROR {s}\n", .{payroll.error_message});
589 continue;
590 }
591 return err;
592 };
593 try output.interface.writeAll(draft.writer.buffered());
594 }
595 try output.interface.flush();
596}
