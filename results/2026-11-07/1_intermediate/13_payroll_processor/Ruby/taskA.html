51def progressive(table, amount)
52 pairs = table.sort
53 numerator = 0
54 pairs.each_with_index do |(lower, rate), i|
55 break if lower >= amount
56 upper = i + 1 < pairs.length ? pairs[i + 1][0] : amount
57 numerator += rate * ([amount, upper].min - lower)
58 end
59 numerator / 10000
60end

111 def employee(text, active = false, kind = nil)
112 e = @employees[text.to_i]
113 raise CommandError, "unknown employee #{text}" unless e
114 raise CommandError, "terminated employee #{text}" if active && !e.active

116 e

119 def day_check(text)
120 day = text.to_i
121 raise CommandError, "invalid day #{text}" unless (1..364).cover?(day)
122 period = period_of(day)
123 raise CommandError, "period closed #{period}" if period <= @closed
124 day
125 end

127 def period_check(text)
128 period = text.to_i
129 raise CommandError, "invalid period #{text}" unless (1..26).cover?(period)
130 period
131 end

133 def rate_check(text)
134 value = cents(text)
135 raise CommandError, "invalid rate #{text}" if value > 10000
136 value
137 end

139 def amount_check(text)
140 value = cents(text)
141 raise CommandError, 'invalid amount' if value.zero?
142 value
143 end

145 def time_check(text)
146 hour, minute = text.split(':').map(&:to_i)
147 raise CommandError, "invalid time #{text}" if hour > 23 || minute > 59
148 hour * 60 + minute
149 end

151 def daily(e, day)
152 shifts = e.shifts.fetch(day, [])
153 worked = shifts.sum { |s, t| t - s }
154 night = shifts.sum { |s, t| [0, [t, 360].min - s].max + [0, t - [s, 1320].max].max }
155 pto = e.pto.fetch(day, 0)
156 return [worked, 0, 0, worked, night, pto] if (day % 7).zero? || @holidays.key?(day)
157 [worked, [worked, 480].min, [[worked - 480, 0].max, 240].min, [worked - 720, 0].max, night, pto]
158 end

160 def weekly(e, week)
161 values = (7 * week - 6..7 * week).map { |day| daily(e, day).drop(1) }.transpose.map(&:sum)
162 overtime = [0, values[0] - 2400].max
163 values[0] -= overtime
164 values[1] += overtime
165 [values, overtime]
166 end
167
168 def breakdown(e, period)
169 a = weekly(e, 2 * period - 1)[0]
170 b = weekly(e, 2 * period)[0]
171 a.zip(b).map { |x, y| x + y }
172 end

174 def calculate(e, period)
175 s = empty_totals
176 c = @settings
177 ytd = e.ytd
178 if e.kind == 'HOURLY'
179 values = breakdown(e, period)
180 %i[R O D N P].zip(values).each { |field, value| s[field] = value }
181 weighted = 2 * (s[:R] + s[:P]) + 3 * s[:O] + 4 * s[:D]
182 s[:earnings] = (weighted * e.base + 2 * s[:N] * c['NIGHT']) / 120
183 elsif e.kind == 'SALARIED'
184 if e.active
185 s[:salary] = e.base / 26
186 s[:leave] = e.leave.keys.count { |day| period_of(day) == period && !@holidays.key?(day) }
187 s[:deduction] = s[:leave] * (e.base / 260)
188 s[:earnings] = s[:salary] - s[:deduction]
189 end
190 else
191 s[:sales] = e.sales.fetch(period, 0)
192 s[:commission] = progressive(@tiers, s[:sales])
193 s[:earnings] = s[:commission] + (e.active ? e.base : 0)
194 end
195 s[:bonus] = e.bonus
196 gross = s[:earnings] + s[:bonus]
197 return nil if gross.zero?
198 s[:gross] = gross
199 remaining = gross
200 deduct = lambda do |field, amount|
201 s[field] = [amount, remaining].min
202 remaining -= s[field]
203 end
204 retirement_room = [0, c['RETIRECAP'] - ytd[:retirement]].max
205 deduct.call(:retirement, [gross * e.retirement / 10000, retirement_room].min)
206 deduct.call(:health, e.health)
207 s[:W] = remaining
208 base_taxable = [0, s[:earnings] - s[:retirement] - s[:health]].max
209 allowance_taxable = [0, base_taxable - e.allowances * c['ALLOWANCE']].max
210 deduct.call(:federal, progressive(@brackets, allowance_taxable) + e.extra)
211 deduct.call(:supplemental, (s[:W] - base_taxable) * c['SUPP'] / 10000)
212 deduct.call(:state, s[:W] * c['STATE'] / 10000)
213 s[:F] = gross - s[:health]
214 s[:Y] = [s[:F], [0, c['SSBASE'] - ytd[:Y]].max].min
215 additional = [0, [s[:F], ytd[:F] + s[:F] - c['ADDBASE']].min].max
216 deduct.call(:social, s[:Y] * c['SSRATE'] / 10000)
217 deduct.call(:medicare, s[:F] * c['MEDRATE'] / 10000 + additional * c['ADDMED'] / 10000)
218 disposable = remaining
219 cap = disposable * c['GARNCAP'] / 10000
220 balances = []
221 garnishment = 0
222 e.garnishments.each do |rate, balance|
223 taken = [disposable * rate / 10000, balance, cap - garnishment].min
224 garnishment += taken
225 balances << balance - taken
226 end
227 s[:garnishment] = garnishment
228 s[:garnishment_balance] = balances.sum
229 s[:net] = remaining - garnishment
230 s[:employer_social] = s[:Y] * c['SSRATE'] / 10000
231 s[:employer_medicare] = s[:F] * c['MEDRATE'] / 10000
232 s[:J] = [s[:F], [0, c['UIBASE'] - ytd[:J]].max].min
233 s[:unemployment] = s[:J] * c['UIRATE'] / 10000
234 s[:match] = [s[:retirement], gross * c['MATCHMAX'] / 10000].min * c['MATCH'] / 10000
235 s[:employer] = %i[employer_social employer_medicare unemployment match].sum { |field| s[field] }
236 s[:accrued] = [960 - e.pto_balance, (s[:R] + s[:O] + s[:D]) / 20].min if e.kind == 'HOURLY'
237 s[:pto_balance] = e.pto_balance + s[:accrued]
238 [s, balances]
239 end

241 def hours_parts(values)
242 %w[reg ot dbl night pto].zip(values).flat_map { |label, value| [label, duration(value)] }
243 end
244
245 def stub_block(e, s, indent)
246 if e.kind == 'HOURLY'
247 emit(indent, 'hours', *hours_parts(%i[R O D N P].map { |field| s[field] }))
248 emit(indent, 'pto', 'accrued', duration(s[:accrued]), 'balance', duration(s[:pto_balance]))
249 elsif e.kind == 'SALARIED'
250 emit(indent, 'salary', money(s[:salary]), 'leave', s[:leave], 'deduction', money(s[:deduction]))
251 else
252 emit(indent, 'sales', money(s[:sales]), 'commission', money(s[:commission]))
253 end
254 %i[earnings bonus gross retirement health federal supplemental state social medicare].each do |field|
255 emit(indent, field == :social ? 'social-security' : field, money(s[field]))
256 end
257 emit(indent, 'garnishment', money(s[:garnishment]), 'balance', money(s[:garnishment_balance]))
258 emit(indent, 'net', money(s[:net]))
259 emit(indent, 'employer', 'social-security', money(s[:employer_social]), 'medicare', money(s[:employer_medicare]),
260 'unemployment', money(s[:unemployment]), 'match', money(s[:match]))
261 end

263 def print_stub(e, period, s, department)
264 emit(2, 'STUB', e.id, department, e.kind, 'period', period)
265 stub_block(e, s, 4)
266 end

268 def show_hours(e, period)
269 emit(0, 'HOURS', e.id, 'period', period)
270 found = false
271 (14 * period - 13..14 * period).each do |day|
272 shifts = e.shifts.fetch(day, []).sort
273 next if shifts.empty? && !e.pto.key?(day)
274 worked, *values = daily(e, day)
275 parts = ['day', day, weekday(day)]
276 parts << 'holiday' if @holidays.key?(day)
277 parts.concat(shifts.map { |s, t| "#{clock(s)}-#{clock(t)}" })
278 emit(2, *parts, 'total', duration(worked), *hours_parts(values))
279 found = true
280 end
281 emit(2, '(none)') unless found
282 [2 * period - 1, 2 * period].each do |week|
283 values, overtime = weekly(e, week)
284 emit(2, 'week', week, *hours_parts(values), 'weekly-ot', duration(overtime))
285 end
286 emit(2, 'total', *hours_parts(breakdown(e, period)))
287 end

289 def run(period)
290 @closed = period
291 emit(0, 'RUN', period, 'days', "#{14 * period - 13}-#{14 * period}")
292 paid = 0
293 skipped = 0
294 stubs = []
295 @employees.values.sort_by(&:id).each do |e|
296 calculated = calculate(e, period)
297 if calculated.nil?
298 skipped += 1 if e.active
299 next
300 end
301 s, balances = calculated
302 print_stub(e, period, s, e.department)
303 e.bonus = 0
304 e.pto_balance = s[:pto_balance]
305 balances.each_with_index { |balance, i| e.garnishments[i][1] = balance }
306 FIELDS.each { |field| e.ytd[field] += s[field] }
307 e.periods += 1
308 e.stubs[period] = [s, e.department]
309 stubs << s
310 paid += 1
311 end
312 t = sum_stubs(stubs)
313 emit(2, 'TOTAL', 'paid', paid, 'skipped', skipped, 'gross', money(t[:gross]),
314 'deductions', money(t[:gross] - t[:net]), 'net', money(t[:net]), 'employer', money(t[:employer]))
315 end

317 def journal(period)
318 stubs = @employees.values.filter_map { |e| e.stubs[period]&.first }
319 t = sum_stubs(stubs)
320 rows = [
321 ['debit', 'wages', t[:gross]],
322 ['debit', 'employer-tax', t[:employer_social] + t[:employer_medicare] + t[:unemployment]],
323 ['debit', 'retirement-match', t[:match]],
324 ['credit', 'retirement', t[:retirement] + t[:match]],
325 ['credit', 'health', t[:health]],
326 ['credit', 'federal', t[:federal] + t[:supplemental]],
327 ['credit', 'state', t[:state]],
328 ['credit', 'fica', t[:social] + t[:medicare] + t[:employer_social] + t[:employer_medicare]],
329 ['credit', 'unemployment', t[:unemployment]],
330 ['credit', 'garnishment', t[:garnishment]],
331 ['credit', 'net-pay', t[:net]]
332 ]
333 emit(0, 'JOURNAL', period, 'stubs', stubs.length)
334 rows.each { |side, label, amount| emit(2, side, label, money(amount)) }
335 debit = rows.select { |side, _, _| side == 'debit' }.sum(&:last)
336 credit = rows.select { |side, _, _| side == 'credit' }.sum(&:last)
337 emit(2, 'total', 'debit', money(debit), 'credit', money(credit))
338 end

340 def report_line(department, stubs)
341 t = sum_stubs(stubs)
342 taxes = %i[federal supplemental state social medicare].sum { |field| t[field] }
343 emit(2, department, 'stubs', stubs.length, 'gross', money(t[:gross]), 'pretax', money(t[:retirement] + t[:health]),
344 'taxes', money(taxes), 'garnishment', money(t[:garnishment]), 'net', money(t[:net]), 'employer', money(t[:employer]))
345 end

347 def command(cmd, a)
348 case cmd
349 when 'SET'
350 name = a[0]
351 raise CommandError, "unknown setting #{name}" unless @settings.key?(name)
352 value = RATE_SETTINGS.include?(name) ? rate_check(a[1]) : cents(a[1])
353 old = @settings[name]
354 @settings[name] = value
355 emit(0, 'SET', name, RATE_SETTINGS.include?(name) ? rate_text(old) : money(old),
356 '->', RATE_SETTINGS.include?(name) ? rate_text(value) : money(value))
357 when 'BRACKET', 'TIER'
358 rate = rate_check(a[1])
359 table = cmd == 'BRACKET' ? @brackets : @tiers
360 lower = cents(a[0])
361 action = table.key?(lower) ? 'replaced' : 'added'
362 table[lower] = rate
363 emit(0, cmd, money(lower), rate_text(rate), action)
364 when 'EMP'
365 id = a[0].to_i
366 raise CommandError, "duplicate employee #{a[0]}" if @employees.key?(id)
367 base = amount_check(a[3])
368 @employees[id] = Employee.new(id, a[1], a[2], base)
369 emit(0, 'EMP', id, a[1], a[2], money(base))
370 when 'RATE', 'DEPT', 'BENEFIT', 'W4', 'GARNISH', 'TERM', 'BONUS'
371 e = employee(a[0], true)
372 case cmd
373 when 'RATE'
374 base = amount_check(a[1])
375 old = e.base
376 e.base = base
377 emit(0, 'RATE', e.id, money(old), '->', money(base))
378 when 'DEPT'
379 raise CommandError, "same department #{a[1]}" if a[1] == e.department
380 old = e.department
381 e.department = a[1]
382 emit(0, 'DEPT', e.id, old, '->', a[1])
383 when 'BENEFIT'
384 rate = rate_check(a[1])
385 e.retirement = rate
386 e.health = cents(a[2])
387 emit(0, 'BENEFIT', e.id, 'retirement', rate_text(rate), 'health', money(e.health))
388 when 'W4'
389 e.allowances = a[1].to_i
390 e.extra = cents(a[2])
391 emit(0, 'W4', e.id, 'allowances', e.allowances, 'extra', money(e.extra))
392 when 'GARNISH'
393 rate = rate_check(a[1])
394 amount = amount_check(a[2])
395 e.garnishments << [rate, amount]
396 emit(0, 'GARNISH', e.id, 'order', e.garnishments.length, rate_text(rate), 'balance', money(amount))
397 when 'TERM'
398 e.active = false
399 emit(0, 'TERM', e.id)
400 when 'BONUS'
401 amount = amount_check(a[1])
402 e.bonus += amount
403 emit(0, 'BONUS', e.id, money(amount), 'pending', money(e.bonus))
404 end
405 when 'HOLIDAY'
406 day = day_check(a[0])
407 raise CommandError, "duplicate holiday #{a[0]}" if @holidays.key?(day)
408 @holidays[day] = true
409 emit(0, 'HOLIDAY', day, weekday(day))
410 when 'TIME', 'UNTIME', 'PTO', 'LEAVE', 'SALES'
411 kind = cmd == 'LEAVE' ? 'SALARIED' : cmd == 'SALES' ? 'COMMISSION' : 'HOURLY'
412 e = employee(a[0], true, kind)
413 day = day_check(a[1])
414 case cmd
415 when 'TIME'
416 start = time_check(a[2])
417 finish = time_check(a[3])
418 raise CommandError, "invalid shift #{a[2]}-#{a[3]}" if start >= finish
419 shifts = e.shifts.fetch(day, []).sort
420 shifts.each do |s, t|
421 raise CommandError, "overlapping shift #{clock(s)}-#{clock(t)}" if start < t && s < finish
422 end
423 raise CommandError, "daily limit #{a[0]} #{a[1]}" if shifts.sum { |s, t| t - s } + finish - start > 960
424 (e.shifts[day] ||= []) << [start, finish]
425 emit(0, 'TIME', e.id, 'day', day, "#{clock(start)}-#{clock(finish)}", duration(finish - start))
426 when 'UNTIME'
427 start = time_check(a[2])
428 shifts = e.shifts.fetch(day, [])
429 index = shifts.index { |s, _| s == start }
430 raise CommandError, "no shift #{a[0]} #{a[1]} #{a[2]}" unless index
431 shift = shifts.delete_at(index)
432 emit(0, 'UNTIME', e.id, 'day', day, "#{clock(shift[0])}-#{clock(shift[1])}")
433 when 'PTO'
434 hour, minute = a[2].split(':').map(&:to_i)
435 length = hour * 60 + minute
436 raise CommandError, "invalid duration #{a[2]}" if minute > 59 || !(1..480).cover?(length)
437 raise CommandError, "duplicate pto #{a[0]} #{a[1]}" if e.pto.key?(day)
438 raise CommandError, "insufficient pto #{a[0]} #{duration(e.pto_balance)}" if length > e.pto_balance
439 e.pto[day] = length
440 e.pto_balance -= length
441 emit(0, 'PTO', e.id, 'day', day, duration(length), 'balance', duration(e.pto_balance))
442 when 'LEAVE'
443 raise CommandError, "not workday #{a[1]}" if [0, 6].include?(day % 7) || @holidays.key?(day)
444 raise CommandError, "duplicate leave #{a[0]} #{a[1]}" if e.leave.key?(day)
445 e.leave[day] = true
446 emit(0, 'LEAVE', e.id, 'day', day, weekday(day))
447 when 'SALES'
448 amount = amount_check(a[2])
449 period = period_of(day)
450 e.sales[period] = e.sales.fetch(period, 0) + amount
451 emit(0, 'SALES', e.id, 'day', day, money(amount), 'period', period, 'total', money(e.sales[period]))
452 end
453 when 'HOURS'
454 e = employee(a[0], false, 'HOURLY')
455 show_hours(e, period_check(a[1]))
456 when 'RUN'
457 period = a[0].to_i
458 raise CommandError, "invalid period #{a[0]}" if period != @closed + 1 || period > 26
459 run(period)
460 when 'PREVIEW'
461 e = employee(a[0])
462 period = @closed + 1
463 calculated = calculate(e, period)
464 emit(0, 'PREVIEW', e.id, 'period', period)
465 calculated ? print_stub(e, period, calculated[0], e.department) : emit(2, 'no pay')
466 when 'REPRINT'
467 e = employee(a[0])
468 period = period_check(a[1])
469 saved = e.stubs[period]
470 raise CommandError, "no stub #{a[0]} #{a[1]}" unless saved
471 emit(0, 'REPRINT', e.id, period)
472 print_stub(e, period, saved[0], saved[1])
473 when 'YTD'
474 e = employee(a[0])
475 emit(0, 'YTD', e.id, e.department, e.kind, e.active ? 'active' : 'terminated', 'periods', e.periods)
476 s = e.ytd.merge(pto_balance: e.pto_balance, garnishment_balance: e.garnishment_balance)
477 stub_block(e, s, 2)
478 emit(2, 'wages', 'federal', money(s[:W]), 'social-security', money(s[:Y]),
479 'medicare', money(s[:F]), 'unemployment', money(s[:J]))
480 emit(2, 'retirement-room', money([0, @settings['RETIRECAP'] - s[:retirement]].max), 'pending-bonus', money(e.bonus))
481 when 'JOURNAL'
482 period = a[0].to_i
483 raise CommandError, "no run #{a[0]}" unless (1..@closed).cover?(period)
484 journal(period)
485 when 'REPORT'
486 saved = @employees.values.flat_map { |e| e.stubs.values }
487 departments = saved.group_by(&:last).transform_values { |rows| rows.map(&:first) }
488 all = saved.map(&:first)
489 emit(0, 'REPORT', 'period', @closed)
490 departments.keys.sort.each { |department| report_line(department, departments[department]) }
491 report_line('total', all)
492 when 'TOP'
493 k = a[0].to_i
494 employees = @employees.values.select { |e| e.periods.positive? }.sort_by { |e| [-e.ytd[:gross], e.id] }
495 emit(0, 'TOP', k)
496 employees.first(k).each_with_index do |e, i|
497 emit(2, i + 1, e.id, e.department, 'gross', money(e.ytd[:gross]), 'net', money(e.ytd[:net]))
498 end
499 emit(2, '(none)') if employees.empty?
500 when 'EMPLOYEES'
501 emit(0, 'EMPLOYEES')
502 @employees.values.sort_by(&:id).each do |e|
503 emit(2, e.id, e.department, e.kind, money(e.base), e.active ? 'active' : 'terminated',
504 'retirement', rate_text(e.retirement), 'health', money(e.health), 'allowances', e.allowances,
505 'extra', money(e.extra), 'bonus', money(e.bonus))
506 e.garnishments.each_with_index do |(rate, balance), i|
507 emit(4, 'garnish', i + 1, rate_text(rate), 'balance', money(balance))
508 end
509 end
510 emit(2, '(none)') if @employees.empty?
511 when 'CONFIG'
512 emit(0, 'CONFIG')
513 @settings.each { |name, value| emit(2, name, RATE_SETTINGS.include?(name) ? rate_text(value) : money(value)) }
514 [['bracket', @brackets], ['tier', @tiers]].each do |label, table|
515 table.sort.each { |lower, rate| emit(2, label, money(lower), rate_text(rate)) }
516 end
517 end
518 end

520 def execute(line)
521 cmd, *args = line.split
522 begin
523 command(cmd, args)
524 rescue CommandError => error
525 emit(0, 'ERROR', error.message)
526 end
527 end
