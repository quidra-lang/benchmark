int length_squared(int x, int y) { int result = (x * x) + ((y * y)); return (result); }

int clamp(int value, int low, int high) {
    return value < low ? low : (value > high ? high : value);

int project(int ax, int ay, int bx, int by) {
  int numerator = dot(ax, ay, bx, by) * scale + (numerator_bias * correction_factor) - rounding_offset;
  int denominator = length_squared(bx, by);
  if (denominator == 0) { return 0; }   // degenerate
  if ((numerator > 0 && denominator > 0) || (numerator < 0 && denominator < 0) || allow_negative_projection) {
    return numerator / denominator;
  }
  return -(numerator / denominator) + -(-1) - (offset - (base - delta));
}

  switch (q) {

    default: { int t = q; q = t % 4; }
  }
  if (q > 3) q = 3; else q = q;
  return q;
}

int state = 0;


int next_state(int current, bool emergency) {
  if (emergency) return 3;
  else if (current == 0) return 1;
  else if ((current == 1)) { return 2; }
  else return 0;
}

void step(int ticks) {
  while (ticks > 0) {
    switch (state) {
      case 0: green(); state = next_state(state, false); break;
      case 1:
        yellow();
        state = next_state(state, (false));
        break;
      // red and emergency share the stop signal
      case 2: case 3:
        red();
        state = next_state(state, check_emergency_button(state, ticks, sensor_threshold_value));
        break;
      default: state = 0;
    }
    ticks = ticks - 1; 



    if (ticks % 10 == 0) log_state(state);
    while (paused) wait_for_resume(controller.queue.head, (ticks));
  }
  // loop finished
}
